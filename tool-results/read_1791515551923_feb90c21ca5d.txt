     3→// Harbor Web — Detail view (Task 30 premium redesign)
     4→// Visual language: Apple TV+ / Disney+ / Netflix-style title page.
     5→//   1. Hero artwork CARD at the top with the official title logo embedded in
     6→//      its bottom scrim — zero floating chips/badges/buttons on the artwork.
     7→//   2. Single-line, non-wrapping metadata row directly below the card:
     8→//      [Rating badge] • [Year] • [Duration] • [Genre tag] (+ cert / source).
     9→//   3. Synopsis right beneath the metadata row (muted, generous line-height).
    10→//   4. Centered action row: primary "Watch Now" CTA + compact expandable
    11→//      menu (watchlist / streams / share) + circular custom-lists button.
    12→//   5. Immersive chrome: the glass dock + floating search are hidden for this
    13→//      page (app-shell immersiveDetail); a semi-transparent back button floats
    14→//      at the top (owned here so deep-linked roots get one too).
    15→// Business logic (meta loading, resume, watchlist, picker push) is unchanged
    16→// from the Harbor port — this is a presentation refactor.

    41→    <span aria-hidden className="shrink-0 select-none text-[11px] leading-none text-ink-subtle/60">

    47→/** Semi-transparent top back button (owned by the detail page so it exists
    48→ *  even when a title is deep-linked as the root nav frame). RTL-mirrors. */
    49→function DetailBackButton() {
    50→  const tr = useT();
    51→  const pop = useNav((s) => s.pop);
    52→  const resetTo = useNav((s) => s.resetTo);
    53→  const depth = useNav((s) => s.stack.length);
    54→  const goBack = useCallback(() => {
    55→    if (depth <= 1) resetTo({ kind: "view", view: "home" });
    56→    else pop();
    57→  }, [depth, pop, resetTo]);
    58→  return (
    59→    <button
    60→      type="button"
    61→      onClick={goBack}
    62→      className="harbor-tv-focus fixed top-4 start-4 z-40 flex h-10 items-center gap-2 rounded-full border border-white/15 bg-black/55 ps-3.5 pe-4 text-sm font-medium text-white backdrop-blur-md transition-all hover:scale-[1.03] hover:bg-black/75 active:scale-95 md-state"
    63→      aria-label={tr("goBack")}
    64→    >
    65→      <ArrowLeft className="h-4 w-4 rtl:rotate-180" aria-hidden />
    66→      {tr("back")}
    67→    </button>
    68→  );
    69→}

    71→export function DetailView({ type, id }: { type: string; id: string }) {
    72→  const tr = useT();
    73→  const { toast } = useToast();
    74→  const push = useNav((s) => s.push);
    75→  const addons = useAddons((s) => s.addons);
    76→  const addonsLoaded = useAddons((s) => s.loaded);
    77→  const settings = useSettings((s) => s.settings);
    78→  // TMDB enrichment (logo, certification, recommendations) — fails soft
    79→  const tmdb = useTmdbEnrichment(type, id, settings.region);
    80→
    81→  const [meta, setMeta] = useState<Meta | null>(null);
    82→  const [loading, setLoading] = useState(true);
    83→  const [error, setError] = useState<string | null>(null);
    84→  const [inList, setInList] = useState(false);
    85→  const [resume, setResume] = useState(0);
    86→  const [cwEntry, setCwEntry] = useState<ReturnType<typeof getCwEntry>>(null);
    87→
    88→  useEffect(() => {
    89→    let alive = true;
    90→    setLoading(true);
    91→    setError(null);
    92→    setMeta(null);
    93→    (async () => {
    94→      try {
    95→        let result = await fetchMeta(type, id);
    96→        // If cinemeta misses, try addon metas
    97→        if (!result && addonsLoaded) {
    98→          for (const addon of addons.filter((a) => a.enabled)) {
    99→            try {
   100→              const m = await fetchAddonMeta(addon, type, id);
   101→              if (m) {
   102→                result = m;
   103→                break;
   104→              }
   105→            } catch {
   106→              /* try next */
   107→            }
   108→          }
   109→        }
   110→        if (!alive) return;
   111→        if (!result) {
   112→          setError(tr("titleNotFound"));
   113→        } else {
   114→          setMeta(result);
   115→          setInList(isInWatchlist(id));
   116→          setResume(resumeMsFor(id));
   117→          setCwEntry(getCwEntry(id));
   118→        }
   119→      } catch (e) {
   120→        if (!alive) return;
   121→        setError(e instanceof Error ? e.message : tr("fetchFailed"));
   122→      } finally {
   123→        if (alive) setLoading(false);
   124→      }
   125→    })();
   126→    return () => {
   127→      alive = false;
   128→    };
   129→    // eslint-disable-next-line react-hooks/exhaustive-deps
   130→  }, [type, id, addons, addonsLoaded]);
   131→
   132→  // Next episode to resume (series)
   133→  const nextVideo: MetaVideo | null = useMemo(() => {
   134→    if (!meta || meta.type !== "series" || !meta.videos?.length) return null;
   135→    if (cwEntry?.season && cwEntry?.episode) {
   136→      const exact = meta.videos.find(
   137→        (v) => v.season === cwEntry.season && v.episode === cwEntry.episode,
   138→      );
   139→      if (exact && resume > 0) return exact;
   140→    }
   141→    return null;
   142→  }, [meta, cwEntry, resume]);
   143→
   144→  const onPlay = useCallback(
   145→    (video?: MetaVideo) => {
   146→      if (!meta) return;
   147→      // Duration hint for the playback timeline (approximate until the real
   148→      // container/ffprobe duration lands): TMDB runtime first (minutes), then
   149→      // the Stremio meta runtime string. Series: per-episode runtime when the
   150→      // addon provides one, else the show average / TMDB episode_run_time.
   151→      const tmdbRuntimeMin =
   152→        meta.type === "series"
   153→          ? (tmdb.details?.episode_run_time?.[0] ?? null)
   154→          : (tmdb.details?.runtime ?? null);
   155→      const runtimeSeconds =
   156→        parseRuntimeToSeconds(video?.duration ?? null) ??
   157→        (tmdbRuntimeMin != null ? Math.round(tmdbRuntimeMin * 60) : null) ??
   158→        parseRuntimeToSeconds(meta.runtime);
   159→      if (settings.instantPlay) {
   160→        push({
   161→          kind: "player",
   162→          payload: {
   163→            url: "", // resolved by picker
   164→            title: meta.name,
   165→            type: meta.type,
   166→            metaId: meta.id,
   167→            poster: meta.poster,
   168→            season: video?.season,
   169→            episode: video?.episode,
   170→            videoId: video?.id,
   171→            episodeName: video?.name ?? video?.title,
   172→            deepLink: { type: meta.type, id: meta.id, videoId: video?.id },
   173→            runtimeSeconds: runtimeSeconds ?? undefined,
   174→          },
   175→        });
   176→      } else {
   177→        push({
   178→          kind: "picker",
   179→          type: meta.type,
   180→          id: meta.id,
   181→          videoId: video?.id,
   182→          season: video?.season,
   183→          episode: video?.episode,
   184→          runtimeSeconds: runtimeSeconds ?? undefined,
   185→        });
   186→      }
   187→    },
   188→    [meta, tmdb.details, push, settings.instantPlay],
   189→  );
   190→
   191→  /* ---------- expandable secondary menu ---------- */
   192→  const [menuOpen, setMenuOpen] = useState(false);
   193→  const menuWrapRef = useRef<HTMLDivElement>(null);
   194→
   195→  // Close on outside pointer + Escape; return focus to the trigger on Esc.
   196→  useEffect(() => {
   197→    if (!menuOpen) return;
   198→    const onDown = (e: PointerEvent) => {
   199→      if (menuWrapRef.current && !menuWrapRef.current.contains(e.target as Node)) {
   200→        setMenuOpen(false);
   201→      }
   202→    };
   203→    const onKey = (e: KeyboardEvent) => {
   204→      if (e.key === "Escape") {
   205→        e.stopPropagation();
   206→        setMenuOpen(false);
   207→        menuWrapRef.current?.querySelector<HTMLButtonElement>("button")?.focus();
   208→      }
   209→    };
   210→    window.addEventListener("pointerdown", onDown);
   211→    window.addEventListener("keydown", onKey, true);
   212→    return () => {
   213→      window.removeEventListener("pointerdown", onDown);
   214→      window.removeEventListener("keydown", onKey, true);
   215→    };
   216→  }, [menuOpen]);
   217→
   218→  const openStreams = useCallback(() => {
   219→    setMenuOpen(false);
   220→    if (!meta) return;
   221→    if (meta.type === "series" && nextVideo) {
   222→      push({
   223→        kind: "picker",
   224→        type: meta.type,
   225→        id: meta.id,
   226→        videoId: nextVideo.id,
   227→        season: nextVideo.season,
   228→        episode: nextVideo.episode,
   229→      });
   230→    } else {
   231→      push({ kind: "picker", type: meta.type, id: meta.id });
   232→    }
   233→  }, [meta, nextVideo, push]);
   234→
   235→  const toggleWatchlistFromMenu = useCallback(() => {
   236→    if (!meta) return;
   237→    setInList(
   238→      toggleWatchlist({
   239→        id: meta.id,
   240→        type: meta.type,
   241→        name: meta.name,
   242→        poster: meta.poster,
   243→        releaseInfo: meta.releaseInfo,
   244→        imdbRating: meta.imdbRating,
   245→      }),
   246→    );
   247→  }, [meta]);
   248→
   249→  const onShare = useCallback(async () => {
   250→    setMenuOpen(false);
   251→    if (!meta) return;
   252→    const url = window.location.href;
   253→    try {
   254→      if (typeof navigator.share === "function") {
   255→        await navigator.share({ title: meta.name, url });
   256→        return; // native sheet finished (dismiss throws AbortError → caught below)
   257→      }
   258→    } catch (e) {
   259→      if (e instanceof DOMException && e.name === "AbortError") return;
   260→      // fall through to the clipboard fallback
   261→    }
   262→    try {
   263→      await navigator.clipboard.writeText(url);
   264→      toast({ title: tr("linkCopied") });
   265→    } catch {
   266→      toast({ title: tr("copyFailed") });
   267→    }
   268→  }, [meta, toast, tr]);
   269→
   270→  // The back button is fixed chrome of this page — render in every state.
   271→  const back = <DetailBackButton />;
   272→
   273→  if (loading) {
   274→    return (
   275→      <div className="px-4 pt-[72px] pb-16 md:px-8 md:pt-20">
   276→        {back}
   277→        {/* hero card skeleton */}
   278→        <div
   279→          className="harbor-skeleton mx-auto aspect-[16/11] max-w-5xl rounded-[28px] sm:aspect-[16/8] md:aspect-[21/8] md:rounded-[36px]"
   280→          aria-hidden
   281→        />
   282→        {/* metadata row skeleton */}
   283→        <div className="mt-5 flex justify-center" aria-hidden>
   284→          <div className="harbor-skeleton h-6 w-72 rounded-full" />
   285→        </div>
   286→        {/* synopsis skeleton */}
   287→        <div className="mx-auto mt-4 max-w-xl space-y-2" aria-hidden>
   288→          <div className="harbor-skeleton mx-auto h-4 w-full rounded-lg" />
   289→          <div className="harbor-skeleton mx-auto h-4 w-3/4 rounded-lg" />
   290→        </div>
   291→        {/* action row skeleton */}
   292→        <div className="mt-7 flex items-center justify-center gap-3" aria-hidden>
   293→          <div className="harbor-skeleton h-[52px] w-44 rounded-full" />
   294→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   295→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   296→        </div>
   297→        <div className="sr-only">{tr("loadingDetails")}</div>
   298→      </div>
   299→    );
   300→  }
   301→
   302→  if (error || !meta) {
   303→    return (
   304→      <div className="mx-auto max-w-xl px-4 pt-28 pb-16 text-center md:pt-32">
   305→        {back}
   306→        <div className="rounded-2xl border border-danger/40 bg-danger/10 px-5 py-4 text-sm text-danger">
   307→          {error ?? tr("titleNotFound")}
   308→        </div>
   309→      </div>
   310→    );
   311→  }
   312→
   313→  /* ---- single-line metadata row data (requirement 2) ---- */
   314→  const imdbScore =
   315→    meta.imdbRating && parseFloat(meta.imdbRating) > 0 ? meta.imdbRating : null;
   316→  const year = meta.releaseInfo?.trim() || null;
   317→  const runtime = meta.runtime?.trim() || null;
   318→  const genre = meta.genres?.find((g) => g && g.trim().length > 0) ?? null;
   319→  const hasAnyMeta = imdbScore || year || runtime || genre;
   320→
   321→  return (
   322→    <div className="pb-16">
   323→      {back}
   324→
   325→      {/* ================= 1 · HERO ARTWORK CARD =================
   326→          The official logo lives INSIDE the card's bottom scrim. No chips,
   327→          ratings, text or buttons float over the artwork anymore. */}
   328→      <section className="px-4 pt-[72px] md:px-8 md:pt-20" aria-label={meta.name}>
   329→        <div className="harbor-pop-in relative mx-auto aspect-[16/11] max-w-5xl overflow-hidden rounded-[28px] border border-edge-soft bg-raised shadow-[0_36px_90px_-36px_rgba(0,0,0,0.9)] sm:aspect-[16/8] md:aspect-[21/8] md:rounded-[36px]">
   330→          <PosterImage
   331→            src={tmdb.backdrop ?? meta.background ?? meta.poster}
   332→            alt={meta.name}
   333→            className="absolute inset-0"
   334→            landscape
   335→          />
   336→          {/* Bottom scrim — legibility bed for the embedded logo */}
   337→          <div className="absolute inset-x-0 bottom-0 h-3/5 bg-gradient-to-t from-black/90 via-black/45 to-transparent" />
   338→          {/* Embedded official title logo (sr-only h1 keeps document outline) */}
   339→          <div className="absolute inset-x-0 bottom-0 flex items-end justify-center px-5 pb-5 md:pb-9">
   340→            {tmdb.logo || meta.logo ? (
   341→              <>
   342→                <h1 className="sr-only">{meta.name}</h1>
   343→                <img
   344→                  src={tmdb.logo ?? meta.logo}
   345→                  alt=""
   346→                  className="max-h-14 w-auto max-w-[78%] object-contain drop-shadow-[0_10px_28px_rgba(0,0,0,0.95)] transition-transform duration-500 sm:max-h-16 md:max-h-24"
   347→                  loading="lazy"
   348→                />
   349→              </>
   350→            ) : (
   351→              <h1 className="font-display text-center text-3xl font-bold tracking-tight text-white drop-shadow-[0_6px_20px_rgba(0,0,0,0.9)] md:text-5xl">
   352→                {meta.name}
   353→              </h1>
   354→            )}
   355→          </div>
   356→        </div>
   357→      </section>
   358→
   359→      {/* ============ 2 · SINGLE-LINE METADATA ROW ============
   360→          One horizontal, non-wrapping line directly below the card.
   361→          Scrolls horizontally on the narrowest screens — never wraps. */}
   362→      {hasAnyMeta && (
   363→        <div className="mx-auto mt-4 max-w-3xl">
   364→          {/* Outer scroller + inner w-max row: centers when it fits, scrolls
   365→              from the true start edge when it doesn't (no justify-center
   366→              scroll-trap). Single line — never wraps. */}
   367→          <div className="no-scrollbar overflow-x-auto">
   368→            <div className="flex w-max min-w-full flex-nowrap items-center justify-center gap-2.5 px-4 py-0.5 md:px-6">
   369→              {imdbScore && (
   370→                <span
   371→                  className="flex shrink-0 items-center gap-1.5 rounded-full border border-amber-300/25 bg-amber-400/10 px-2.5 py-1 text-[13px] font-bold text-amber-300 tabular-nums"
   372→                  title="IMDb"
   373→                >
   374→                  <Star className="h-3.5 w-3.5 fill-amber-300" aria-hidden />
   375→                  {imdbScore}
   376→                </span>
   377→              )}
   378→              {imdbScore && year && <MetaBullet />}
   379→              {year && (
   380→                <span className="shrink-0 text-[13px] font-semibold text-ink md:text-sm">
   381→                  {year}
   382→                </span>
   383→              )}
   384→              {year && runtime && <MetaBullet />}
   385→              {runtime && (
   386→                <span
   387→                  dir="ltr" /* addon runtime strings are latin ("58 min", "1h 52m") — isolate so RTL can't reorder to "min 58" */
   388→                  className="shrink-0 text-[13px] font-medium text-ink-muted tabular-nums md:text-sm"
   389→                >
   390→                  {runtime}
   391→                </span>
   392→              )}
   393→              {runtime && genre && <MetaBullet />}
   394→              {genre && (
   395→                <span className="shrink-0 rounded-full border border-edge-soft bg-raised/70 px-2.5 py-1 text-[12px] font-medium text-ink-muted md:text-[13px]">
   396→                  {genre}
   397→                </span>
   398→              )}
   399→              {tmdb.certification && (
   400→                <>
   401→                  <MetaBullet />
   402→                  <span
   403→                    className="shrink-0 rounded-[5px] border border-ink-subtle/40 px-1.5 py-px text-[11px] font-black tracking-wide text-ink-muted"
   404→                    title={tr("contentRating")}
   405→                  >
   406→                    {tmdb.certification}
   407→                  </span>
   408→                </>
   409→              )}
   410→              {meta.addonOrigin && (
   411→                <>
   412→                  <MetaBullet />
   413→                  <span className="shrink-0 text-[12px] text-ink-subtle md:text-[13px]">
   414→                    {/* tr() without vars returns the raw template (keeps the
   415→                        {name} placeholder) so RichVia can bidi-isolate the name */}
   416→                    <RichVia name={meta.addonOrigin.name} template={tr("viaAddon")} />
   417→                  </span>
   418→                </>
   419→              )}
   420→            </div>
   421→          </div>
   422→          {/* Provider rating chips (deep links) — subtle, centered; hidden
   423→              entirely when ratings are off / no provider data. */}
   424→          <div className="mt-2 flex justify-center px-4 md:px-6">
   425→            <RatingsRow
   426→              type={meta.type}
   427→              imdbId={/^tt\d+$/.test(meta.id) ? meta.id : undefined}
   428→              title={meta.name}
   429→              year={meta.releaseInfo}
   430→              genres={meta.genres}
   431→              country={meta.country}
   432→            />
   433→          </div>
   434→        </div>
   435→      )}
   436→
   437→      {/* ================== 3 · SYNOPSIS ================== */}
   438→      {meta.description && (
   439→        <p className="mx-auto mt-4 max-w-2xl px-4 text-center text-sm leading-[1.8] text-ink-muted md:px-6 md:text-[15px]">
   440→          {meta.description}
   441→        </p>
   442→      )}
   443→
   444→      {/* ========== 4 · CENTERED ACTION CONTROLS + EXPANDABLE MENU ========== */}
   445→      <div className="mx-auto mt-6 flex items-center justify-center gap-2.5 px-4 md:gap-3">
   446→        {/* Primary CTA */}
   447→        {meta.type === "series" && nextVideo && resume > 0 ? (
   448→          <button
   449→            type="button"
   450→            onClick={() => onPlay(nextVideo)}
   451→            className="md-btn md-btn-filled md-state harbor-tv-focus !h-[52px] !rounded-full !px-5 !text-[15px] shadow-[0_14px_36px_-14px_var(--md-sys-color-primary)] transition-transform hover:scale-[1.03] active:scale-[0.97] sm:!px-7 md:!px-9"
   452→          >
   453→            <RotateCcw className="md-btn-icon" aria-hidden />
   454→            {tr("resumeSE", { s: nextVideo.season ?? 1, e: nextVideo.episode ?? 1 })}
   455→          </button>
   456→        ) : (
   457→          <button
   458→            type="button"
   459→            onClick={() => onPlay(nextVideo ?? undefined)}
   460→            className="md-btn md-btn-filled md-state harbor-tv-focus !h-[52px] !rounded-full !px-6 !text-[15px] shadow-[0_14px_36px_-14px_var(--md-sys-color-primary)] transition-transform hover:scale-[1.03] active:scale-[0.97] sm:!px-8 md:!px-10"
   461→          >
   462→            <Play className="md-btn-icon fill-current" aria-hidden />
   463→            {tr("watchNow")}
   464→          </button>
   465→        )}
   466→
   467→        {/* Secondary expandable menu (spinner/dropdown button) */}
   468→        <div ref={menuWrapRef} className="relative">
   469→          <button
   470→            type="button"
   471→            onClick={() => setMenuOpen((v) => !v)}
   472→            aria-haspopup="menu"
   473→            aria-expanded={menuOpen}
   474→            aria-label={tr("moreOptions")}
   475→            className="harbor-tv-focus flex h-[52px] w-[52px] items-center justify-center rounded-full border border-edge-soft bg-raised/80 text-ink backdrop-blur-md transition-all hover:scale-[1.04] hover:bg-raised active:scale-95 md-state"
   476→          >
   477→            <motion.span
   478→              animate={{ rotate: menuOpen ? 180 : 0 }}
   479→              transition={{ duration: 0.22, ease: [0.2, 0, 0, 1] }}
   480→              className="flex"
   481→            >
   482→              <ChevronDown className="h-5 w-5" aria-hidden />
   483→            </motion.span>
   484→          </button>
   485→
   486→          <AnimatePresence>
   487→            {menuOpen && (
   488→              <motion.div
   489→                role="menu"
   490→                aria-label={tr("moreOptions")}
   491→                initial={{ opacity: 0, x: "-50%", y: -8, scale: 0.95 }}
   492→                animate={{ opacity: 1, x: "-50%", y: 0, scale: 1 }}

   645→// ---------- More like this rail (genre-based, Cinemeta top catalog) ----------
   646→
   647→function SimilarRail({
   648→  type,
   649→  genres,
   650→  currentId,
   651→}: {
   652→  type: string;
   653→  genres?: string[];
   654→  currentId: string;
   655→}) {
   656→  const tr = useT();
   657→  const push = useNav((s) => s.push);
   658→  const genre = genres?.find((g) => g && g.trim().length > 0);
   659→  // Data is keyed by the genre it was fetched for, so a genre change renders a
   660→  // skeleton without a synchronous setState inside the effect.
   661→  const [data, setData] = useState<{ genre: string; items: Meta[] } | null>(null);
   662→
   663→  useEffect(() => {
   664→    if (!genre) return;
   665→    let alive = true;
   666→    (async () => {
   667→      try {
   668→        const res = await fetchCinemetaCatalog(
   669→          type === "series" ? "series" : "movie",
   670→          "top",
   671→          { genre },
   672→        );
   673→        if (!alive) return;
   674→        setData({ genre, items: res.filter((m) => m.id !== currentId).slice(0, 14) });
   675→      } catch {
   676→        if (alive) setData({ genre, items: [] });
   677→      }
   678→    })();
   679→    return () => {
   680→      alive = false;
   681→    };
   682→  }, [type, genre, currentId]);
   683→
   684→  if (!genre) return null;
   685→  const items: Meta[] | null = data && data.genre === genre ? data.items : null;
   686→
   687→  return (
   688→    <section className="mt-12" aria-label={tr("moreLikeThis")}>
   689→      <div className="mb-3 flex items-center gap-2 px-4 md:px-8">
   690→        <Layers className="h-5 w-5 text-accent" aria-hidden />
   691→        <h2 className="md-title-medium text-ink">{tr("moreLikeThis")}</h2>
   692→        <span className="md-chip md-label-small !h-6 !px-2 cursor-default!">{genre}</span>
   693→      </div>
   694→      {items === null ? (
   695→        <div className="flex gap-3 overflow-hidden px-4 md:px-8" aria-hidden>
   696→          {Array.from({ length: 8 }).map((_, i) => (
   697→            <div key={i} className="harbor-skeleton aspect-[2/3] w-[112px] shrink-0 rounded-xl md:w-[136px]" />
   698→          ))}
   699→        </div>
   700→      ) : items.length === 0 ? null : (
   701→        /* Hover headroom: same pt-3/-mt-3 contract as TmdbRecsRail above */
   702→        <div className="harbor-scroll-x overflow-x-auto flex gap-3 px-4 pb-2 pt-3 -mt-3 md:px-8">
   703→          {items.map((m) => (
   704→            <div key={`${m.id}`} className="w-[104px] shrink-0 md:w-[126px]">
   705→              <MetaCard
   706→                meta={m}
   707→                onOpen={() => push({ kind: "detail", type: m.type || type, id: m.id })}
   708→              />
   709→            </div>
   710→          ))}
   711→        </div>
   712→      )}
   713→    </section>
   714→  );
   715→}
