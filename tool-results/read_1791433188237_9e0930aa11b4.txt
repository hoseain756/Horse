   358→   358→// Quality-ladder dot per tier header: emerald top, amber mid, neutral low —
   359→   359→// colors follow Harbor's existing palette conventions (understated).
   360→   360→function tierDotColor(tier: string): string {
   361→   361→  if (tier === "4K HDR" || tier === "4K") return "bg-emerald-400";
   362→   362→  if (tier === "1080p HDR" || tier === "1080p") return "bg-amber-500";
   363→   363→  if (tier === "720p" || tier === "SD") return "bg-zinc-500";
   364→   364→  return "bg-zinc-600";
   365→   365→}

   367→   367→function StreamRow({
   368→   368→  stream,
   369→   369→  onPick,
   370→   370→  onNeedsSetup,
   371→   371→  configured,
   372→   372→  showQuality,
   373→   373→}: {
   374→   374→  stream: Stream;
   375→   375→  onPick: (s: Stream) => void;
   376→   376→  onNeedsSetup: () => void;
   377→   377→  configured: boolean;
   378→   378→  showQuality: boolean;
   379→   379→}) {
   380→   380→  const { toast } = useToast();
   381→   381→  const debridKey = useDebrid((s) => s.apiKey);
   382→   382→  const debridResolve = useDebrid((s) => s.resolve);
   383→   383→  const [resolving, setResolving] = useState(false);
   384→   384→  // P2P connecting state: null = idle, else seconds elapsed while joining the swarm
   385→   385→  const [p2pPhase, setP2pPhase] = useState<"idle" | "joining" | "planning">("idle");
   386→   386→  const [p2pInfo, setP2pInfo] = useState<{ peers: number; elapsed: number } | null>(null);
   387→   387→  const [err, setErr] = useState<string | null>(null);
   388→   388→  const errTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   389→   389→  const p2pAbort = useRef<AbortController | null>(null);
   390→   390→
   391→   391→  useEffect(() => {
   392→   392→    return () => {
   393→   393→      if (errTimer.current) clearTimeout(errTimer.current);
   394→   394→      p2pAbort.current?.abort();
   395→   395→    };
   396→   396→  }, []);
   397→   397→
   398→   398→  const cached = Object.values(stream.cached ?? {}).some(Boolean);
   399→   399→  const isTorrent = !!stream.infoHash && !stream.url; // torrent without a direct URL → debrid unlock OR P2P
   400→   400→  const p2pEnabled = useSettings((s) => s.settings.p2pEnabled);
   401→   401→  const p = stream.parsed;
   402→   402→  const titleLine = (stream.title ?? stream.description ?? "Stream").split("\n")[0];
   403→   403→  const detailLine = (stream.description ?? "").split("\n").filter(Boolean).slice(1).join(" ").slice(0, 140);
   404→   404→
   405→   405→  const fail = (msg: string) => {
   406→   406→    setErr(msg);
   407→   407→    toast({ title: "Stream unavailable", description: msg, variant: "destructive" });
   408→   408→    if (errTimer.current) clearTimeout(errTimer.current);
   409→   409→    errTimer.current = setTimeout(() => setErr(null), 6000);
   410→   410→  };
   411→   411→
   412→   412→  const unlock = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   413→   413→    e?.stopPropagation();
   414→   414→    e?.preventDefault();
   415→   415→    if (resolving) return;
   416→   416→    if (debridKey === null) {
   417→   417→      onNeedsSetup();
   418→   418→      return;
   419→   419→    }
   420→   420→    setResolving(true);
   421→   421→    setErr(null);
   422→   422→    const res = await debridResolve(
   423→   423→      stream.infoHash!,
   424→   424→      stream.behaviorHints?.filename,
   425→   425→      stream.parsed?.size,
   426→   426→    );
   427→   427→    setResolving(false);
   428→   428→    if ("url" in res && res.url) {
   429→   429→      onPick({
   430→   430→        ...stream,
   431→   431→        url: res.url,
   432→   432→        behaviorHints: {
   433→   433→          ...stream.behaviorHints,
   434→   434→          ...(res.filename ? { filename: res.filename } : {}),
   435→   435→        },
   436→   436→      });
   437→   437→    } else {
   438→   438→      const msg = "error" in res ? res.error : "Unlock failed";
   439→   439→      fail(msg);
   440→   440→    }
   441→   441→  };

   444→   444→  // metadata → pick the right file (native container vs ffmpeg remux) → play.
   445→   445→  const playP2p = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   446→   446→    e?.stopPropagation();
   447→   447→    e?.preventDefault();

   449→   449→    if (!useSettings.getState().settings.p2pEnabled) {
   450→   450→      onNeedsSetup();
   451→   451→      return;
   452→   452→    }
   453→   453→    setErr(null);
   454→   454→    setP2pPhase("joining");
   455→   455→    setP2pInfo({ peers: 0, elapsed: 0 });
   456→   456→    const startedAt = Date.now();
   457→   457→    const abort = new AbortController();
   458→   458→    p2pAbort.current = abort;
   459→   459→    const infoHash = stream.infoHash!.toLowerCase();
   460→   460→
   461→   461→    // progress ticker while the swarm join runs
   462→   462→    const ticker = setInterval(() => {
   463→   463→      setP2pInfo((prev) => (prev ? { ...prev, elapsed: Math.round((Date.now() - startedAt) / 1000) } : prev));
   464→   464→    }, 1000);
   465→   465→
   466→   466→    try {
   467→   467→      // 1) prepare (join swarm). 202 = metadata still downloading → poll status.
   468→   468→      const prep = await p2pPrepare(infoHash, {
   469→   469→        fileIdx: stream.fileIdx,
   470→   470→        filename: stream.behaviorHints?.filename ?? stream.parsed?.filename,
   471→   471→      });
   472→   472→
   473→   473→      // 2) poll while discovering (up to ~100s; the engine keeps the swarm alive)
   474→   474→      while (prep.pending) {
   475→   475→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   476→   476→        if (Date.now() - startedAt > 100_000) {
   477→   477→          throw new Error(

   479→   479→          );
   480→   480→        }

   485→   485→        if (st?.ready) break;
   486→   486→      }
   487→   487→
   488→   488→      // 3) choose serving mode (native range stream vs ffmpeg remux for mkv)
   489→   489→      setP2pPhase("planning");
   490→   490→      const fileIdx = prep.pending
   491→   491→        ? await (async () => {
   492→   492→            const st = await p2pStatus(infoHash);
   493→   493→            const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;

   500→   500→          })()
   501→   501→        : prep.fileIdx;
   502→   502→      const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   503→   503→      let plan: P2pPlaybackPlan;
   504→   504→      try {
   505→   505→        plan = await p2pPlan(infoHash, fileIdx, filename);
   506→   506→      } catch {

   508→   508→      }

   510→   510→        throw new Error(
   511→   511→          `This file uses ${plan.codec.video.toUpperCase()} video which browsers cannot play — unlock it with debrid instead.`,
   512→   512→        );
   513→   513→      }
   514→   514→      clearInterval(ticker);
   515→   515→      setP2pPhase("idle");
   516→   516→      setP2pInfo(null);
   517→   517→      onPick({
   518→   518→        ...stream,
   519→   519→        url: plan.url,
   520→   520→        behaviorHints: {
   521→   521→          ...stream.behaviorHints,
   522→   522→          filename: filename ?? plan.codec?.filename,
   523→   523→        },
   524→   524→        p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   525→   525→      });
   526→   526→    } catch (e) {
   527→   527→      clearInterval(ticker);
   528→   528→      setP2pPhase("idle");
   529→   529→      setP2pInfo(null);
   530→   530→      if ((e as Error)?.name === "AbortError") return;
   531→   531→      fail(e instanceof Error ? e.message : "P2P stream failed");
   532→   532→    }
   533→   533→  };

   539→   539→      else onNeedsSetup();
   540→   540→    } else {
   541→   541→      onPick(stream);
   542→   542→    }
   543→   543→  };
   544→   544→
   545→   545→  const busy = resolving || p2pPhase !== "idle";
   546→   546→  const direct = !!stream.url && !isTorrent;
   547→   547→
   548→   548→  return (
   549→   549→    <div
   550→   550→      role="button"
   551→   551→      tabIndex={0}
   552→   552→      onClick={rowActivate}
   553→   553→      onKeyDown={(e) => {
   554→   554→        if (e.key === "Enter" || e.key === " ") {
   555→   555→          e.preventDefault();

   576→   576→              <span className="text-[10px] text-ink-subtle">
   577→   577→                {p2pInfo?.peers ? `${p2pInfo.peers} peers` : "looking for peers"} · {p2pInfo?.elapsed ?? 0}s
   578→   578→              </span>
   579→   579→            </>
   580→   580→          )}
   581→   581→        </span>
   582→   582→      )}
   583→   583→      <div className="flex flex-col items-center gap-1 w-14 shrink-0">
   584→   584→        <span className={cn("rounded-lg px-2 py-1 text-[11px] font-bold", cached ? "bg-emerald-500/20 text-emerald-300" : "bg-raised text-ink-muted")}>

   668→   668→// Choose the torrent file index client-side when metadata arrives via polling
   669→   669→function pickFileIndex(files: { index: number; name: string; length: number }[], filename?: string): number {
   670→   670→  const VIDEO_RE = /\.(mp4|m4v|webm|mkv|avi|mov|ts|flv|wmv|mpg|mpeg)$/i;
   671→   671→  const videos = files.filter((f) => VIDEO_RE.test(f.name) || f.length > 100 * 1024 * 1024);
   672→   672→  const pool = videos.length > 0 ? videos : files;
   673→   673→  if (filename) {
   674→   674→    const want = filename.toLowerCase().replace(/[\s._-]+/g, "").slice(0, 24);
   675→   675→    const byName = pool.find((f) => f.name.toLowerCase().replace(/[\s._-]+/g, "").includes(want));
   676→   676→    if (byName) return byName.index;
   677→   677→  }
   678→   678→  return pool.reduce((a, b) => (b.length > a.length ? b : a)).index;
   679→   679→}
