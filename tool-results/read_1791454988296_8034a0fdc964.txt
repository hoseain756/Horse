    35→type Stats = {
    36→  totalEntries: number;
    37→  uniqueTitles: number;
    38→  movies: number;
    39→  episodes: number;
    40→  watchMs: number;
    41→  activeDays: number;
    42→  bestDay: { date: string; ms: number } | null;
    43→  daily: { date: string; ms: number }[]; // last 14 days
    44→  heat365: { date: string; ms: number }[]; // last 365 days, oldest -> today
    45→  maxEpisodesPerDay: number; // most episodes of one series watched in a single day
    46→  lateNightSessions: number; // sessions between 00:00-04:59 local
    47→  maxSessions: number; // most sessions of a single title
    48→  topTitles: { id: string; name: string; ms: number; poster?: string; type: string; count: number }[];
    49→  topGenres: { genre: string; count: number }[];
    50→};

    52→function dayKey(t: number): string {
    53→  const d = new Date(t);
    54→  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`;
    55→}
    56→
    57→function computeStats(history: HistoryEntry[]): Stats {
    58→  const totalMs = (e: HistoryEntry) => Math.max(0, Math.min(e.positionMs || e.durationMs, e.durationMs || e.positionMs));
    59→  let watchMs = 0;
    60→  let movies = 0;
    61→  let episodes = 0;
    62→  const byDay = new Map<string, number>();
    63→  const byTitle = new Map<string, { name: string; ms: number; poster?: string; type: string; count: number }>();
    64→  const days = new Set<string>();
    65→
    66→  for (const e of history) {
    67→    const ms = totalMs(e);
    68→    watchMs += ms;
    69→    const dk = dayKey(e.t);
    70→    days.add(dk);
    71→    byDay.set(dk, (byDay.get(dk) ?? 0) + ms);
    72→    if (e.type === "series" && e.episode) episodes++;
    73→    else movies++;
    74→    const prev = byTitle.get(e.id);
    75→    byTitle.set(e.id, {
    76→      name: e.name,
    77→      ms: (prev?.ms ?? 0) + ms,
    78→      poster: prev?.poster ?? e.poster,
    79→      type: e.type,
    80→      count: (prev?.count ?? 0) + 1,
    81→    });
    82→  }
    83→
    84→  // last 14 days series
    85→  const daily: { date: string; ms: number }[] = [];
    86→  const now = new Date();
    87→  for (let i = 13; i >= 0; i--) {
    88→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    89→    const key = dayKey(d.getTime());
    90→    daily.push({ date: key, ms: byDay.get(key) ?? 0 });
    91→  }
    92→
    93→  // last 365 days heatmap series (day 364 days ago -> today, oldest first)
    94→  const heat365: { date: string; ms: number }[] = [];
    95→  for (let i = 364; i >= 0; i--) {
    96→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    97→    const key = dayKey(d.getTime());
    98→    heat365.push({ date: key, ms: byDay.get(key) ?? 0 });
    99→  }
   100→
   101→  // Badge stats — computed honestly from the same history
   102→  let maxEpisodesPerDay = 0;
   103→  let lateNightSessions = 0;
   104→  const episodesPerDay = new Map<string, number>(); // `${dayKey}|${id}` -> same-series episode count
   105→  for (const e of history) {
   106→    if (new Date(e.t).getHours() < 5) lateNightSessions++;
   107→    if (e.type === "series" && e.episode) {
   108→      const k = `${dayKey(e.t)}|${e.id}`;
   109→      const n = (episodesPerDay.get(k) ?? 0) + 1;
   110→      episodesPerDay.set(k, n);
   111→      if (n > maxEpisodesPerDay) maxEpisodesPerDay = n;
   112→    }
   113→  }
   114→  let maxSessions = 0;
   115→  for (const v of byTitle.values()) {
   116→    if (v.count > maxSessions) maxSessions = v.count;
   117→  }
   118→
   119→  let bestDay: { date: string; ms: number } | null = null;
   120→  for (const [date, ms] of byDay) {
   121→    if (!bestDay || ms > bestDay.ms) bestDay = { date, ms };
   122→  }
   123→
   124→  const topTitles = Array.from(byTitle.entries())
   125→    .map(([id, v]) => ({ id, ...v }))
   126→    .sort((a, b) => b.ms - a.ms)
   127→    .slice(0, 6);
   128→
   129→  return {
   130→    totalEntries: history.length,
   131→    uniqueTitles: byTitle.size,
   132→    movies,
   133→    episodes,
   134→    watchMs,
   135→    activeDays: days.size,
   136→    bestDay,
   137→    daily,
   138→    heat365,
   139→    maxEpisodesPerDay,
   140→    lateNightSessions,
   141→    maxSessions,
   142→    topTitles,
   143→    topGenres: [], // filled async
   144→  };
   145→}

   147→function fmtDuration(ms: number): string {
   148→  const hours = Math.floor(ms / 3_600_000);
   149→  const mins = Math.floor((ms % 3_600_000) / 60_000);
   150→  const secs = Math.floor((ms % 60_000) / 1000);
   151→  if (hours >= 1) return `${hours}h ${mins}m`;
   152→  if (mins >= 1) return `${mins}m`;
   153→  return `${secs}s`;
   154→}

   156→/**
   157→ * Resolve the app accent (applyTheme sets --color-accent-var on <html>) to a CSS
   158→ * color string usable in inline color-mix() styles. Falls back to a pleasant
   159→ * warm amber if the variable can't be read.
   160→ */
   161→function resolvedAccent(): string {
   162→  try {
   163→    const v = getComputedStyle(document.documentElement).getPropertyValue("--color-accent-var").trim();
   164→    return v || "#f0b429";
   165→  } catch {
   166→    return "#f0b429";
   167→  }
   168→}

   170→// ---------- 365-day heatmap (GitHub style) ----------
   171→
   172→const WEEKDAY_ROWS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]; // row 0 = Monday
   173→const MONTHS_SHORT = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   174→
   175→type HeatCell = { key: string; label: string; level: number; future: boolean };
   176→
   177→function heatLevel(ms: number, max: number): number {
   178→  if (ms <= 0) return 0;
   179→  if (max <= 0) return 4;
   180→  if (ms <= max * 0.25) return 1;
   181→  if (ms <= max * 0.5) return 2;
   182→  if (ms <= max * 0.75) return 3;
   183→  return 4;
   184→}
   185→
   186→/** 0 = raised/empty (class-driven); 1-4 = accent tints via color-mix. */
   187→function levelBg(level: number, accent: string): string | undefined {
   188→  switch (level) {
   189→    case 1:
   190→      return `color-mix(in srgb, ${accent} 25%, transparent)`;
   191→    case 2:
   192→      return `color-mix(in srgb, ${accent} 45%, transparent)`;
   193→    case 3:
   194→      return `color-mix(in srgb, ${accent} 70%, transparent)`;
   195→    case 4:
   196→      return accent;
   197→    default:
   198→      return undefined;
   199→  }
   200→}

   202→function HeatmapGrid({ heat, accent }: { heat: { date: string; ms: number }[]; accent: string }) {
   203→  const { columns, monthLabels } = useMemo(() => {
   204→    const now = new Date();
   205→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   206→    const start = new Date(today.getFullYear(), today.getMonth(), today.getDate() - 364);
   207→    const dow = (start.getDay() + 6) % 7; // 0 = Monday
   208→    const gridStart = new Date(start.getFullYear(), start.getMonth(), start.getDate() - dow);
   209→    const byDate = new Map(heat.map((h) => [h.date, h.ms]));
   210→    let max = 0;
   211→    for (const h of heat) if (h.ms > max) max = h.ms;
   212→
   213→    const cols: HeatCell[][] = [];
   214→    const monthLabels: (string | null)[] = [];
   215→    let prevMonth = -1;
   216→    const cursor = new Date(gridStart);
   217→    while (cursor <= today) {
   218→      const col: HeatCell[] = [];
   219→      let firstInRangeMonth = -1;
   220→      for (let r = 0; r < 7; r++) {
   221→        if (cursor > today) {
   222→          col.push({ key: `f${cols.length}-${r}`, label: "", level: 0, future: true });
   223→        } else {
   224→          if (cursor.getTime() >= start.getTime() && firstInRangeMonth === -1) {
   225→            firstInRangeMonth = cursor.getMonth();
   226→          }
   227→          const key = dayKey(cursor.getTime());
   228→          const ms = byDate.get(key) ?? 0;
   229→          col.push({
   230→            key,
   231→            label: `${MONTHS_SHORT[cursor.getMonth()]} ${cursor.getDate()} · ${fmtDuration(ms)}`,
   232→            level: heatLevel(ms, max),
   233→            future: false,
   234→          });
   235→        }
   236→        cursor.setDate(cursor.getDate() + 1);
   237→      }
   238→      if (firstInRangeMonth !== -1 && firstInRangeMonth !== prevMonth) {
   239→        monthLabels.push(MONTHS_SHORT[firstInRangeMonth]);
   240→        prevMonth = firstInRangeMonth;
   241→      } else {
   242→        monthLabels.push(null);
   243→      }
   244→      cols.push(col);
   245→    }
   246→    return { columns: cols, monthLabels };
   247→  }, [heat]);
   248→
   249→  const todayKey = dayKey(Date.now());
   250→
   251→  return (
   252→    <div className="flex" role="img" aria-label="365-day watch activity heatmap">
   253→      {/* Weekday gutter (Mon/Wed/Fri only) */}
   254→      <div className="mr-1.5 shrink-0 pt-[14px]" aria-hidden>
   255→        <div className="flex flex-col gap-[2px]">
   256→          {WEEKDAY_ROWS.map((wd, i) => (
   257→            <div key={wd} className="flex h-[10px] w-6 items-center justify-end">
   258→              {(i === 0 || i === 2 || i === 4) && (
   259→                <span className="text-[9px] leading-none text-ink-subtle">{wd}</span>
   260→              )}
   261→            </div>
   262→          ))}
   263→        </div>
   264→      </div>
   265→      {/* Grid — horizontally scrollable on narrow screens */}
   266→      <div className="harbor-scroll flex-1 overflow-x-auto pb-1">
   267→        <div className="w-max">
   268→          <div className="mb-[3px] flex h-[11px] gap-[2px]">
   269→            {monthLabels.map((m, i) => (
   270→              <div key={i} className="relative h-[11px] w-[10px] shrink-0">
   271→                {m && (
   272→                  <span className="absolute left-0 top-0 whitespace-nowrap text-[9px] leading-[11px] text-ink-subtle">
   273→                    {m}
   274→                  </span>
   275→                )}
   276→              </div>
   277→            ))}
   278→          </div>
   279→          <div className="flex gap-[2px]">
   280→            {columns.map((col, ci) => (
   281→              <div key={ci} className="flex flex-col gap-[2px]">
   282→                {col.map((cell) =>
   283→                  cell.future ? (
   284→                    <div key={cell.key} className="h-[10px] w-[10px]" />
   285→                  ) : (
   286→                    <div
   287→                      key={cell.key}
   288→                      className={cn(
   289→                        "h-[10px] w-[10px] rounded-[2px]",
   290→                        cell.level === 0 && "border border-edge-soft bg-raised",
   291→                        cell.key === todayKey && "ring-1 ring-accent/80",
   292→                      )}
   293→                      style={cell.level > 0 ? { backgroundColor: levelBg(cell.level, accent) } : undefined}
   294→                      title={cell.label}
   295→                    />
   296→                  ),
   297→                )}
   298→              </div>
   299→            ))}
   300→          </div>
   301→        </div>
   302→      </div>
   303→    </div>
   304→  );
   305→}

   416→type GenreStatus = "idle" | "resolving" | "done";

   418→export function WrappedView() {
   419→  const push = useNav((s) => s.push);
   420→  const { toast } = useToast();
   421→  const [stats, setStats] = useState<Stats | null>(null);
   422→  const [genreMap, setGenreMap] = useState<Map<string, string[]>>(new Map());
   423→  const [genreStatus, setGenreStatus] = useState<GenreStatus>("idle");
   424→  const [genreMisses, setGenreMisses] = useState(0);
   425→  const [sharing, setSharing] = useState(false);
   426→  const [shareError, setShareError] = useState(false);
   427→
   428→  useEffect(() => {
   429→    const t = setTimeout(() => setStats(computeStats(getHistory())), 0);
   430→    return () => clearTimeout(t);
   431→  }, []);
   432→
   433→  const accent = useMemo(() => (stats ? resolvedAccent() : "#f0b429"), [stats]);
   434→
   435→  const handleShare = useCallback(async () => {
   436→    if (!stats || sharing || stats.totalEntries === 0) return;
   437→    setSharing(true);
   438→    setShareError(false);
   439→    try {
   440→      const blob = await generateShareCard({
   441→        watchMs: stats.watchMs,
   442→        movies: stats.movies,
   443→        episodes: stats.episodes,
   444→        activeDays: stats.activeDays,
   445→        topTitles: stats.topTitles
   446→          .slice(0, 3)
   447→          .map((t) => ({ name: t.name, poster: t.poster, type: t.type })),
   448→        accent: resolvedAccent(),
   449→      });
   450→      const url = URL.createObjectURL(blob);
   451→      const a = document.createElement("a");
   452→      a.href = url;
   453→      a.download = "harbor-wrapped.png";
   454→      document.body.appendChild(a);
   455→      a.click();
   456→      a.remove();
   457→      window.setTimeout(() => URL.revokeObjectURL(url), 5000);
   458→      toast({ title: "Share card downloaded" });
   459→    } catch (e) {
   460→      console.error("[wrapped] share card generation failed", e);
   461→      setShareError(true);
   462→    } finally {
   463→      setSharing(false);
   464→    }
   465→  }, [stats, sharing, toast]);
   466→
   467→  // Resolve genres for top titles via Cinemeta (bounded time; never hangs the panel)
   468→  useEffect(() => {
   469→    if (!stats || stats.topTitles.length === 0) return;
   470→    let alive = true;
   471→    (async () => {
   472→      setGenreStatus("resolving");
   473→      const results = await Promise.allSettled(
   474→        stats.topTitles.map((t) =>
   475→          Promise.race([
   476→            fetchMeta(t.type === "series" ? "series" : "movie", t.id),
   477→            new Promise<null>((resolve) => setTimeout(() => resolve(null), 8_000)),
   478→          ]),
   479→        ),
   480→      );
   481→      if (!alive) return;
   482→      const map = new Map<string, string[]>();
   483→      let misses = 0;
   484→      for (let i = 0; i < results.length; i++) {
   485→        const r = results[i];
   486→        if (r.status === "fulfilled" && r.value) {
   487→          map.set(stats.topTitles[i].id, (r.value.genres ?? []).slice(0, 4));
   488→        } else {
   489→          misses++;
   490→        }
   491→      }
   492→      setGenreMap(map);
   493→      setGenreMisses(misses);
   494→      setGenreStatus("done");
   495→    })();
   496→    return () => {
   497→      alive = false;
   498→    };
   499→  }, [stats]);
   500→
   501→  const topGenres = useMemo(() => {
   502→    const counts = new Map<string, number>();
   503→    for (const t of stats?.topTitles ?? []) {
   504→      for (const g of genreMap.get(t.id) ?? []) {
   505→        counts.set(g, (counts.get(g) ?? 0) + 1);
   506→      }
   507→    }
   508→    return Array.from(counts.entries())
   509→      .map(([genre, count]) => ({ genre, count }))
   510→      .sort((a, b) => b.count - a.count)
   511→      .slice(0, 5);
   512→  }, [stats, genreMap]);
   513→
   514→  const heatTotalMs = useMemo(() => (stats?.heat365 ?? []).reduce((acc, d) => acc + d.ms, 0), [stats]);
   515→  const heatActiveDays = useMemo(() => (stats?.heat365 ?? []).filter((d) => d.ms > 0).length, [stats]);
   516→  const achievements = useMemo(() => buildAchievements(stats), [stats]);
   517→  const earnedCount = achievements.filter((a) => a.earned).length;
   518→
   519→  if (!stats) {
   520→    return (
   521→      <div className="pt-24 px-4 md:px-8">
   522→        <div className="harbor-skeleton h-40 rounded-3xl mb-6" aria-hidden />
   523→        <div className="harbor-skeleton h-24 rounded-3xl" aria-hidden />
   524→        <div className="sr-only">Loading your stats…</div>
   525→      </div>
   526→    );
   527→  }
   528→
   529→  const maxDaily = Math.max(...stats.daily.map((d) => d.ms), 1);
   530→
   531→  return (
   532→    <div className="pb-16 px-4 md:px-8 max-w-5xl">
   533→      <PageHeader view="wrapped" />
   534→      <div className="flex flex-wrap items-center gap-3 mb-1">
   535→        <div className="ms-auto flex items-center gap-2.5">
   536→          {shareError && (
   537→            <span role="alert" className="md-body-small text-danger">
   538→              Couldn&apos;t generate card
   539→            </span>
   540→          )}
   541→          <button
   542→            type="button"
   543→            onClick={handleShare}
   544→            disabled={sharing || !stats || stats.totalEntries === 0}
   545→            aria-label="Share stats"
   546→            className="md-btn-outlined harbor-tv-focus !h-10 disabled:pointer-events-none disabled:opacity-50"
   547→          >
   548→            {sharing ? (
   549→              <Loader2 className="w-4 h-4 animate-spin" aria-hidden />
   550→            ) : (
   551→              <Share2 className="w-4 h-4" aria-hidden />
   552→            )}
   553→            Share stats
   554→          </button>
   555→        </div>
   556→      </div>
   557→      <p className="md-body-medium text-ink-muted mb-8">
   558→        Everything you&apos;ve watched on this device, computed locally.
   559→      </p>
   560→
   561→      {stats.totalEntries === 0 && <WrappedEmptyState onNavigate={(view) => push({ kind: "view", view })} />}
   562→
   563→      {stats.totalEntries > 0 && (
   564→        <>
   565→          {/* Stat tiles */}
   566→          <div className="grid grid-cols-2 md:grid-cols-4 gap-3 mb-8">
   567→            <StatTile icon={<Clock className="w-5 h-5" />} label="Total watch time" value={fmtDuration(stats.watchMs)} accent />
   568→            <StatTile icon={<Film className="w-5 h-5" />} label="Movies" value={String(stats.movies)} />
   569→            <StatTile icon={<Tv className="w-5 h-5" />} label="Episodes" value={String(stats.episodes)} />
   570→            <StatTile icon={<Flame className="w-5 h-5" />} label="Active days" value={String(stats.activeDays)} />
   571→          </div>
   572→
   573→          {/* 14-day activity */}
   574→          <div className="md-card-outlined rounded-[var(--md-sys-shape-corner-large)] p-5 mb-8">
   575→            <div className="flex items-center gap-2 mb-4">
   576→              <CalendarDays className="w-4 h-4 text-accent" />
   577→              <h2 className="md-title-small text-ink">Last 14 days</h2>
   578→              {stats.bestDay && (
   579→                <span className="ms-auto md-body-small text-ink-muted">
   580→                  Best day: {fmtDuration(stats.bestDay.ms)}
   581→                </span>
   582→              )}
   583→            </div>
   584→            <div className="flex items-end gap-1.5 h-28" role="img" aria-label="Watch activity chart">
   585→              {stats.daily.map((d) => {
   586→                const isToday = d.date === stats.daily[stats.daily.length - 1].date;
   587→                return (
   588→                  <div key={d.date} className="flex-1 flex flex-col items-center gap-1.5 group">
   589→                    <div
   590→                      className={cn(
   591→                        "w-full rounded-t-md transition-colors bg-gradient-to-t from-accent/50 to-accent group-hover:brightness-110",
   592→                        isToday && "ring-1 ring-accent/60",
   593→                      )}
   594→                      style={{ height: `${Math.max(4, (d.ms / maxDaily) * 100)}%` }}
   595→                      title={`${d.date}: ${fmtDuration(d.ms)}`}
   596→                    />
   597→                    <span className={cn("text-[9px]", isToday ? "text-ink font-semibold" : "text-ink-subtle")}>
   598→                      {d.date.split("-")[2]}
   599→                    </span>
   600→                  </div>
   601→                );
   602→              })}
   603→            </div>
   604→          </div>
   605→
   606→          {/* 365-day heatmap */}
   607→          <div className="md-card-outlined harbor-pop-in rounded-[var(--md-sys-shape-corner-large)] p-5 mb-8" style={{ animationDelay: "60ms" }}>
   608→            <div className="flex items-center gap-2 mb-4">
   609→              <CalendarDays className="w-4 h-4 text-accent" />
   610→              <h2 className="md-title-small text-ink">Last 365 days</h2>
   611→              <span className="ms-auto md-body-small text-ink-muted">{heatActiveDays} active days</span>
   612→            </div>
   613→            <HeatmapGrid heat={stats.heat365} accent={accent} />
   614→            <p className="sr-only">
   615→              {`Over the last 365 days you watched ${fmtDuration(heatTotalMs)} across ${heatActiveDays} active days.`}
   616→            </p>
   617→            <div className="mt-3 flex items-center justify-end gap-1.5" aria-hidden>
   618→              <span className="text-[9px] text-ink-subtle">Less</span>
   619→              {[0, 1, 2, 3, 4].map((l) => (
   620→                <div
   621→                  key={l}
   622→                  className={cn("h-[9px] w-[9px] rounded-[2px]", l === 0 && "border border-edge-soft bg-raised")}
   623→                  style={l > 0 ? { backgroundColor: levelBg(l, accent) } : undefined}

   625→              ))}
   626→              <span className="text-[9px] text-ink-subtle">More</span>
   627→            </div>
   628→          </div>
   629→
   630→          {/* Achievements */}
   631→          <div className="harbor-pop-in mb-8" style={{ animationDelay: "120ms" }}>
   632→            <div className="flex items-center gap-2 mb-3">
   633→              <Medal className="w-4 h-4 text-accent" />
   634→              <h2 className="md-title-small text-ink">Achievements</h2>

   636→                {earnedCount}/{achievements.length} earned
   637→              </span>
   638→            </div>
   639→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   640→              {achievements.map((a, i) => (
   641→                <AchievementCard key={a.id} a={a} delay={i * 45} />

   644→          </div>

   646→          {/* Top titles + collage */}
   647→          <div className="grid gap-6 md:grid-cols-2 mb-8">
   648→            <div className="md-card-outlined rounded-[var(--md-sys-shape-corner-large)] p-5">
   649→              <div className="flex items-center gap-2 mb-4">
   650→                <Trophy className="w-4 h-4 text-accent" />
   651→                <h2 className="md-title-small text-ink">Most watched</h2>
   652→              </div>
   653→              <div className="space-y-1">

   665→                    <div className="min-w-0 flex-1">
   666→                      <p className="harbor-clamp-1 text-sm font-medium text-ink group-hover:text-accent transition-colors">
   667→                        {t.name}
   668→                      </p>
   669→                      <p className="text-xs text-ink-subtle">
   670→                        {t.count > 1 ? `${t.count} sessions · ` : ""}{fmtDuration(t.ms)}
   671→                      </p>
   672→                      {genreMap.has(t.id) && (
   673→                        <p className="harbor-clamp-1 text-[10px] text-ink-subtle">
   674→                          {(genreMap.get(t.id) ?? []).join(" · ")}
   675→                        </p>
   676→                      )}
   677→                    </div>
   678→                  </button>
   679→                ))}
   680→              </div>
   681→            </div>
   682→
   683→            <div className="md-card-outlined rounded-[var(--md-sys-shape-corner-large)] p-5 flex flex-col">
   684→              <h2 className="md-title-small text-ink mb-4">Top genres</h2>
   685→              {topGenres.length === 0 && genreStatus === "resolving" ? (
   686→                <div className="space-y-3 flex-1" aria-label="Resolving genres">
   687→                  {[0, 1, 2].map((i) => (
   688→                    <div key={i}>
   689→                      <div className="harbor-skeleton h-3 w-24 rounded mb-1.5" />
   690→                      <div className="harbor-skeleton h-2 rounded-full" style={{ width: `${80 - i * 20}%` }} />
   691→                    </div>
   692→                  ))}
   693→                </div>
   694→              ) : topGenres.length === 0 ? (
   695→                <p className="text-xs text-ink-subtle">
   696→                  {genreMisses > 0
   697→                    ? "Genres unavailable — could not reach Cinemeta for these titles."
   698→                    : "No genre data for your watch history yet."}
   699→                </p>
   700→              ) : (
   701→                <div className="space-y-3 flex-1">
   702→                  {topGenres.map((g) => {
   703→                    const max = topGenres[0].count || 1;
   704→                    return (
   705→                      <div key={g.genre}>
   706→                        <div className="flex justify-between text-xs mb-1">
   707→                          <span className="text-ink font-medium">{g.genre}</span>
   708→                          <span className="text-ink-subtle">{g.count}</span>
   709→                        </div>
   710→                        <div className="h-2 rounded-full bg-raised overflow-hidden">
   711→                          <div
   712→                            className="h-full rounded-full bg-gradient-to-r from-accent/60 to-accent"
   713→                            style={{ width: `${(g.count / max) * 100}%` }}
   714→                          />
   715→                        </div>
   716→                      </div>
   717→                    );
   718→                  })}
   719→                </div>
   720→              )}
   721→              {/* Poster collage */}
   722→              <div className="grid grid-cols-3 gap-1.5 mt-5">
   723→                {stats.topTitles.slice(0, 6).map((t) => (

   725→                    <PosterImage src={t.poster} alt="" className="absolute inset-0" />
   726→                  </div>
   727→                ))}
   728→              </div>
   729→            </div>
   730→          </div>
   731→
   732→          <p className="text-[11px] text-ink-subtle text-center">
   733→            Stats are computed locally from your watch history. Private by design — nothing leaves
   734→            this device unless you sign in to Stremio.
   735→          </p>
   736→        </>
   737→      )}
   738→    </div>
   739→  );
   740→}
