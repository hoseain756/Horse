    30→    30→
    31→    31→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    32→    32→  const pop = useNav((s) => s.pop);
    33→    33→  const replace = useNav((s) => s.replace);
    34→    34→  const settings = useSettings((s) => s.settings);
    35→    35→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    36→    36→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    37→    37→  const [error, setError] = useState<string | null>(null);
    38→    38→  const [switcherOpen, setSwitcherOpen] = useState(false);
    39→    39→  const [resolveStatus, setResolveStatus] = useState<string>("Finding the best stream…");
    40→    40→
    41→    41→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    42→    42→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    43→    43→  const handleError = useCallback((msg: string) => {
    44→    44→    setPhase("error");
    45→    45→    setError(msg);
    46→    46→  }, []);
    47→    47→
    48→    48→  // Resolve stream via addons when payload.url is empty (instant play path)
    49→    49→  useEffect(() => {
    50→    50→    if (current.url || phase !== "resolving") return;
    51→    51→    let alive = true;
    52→    52→    setResolveStatus("Finding the best stream…");
    53→    53→    (async () => {
    54→    54→      const { useAddons: addons } = await import("@/lib/harbor/store");
    55→    55→      try {
    56→    56→        // Series opened without an episode id (bare tt-id): stream addons need
    57→    57→        // "id:season:episode" — resolve the default video from meta first.
    58→    58→        let targetId = current.videoId ?? current.metaId;
    59→    59→        if (current.type === "series" && !targetId.includes(":")) {
    60→    60→          const meta = await fetchMeta("series", current.metaId).catch(() => null);
    61→    61→          const vid = defaultVideoId(meta);
    62→    62→          if (vid) {
    63→    63→            targetId = vid;
    64→    64→            if (alive) {
    65→    65→              const vidMeta = (meta?.videos ?? []).find((v) => v.id === vid);
    66→    66→              setCurrent((c) => ({
    67→    67→                ...c,
    68→    68→                videoId: vid,
    69→    69→                season: c.season ?? vidMeta?.season,
    70→    70→                episode: c.episode ?? vidMeta?.episode,
    71→    71→                episodeName: c.episodeName ?? vidMeta?.name ?? vidMeta?.title,
    72→    72→              }));
    73→    73→            }
    74→    74→          }
    75→    75→        }
    76→    76→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
    77→    77→        if (!alive) return;
    78→    78→        const ranked = runPipeline(results);
    79→    79→        // Prefer browser-playable candidates: direct URLs first, then torrents whose
    80→    80→        // parsed video codec isn't HEVC (browsers can't decode it — pick the best
    81→    81→        // H.264/VP9/AV1 release instead of a 4K HEVC one that would dead-end).
    82→    82→        const p2pCandidate = (s: Stream) =>
    83→    83→          !!s.infoHash && !s.url && s.parsed?.codec !== "HEVC";
    84→    84→        const best = ranked.find((s) => s.url) ?? ranked.find(p2pCandidate) ?? ranked[0];
    85→    85→        if (!best) {
    86→    86→          setPhase("error");
    87→    87→          setError("No streams found from your addons.");
    88→    88→          return;
    89→    89→        }
    90→    90→        if (best.url) {
    91→    91→          setCurrent((c) => ({
    92→    92→            ...c,
    93→    93→            url: best.url!,
    94→    94→            streamTitle: best.title ?? c.streamTitle,
    95→    95→            streamDescription: best.description,
    96→    96→          }));
    97→    97→          setPhase("loading");
    98→    98→          return;
    99→    99→        }
   100→   100→        // Best candidate is a torrent → try the server-side P2P engine before giving up
   101→   101→        if (!best.infoHash || !useSettings.getState().settings.p2pEnabled) {
   102→   102→          setPhase("error");
   103→   103→          setError(
   104→   104→            ranked.length > 0
   105→   105→              ? "No browser-playable stream found. Open the stream picker to choose manually — torrent streams play via P2P or unlock with debrid."
   106→   106→              : "No streams found from your addons.",
   107→   107→          );
   108→   108→          return;
   109→   109→        }
   110→   110→        const { p2pPrepare, p2pStatus, p2pPlan } = await import("@/lib/harbor/p2p");
   111→   111→        const infoHash = best.infoHash.toLowerCase();
   112→   112→        setResolveStatus("Joining the torrent swarm…");
   113→   113→        const prep = await p2pPrepare(infoHash, {
   114→   114→          fileIdx: best.fileIdx,
   115→   115→          filename: best.behaviorHints?.filename ?? best.parsed?.filename,
   116→   116→        });
   117→   117→        for (let i = 0; prep.pending && i < 40 && alive; i++) {
   118→   118→          await new Promise((r) => setTimeout(r, 2500));
   119→   119→          setResolveStatus((s) => (s.startsWith("Finding peers") ? s : `Finding peers… (${i * 2.5 | 0}s)`));
   120→   120→          const st = await p2pStatus(infoHash).catch(() => null);
   121→   121→          if (st?.peers) setResolveStatus(`Finding peers… (${st.peers} found)`);
   122→   122→          if (st?.ready) break;
   123→   123→        }
   124→   124→        if (!alive) return;
   125→   125→        const st = prep.pending ? await p2pStatus(infoHash).catch(() => null) : null;
   126→   126→        if (prep.pending && !st?.ready) {
   127→   127→          setPhase("error");
   128→   128→          setError(
   129→   129→            "No peers responded for the best torrent. Open the stream picker to pick another source, or connect debrid for instant cached streams.",
   130→   130→          );
   131→   131→          return;
   132→   132→        }
   133→   133→        const fileIdx = prep.pending
   134→   134→          ? (Number.isInteger(best.fileIdx)
   135→   135→              ? best.fileIdx!
   136→   136→              : (st?.files ?? []).length > 0
   137→   137→                ? (st!.files ?? []).reduce(
   138→   138→                    (a, b) => (b.length > a.length ? b : a),
   139→   139→                    (st!.files ?? [{ index: 0, name: "", length: 0 }])[0],
   140→   140→                  ).index
   141→   141→                : 0)
   142→   142→          : prep.fileIdx;
   143→   143→        const filename = best.behaviorHints?.filename ?? best.parsed?.filename;
   144→   144→        setResolveStatus("Checking video codecs…");
   145→   145→        const plan = await p2pPlan(infoHash, fileIdx, filename).catch(() => null);
   146→   146→        if (!alive) return;
   147→   147→        if (!plan) {
   148→   148→          setPhase("error");
   149→   149→          setError("Torrent engine could not serve this file. Try another stream in the picker.");
   150→   150→          return;
   151→   151→        }
   152→   152→        setCurrent((c) => ({
   153→   153→          ...c,
   154→   154→          url: plan.url,
   155→   155→          streamTitle: best.title ?? c.streamTitle,
   156→   156→          streamDescription: best.description,
   157→   157→          p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   158→   158→        }));
   159→   159→        setPhase("loading");
   160→   160→      } catch (e) {
   161→   161→        if (!alive) return;
   162→   162→        setPhase("error");
   163→   163→        setError(e instanceof Error ? e.message : "Failed to resolve streams");
   164→   164→      }
   165→   165→    })();
   166→   166→    return () => {
   167→   167→      alive = false;
   168→   168→    };
   169→   169→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
   170→   170→
   171→   171→  const setPayload = useCallback((p: PlayerPayload) => {
   172→   172→    setCurrent(p);
   173→   173→    setPhase(p.url ? "loading" : "resolving");
   174→   174→    setError(null);
   175→   175→  }, []);
   176→   176→
   177→   177→  const openSwitcher = useCallback(() => setSwitcherOpen(true), []);
   178→   178→
   179→   179→  // Advance to the next episode in-place (Up Next card / auto-advance / "n" key).
   180→   180→  // Empty url re-enters the resolving path so addons fetch streams for the new episode.
   181→   181→  const handlePlayNext = useCallback(
   182→   182→    (next: { season: number; episode: number; videoId: string; episodeName?: string }) => {
   183→   183→      setCurrent((c) => ({
   184→   184→        ...c,
   185→   185→        url: "",
   186→   186→        season: next.season,
   187→   187→        episode: next.episode,
   188→   188→        videoId: next.videoId,
   189→   189→        episodeName: next.episodeName,
   190→   190→        streamTitle: undefined,
   191→   191→        streamDescription: undefined,
   192→   192→        resumeAt: 0,
   193→   193→      }));
   194→   194→      setPhase("resolving");
   195→   195→      setError(null);
   196→   196→    },
   197→   197→    [],
   198→   198→  );
   199→   199→
   200→   200→  return (
   201→   201→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
   202→   202→      {phase === "resolving" && (
   203→   203→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
   204→   204→          <div className="relative">
   205→   205→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
   206→   206→          </div>
   207→   207→          <p className="text-sm text-ink-muted" aria-live="polite">{resolveStatus}</p>
   208→   208→        </div>
   209→   209→      )}
   210→   210→      {phase === "error" ? (
   211→   211→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   212→   212→          <AlertTriangle className="w-12 h-12 text-danger" />
   213→   213→          <p className="text-sm text-ink-muted max-w-md">{error}</p>
   214→   214→          <div className="flex gap-3">
   215→   215→            <button
   216→   216→              type="button"
   217→   217→              onClick={() => setSwitcherOpen(true)}
   218→   218→              className="rounded-xl bg-accent px-5 py-2.5 text-sm font-bold text-black hover:brightness-110"
   219→   219→            >
   220→   220→              Open stream picker
   221→   221→            </button>
   222→   222→            <button
   223→   223→              type="button"
   224→   224→              onClick={pop}
   225→   225→              className="rounded-xl bg-raised px-5 py-2.5 text-sm font-semibold text-ink hover:bg-raised/70"
   226→   226→            >
   227→   227→              Back
   228→   228→            </button>
   229→   229→          </div>
   230→   230→        </div>
   231→   231→      ) : (
   232→   232→        current.url && (
   233→   233→          <VideoStage
   234→   234→            key={current.url}
   235→   235→            payload={current}
   236→   236→            settings={settings}
   237→   237→            onPhase={handlePhase}
   238→   238→            onError={handleError}
   239→   239→            onPlayNext={handlePlayNext}
   240→   240→            onOpenSwitcher={openSwitcher}
   241→   241→          />
   242→   242→        )
   243→   243→      )}
   244→   244→
   245→   245→      {/* Top bar */}
   246→   246→      {phase !== "error" && (
   247→   247→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   248→   248→          <button
   249→   249→            type="button"
   250→   250→            onClick={() => {
   251→   251→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   252→   252→              pop();
   253→   253→            }}
   254→   254→            className="w-9 h-9 rounded-full bg-black/50 flex items-center justify-center text-ink hover:bg-black/70"
   255→   255→            aria-label="Close player (Esc)"
   256→   256→          >
   257→   257→            <X className="w-5 h-5" />
   258→   258→          </button>
   259→   259→          <div className="min-w-0">
   260→   260→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   261→   261→            {current.episode && (
   262→   262→              <p className="text-xs text-ink-muted">
   263→   263→                S{current.season}:E{current.episode}
   264→   264→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   265→   265→              </p>
   266→   266→            )}
   267→   267→          </div>
   268→   268→          <button
   269→   269→            type="button"
   270→   270→            onClick={() => setSwitcherOpen(true)}
   271→   271→            className="ml-auto flex items-center gap-1.5 rounded-full bg-black/50 hover:bg-black/70 px-3 py-1.5 text-xs font-medium text-ink"
   272→   272→            title="Switch stream (W)"
   273→   273→          >
   274→   274→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   275→   275→          </button>
   276→   276→        </div>
   277→   277→      )}
   278→   278→
   279→   279→      {switcherOpen && (
   280→   280→        <PickerOverlay
   281→   281→          type={current.type}
   282→   282→          id={current.deepLink?.id ?? current.metaId}
   283→   283→          videoId={current.videoId ?? current.deepLink?.videoId}
   284→   284→          season={current.season}
   285→   285→          episode={current.episode}
   286→   286→        />
   287→   287→      )}
   288→   288→    </div>
   289→   289→  );
   290→   290→}

   292→   292→// ---------------- Video stage ----------------
   293→   293→function VideoStage({
   294→   294→  payload,
   295→   295→  settings,
   296→   296→  onPhase,
   297→   297→  onError,
   298→   298→  onPlayNext,
   299→   299→  onOpenSwitcher,
   300→   300→}: {
   301→   301→  payload: PlayerPayload;
   302→   302→  settings: ReturnType<typeof useSettings.getState>["settings"];
   303→   303→  onPhase: (p: "loading" | "playing") => void;
   304→   304→  onError: (msg: string) => void;
   305→   305→  onPlayNext: (next: { season: number; episode: number; videoId: string; episodeName?: string }) => void;
   306→   306→  onOpenSwitcher: () => void;
   307→   307→}) {
   308→   308→  const videoRef = useRef<HTMLVideoElement>(null);
   309→   309→  const containerRef = useRef<HTMLDivElement>(null);
   310→   310→  const hlsRef = useRef<Hls | null>(null);
   311→   311→  const [ready, setReady] = useState(false);
   312→   312→  const [playing, setPlaying] = useState(false);
   313→   313→  const [muted, setMuted] = useState(false);
   314→   314→  const [volume, setVolume] = useState(1);
   315→   315→  const [time, setTime] = useState(0);
   316→   316→  const [duration, setDuration] = useState(0);
   317→   317→  const [buffering, setBuffering] = useState(true);
   318→   318→  const [fullscreen, setFullscreen] = useState(false);
   319→   319→  const [speed, setSpeed] = useState(1);
   320→   320→  const [controlsVisible, setControlsVisible] = useState(true);
   321→   321→  const [volumeHudVisible, setVolumeHudVisible] = useState(false);
   322→   322→  const volumeHudTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   323→   323→  const [subMenuOpen, setSubMenuOpen] = useState(false);
   324→   324→  const [settingsMenuOpen, setSettingsMenuOpen] = useState(false);
   325→   325→  // P2P engine: live swarm stats + one-shot remux fallback when the native stream can't demux
   326→   326→  const [p2pStats, setP2pStats] = useState<{ progress: number; peers: number; downloadSpeed: number; ready: boolean } | null>(null);
   327→   327→  const [srcOverride, setSrcOverride] = useState<string | null>(null);
   328→   328→  const remuxTriedRef = useRef(false);
   329→   329→  const [subtitles, setSubtitles] = useState<LoadedSubtitle[]>([]);
   330→   330→  // Full ranked subtitle pool + how many are loaded — powers the "Load more" affordance
   331→   331→  const subPoolRef = useRef<RawSubtitle[]>([]); // ranked by preferred languages
   332→   332→  const subEffectiveRef = useRef<RawSubtitle[]>([]); // currently displayed ranking (lang filter)
   333→   333→  const subAttemptedRef = useRef(0);
   334→   334→  const [subPoolSize, setSubPoolSize] = useState(0);
   335→   335→  const [subLoadingMore, setSubLoadingMore] = useState(false);
   336→   336→  const [subLangFilter, setSubLangFilter] = useState<string | null>(null);
   337→   337→  // Free-text search over the ranked pool (label / language / provider) — composes
   338→   338→  // with the language chips: chip re-ranks, search then narrows.
   339→   339→  const [subSearch, setSubSearch] = useState(""); // committed (drives the pool view)
   340→   340→  const [subSearchInput, setSubSearchInput] = useState(""); // typing value
   341→   341→  const subViewTokenRef = useRef(0); // invalidates stale parse batches when the view changes
   342→   342→  const subSearchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   343→   343→  // Language chips derived from the pool: chip code + available count
   344→   344→  const subLangChips = useMemo(() => {
   345→   345→    const counts = new Map<string, number>();
   346→   346→    for (const s of subPoolRef.current) {
   347→   347→      const chip = langChip(s.lang);
   348→   348→      counts.set(chip, (counts.get(chip) ?? 0) + 1);
   349→   349→    }
   350→   350→    return Array.from(counts.entries())
   351→   351→      .sort((a, b) => b[1] - a[1])
   352→   352→      .slice(0, 6);
   353→   353→  }, [subPoolSize, subLangFilter]);
   354→   354→  // How many entries the CURRENT view (chip re-rank + search filter) contains
   355→   355→  const [subEffectiveCount, setSubEffectiveCount] = useState(0);
   356→   356→  const subtitlesRef = useRef<LoadedSubtitle[]>([]);
   357→   357→  subtitlesRef.current = subtitles; // latest committed list for stable index math
   358→   358→  const [activeSub, setActiveSub] = useState<number>(-1); // -1 = off
   359→   359→  const [subCue, setSubCue] = useState<SubCue | null>(null);
   360→   360→  const [availableTracks, setAvailableTracks] = useState<{ label: string }[]>([]);
   361→   361→  const [levels, setLevels] = useState<{ index: number; label: string }[]>([]);
   362→   362→  const [levelSel, setLevelSel] = useState<number>(-1); // -1 = adaptive (auto)
   363→   363→  const [autoLevel, setAutoLevel] = useState<number>(-1); // level currently playing in auto mode
   364→   364→  const [activeAudio, setActiveAudio] = useState(0);
   365→   365→  const [bufferedEnd, setBufferedEnd] = useState(0);
   366→   366→  const [nextEp, setNextEp] = useState<{
   367→   367→    season: number;
   368→   368→    episode: number;
   369→   369→    videoId: string;
   370→   370→    episodeName?: string;
   371→   371→  } | null>(null);
   372→   372→  const [upNextDismissed, setUpNextDismissed] = useState(false);
   373→   373→  const controlsTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   374→   374→  const seekHandled = useRef(false);
   375→   375→  // Trakt scrobble state (dedupe: one scrobble per action per 10s)
   376→   376→  const scrobbleStartSent = useRef(false);
   377→   377→  const scrobbleLastSent = useRef<Record<"start" | "pause" | "stop", number>>({
   378→   378→    start: 0,
   379→   379→    pause: 0,
   380→   380→    stop: 0,
   381→   381→  });
   382→   382→
   383→   383→  const resumeAt = useRef(
   384→   384→    payload.resumeAt ??
   385→   385→      (settings.resumePlayback && payload.metaId
   386→   386→        ? resumeMsFor(payload.metaId, payload.videoId) / 1000
   387→   387→        : 0),
   388→   388→  );
   389→   389→
   390→   390→  // ---- P2P swarm stats polling ----
   391→   391→  useEffect(() => {
   392→   392→    if (!payload.p2p) {
   393→   393→      setP2pStats(null);
   394→   394→      return;
   395→   395→    }
   396→   396→    let alive = true;
   397→   397→    const key = payload.p2p.key;
   398→   398→    const tick = async () => {
   399→   399→      try {
   400→   400→        const st = await p2pStatus(key);
   401→   401→        if (!alive) return;
   402→   402→        setP2pStats({
   403→   403→          progress: st.progress ?? 0,
   404→   404→          peers: st.peers ?? 0,
   405→   405→          downloadSpeed: st.downloadSpeed ?? 0,
   406→   406→          ready: st.ready === true,
   407→   407→        });
   408→   408→      } catch {
   409→   409→        if (alive) setP2pStats(null);
   410→   410→      }
   411→   411→    };
