    25→export function PickerOverlay({
    26→  type,
    27→  id,
    28→  videoId,
    29→  season,
    30→  episode,

    32→  type: string;
    33→  id: string;

    35→  season?: number;

    39→  const push = useNav((s) => s.push);
    40→  const addons = useAddons((s) => s.addons);
    41→  const addonsLoaded = useAddons((s) => s.loaded);
    42→  const settings = useSettings((s) => s.settings);
    43→  const { toast } = useToast();
    44→  const debridConfigured = useDebrid((s) => s.apiKey !== null);
    45→  const [debridDialogOpen, setDebridDialogOpen] = useState(false);
    46→

    48→  // into a picker after a reload (idempotent; settings loads it too).

    50→    useDebrid.getState().load();
    51→  }, []);
    52→
    53→  // Torrent rows: no debrid key → offer the inline setup dialog (does not leave the picker).

    61→  const [totalAddons, setTotalAddons] = useState(0);
    62→  const [error, setError] = useState<string | null>(null);

    65→  const [playing, setPlaying] = useState<Stream | null>(null);
    66→
    67→  // TV-mode auto focus: keyboard users landing in the picker should get focus moved
    68→  // to the dialog without touching the mouse. Mouse use (dataset.tv flips off in
    69→  // tvnav) never steals focus. The check happens inside the delayed callback so a

    71→  const bodyRef = useRef<HTMLDivElement | null>(null);
    72→  const closeRef = useRef<HTMLButtonElement | null>(null);
    73→  const focusTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    74→
    75→  const scheduleTvFocus = useCallback(() => {
    76→    if (focusTimer.current) clearTimeout(focusTimer.current);

    80→        const row = bodyRef.current?.querySelector<HTMLElement>(".harbor-picker-row");
    81→        const target = row ?? closeRef.current;
    82→        target?.focus({ preventScroll: true });
    83→        row?.scrollIntoView({ block: "nearest", behavior: "smooth" });
    84→      });
    85→    }, 350);
    86→  }, []);
    87→
    88→  // On open: focus the first row if streams are already there, else the close button.
    89→  useEffect(() => {
    90→    scheduleTvFocus();
    91→    return () => {
    92→      if (focusTimer.current) clearTimeout(focusTimer.current);
    93→    };
    94→  }, [scheduleTvFocus]);
    95→
    96→  // Once streams finish loading (or fail), settle for ~350ms then move focus in.

    98→    if (!loading) scheduleTvFocus();

   107→    setError(null);

   113→    setProgressAddons(0);

   178→          }
   179→          return true;
   180→        }),
   181→      }))
   182→      .filter((t) => t.items.length > 0);
   183→  }, [tiers, filter, query]);
   184→
   185→  if (playing) {
   186→    const metaName = playing.title?.split("\n")[0] ?? "";
   187→    return (
   188→      <PlayerOverlay
   189→        payload={{
   190→          url: playing.url ?? "",
   191→          title: metaName,
   192→          type,
   193→          metaId: id,
   194→          season,
   195→          episode,
   196→          videoId: targetId,
   197→          streamTitle: playing.title,
   198→          streamDescription: playing.description,
   199→          deepLink: { type, id, videoId: targetId },
   200→          ...(playing.p2p ? { p2p: playing.p2p } : {}),
   201→        }}
   202→      />
   203→    );
   204→  }
   205→
   206→  return (
   207→    <div className="fixed inset-0 z-[150] bg-black/80 backdrop-blur-sm flex items-end md:items-center justify-center" role="dialog" aria-modal="true" aria-label="Stream picker">
   208→      <div className="w-full md:max-w-3xl md:max-h-[82vh] max-h-[88vh] bg-canvas border border-edge rounded-t-3xl md:rounded-3xl flex flex-col overflow-hidden shadow-2xl">
   209→        {/* Header */}
   210→        <div className="flex items-center justify-between gap-3 px-5 py-4 border-b border-edge-soft shrink-0">
   211→          <div className="min-w-0">
   212→            <h2 className="font-display text-lg font-bold text-ink">Choose a stream</h2>
   213→            <p className="text-xs text-ink-subtle truncate">
   214→              {type === "series" ? `Series · ${targetId}` : targetId}
   215→              {season ? ` · S${season}:E${episode}` : ""}
   216→            </p>
   217→          </div>
   218→          <div className="flex items-center gap-1.5 shrink-0">
   219→            <button
   220→              type="button"
   221→              onClick={load}

   223→              aria-label="Refresh streams"
   224→              title="Refresh"
   225→            >
   226→              <RefreshCw className={cn("w-4 h-4", loading && "animate-spin")} />
   227→            </button>
   228→            <button
   229→              type="button"
   230→              ref={closeRef}
   231→              onClick={pop}
   232→              className="harbor-tv-focus w-9 h-9 rounded-xl flex items-center justify-center text-ink-muted hover:text-ink hover:bg-raised"

   240→        {/* Filters */}

   242→          {(
   243→            [
   244→              ["all", "All"],
   245→              ["free", "Free"],
   246→              ["cached", "Cached"],
   247→              ["4k", "4K"],
   248→              ["1080p", "1080p"],
   249→            ] as const
   250→          ).map(([key, label]) => (
   251→            <button
   252→              key={key}
   253→              type="button"
   254→              onClick={() => setFilter(key)}
   255→              className={cn(
   256→                "rounded-full px-3 py-1.5 text-xs font-medium transition-colors",
   257→                filter === key ? "bg-accent text-black" : "bg-raised text-ink-muted hover:text-ink",
   258→              )}
   259→            >
   260→              {label}
   261→            </button>

   263→          <div className="relative ml-auto w-40">
   264→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-ink-subtle" />

   266→              value={query}
   267→              onChange={(e) => setQuery(e.target.value)}

   284→              type="button"
   285→              onClick={handleDebridNotConfigured}

   304→                  <div className="w-14 h-9 rounded-lg bg-white/10 shrink-0" />
   305→                  <div className="flex-1 space-y-2">
   306→                    <div className="h-3 w-2/3 rounded bg-white/10" />
   307→                    <div className="h-2.5 w-1/3 rounded bg-white/10" />
   308→                  </div>
   309→                  <div className="w-16 h-7 rounded-lg bg-white/10 shrink-0" />

   311→              ))}
   312→            </div>

   314→

   320→

   327→
   328→          {!loading &&
   329→            filteredTiers.map(({ tier, items }) => (
   330→              <section key={tier}>
   331→                <div className="flex items-center gap-2 mb-2.5">
   332→                  <span aria-hidden className={cn("h-1.5 w-1.5 rounded-full shrink-0", tierDotColor(tier))} />
   333→                  <h3 className="text-xs font-bold uppercase tracking-widest text-ink-muted">{tier}</h3>

   344→                      configured={debridConfigured}

   347→                  ))}
   348→                </div>
   349→              </section>
   350→            ))}
   351→        </div>
   352→      </div>
   353→      {debridDialogOpen && <DebridSetupDialog onClose={() => setDebridDialogOpen(false)} />}
   354→    </div>
   355→  );

   358→// Quality-ladder dot per tier header: emerald top, amber mid, neutral low —
   359→// colors follow Harbor's existing palette conventions (understated).
   360→function tierDotColor(tier: string): string {
   361→  if (tier === "4K HDR" || tier === "4K") return "bg-emerald-400";
   362→  if (tier === "1080p HDR" || tier === "1080p") return "bg-amber-500";
   363→  if (tier === "720p" || tier === "SD") return "bg-zinc-500";
   364→  return "bg-zinc-600";
   365→}
   366→
   367→function StreamRow({
   368→  stream,
   369→  onPick,
   370→  onNeedsSetup,
   371→  configured,
   372→  showQuality,
   373→}: {
   374→  stream: Stream;
   375→  onPick: (s: Stream) => void;
   376→  onNeedsSetup: () => void;
   377→  configured: boolean;
   378→  showQuality: boolean;
   379→}) {
   380→  const { toast } = useToast();
   381→  const debridKey = useDebrid((s) => s.apiKey);
   382→  const debridResolve = useDebrid((s) => s.resolve);
   383→  const [resolving, setResolving] = useState(false);
   384→  // P2P connecting state: null = idle, else seconds elapsed while joining the swarm
   385→  const [p2pPhase, setP2pPhase] = useState<"idle" | "joining" | "planning">("idle");
   386→  const [p2pInfo, setP2pInfo] = useState<{ peers: number; elapsed: number } | null>(null);
   387→  const [err, setErr] = useState<string | null>(null);
   388→  const errTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   389→  const p2pAbort = useRef<AbortController | null>(null);
   390→
   391→  useEffect(() => {
   392→    return () => {
   393→      if (errTimer.current) clearTimeout(errTimer.current);
   394→      p2pAbort.current?.abort();
   395→    };
   396→  }, []);
   397→
   398→  const cached = Object.values(stream.cached ?? {}).some(Boolean);
   399→  const isTorrent = !!stream.infoHash && !stream.url; // torrent without a direct URL → debrid unlock OR P2P
   400→  const p2pEnabled = useSettings((s) => s.settings.p2pEnabled);
   401→  const p = stream.parsed;
   402→  const titleLine = (stream.title ?? stream.description ?? "Stream").split("\n")[0];
   403→  const detailLine = (stream.description ?? "").split("\n").filter(Boolean).slice(1).join(" ").slice(0, 140);
   404→
   405→  const fail = (msg: string) => {
   406→    setErr(msg);
   407→    toast({ title: "Stream unavailable", description: msg, variant: "destructive" });
   408→    if (errTimer.current) clearTimeout(errTimer.current);
   409→    errTimer.current = setTimeout(() => setErr(null), 6000);
   410→  };
   411→
   412→  const unlock = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   413→    e?.stopPropagation();
   414→    e?.preventDefault();
   415→    if (resolving) return;
   416→    if (debridKey === null) {
   417→      onNeedsSetup();
   418→      return;
   419→    }
   420→    setResolving(true);
   421→    setErr(null);
   422→    const res = await debridResolve(
   423→      stream.infoHash!,
   424→      stream.behaviorHints?.filename,
   425→      stream.parsed?.size,
   426→    );
   427→    setResolving(false);
   428→    if ("url" in res && res.url) {
   429→      onPick({
   430→        ...stream,
   431→        url: res.url,
   432→        behaviorHints: {
   433→          ...stream.behaviorHints,
   434→          ...(res.filename ? { filename: res.filename } : {}),
   435→        },
   436→      });
   437→    } else {
   438→      const msg = "error" in res ? res.error : "Unlock failed";
   439→      fail(msg);
   440→    }
   441→  };
   442→
   443→  // Play a torrent through the server-side P2P engine: join swarm → wait for
   444→  // metadata → pick the right file (native container vs ffmpeg remux) → play.
   445→  const playP2p = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   446→    e?.stopPropagation();
   447→    e?.preventDefault();
   448→    if (p2pPhase !== "idle") return;
   449→    if (!useSettings.getState().settings.p2pEnabled) {
   450→      onNeedsSetup();
   451→      return;
   452→    }
   453→    setErr(null);
   454→    setP2pPhase("joining");
   455→    setP2pInfo({ peers: 0, elapsed: 0 });
   456→    const startedAt = Date.now();
   457→    const abort = new AbortController();
   458→    p2pAbort.current = abort;
   459→    const infoHash = stream.infoHash!.toLowerCase();
   460→
   461→    // progress ticker while the swarm join runs
   462→    const ticker = setInterval(() => {
   463→      setP2pInfo((prev) => (prev ? { ...prev, elapsed: Math.round((Date.now() - startedAt) / 1000) } : prev));
   464→    }, 1000);
   465→
   466→    try {
   467→      // 1) prepare (join swarm). 202 = metadata still downloading → poll status.
   468→      const prep = await p2pPrepare(infoHash, {
   469→        fileIdx: stream.fileIdx,
   470→        filename: stream.behaviorHints?.filename ?? stream.parsed?.filename,
   471→      });
   472→
   473→      // 2) poll while discovering (up to ~100s; the engine keeps the swarm alive)
   474→      while (prep.pending) {
   475→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   476→        if (Date.now() - startedAt > 100_000) {
   477→          throw new Error(
   478→            "Could not find peers for this torrent. The swarm may be dead or this network blocks BitTorrent — a debrid service would unlock it instantly.",
   479→          );
   480→        }
   481→        await new Promise((r) => setTimeout(r, 2500));
   482→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   483→        const st = await p2pStatus(infoHash).catch(() => null);
   484→        setP2pInfo((prev) => (prev ? { ...prev, peers: st?.peers ?? prev.peers } : prev));
   485→        if (st?.ready) break;
   486→      }
   487→
   488→      // 3) choose serving mode (native range stream vs ffmpeg remux for mkv)
   489→      setP2pPhase("planning");
   490→      const fileIdx = prep.pending
   491→        ? await (async () => {
   492→            const st = await p2pStatus(infoHash);
   493→            const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   494→            const idx = Number.isInteger(stream.fileIdx)
   495→              ? stream.fileIdx!
   496→              : st.files
   497→                ? pickFileIndex(st.files, filename)
   498→                : 0;
   499→            return idx;
   500→          })()
   501→        : prep.fileIdx;
   502→      const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   503→      let plan: P2pPlaybackPlan;
   504→      try {
   505→        plan = await p2pPlan(infoHash, fileIdx, filename);
   506→      } catch {
   507→        plan = { key: infoHash, fileIdx, url: `/stream/${infoHash}/${fileIdx}`, mode: "unknown" };
   508→      }
   509→      if (plan.mode === "unknown" && plan.codec?.video === "hevc") {
   510→        throw new Error(
   511→          `This file uses ${plan.codec.video.toUpperCase()} video which browsers cannot play — unlock it with debrid instead.`,
   512→        );
   513→      }
   514→      clearInterval(ticker);
   515→      setP2pPhase("idle");
   516→      setP2pInfo(null);
   517→      onPick({
   518→        ...stream,
   519→        url: plan.url,
   520→        behaviorHints: {
   521→          ...stream.behaviorHints,
   522→          filename: filename ?? plan.codec?.filename,
   523→        },
   524→        p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   525→      });
   526→    } catch (e) {
   527→      clearInterval(ticker);
   528→      setP2pPhase("idle");
   529→      setP2pInfo(null);
   530→      if ((e as Error)?.name === "AbortError") return;
   531→      fail(e instanceof Error ? e.message : "P2P stream failed");
   532→    }
   533→  };
   534→
   535→  const rowActivate = () => {
   536→    if (isTorrent) {
   537→      if (debridKey !== null) void unlock();
   538→      else if (p2pEnabled) void playP2p();
   539→      else onNeedsSetup();
   540→    } else {
   541→      onPick(stream);
   542→    }
   543→  };
   544→
   545→  const busy = resolving || p2pPhase !== "idle";
   546→  const direct = !!stream.url && !isTorrent;
   547→
   548→  return (
   549→    <div
   550→      role="button"
   551→      tabIndex={0}
   552→      onClick={rowActivate}
   553→      onKeyDown={(e) => {
   554→        if (e.key === "Enter" || e.key === " ") {
   555→          e.preventDefault();
   556→          rowActivate();
   557→        }
   558→      }}
   559→      aria-label={`${titleLine} — play stream`}
   560→      className={cn(
   561→        "harbor-tv-focus harbor-picker-row relative w-full flex flex-wrap items-center gap-3.5 rounded-xl border p-3 text-left transition-all hover:border-accent/60 cursor-pointer",
   562→        cached ? "border-emerald-500/30 bg-emerald-500/5" : "border-edge-soft bg-elevated",
   563→        direct && "border-l-2 border-l-emerald-500/50",
   564→      )}
   565→    >
   566→      {busy && (
   567→        <span className="absolute inset-0 z-10 rounded-xl bg-black/60 backdrop-blur-[2px] flex flex-col items-center justify-center gap-1" aria-live="polite">
   568→          {resolving ? (
   569→            <Loader2 className="w-5 h-5 text-accent animate-spin" />
   570→          ) : (
   571→            <>
   572→              <span className="flex items-center gap-2 text-xs font-semibold text-ink">
   573→                <Wifi className="w-4 h-4 text-accent animate-pulse" aria-hidden />
   574→                {p2pPhase === "planning" ? "Preparing file…" : "Joining swarm…"}
   575→              </span>
   576→              <span className="text-[10px] text-ink-subtle">
   577→                {p2pInfo?.peers ? `${p2pInfo.peers} peers` : "looking for peers"} · {p2pInfo?.elapsed ?? 0}s
   578→              </span>
   579→            </>
   580→          )}
   581→        </span>
   582→      )}
   583→      <div className="flex flex-col items-center gap-1 w-14 shrink-0">
   584→        <span className={cn("rounded-lg px-2 py-1 text-[11px] font-bold", cached ? "bg-emerald-500/20 text-emerald-300" : "bg-raised text-ink-muted")}>
   585→          {p?.resolution ?? "—"}
   586→        </span>
   587→        {p?.hdrFormat && (
   588→          <span className="rounded px-1.5 py-0.5 text-[9px] font-bold bg-accent-soft text-accent">{p.hdrFormat}</span>
   589→        )}
   590→      </div>
   591→      <div className="min-w-0 flex-1">
   592→        <p className="harbor-clamp-1 text-sm font-medium text-ink">{titleLine}</p>
   593→        {detailLine && !showQuality && (
   594→          <p className="harbor-clamp-1 text-xs text-ink-subtle mt-0.5">{detailLine}</p>
   595→        )}
   596→        {showQuality && (
   597→          <div className="flex items-center gap-2.5 mt-1 text-[10px] text-ink-subtle flex-wrap">
   598→            {p?.codec && (
   599→              <span className="flex items-center gap-1"><FileVideo className="w-3 h-3" />{p.codec}</span>

   604→          </div>
   605→        )}
   606→        <p className="text-[10px] text-ink-subtle mt-0.5 flex items-center gap-1.5 flex-wrap">
   607→          {cached && <Zap className="w-3 h-3 text-emerald-400" />}
   608→          {isTorrent && (
   609→            <span className="rounded border border-edge-soft bg-raised px-1 py-0.5 text-[9px] font-bold text-ink-muted">
   610→              DEBRID
   611→            </span>
   612→          )}
   613→          {isTorrent && p2pEnabled && (
   614→            <span className="rounded border border-accent/40 bg-accent/10 px-1 py-0.5 text-[9px] font-bold text-accent flex items-center gap-1">
   615→              <Network className="w-2.5 h-2.5" aria-hidden /> P2P
   616→            </span>
   617→          )}
   618→          {stream.addonName ?? "addon"}
   619→        </p>
   620→        {err && (
   621→          <p className="basis-full w-full text-xs text-danger mt-1" role="alert">
   622→            {err}
   623→          </p>
   624→        )}
   625→      </div>
   626→      {isTorrent && (
   627→        <div className="flex items-center gap-1.5 shrink-0">
   628→          {p2pEnabled && (
   629→            <button
   630→              type="button"
   631→              onClick={(e) => void playP2p(e)}
   632→              disabled={busy}
   633→              className={cn(
   634→                "harbor-tv-focus flex items-center gap-1.5 rounded-xl px-3 py-2 text-xs font-bold transition-all",
   635→                debridKey === null
   636→                  ? "bg-accent text-black hover:brightness-110 disabled:opacity-60"
   637→                  : "bg-raised text-ink-muted hover:text-ink disabled:opacity-60",
   638→              )}
   639→              aria-label={debridKey === null ? "Play via P2P" : "Play via P2P torrent engine"}
   640→              title={debridKey === null ? "Play free via the built-in P2P engine" : "Play via the built-in P2P engine (slower)"}
   641→            >
   642→              {p2pPhase !== "idle" ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Network className="w-3.5 h-3.5" aria-hidden />}
   643→              {p2pPhase !== "idle" ? "Connecting…" : "Play"}
   644→            </button>
   645→          )}
   646→          <button
   647→            type="button"
   648→            onClick={(e) => void unlock(e)}
   649→            disabled={resolving}
   650→            className={cn(
   651→              "harbor-tv-focus flex items-center gap-1.5 rounded-xl px-3 py-2 text-xs font-bold transition-all",
   652→              debridKey === null
   653→                ? "bg-raised text-ink-muted hover:text-ink"
   654→                : "bg-accent-soft text-accent hover:brightness-110 disabled:opacity-60",
   655→            )}
   656→            aria-label="Unlock with debrid"
   657→            title={debridKey === null ? "Connect a debrid service for instant links" : "Unlock with your debrid service"}
   658→          >
   659→            {resolving ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <KeyRound className="w-3.5 h-3.5" aria-hidden />}
   660→            {resolving ? "Unlocking…" : debridKey === null ? "Debrid" : "Unlock"}
   661→          </button>
   662→        </div>
   663→      )}
   664→    </div>
   665→  );

   668→// Choose the torrent file index client-side when metadata arrives via polling
   669→function pickFileIndex(files: { index: number; name: string; length: number }[], filename?: string): number {
   670→  const VIDEO_RE = /\.(mp4|m4v|webm|mkv|avi|mov|ts|flv|wmv|mpg|mpeg)$/i;
   671→  const videos = files.filter((f) => VIDEO_RE.test(f.name) || f.length > 100 * 1024 * 1024);
   672→  const pool = videos.length > 0 ? videos : files;
   673→  if (filename) {
   674→    const want = filename.toLowerCase().replace(/[\s._-]+/g, "").slice(0, 24);
   675→    const byName = pool.find((f) => f.name.toLowerCase().replace(/[\s._-]+/g, "").includes(want));
   676→    if (byName) return byName.index;
   677→  }
   678→  return pool.reduce((a, b) => (b.length > a.length ? b : a)).index;
   679→}
