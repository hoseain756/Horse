    62→    62→/** Cap the ladder for data-saver / slow connections (never below medium so
    63→    63→ *  quality stays usable; the browser still picks by rendered×DPR). */
    64→    64→function tierCap(): number {
    65→    65→  if (typeof navigator === "undefined") return Infinity;
    66→    66→  const c = (navigator as unknown as { connection?: { saveData?: boolean; effectiveType?: string } }).connection;
    67→    67→  if (!c) return Infinity;
    68→    68→  if (c.saveData) return 500;
    69→    69→  if (c.effectiveType && /(^|\b)2g(\b|$)/i.test(c.effectiveType)) return 500;
    70→    70→  return Infinity;
    71→    71→}

    73→    73→function metahubLadder(base: string, kind: "poster" | "background" | "logo"): ArtTier[] {
    74→    74→  // https://images.metahub.space/<kind>/<size>/<id>/img — sizes verified:
    75→    75→  // poster small 300w / medium 500w / large 780w / original 2000w;
    76→    76→  // background medium 1280w / large 3840w; logo medium ~780w / large ~1526w.
    77→    77→  const sizes = kind === "poster"
    78→    78→    ? [
    79→    79→        { slug: "small", w: 300 },
    80→    80→        { slug: "medium", w: 500 },
    81→    81→        { slug: "large", w: 780 },
    82→    82→        { slug: "original", w: 2000 },
    83→    83→      ]
    84→    84→    : kind === "background"
    85→    85→      ? [
    86→    86→          { slug: "medium", w: 1280 },
    87→    87→          { slug: "large", w: 3840 },
    88→    88→        ]
    89→    89→      : [
    90→    90→          { slug: "medium", w: 780 },
    91→    91→          { slug: "large", w: 1526 },
    92→    92→        ];
    93→    93→  const cap = tierCap();
    94→    94→  const out: ArtTier[] = [];
    95→    95→  for (const s of sizes) {
    96→    96→    if (s.w > cap) break;
    97→    97→    out.push({ w: s.w, url: base.replace(/\/(small|medium|large|xlarge|original)\//, `/${s.slug}/`) });
    98→    98→  }
    99→    99→  return out.length > 0 ? out : [{ w: 500, url: base.replace(/\/(small|medium|large|xlarge|original)\//, "/medium/") }];
   100→   100→}

   102→   102→function tmdbLadder(base: string, poster: boolean): ArtTier[] {
   103→   103→  // .../t/p/<size>/<path> — poster w185…w780/original; backdrop w780…original.
   104→   104→  const sizes = poster
   105→   105→    ? [
   106→   106→        { slug: "w342", w: 342 },
   107→   107→        { slug: "w500", w: 500 },
   108→   108→        { slug: "w780", w: 780 },
   109→   109→        { slug: "original", w: 2000 },
   110→   110→      ]
   111→   111→    : [
   112→   112→        { slug: "w780", w: 780 },
   113→   113→        { slug: "w1280", w: 1280 },
   114→   114→        { slug: "original", w: 3840 },
   115→   115→      ];
   116→   116→  const cap = tierCap();

   118→   118→  for (const s of sizes) {

   129→   129→/** Resolved art for one slide: portrait (poster) + landscape (backdrop)
   130→   130→ *  ladders, plus a stable failure key (per title, not per size). */
   131→   131→function slideArt(meta: Meta): {
   132→   132→  portrait: { src: string; srcSet: string } | null;
   133→   133→  landscape: { src: string; srcSet: string } | null;
   134→   134→  failKey: string;
   135→   135→} | null {
   136→   136→  const poster = meta.poster;
   137→   137→  const bg = meta.background;
   138→   138→  if (!poster && !bg) return null;
   139→   139→  // Failure key is per TITLE (all size variants of one artwork 404 together).
   140→   140→  const failKey = `${meta.type}:${meta.id}`;
   141→   141→  const ladderFor = (url: string, kind: "poster" | "background" | "logo") => {
   142→   142→    if (!url) return null;
   143→   143→    if (/images\.metahub\.space\//.test(url)) return metahubLadder(url, kind);
   144→   144→    if (/image\.tmdb\.org\//.test(url)) return tmdbLadder(url, kind === "poster");
   145→   145→    return null; // unknown CDN → single-src
   146→   146→  };
   147→   147→  const pTiers = ladderFor(poster ?? "", "poster");
   148→   148→  const bTiers = ladderFor(bg ?? "", "background");
   149→   149→  return {
   150→   150→    portrait: poster
   151→   151→      ? { src: (pTiers ? pTiers[Math.min(1, pTiers.length - 1)] : { url: poster }).url, srcSet: pTiers ? srcSetOf(pTiers) : "" }
   152→   152→      : null,
   153→   153→    landscape: bg
   154→   154→      ? { src: (bTiers ? bTiers[0] : { url: bg }).url, srcSet: bTiers ? srcSetOf(bTiers) : "" }
   155→   155→      : null,
   156→   156→    failKey,
   157→   157→  };
   158→   158→}

   165→   165→/** Physical scrollLeft that aligns the given slide to the inline-start edge.
   166→   166→ *  Works in both directions: LTR slide.offsetLeft is i*width; RTL scrollLeft
   167→   167→ *  is negative (modern spec) and slide 0 rests at the right edge. */
   168→   168→function slideScrollLeft(track: HTMLElement, slide: HTMLElement): number {
   169→   169→  const dir = getComputedStyle(track).direction;
   170→   170→  if (dir === "rtl") {
   171→   171→    const slideStart = slide.offsetLeft + slide.offsetWidth; // physical right edge
   172→   172→    return -(track.clientWidth - slideStart);
   173→   173→  }
   174→   174→  return slide.offsetLeft;
   175→   175→}

   177→   177→/** One art element per slide — <picture> swaps portrait (phones) ↔ backdrop
   178→   178→ *  (md+) at the CSS breakpoint so ONLY the matching layer downloads. The img
   179→   179→ *  fades in after decode() resolves; failure falls back to the theme
   180→   180→ *  gradient (never a blank or broken icon). */
   181→   181→const HeroArt = memo(function HeroArt({
   182→   182→  art,
   183→   183→  alt,
   184→   184→  eager,
   185→   185→  active,
   186→   186→}: {
   187→   187→  art: NonNullable<ReturnType<typeof slideArt>>;
   188→   188→  alt: string;
   189→   189→  eager: boolean;
   190→   190→  active: boolean;
   191→   191→}) {
   192→   192→  const [failed, setFailed] = useState(FAILED_ART.has(art.failKey));
   193→   193→  const [ready, setReady] = useState(false);
   194→   194→  const [seenKey, setSeenKey] = useState(art.failKey);
   195→   195→  const imgRef = useRef<HTMLImageElement>(null);
   196→   196→  // Render-phase reset when the artwork changes (React-recommended pattern;
   197→   197→  // keeps the codebase's no-sync-setState-in-effect rule intact).
   198→   198→  if (seenKey !== art.failKey) {
   199→   199→    setSeenKey(art.failKey);
   200→   200→    setReady(false);
   201→   201→  }
   202→   202→
   203→   203→  // Decode gate: reveal only a fully-decoded frame (no half-painted flash).
   204→   204→  // Cached images are detected on the next tick (codebase lint pattern);
   205→   205→  // in-flight ones resolve via decode()/onLoad.
   206→   206→  useEffect(() => {
   207→   207→    const img = imgRef.current;
   208→   208→    if (!img) return;
   209→   209→    if (img.complete && img.naturalWidth > 0) {
   210→   210→      const h = setTimeout(() => setReady(true), 0);
   211→   211→      return () => clearTimeout(h);
   212→   212→    }
   213→   213→    img.decode?.().catch(() => {
   214→   214→      /* decode rejects on empty/undecodable sources; onLoad/onError take over */
   215→   215→    });
   216→   216→    return undefined;
   217→   217→  }, [art.portrait?.src, art.landscape?.src]);
   218→   218→
   219→   219→  if (failed || (!art.portrait && !art.landscape)) {
   220→   220→    return (
   221→   221→      <div
   222→   222→        aria-label={alt || undefined}
   223→   223→        role="img"
   224→   224→        className="home-hero-art home-hero-art-fallback"
   225→   225→      />
   226→   226→    );
   227→   227→  }
   228→   228→
   229→   229→  return (
   230→   230→    <picture
   231→   231→      className={`home-hero-art home-hero-art-picture${ready ? " home-hero-art-ready" : ""}`}
   232→   232→    >
   233→   233→      {art.landscape && (
   234→   234→        <source
   235→   235→          media={MD_MIN}
   236→   236→          srcSet={art.landscape.srcSet || undefined}
   237→   237→          sizes="100vw"
   238→   238→        />
   239→   239→      )}
   240→   240→      <img
   241→   241→        ref={imgRef}
   242→   242→        src={art.portrait?.src ?? art.landscape?.src}
   243→   243→        srcSet={art.portrait?.srcSet || undefined}
   244→   244→        sizes="(max-width: 599px) 100vw"
   245→   245→        alt={alt}
   246→   246→        loading={eager ? "eager" : "lazy"}
   247→   247→        decoding={eager ? "sync" : "async"}
   248→   248→        fetchPriority={eager ? (active ? "high" : "low") : "auto"}
   249→   249→        draggable={false}
   250→   250→        onLoad={() => setReady(true)}
   251→   251→        onError={() => {
   252→   252→          FAILED_ART.add(art.failKey);
   253→   253→          setFailed(true);
   254→   254→        }}
   255→   255→      />
   256→   256→    </picture>
   257→   257→  );
   258→   258→});

   273→   273→/** Sequential TMDB enrichment: fills title logos + a localized first genre
   274→   274→ *  for slides that lack them. One slide at a time, ~150ms apart; every step
   275→   275→ *  fails soft (cached by tmdbGet, so revisits are free). */
   276→   276→function useHeroEnrichment(slides: HeroSlide[]) {
   277→   277→  const [info, setInfo] = useState<Record<string, SlideInfo>>({});
   278→   278→  const infoRef = useRef(info);
   279→   279→  useEffect(() => {
   280→   280→    infoRef.current = info;
   281→   281→  }, [info]);
   282→   282→  const sig = slides.map((s) => s.meta.id).join("|");
   283→   283→  useEffect(() => {
   284→   284→    if (slides.length === 0) return;
   285→   285→    let alive = true;
   286→   286→    (async () => {
   287→   287→      for (const { meta } of slides) {
   288→   288→        if (!alive) return;
   289→   289→        if (!/^tt\d+$/.test(meta.id)) continue;
   290→   290→        if (meta.logo && meta.genres?.length) continue;
   291→   291→        if (infoRef.current[meta.id]) continue;
   292→   292→        try {
   293→   293→          const tmdbId = await tmdbIdFromImdb(meta.id, meta.type === "series" ? "series" : "movie");
   294→   294→          if (!alive) return;
   295→   295→          if (!tmdbId) continue;
   296→   296→          const details = await tmdbDetails(meta.type === "series" ? "series" : "movie", tmdbId);
   297→   297→          if (!alive || !details) continue;
   298→   298→          const logo = tmdbLogo(details);
   299→   299→          const genre = details.genres?.[0]?.name;
   300→   300→          if (logo || genre) {
   301→   301→            setInfo((prev) => ({ ...prev, [meta.id]: { logo, genre } }));
   302→   302→          }
   303→   303→        } catch {
   304→   304→          /* fail soft */
   305→   305→        }
   306→   306→        await new Promise((r) => setTimeout(r, 150));
   307→   307→      }
   308→   308→    })();
   309→   309→    return () => {
   310→   310→      alive = false;
   311→   311→    };
   312→   312→  }, [sig, slides]);
   313→   313→  return info;
   314→   314→}

   316→   316→export function HomeHero({ slides, loading }: { slides: HeroSlide[]; loading: boolean }) {
   317→   317→  const settings = useSettings((s) => s.settings);
   318→   318→  const lang = settings.uiLanguage || "en";
   319→   319→  const rtl = isArabic(lang);
   320→   320→  const push = useNav((s) => s.push);
   321→   321→
   322→   322→  const [idx, setIdx] = useState(0);
   323→   323→  const [userPaused, setUserPaused] = useState(false);
   324→   324→  const [hovering, setHovering] = useState(false);
   325→   325→  const [touching, setTouching] = useState(false);
   326→   326→  const [focusPaused, setFocusPaused] = useState(false);
   327→   327→  const [onScreen, setOnScreen] = useState(true);
   328→   328→  const [reduced, setReduced] = useState(false);
   329→   329→  // Burst-tolerant art window (round-23): steady state is the strict
   330→   330→  // [idx-1, idx, idx+1] window; during rapid navigation (fast dot jumps,
   331→   331→  // keyboard repeat) the set ACCUMULATES visited slides and prunes back
   332→   332→  // 1.6s after the burst ends — mount/unmount thrash during a burst is zero,
   333→   333→  // the resting DOM stays virtualized (3 arts). Standard overscan practice.
   334→   334→  const [burstSet, setBurstSet] = useState<Set<number>>(() => new Set());
   335→   335→
   336→   336→  const trackRef = useRef<HTMLDivElement>(null);
   337→   337→  const contentRef = useRef<HTMLDivElement>(null);
   338→   338→  const rafRef = useRef(0);
   339→   339→  const touchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   340→   340→
   341→   341→  const info = useHeroEnrichment(slides);
   342→   342→
   343→   343→  // Reduced-motion: no autoplay, no smooth scrolling, no crossfade (CSS too)
   344→   344→  useEffect(() => {
   345→   345→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   346→   346→    const update = () => setReduced(mq.matches);
   347→   347→    update();
   348→   348→    mq.addEventListener("change", update);
   349→   349→    return () => mq.removeEventListener("change", update);
   350→   350→  }, []);
   351→   351→
   352→   352→  // Off-screen gate (also prevents programmatic scrolling from yanking the page)
   353→   353→  useEffect(() => {
   354→   354→    const el = trackRef.current;
   355→   355→    if (!el || typeof IntersectionObserver === "undefined") return;
   356→   356→    const io = new IntersectionObserver(([e]) => setOnScreen(e.isIntersecting), {
   357→   357→      threshold: 0.35,
   358→   358→    });
   359→   359→    io.observe(el);
   360→   360→    return () => io.disconnect();
   361→   361→  }, []);
   362→   362→
   363→   363→  // Scroll → active index (exact 1-slide-per-width snapping). rAF-coalesced;
   364→   364→  // setIdx with an unchanged number is bailed out by React, so swiping costs
   365→   365→  // one cheap re-render per slide boundary only.
   366→   366→  const onScroll = useCallback(() => {
   367→   367→    if (rafRef.current) return;
   368→   368→    rafRef.current = requestAnimationFrame(() => {
   369→   369→      rafRef.current = 0;
   370→   370→      const el = trackRef.current;
   371→   371→      if (!el || el.clientWidth === 0) return;
   372→   372→      const i = Math.round(Math.abs(el.scrollLeft) / el.clientWidth);
   373→   373→      setIdx(Math.max(0, Math.min(slides.length - 1, i)));
   374→   374→    });
   375→   375→  }, [slides.length]);
   376→   376→
   377→   377→  const goTo = useCallback(
   378→   378→    (i: number, smooth = true) => {
   379→   379→      const el = trackRef.current;
   380→   380→      if (!el) return;
   381→   381→      const slide = el.querySelector<HTMLElement>(`[data-slide="${i}"]`);
   382→   382→      if (!slide) return;
   383→   383→      el.scrollTo({
   384→   384→        left: slideScrollLeft(el, slide),
   385→   385→        behavior: smooth && !reduced ? "smooth" : "auto",
   386→   386→      });
   387→   387→      setIdx(i);
   388→   388→    },
   389→   389→    [reduced],
   390→   390→  );
   391→   391→
   392→   392→  const total = slides.length;
   393→   393→
   394→   394→  // Wrap-aware art window + per-slide art resolution. Hooks stay ABOVE the
   395→   395→  // early returns (rules-of-hooks); the early returns below only choose the
   396→   396→  // skeleton/empty presentation.
   397→   397→  // Resolved ONCE per slides-array identity so HeroArt's memo holds.
   398→   398→  const arts = useMemo(() => slides.map((s) => slideArt(s.meta)), [slides]);
   399→   399→  const artWindow = useMemo(() => {
   400→   400→    const set = new Set<number>();
   401→   401→    if (total > 0) {
   402→   402→      set.add(idx);
   403→   403→      set.add((idx + 1) % total);
   404→   404→      set.add((idx - 1 + total) % total);
   405→   405→    }
   406→   406→    return set;
   407→   407→  }, [idx, total]);
   408→   408→
   409→   409→  // Burst accumulation + delayed prune (see burstSet above). Both state
   410→   410→  // updates run inside timers (codebase no-sync-setState-in-effect rule).
   411→   411→  useEffect(() => {
   412→   412→    if (total <= 0) return;
   413→   413→    const grow = setTimeout(() => {
   414→   414→      setBurstSet((prev) => {
   415→   415→        const next = new Set(prev);
   416→   416→        next.add(idx);
   417→   417→        next.add((idx + 1) % total);
   418→   418→        next.add((idx - 1 + total) % total);
   419→   419→        return next.size === prev.size ? prev : next;
   420→   420→      });
   421→   421→    }, 0);
   422→   422→    const prune = setTimeout(() => {
   423→   423→      setBurstSet(new Set([idx, (idx + 1) % total, (idx - 1 + total) % total]));
   424→   424→    }, 1600);
   425→   425→    return () => {

   572→   572→    else setTimeout(step, 300);

   574→   574→      cancelled = true;
   575→   575→    };
