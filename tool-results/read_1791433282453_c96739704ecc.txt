    34→type Stats = {
    35→  totalEntries: number;
    36→  uniqueTitles: number;
    37→  movies: number;
    38→  episodes: number;
    39→  watchMs: number;
    40→  activeDays: number;
    41→  bestDay: { date: string; ms: number } | null;
    42→  daily: { date: string; ms: number }[]; // last 14 days
    43→  heat365: { date: string; ms: number }[]; // last 365 days, oldest -> today
    44→  maxEpisodesPerDay: number; // most episodes of one series watched in a single day
    45→  lateNightSessions: number; // sessions between 00:00-04:59 local
    46→  maxSessions: number; // most sessions of a single title
    47→  topTitles: { id: string; name: string; ms: number; poster?: string; type: string; count: number }[];
    48→  topGenres: { genre: string; count: number }[];
    49→};

    51→function dayKey(t: number): string {
    52→  const d = new Date(t);
    53→  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`;
    54→}
    55→
    56→function computeStats(history: HistoryEntry[]): Stats {
    57→  const totalMs = (e: HistoryEntry) => Math.max(0, Math.min(e.positionMs || e.durationMs, e.durationMs || e.positionMs));
    58→  let watchMs = 0;
    59→  let movies = 0;
    60→  let episodes = 0;
    61→  const byDay = new Map<string, number>();
    62→  const byTitle = new Map<string, { name: string; ms: number; poster?: string; type: string; count: number }>();
    63→  const days = new Set<string>();
    64→
    65→  for (const e of history) {
    66→    const ms = totalMs(e);
    67→    watchMs += ms;
    68→    const dk = dayKey(e.t);
    69→    days.add(dk);
    70→    byDay.set(dk, (byDay.get(dk) ?? 0) + ms);
    71→    if (e.type === "series" && e.episode) episodes++;
    72→    else movies++;
    73→    const prev = byTitle.get(e.id);
    74→    byTitle.set(e.id, {
    75→      name: e.name,
    76→      ms: (prev?.ms ?? 0) + ms,
    77→      poster: prev?.poster ?? e.poster,
    78→      type: e.type,
    79→      count: (prev?.count ?? 0) + 1,
    80→    });
    81→  }
    82→
    83→  // last 14 days series
    84→  const daily: { date: string; ms: number }[] = [];
    85→  const now = new Date();
    86→  for (let i = 13; i >= 0; i--) {
    87→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    88→    const key = dayKey(d.getTime());
    89→    daily.push({ date: key, ms: byDay.get(key) ?? 0 });
    90→  }
    91→
    92→  // last 365 days heatmap series (day 364 days ago -> today, oldest first)
    93→  const heat365: { date: string; ms: number }[] = [];
    94→  for (let i = 364; i >= 0; i--) {
    95→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    96→    const key = dayKey(d.getTime());
    97→    heat365.push({ date: key, ms: byDay.get(key) ?? 0 });
    98→  }
    99→
   100→  // Badge stats — computed honestly from the same history
   101→  let maxEpisodesPerDay = 0;
   102→  let lateNightSessions = 0;
   103→  const episodesPerDay = new Map<string, number>(); // `${dayKey}|${id}` -> same-series episode count
   104→  for (const e of history) {
   105→    if (new Date(e.t).getHours() < 5) lateNightSessions++;
   106→    if (e.type === "series" && e.episode) {
   107→      const k = `${dayKey(e.t)}|${e.id}`;
   108→      const n = (episodesPerDay.get(k) ?? 0) + 1;
   109→      episodesPerDay.set(k, n);
   110→      if (n > maxEpisodesPerDay) maxEpisodesPerDay = n;
   111→    }
   112→  }
   113→  let maxSessions = 0;
   114→  for (const v of byTitle.values()) {
   115→    if (v.count > maxSessions) maxSessions = v.count;
   116→  }
   117→
   118→  let bestDay: { date: string; ms: number } | null = null;
   119→  for (const [date, ms] of byDay) {
   120→    if (!bestDay || ms > bestDay.ms) bestDay = { date, ms };
   121→  }
   122→
   123→  const topTitles = Array.from(byTitle.entries())
   124→    .map(([id, v]) => ({ id, ...v }))
   125→    .sort((a, b) => b.ms - a.ms)
   126→    .slice(0, 6);
   127→
   128→  return {
   129→    totalEntries: history.length,
   130→    uniqueTitles: byTitle.size,
   131→    movies,
   132→    episodes,
   133→    watchMs,
   134→    activeDays: days.size,
   135→    bestDay,
   136→    daily,
   137→    heat365,
   138→    maxEpisodesPerDay,
   139→    lateNightSessions,
   140→    maxSessions,
   141→    topTitles,
   142→    topGenres: [], // filled async
   143→  };
   144→}

   146→function fmtDuration(ms: number): string {
   147→  const hours = Math.floor(ms / 3_600_000);
   148→  const mins = Math.floor((ms % 3_600_000) / 60_000);
   149→  const secs = Math.floor((ms % 60_000) / 1000);
   150→  if (hours >= 1) return `${hours}h ${mins}m`;
   151→  if (mins >= 1) return `${mins}m`;
   152→  return `${secs}s`;
   153→}

   155→/**
   156→ * Resolve the app accent (applyTheme sets --color-accent-var on <html>) to a CSS
   157→ * color string usable in inline color-mix() styles. Falls back to a pleasant
   158→ * warm amber if the variable can't be read.
   159→ */
   160→function resolvedAccent(): string {
   161→  try {
   162→    const v = getComputedStyle(document.documentElement).getPropertyValue("--color-accent-var").trim();
   163→    return v || "#f0b429";
   164→  } catch {
   165→    return "#f0b429";
   166→  }
   167→}

   169→// ---------- 365-day heatmap (GitHub style) ----------
   170→
   171→const WEEKDAY_ROWS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]; // row 0 = Monday
   172→const MONTHS_SHORT = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   173→
   174→type HeatCell = { key: string; label: string; level: number; future: boolean };
   175→
   176→function heatLevel(ms: number, max: number): number {
   177→  if (ms <= 0) return 0;
   178→  if (max <= 0) return 4;
   179→  if (ms <= max * 0.25) return 1;
   180→  if (ms <= max * 0.5) return 2;
   181→  if (ms <= max * 0.75) return 3;
   182→  return 4;
   183→}
   184→
   185→/** 0 = raised/empty (class-driven); 1-4 = accent tints via color-mix. */
   186→function levelBg(level: number, accent: string): string | undefined {
   187→  switch (level) {
   188→    case 1:
   189→      return `color-mix(in srgb, ${accent} 25%, transparent)`;
   190→    case 2:
   191→      return `color-mix(in srgb, ${accent} 45%, transparent)`;
   192→    case 3:
   193→      return `color-mix(in srgb, ${accent} 70%, transparent)`;
   194→    case 4:
   195→      return accent;
   196→    default:
   197→      return undefined;
   198→  }
   199→}

   201→function HeatmapGrid({ heat, accent }: { heat: { date: string; ms: number }[]; accent: string }) {
   202→  const { columns, monthLabels } = useMemo(() => {
   203→    const now = new Date();
   204→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   205→    const start = new Date(today.getFullYear(), today.getMonth(), today.getDate() - 364);
   206→    const dow = (start.getDay() + 6) % 7; // 0 = Monday
   207→    const gridStart = new Date(start.getFullYear(), start.getMonth(), start.getDate() - dow);
   208→    const byDate = new Map(heat.map((h) => [h.date, h.ms]));
   209→    let max = 0;
   210→    for (const h of heat) if (h.ms > max) max = h.ms;
   211→
   212→    const cols: HeatCell[][] = [];
   213→    const monthLabels: (string | null)[] = [];
   214→    let prevMonth = -1;
   215→    const cursor = new Date(gridStart);
   216→    while (cursor <= today) {
   217→      const col: HeatCell[] = [];
   218→      let firstInRangeMonth = -1;
   219→      for (let r = 0; r < 7; r++) {
   220→        if (cursor > today) {
   221→          col.push({ key: `f${cols.length}-${r}`, label: "", level: 0, future: true });
   222→        } else {
   223→          if (cursor.getTime() >= start.getTime() && firstInRangeMonth === -1) {
   224→            firstInRangeMonth = cursor.getMonth();
   225→          }
   226→          const key = dayKey(cursor.getTime());
   227→          const ms = byDate.get(key) ?? 0;
   228→          col.push({
   229→            key,
   230→            label: `${MONTHS_SHORT[cursor.getMonth()]} ${cursor.getDate()} · ${fmtDuration(ms)}`,
   231→            level: heatLevel(ms, max),
   232→            future: false,
   233→          });
   234→        }
   235→        cursor.setDate(cursor.getDate() + 1);
   236→      }
   237→      if (firstInRangeMonth !== -1 && firstInRangeMonth !== prevMonth) {
   238→        monthLabels.push(MONTHS_SHORT[firstInRangeMonth]);
   239→        prevMonth = firstInRangeMonth;
   240→      } else {
   241→        monthLabels.push(null);
   242→      }
   243→      cols.push(col);
   244→    }
   245→    return { columns: cols, monthLabels };
   246→  }, [heat]);
   247→
   248→  const todayKey = dayKey(Date.now());
   249→
   250→  return (
   251→    <div className="flex" role="img" aria-label="365-day watch activity heatmap">
   252→      {/* Weekday gutter (Mon/Wed/Fri only) */}
   253→      <div className="mr-1.5 shrink-0 pt-[14px]" aria-hidden>
   254→        <div className="flex flex-col gap-[2px]">
   255→          {WEEKDAY_ROWS.map((wd, i) => (
   256→            <div key={wd} className="flex h-[10px] w-6 items-center justify-end">
   257→              {(i === 0 || i === 2 || i === 4) && (
   258→                <span className="text-[9px] leading-none text-ink-subtle">{wd}</span>
   259→              )}
   260→            </div>
   261→          ))}
   262→        </div>
   263→      </div>
   264→      {/* Grid — horizontally scrollable on narrow screens */}
   265→      <div className="harbor-scroll flex-1 overflow-x-auto pb-1">
   266→        <div className="w-max">
   267→          <div className="mb-[3px] flex h-[11px] gap-[2px]">
   268→            {monthLabels.map((m, i) => (
   269→              <div key={i} className="relative h-[11px] w-[10px] shrink-0">
   270→                {m && (
   271→                  <span className="absolute left-0 top-0 whitespace-nowrap text-[9px] leading-[11px] text-ink-subtle">
   272→                    {m}
   273→                  </span>
   274→                )}
   275→              </div>
   276→            ))}
   277→          </div>
   278→          <div className="flex gap-[2px]">
   279→            {columns.map((col, ci) => (
   280→              <div key={ci} className="flex flex-col gap-[2px]">
   281→                {col.map((cell) =>
   282→                  cell.future ? (
   283→                    <div key={cell.key} className="h-[10px] w-[10px]" />
   284→                  ) : (
   285→                    <div
   286→                      key={cell.key}
   287→                      className={cn(
   288→                        "h-[10px] w-[10px] rounded-[2px]",
   289→                        cell.level === 0 && "border border-edge-soft bg-raised",
   290→                        cell.key === todayKey && "ring-1 ring-accent/80",
   291→                      )}
   292→                      style={cell.level > 0 ? { backgroundColor: levelBg(cell.level, accent) } : undefined}
   293→                      title={cell.label}
   294→                    />
   295→                  ),
   296→                )}
   297→              </div>
   298→            ))}
   299→          </div>
   300→        </div>
   301→      </div>
   302→    </div>
   303→  );
   304→}

   415→type GenreStatus = "idle" | "resolving" | "done";

   417→export function WrappedView() {
   418→  const push = useNav((s) => s.push);
   419→  const { toast } = useToast();
   420→  const [stats, setStats] = useState<Stats | null>(null);
   421→  const [genreMap, setGenreMap] = useState<Map<string, string[]>>(new Map());
   422→  const [genreStatus, setGenreStatus] = useState<GenreStatus>("idle");
   423→  const [genreMisses, setGenreMisses] = useState(0);
   424→  const [sharing, setSharing] = useState(false);
   425→  const [shareError, setShareError] = useState(false);
   426→
   427→  useEffect(() => {
   428→    const t = setTimeout(() => setStats(computeStats(getHistory())), 0);
   429→    return () => clearTimeout(t);
   430→  }, []);
   431→
   432→  const accent = useMemo(() => (stats ? resolvedAccent() : "#f0b429"), [stats]);
   433→
   434→  const handleShare = useCallback(async () => {
   435→    if (!stats || sharing || stats.totalEntries === 0) return;
   436→    setSharing(true);
   437→    setShareError(false);
   438→    try {
   439→      const blob = await generateShareCard({
   440→        watchMs: stats.watchMs,
   441→        movies: stats.movies,
   442→        episodes: stats.episodes,
   443→        activeDays: stats.activeDays,
   444→        topTitles: stats.topTitles
   445→          .slice(0, 3)
   446→          .map((t) => ({ name: t.name, poster: t.poster, type: t.type })),
   447→        accent: resolvedAccent(),
   448→      });
   449→      const url = URL.createObjectURL(blob);
   450→      const a = document.createElement("a");
   451→      a.href = url;
   452→      a.download = "harbor-wrapped.png";
   453→      document.body.appendChild(a);
   454→      a.click();
   455→      a.remove();
   456→      window.setTimeout(() => URL.revokeObjectURL(url), 5000);
   457→      toast({ title: "Share card downloaded" });
   458→    } catch (e) {
   459→      console.error("[wrapped] share card generation failed", e);
   460→      setShareError(true);
   461→    } finally {
   462→      setSharing(false);
   463→    }
   464→  }, [stats, sharing, toast]);
   465→
   466→  // Resolve genres for top titles via Cinemeta (bounded time; never hangs the panel)
   467→  useEffect(() => {
   468→    if (!stats || stats.topTitles.length === 0) return;
   469→    let alive = true;
   470→    (async () => {
   471→      setGenreStatus("resolving");
   472→      const results = await Promise.allSettled(
   473→        stats.topTitles.map((t) =>
   474→          Promise.race([
   475→            fetchMeta(t.type === "series" ? "series" : "movie", t.id),
   476→            new Promise<null>((resolve) => setTimeout(() => resolve(null), 8_000)),
   477→          ]),
   478→        ),
   479→      );
   480→      if (!alive) return;
   481→      const map = new Map<string, string[]>();
   482→      let misses = 0;
   483→      for (let i = 0; i < results.length; i++) {
   484→        const r = results[i];
   485→        if (r.status === "fulfilled" && r.value) {
   486→          map.set(stats.topTitles[i].id, (r.value.genres ?? []).slice(0, 4));
   487→        } else {
   488→          misses++;
   489→        }
   490→      }
   491→      setGenreMap(map);
   492→      setGenreMisses(misses);
   493→      setGenreStatus("done");
   494→    })();
   495→    return () => {
   496→      alive = false;
   497→    };
   498→  }, [stats]);
   499→
   500→  const topGenres = useMemo(() => {
   501→    const counts = new Map<string, number>();
   502→    for (const t of stats?.topTitles ?? []) {
   503→      for (const g of genreMap.get(t.id) ?? []) {
   504→        counts.set(g, (counts.get(g) ?? 0) + 1);
   505→      }
   506→    }
   507→    return Array.from(counts.entries())
   508→      .map(([genre, count]) => ({ genre, count }))
   509→      .sort((a, b) => b.count - a.count)
   510→      .slice(0, 5);
   511→  }, [stats, genreMap]);
   512→
   513→  const heatTotalMs = useMemo(() => (stats?.heat365 ?? []).reduce((acc, d) => acc + d.ms, 0), [stats]);
   514→  const heatActiveDays = useMemo(() => (stats?.heat365 ?? []).filter((d) => d.ms > 0).length, [stats]);
   515→  const achievements = useMemo(() => buildAchievements(stats), [stats]);
   516→  const earnedCount = achievements.filter((a) => a.earned).length;
   517→
   518→  if (!stats) {
   519→    return (
   520→      <div className="pt-24 px-4 md:px-8">
   521→        <div className="harbor-skeleton h-40 rounded-3xl mb-6" aria-hidden />
   522→        <div className="harbor-skeleton h-24 rounded-3xl" aria-hidden />
   523→        <div className="sr-only">Loading your stats…</div>
   524→      </div>
   525→    );
   526→  }
   527→
   528→  const maxDaily = Math.max(...stats.daily.map((d) => d.ms), 1);
   529→
   530→  return (
   531→    <div className="pt-20 md:pt-14 pb-16 px-4 md:px-8 max-w-5xl">
   532→      <div className="flex flex-wrap items-center gap-3 mb-1">
   533→        <Sparkles className="w-6 h-6 text-accent" />
   534→        <h1 className="font-display text-2xl md:text-3xl font-bold text-ink">Your Wrapped</h1>
