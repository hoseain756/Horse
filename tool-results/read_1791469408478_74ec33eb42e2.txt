    74→    74→  const [query, setQuery] = useState("");

    84→    84→
    85→    85→  // TV-mode auto focus: keyboard users landing in the picker should get focus moved

    99→    99→        const target = row ?? closeRef.current;
   100→   100→        target?.focus({ preventScroll: true });
   101→   101→        row?.scrollIntoView({ block: "nearest", behavior: "smooth" });
   102→   102→      });
   103→   103→    }, 350);
   104→   104→  }, []);
   105→   105→
   106→   106→  // On open: focus the first row if streams are already there, else the close button.

   109→   109→    return () => {

   131→   131→    setProgressAddons(0);

   187→   187→  }, [streams]);

   208→   208→          }
   209→   209→          return true;

   211→   211→      }))

   229→   229→          metaId: id,
   230→   230→          season,
   231→   231→          episode,
   232→   232→          videoId: targetId,
   233→   233→          streamTitle: playing.title,
   234→   234→          streamDescription: playing.description,

   236→   236→          stream: playing,

   239→   239→        }}

   241→   241→    );
   242→   242→  }
   243→   243→

   246→   246→      <div className="md-sheet rounded-b-none! md:rounded-b-[28px]! w-full md:max-w-3xl md:max-h-[82vh] max-h-[88vh] flex flex-col overflow-hidden shadow-[var(--md-sys-elevation-3)]">
   247→   247→        {/* Header */}
   248→   248→        <div className="flex items-center justify-between gap-3 px-5 py-4 border-b border-edge-soft shrink-0">
   249→   249→          <div className="min-w-0">
   250→   250→            <h2 className="md-title-large text-ink">Choose a stream</h2>
   251→   251→            <p className="text-xs text-ink-subtle truncate">
   252→   252→              {type === "series" ? `Series · ${targetId}` : targetId}
   253→   253→              {season ? ` · S${season}:E${episode}` : ""}
   254→   254→            </p>
   255→   255→          </div>
   256→   256→          <div className="flex items-center gap-1.5 shrink-0">
   257→   257→            <button
   258→   258→              type="button"
   259→   259→              onClick={load}
   260→   260→              className="md-icon-btn md-state harbor-tv-focus"
   261→   261→              aria-label="Refresh streams"
   262→   262→              title="Refresh"
   263→   263→            >
   264→   264→              <RefreshCw className={cn("w-4 h-4", loading && "animate-spin")} />
   265→   265→            </button>
   266→   266→            <button
   267→   267→              type="button"
   268→   268→              ref={closeRef}
   269→   269→              onClick={pop}
   270→   270→              className="md-icon-btn md-state harbor-tv-focus"
   271→   271→              aria-label="Close picker"
   272→   272→            >
   273→   273→              <X className="w-5 h-5" />
   274→   274→            </button>
   275→   275→          </div>
   276→   276→        </div>
   277→   277→

   284→   284→              ["cached", "Cached"],

   312→   312→            {!showAll && hiddenCount > 0 ? ` (${hiddenCount} hidden)` : ""}
   313→   313→          </button>
   314→   314→          <div className="relative ml-auto w-40">
   315→   315→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-ink-subtle" />
   316→   316→            <input
   317→   317→              value={query}
   318→   318→              onChange={(e) => setQuery(e.target.value)}
   319→   319→              placeholder="Filter…"
   320→   320→              className="md-field-outlined w-full pl-8 pr-2 py-1.5 text-xs placeholder:text-ink-subtle"
   321→   321→              aria-label="Filter streams"
   322→   322→            />
   323→   323→          </div>
   324→   324→        </div>
   325→   325→
   326→   326→        {/* No-debrid hint: torrents play via the built-in P2P engine, debrid adds instant cached links */}
   327→   327→        {!loading && !debridConfigured && filteredTiers.length > 0 && (
   328→   328→          <div className="mx-5 mt-3 flex items-center gap-2.5 rounded-xl border border-accent/30 bg-accent/10 px-3.5 py-2.5 text-xs text-ink-muted shrink-0">
   329→   329→            <Network className="w-4 h-4 text-accent shrink-0" aria-hidden />
   330→   330→            <p className="min-w-0 flex-1">
   331→   331→              Torrent streams play through the <span className="font-semibold text-ink">built-in P2P engine</span> —

   333→   333→            </p>
   334→   334→            <button
   335→   335→              type="button"
   336→   336→              onClick={handleDebridNotConfigured}
   337→   337→              className="harbor-tv-focus md-state shrink-0 rounded-full bg-accent-soft px-3 py-1 text-[11px] font-bold text-accent"

   339→   339→              Connect
   340→   340→            </button>

   358→   358→                    <div className="h-2.5 w-1/3 rounded bg-white/10" />

   360→   360→                  <div className="w-16 h-7 rounded-lg bg-white/10 shrink-0" />

   362→   362→              ))}
   363→   363→            </div>
   364→   364→          )}
   365→   365→
   366→   366→          {!loading && error && (

   369→   369→            </div>
   370→   370→          )}
   371→   371→
   372→   372→          {!loading && !error && filteredTiers.length === 0 && (

   374→   374→              <Filter className="w-9 h-9 mx-auto mb-2 opacity-40" />

   376→   376→            </div>
   377→   377→          )}
   378→   378→
   379→   379→          {!loading &&
   380→   380→            filteredTiers.map(({ tier, items }) => (
   381→   381→              <section key={tier}>
   382→   382→                <div className="flex items-center gap-2 mb-2.5">

   385→   385→                  <span className="text-[10px] text-ink-subtle">{items.length} streams</span>
   386→   386→                  <div className="flex-1 h-px bg-edge-soft" />
   387→   387→                </div>
   388→   388→                <div className="space-y-2">
   389→   389→                  {items.slice(0, 40).map((s, i) => (
   390→   390→                    <StreamRow
   391→   391→                      key={`${s.addonId}-${i}`}
   392→   392→                      stream={s}
   393→   393→                      onPick={setPlaying}
   394→   394→                      onNeedsSetup={handleDebridNotConfigured}
   395→   395→                      configured={debridConfigured}
   396→   396→                      showQuality={settings.showQualityInfo}
   397→   397→                    />
   398→   398→                  ))}
   399→   399→                </div>
   400→   400→              </section>
   401→   401→            ))}
   402→   402→        </div>
   403→   403→      </div>
   404→   404→      {debridDialogOpen && <DebridSetupDialog onClose={() => setDebridDialogOpen(false)} />}
   405→   405→    </div>
   406→   406→  );
   407→   407→}

   409→   409→// Quality-ladder dot per tier header: emerald top, amber mid, neutral low —
   410→   410→// colors follow Harbor's existing palette conventions (understated).
   411→   411→function tierDotColor(tier: string): string {
   412→   412→  if (tier === "4K HDR" || tier === "4K") return "bg-emerald-400";
   413→   413→  if (tier === "1080p HDR" || tier === "1080p") return "bg-amber-500";
   414→   414→  if (tier === "720p" || tier === "SD") return "bg-zinc-500";
   415→   415→  return "bg-zinc-600";
   416→   416→}
   417→   417→
   418→   418→// Compatibility badge styles per verdict (green/yellow/gray per spec)
   419→   419→const BADGE_STYLES: Record<
   420→   420→  StreamClass["badge"],
   421→   421→  { label: string; title: string; className: string; dot: string }
   422→   422→> = {
   423→   423→  "plays-here": {
   424→   424→    label: "Plays here",
   425→   425→    title: "Direct play — this stream matches your browser",
   426→   426→    className: "bg-emerald-500/15 text-emerald-300 border border-emerald-500/30",
   427→   427→    dot: "bg-emerald-400",

   775→   775→// Choose the torrent file index client-side when metadata arrives via polling
   776→   776→function pickFileIndex(files: { index: number; name: string; length: number }[], filename?: string): number {
   777→   777→  const VIDEO_RE = /\.(mp4|m4v|webm|mkv|avi|mov|ts|flv|wmv|mpg|mpeg)$/i;
   778→   778→  const videos = files.filter((f) => VIDEO_RE.test(f.name) || f.length > 100 * 1024 * 1024);
   779→   779→  const pool = videos.length > 0 ? videos : files;
   780→   780→  if (filename) {
   781→   781→    const want = filename.toLowerCase().replace(/[\s._-]+/g, "").slice(0, 24);
   782→   782→    const byName = pool.find((f) => f.name.toLowerCase().replace(/[\s._-]+/g, "").includes(want));
   783→   783→    if (byName) return byName.index;
   784→   784→  }
   785→   785→  return pool.reduce((a, b) => (b.length > a.length ? b : a)).index;
   786→   786→}
