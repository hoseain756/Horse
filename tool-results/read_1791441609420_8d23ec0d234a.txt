    37→    37→// Focus bridge: app-shell hotkeys dispatch this; the bar listens for it.
    38→    38→export function focusFloatingSearch() {
    39→    39→  if (typeof window === "undefined") return;

    43→    43→function readRecents(): string[] {
    44→    44→  if (typeof window === "undefined") return [];
    45→    45→  try {
    46→    46→    const arr = JSON.parse(window.localStorage.getItem(RECENTS_KEY) ?? "[]") as unknown;
    47→    47→    return Array.isArray(arr) ? arr.filter((x): x is string => typeof x === "string").slice(0, MAX_RECENTS) : [];
    48→    48→  } catch {
    49→    49→    return [];
    50→    50→  }
    51→    51→}
    52→    52→
    53→    53→function writeRecent(q: string) {
    54→    54→  if (typeof window === "undefined") return;
    55→    55→  const trimmed = q.trim();
    56→    56→  if (!trimmed) return;
    57→    57→  const next = [trimmed, ...readRecents().filter((r) => r.toLowerCase() !== trimmed.toLowerCase())].slice(0, MAX_RECENTS);
    58→    58→  try {
    59→    59→    window.localStorage.setItem(RECENTS_KEY, JSON.stringify(next));
    60→    60→  } catch {
    61→    61→    /* quota — ignore */
    62→    62→  }
    63→    63→}

    65→    65→// A flattened result row (used for keyboard navigation)
    66→    66→type FlatRow =
    67→    67→  | { kind: "meta"; id: string; meta: Meta; group: string }
    68→    68→  | { kind: "person"; id: string; name: string; sub?: string | null; img?: string | null; url?: string }
    69→    69→  | { kind: "action"; id: string; action: "ai" | "see-all" }
    70→    70→  | { kind: "recent"; id: string; query: string }
    71→    71→  | { kind: "destination"; id: string; entry: HubEntry };

    73→    73→/** Quick Access destinations matching the query (EN + Arabic labels/keywords).
    74→    74→ *  Second way to reach the pages moved from the sidebar into Settings. */
    75→    75→function matchingDestinations(q: string, lang: string): HubEntry[] {
    76→    76→  const needle = q.trim().toLowerCase();
    77→    77→  if (needle.length < 2) return [];
    78→    78→  const ar = /^ar(-|_|$)/i.test(lang);
    79→    79→  return HUB_ENTRIES.filter((e) => {
    80→    80→    const label = hubLabel(e, lang).toLowerCase();
    81→    81→    const words = [...e.keywordsEn, ...e.keywordsAr, label];
    82→    82→    return words.some((w) => w.toLowerCase().includes(needle));
    83→    83→  }).slice(0, ar ? 4 : 3);
    84→    84→}

    86→    86→export function FloatingSearch() {
    87→    87→  const push = useNav((s) => s.push);
    88→    88→  const openFullSearch = useNav((s) => s.setSearchOpen);
    89→    89→  const uiLanguage = useSettings((s) => s.settings.uiLanguage);
    90→    90→
    91→    91→  const [expanded, setExpanded] = useState(false);
    92→    92→  const [hidden, setHidden] = useState(false); // scroll-direction visibility
    93→    93→  const [query, setQuery] = useState("");
    94→    94→  const [phase, setPhase] = useState<Phase>("idle");
    95→    95→  const [movies, setMovies] = useState<Meta[]>([]);
    96→    96→  const [series, setSeries] = useState<Meta[]>([]);
    97→    97→  const [addonHits, setAddonHits] = useState<Meta[]>([]);
    98→    98→  const [people, setPeople] = useState<TmdbMultiHit[]>([]);
    99→    99→  const [recents, setRecents] = useState<string[]>([]);
   100→   100→  const [trending, setTrending] = useState<Meta[]>([]);
   101→   101→  const [activeIdx, setActiveIdx] = useState(-1);
   102→   102→  const runToken = useRef(0);
   103→   103→  const debounce = useRef<ReturnType<typeof setTimeout> | null>(null);
   104→   104→  const wrapRef = useRef<HTMLDivElement>(null);
   105→   105→  const inputRef = useRef<HTMLInputElement>(null);
   106→   106→  const listRef = useRef<HTMLDivElement>(null);
   107→   107→
   108→   108→  // Focus bridge + outside click + Esc handled here
   109→   109→  useEffect(() => {
   110→   110→    const onFocus = () => {
   111→   111→      setExpanded(true);
   112→   112→      setHidden(false); // hotkey reveal: never focus an off-screen bar
   113→   113→      setRecents(readRecents());
   114→   114→      setTimeout(() => inputRef.current?.focus(), 20);
   115→   115→    };
   116→   116→    const onPointer = (e: MouseEvent) => {
   117→   117→      if (!wrapRef.current?.contains(e.target as Node)) {
   118→   118→        setExpanded(false);
   119→   119→        setActiveIdx(-1);
   120→   120→      }
   121→   121→    };
   122→   122→    window.addEventListener("harbor:focus-floating-search", onFocus);
   123→   123→    document.addEventListener("mousedown", onPointer);
   124→   124→    return () => {
   125→   125→      window.removeEventListener("harbor:focus-floating-search", onFocus);
   126→   126→      document.removeEventListener("mousedown", onPointer);
   127→   127→    };
   128→   128→  }, []);
   129→   129→
   130→   130→  // Scroll-direction visibility: scrolling DOWN hides the bar (content first),
   131→   131→  // scrolling UP (or returning near the top) reveals it again. While the bar
   132→   132→  // is expanded/focused it always stays visible.
   133→   133→  useEffect(() => {
   134→   134→    let lastY = window.scrollY;
   135→   135→    const onScroll = () => {
   136→   136→      const y = window.scrollY;
   137→   137→      const dy = y - lastY;
   138→   138→      lastY = y;
   139→   139→      if (expanded) return; // never hide while the user is searching
   140→   140→      if (y > 90 && dy > 4) setHidden(true);
   141→   141→      else if (dy < -4 || y <= 90) setHidden(false);
   142→   142→    };
   143→   143→    window.addEventListener("scroll", onScroll, { passive: true });
   144→   144→    return () => window.removeEventListener("scroll", onScroll);
   145→   145→  }, [expanded]);
   146→   146→
   147→   147→  // Trending suggestions (idle state) — fetched once per session, errors are silent
   148→   148→  useEffect(() => {
   149→   149→    let alive = true;
   150→   150→    (async () => {
   151→   151→      const [mv, sr] = await Promise.allSettled([
   152→   152→        fetchCinemetaCatalog("movie", "trending"),
   153→   153→        fetchCinemetaCatalog("series", "trending"),
   154→   154→      ]);
   155→   155→      if (!alive) return;
   156→   156→      const items = [
   157→   157→        ...(mv.status === "fulfilled" ? mv.value : []),
   158→   158→        ...(sr.status === "fulfilled" ? sr.value : []),
   159→   159→      ].slice(0, 8);
   160→   160→      setTrending(items);
   161→   161→    })();
   162→   162→    return () => {
   163→   163→      alive = false;
   164→   164→    };
   165→   165→  }, []);
   166→   166→
   167→   167→  const runInstant = useCallback(async (q: string) => {
   168→   168→    const token = ++runToken.current;
   169→   169→    setPhase("searching");
   170→   170→    try {
   171→   171→      const jobs: Promise<void>[] = [];
   172→   172→      jobs.push(
   173→   173→        searchCinemeta(q).then((r) => {
   174→   174→          if (token !== runToken.current) return;
   175→   175→          setMovies(r.movies);
   176→   176→          setSeries(r.series);
   177→   177→        }),
   178→   178→      );
   179→   179→      jobs.push(
   180→   180→        searchAddonCatalogs(installedAddons(), q).then((r) => {
   181→   181→          if (token !== runToken.current) return;
   182→   182→          setAddonHits(r);
   183→   183→        }),
   184→   184→      );
   185→   185→      if (tmdbEnabled()) {
   186→   186→        jobs.push(
   187→   187→          tmdbMultiSearch(q).then((r) => {
   188→   188→            if (token !== runToken.current) return;
   189→   189→            setPeople(r.person.slice(0, MAX_PER_GROUP));
   190→   190→          }),
   191→   191→        );
   192→   192→      } else {
   193→   193→        setPeople([]);
   194→   194→      }
   195→   195→      const res = await Promise.allSettled(jobs);
   196→   196→      if (token !== runToken.current) return;
   197→   197→      const allFailed = res.every((r) => r.status === "rejected");
   198→   198→      setPhase(allFailed && q.trim().length >= 2 ? "error" : "idle");
   199→   199→    } catch {
   200→   200→      if (token === runToken.current) setPhase("error");
   201→   201→    }
   202→   202→  }, []);
   203→   203→
   204→   204→  // Debounced instant search (250ms). State resets also run inside the timer
   205→   205→  // (never synchronously in the effect body) to satisfy the set-state lint rule.
   206→   206→  useEffect(() => {
   207→   207→    if (debounce.current) clearTimeout(debounce.current);
   208→   208→    const q = query.trim();
   209→   209→    const token = ++runToken.current; // invalidate any in-flight run
   210→   210→    if (q.length < 2) {
   211→   211→      debounce.current = setTimeout(() => {
   212→   212→        if (token !== runToken.current) return;
   213→   213→        setMovies([]);
   214→   214→        setSeries([]);
   215→   215→        setAddonHits([]);
   216→   216→        setPeople([]);
   217→   217→        setPhase("idle");
   218→   218→      }, 0);
   219→   219→    } else {
   220→   220→      debounce.current = setTimeout(() => void runInstant(q), 250);
   221→   221→    }
   222→   222→    return () => {
   223→   223→      if (debounce.current) clearTimeout(debounce.current);
   224→   224→    };
   225→   225→  }, [query, runInstant]);
   226→   226→
   227→   227→  const openMeta = useCallback(
   228→   228→    (m: Meta) => {
   229→   229→      writeRecent(query || m.name);
   230→   230→      setRecents(readRecents());
   231→   231→      setExpanded(false);
   232→   232→      setActiveIdx(-1);
   233→   233→      push({ kind: "detail", type: m.type, id: m.id });
   234→   234→    },
   235→   235→    [push, query],
   236→   236→  );
   237→   237→
   238→   238→  const openFull = useCallback(
   239→   239→    (q: string) => {
   240→   240→      writeRecent(q);
   241→   241→      setExpanded(false);
   242→   242→      setActiveIdx(-1);
   243→   243→      window.dispatchEvent(new CustomEvent("harbor:prefill-search", { detail: q }));
   244→   244→      openFullSearch(true);
   245→   245→    },
   246→   246→    [openFullSearch],
   247→   247→  );
   248→   248→
   249→   249→  // Flatten the dropdown for keyboard navigation
   250→   250→  const rows = useMemo<FlatRow[]>(() => {
   251→   251→    const out: FlatRow[] = [];
   252→   252→    const q = query.trim();
   253→   253→    if (q.length < 2) {
   254→   254→      for (const r of recents) out.push({ kind: "recent", id: `recent:${r}`, query: r });
   255→   255→      for (const m of trending) out.push({ kind: "meta", id: `trend:${m.type}:${m.id}`, meta: m, group: "Trending" });
   256→   256→      return out;
   257→   257→    }
   258→   258→    for (const m of movies.slice(0, MAX_PER_GROUP)) out.push({ kind: "meta", id: `m:${m.id}`, meta: m, group: "Movies" });
   259→   259→    for (const m of series.slice(0, MAX_PER_GROUP)) out.push({ kind: "meta", id: `s:${m.id}`, meta: m, group: "Series" });
   260→   260→    for (const e of matchingDestinations(q, uiLanguage)) out.push({ kind: "destination", id: `dest:${e.id}`, entry: e });
   261→   261→    for (const m of people) out.push({ kind: "person", id: `p:${m.id}`, name: m.name, sub: m.known_for_department, img: m.profile_path, url: m.url });
   262→   262→    for (const m of addonHits.slice(0, MAX_PER_GROUP)) out.push({ kind: "meta", id: `a:${m.id}`, meta: m, group: "From your addons" });
   263→   263→    if (q.length >= 2) out.push({ kind: "action", id: "see-all", action: "see-all" });
   264→   264→    return out;
   265→   265→  }, [query, movies, series, addonHits, people, recents, trending, uiLanguage]);
   266→   266→
   267→   267→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   268→   268→    if (e.key === "ArrowDown") {
   269→   269→      e.preventDefault();
   270→   270→      e.stopPropagation();
   271→   271→      setActiveIdx((i) => Math.min(rows.length - 1, i + 1));
   272→   272→    } else if (e.key === "ArrowUp") {
   273→   273→      e.preventDefault();
   274→   274→      e.stopPropagation();
   275→   275→      setActiveIdx((i) => Math.max(-1, i - 1));
   276→   276→    } else if (e.key === "Enter") {
   277→   277→      e.preventDefault();
   278→   278→      const row = rows[activeIdx];
   279→   279→      if (row?.kind === "meta") openMeta(row.meta);
   280→   280→      else if (row?.kind === "recent") {
   281→   281→        setQuery(row.query);
   282→   282→        inputRef.current?.focus();
   283→   283→      } else if (row?.kind === "destination") {
   284→   284→        setExpanded(false);
   285→   285→        setActiveIdx(-1);
   286→   286→        push({ kind: "view", view: row.entry.view });
   287→   287→      } else if (row?.kind === "action" && row.action === "see-all") {
   288→   288→        openFull(query);
   289→   289→      } else if (row?.kind === "person" && row.url) {
   290→   290→        window.open(row.url, "_blank", "noopener");
   291→   291→      } else if (query.trim()) {
   292→   292→        openFull(query);
   293→   293→      }
   294→   294→    } else if (e.key === "Escape") {
   295→   295→      e.preventDefault();
   296→   296→      e.stopPropagation();
   297→   297→      if (query) {
   298→   298→        setQuery("");
   299→   299→      } else {
   300→   300→        setExpanded(false);
   301→   301→        setActiveIdx(-1);
   302→   302→        inputRef.current?.blur();
   303→   303→      }
   304→   304→    }
   305→   305→  };
   306→   306→
   307→   307→  // Keep the active option in view
   308→   308→  useEffect(() => {
   309→   309→    if (activeIdx < 0 || !listRef.current) return;
   310→   310→    const el = listRef.current.querySelector(`#fs-opt-${activeIdx}`);
   311→   311→    el?.scrollIntoView({ block: "nearest" });
   312→   312→  }, [activeIdx]);
   313→   313→
   314→   314→  const hasResults = movies.length + series.length + addonHits.length + people.length > 0;
   315→   315→  const searching = phase === "searching";
   316→   316→  const showDropdown = expanded;
   317→   317→
   318→   318→  return (
   319→   319→    <div
   320→   320→      ref={wrapRef}
   321→   321→      className={cn(
   322→   322→        // Fixed top-center; mobile respects safe-area inset; below modals, above content
   323→   323→        "fixed z-[85] top-[max(0.75rem,env(safe-area-inset-top))] left-1/2 -translate-x-1/2",
   324→   324→        "w-[min(calc(100vw-6rem),3rem)] md:w-60",
   325→   325→        // Expand on focus + hide/reveal on scroll direction (smooth; motion-reduce users get instant snap)
   326→   326→        "transition-[width,transform,opacity] duration-300 ease-[var(--md-sys-motion-easing-emphasized)] motion-reduce:transition-none",
   327→   327→        expanded && "w-[min(calc(100vw-1.5rem),40rem)] md:w-[42rem]",
   328→   328→        // Scrolled-down: slide the bar fully off-screen (keeps -translate-x-1/2 centering)
   329→   329→        hidden && "-translate-y-[160%] opacity-0 pointer-events-none",
   330→   330→      )}
   331→   331→      aria-hidden={hidden || undefined}
   332→   332→    >
   333→   333→      <div
   334→   334→        className={cn(
   335→   335→          // M3 search bar shape: corner-full. Glass translucency kept (Harbor identity).
   336→   336→          "harbor-glass flex items-center gap-2 rounded-full border border-edge-soft",
   337→   337→          "shadow-[0_10px_36px_-12px_rgba(0,0,0,0.65),0_2px_10px_rgba(0,0,0,0.3)]",
   338→   338→          "transition-shadow duration-300",
   339→   339→          expanded && "shadow-[0_18px_50px_-12px_rgba(0,0,0,0.75)]",
   340→   340→        )}
   341→   341→      >
   342→   342→        <div
   343→   343→          className="flex items-center gap-2 px-2.5 md:px-4 h-11 md:h-14 w-full"
   344→   344→          role="combobox"
   345→   345→          aria-expanded={showDropdown && rows.length >= 0}
   346→   346→          aria-controls="harbor-float-search-list"
   347→   347→          aria-haspopup="listbox"
   348→   348→        >
   349→   349→          <Search className="w-4 h-4 md:w-5 md:h-5 text-ink-muted shrink-0" aria-hidden />
   350→   350→          <input
   351→   351→            ref={inputRef}
   352→   352→            value={query}
   353→   353→            onChange={(e) => setQuery(e.target.value)}
   354→   354→            onFocus={() => {
   355→   355→              setExpanded(true);
   356→   356→              setHidden(false);
   357→   357→              setRecents(readRecents());
   358→   358→            }}
   359→   359→            onKeyDown={onInputKeyDown}
   360→   360→            placeholder="Search…"
   361→   361→            aria-label="Search movies, series, people and addons"
   362→   362→            aria-autocomplete="list"
   363→   363→            className={cn(
   364→   364→              "min-w-0 flex-1 bg-transparent text-sm md:text-base text-ink placeholder:text-ink-muted outline-none",
   365→   365→              !expanded && "md:opacity-100 opacity-0 md:pointer-events-auto pointer-events-none w-0 md:w-auto",
   366→   366→            )}
   367→   367→            tabIndex={expanded ? 0 : -1}
   368→   368→          />
   369→   369→          {searching && <Loader2 className="w-4 h-4 text-accent animate-spin shrink-0" aria-hidden />}
   370→   370→          {/* AI mode hand-off (full overlay keeps the AI flow) */}
   371→   371→          <button
   372→   372→            type="button"
   373→   373→            tabIndex={expanded ? 0 : -1}
   374→   374→            onClick={() => openFull(query)}
   375→   375→            className={cn(
   376→   376→              "hidden md:flex shrink-0 items-center gap-1 rounded-full border border-edge-soft px-2 py-1 text-[10px] font-semibold text-ink-muted hover:text-accent hover:border-accent/50 transition-colors",
   377→   377→              !expanded && "opacity-0 pointer-events-none",
   378→   378→            )}
   379→   379→            title="AI search: describe what you feel like watching"
   380→   380→          >
   381→   381→            <Sparkles className="w-3 h-3" /> AI
   382→   382→          </button>
   383→   383→          <kbd
   384→   384→            className={cn(
   385→   385→              "harbor-kbd shrink-0 hidden md:inline-flex",
   386→   386→              (expanded || query) && "hidden",
   387→   387→            )}
   388→   388→            aria-hidden
   389→   389→          >
   390→   390→            /
   391→   391→          </kbd>
   392→   392→          {(query || expanded) && (
   393→   393→            <button
   394→   394→              type="button"
   395→   395→              onClick={() => {
   396→   396→                setQuery("");
   397→   397→                setExpanded(false);
   398→   398→                setActiveIdx(-1);
   399→   399→              }}
   400→   400→              className="md-state md-icon-btn shrink-0 w-7! h-7!"
   401→   401→              aria-label="Clear search"
   402→   402→            >
   403→   403→              <X className="w-3.5 h-3.5" />
   404→   404→            </button>
   405→   405→          )}
   406→   406→        </div>
   407→   407→      </div>
   408→   408→
   409→   409→      {/* Dropdown */}
   410→   410→      {showDropdown && (
   411→   411→        <div
   412→   412→          id="harbor-float-search-list"
   413→   413→          role="listbox"
   414→   414→          aria-label="Search suggestions"
   415→   415→          ref={listRef}
   416→   416→          className={cn(
