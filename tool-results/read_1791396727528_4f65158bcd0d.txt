    31→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    32→  const pop = useNav((s) => s.pop);
    33→  const replace = useNav((s) => s.replace);
    34→  const settings = useSettings((s) => s.settings);
    35→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    36→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    37→  const [error, setError] = useState<string | null>(null);
    38→  const [switcherOpen, setSwitcherOpen] = useState(false);
    39→  const [resolveStatus, setResolveStatus] = useState<string>("Finding the best stream…");
    40→
    41→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    42→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    43→  const handleError = useCallback((msg: string) => {
    44→    setPhase("error");
    45→    setError(msg);
    46→  }, []);
    47→
    48→  // Resolve stream via addons when payload.url is empty (instant play path)
    49→  useEffect(() => {
    50→    if (current.url || phase !== "resolving") return;
    51→    let alive = true;
    52→    setResolveStatus("Finding the best stream…");
    53→    (async () => {
    54→      const { useAddons: addons } = await import("@/lib/harbor/store");
    55→      try {
    56→        // Series opened without an episode id (bare tt-id): stream addons need
    57→        // "id:season:episode" — resolve the default video from meta first.
    58→        let targetId = current.videoId ?? current.metaId;
    59→        if (current.type === "series" && !targetId.includes(":")) {
    60→          const meta = await fetchMeta("series", current.metaId).catch(() => null);
    61→          const vid = defaultVideoId(meta);
    62→          if (vid) {
    63→            targetId = vid;
    64→            if (alive) {
    65→              const vidMeta = (meta?.videos ?? []).find((v) => v.id === vid);
    66→              setCurrent((c) => ({
    67→                ...c,
    68→                videoId: vid,
    69→                season: c.season ?? vidMeta?.season,
    70→                episode: c.episode ?? vidMeta?.episode,
    71→                episodeName: c.episodeName ?? vidMeta?.name ?? vidMeta?.title,
    72→              }));
    73→            }
    74→          }
    75→        }
    76→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
    77→        if (!alive) return;
    78→        const ranked = runPipeline(results);
    79→        // Prefer browser-playable candidates: direct URLs first, then torrents whose
    80→        // parsed video codec isn't HEVC (browsers can't decode it — pick the best
    81→        // H.264/VP9/AV1 release instead of a 4K HEVC one that would dead-end).
    82→        const p2pCandidate = (s: Stream) =>
    83→          !!s.infoHash && !s.url && s.parsed?.codec !== "HEVC";
    84→        const best = ranked.find((s) => s.url) ?? ranked.find(p2pCandidate) ?? ranked[0];
    85→        if (!best) {
    86→          setPhase("error");
    87→          setError("No streams found from your addons.");
    88→          return;
    89→        }
    90→        if (best.url) {
    91→          setCurrent((c) => ({
    92→            ...c,
    93→            url: best.url!,
    94→            streamTitle: best.title ?? c.streamTitle,
    95→            streamDescription: best.description,
    96→          }));
    97→          setPhase("loading");
    98→          return;
    99→        }
   100→        // Best candidate is a torrent → try the server-side P2P engine before giving up
   101→        if (!best.infoHash || !useSettings.getState().settings.p2pEnabled) {
   102→          setPhase("error");
   103→          setError(
   104→            ranked.length > 0
   105→              ? "No browser-playable stream found. Open the stream picker to choose manually — torrent streams play via P2P or unlock with debrid."
   106→              : "No streams found from your addons.",
   107→          );
   108→          return;
   109→        }
   110→        const { p2pPrepare, p2pStatus, p2pPlan } = await import("@/lib/harbor/p2p");
   111→        const infoHash = best.infoHash.toLowerCase();
   112→        setResolveStatus("Joining the torrent swarm…");
   113→        const prep = await p2pPrepare(infoHash, {
   114→          fileIdx: best.fileIdx,
   115→          filename: best.behaviorHints?.filename ?? best.parsed?.filename,
   116→        });
   117→        for (let i = 0; prep.pending && i < 40 && alive; i++) {
   118→          await new Promise((r) => setTimeout(r, 2500));
   119→          setResolveStatus((s) => (s.startsWith("Finding peers") ? s : `Finding peers… (${i * 2.5 | 0}s)`));
   120→          const st = await p2pStatus(infoHash).catch(() => null);
   121→          if (st?.peers) setResolveStatus(`Finding peers… (${st.peers} found)`);
   122→          if (st?.ready) break;
   123→        }
   124→        if (!alive) return;
   125→        const st = prep.pending ? await p2pStatus(infoHash).catch(() => null) : null;
   126→        if (prep.pending && !st?.ready) {
   127→          setPhase("error");
   128→          setError(
   129→            "No peers responded for the best torrent. Open the stream picker to pick another source, or connect debrid for instant cached streams.",
   130→          );
   131→          return;
   132→        }
   133→        const fileIdx = prep.pending
   134→          ? (Number.isInteger(best.fileIdx)
   135→              ? best.fileIdx!
   136→              : (st?.files ?? []).length > 0
   137→                ? (st!.files ?? []).reduce(
   138→                    (a, b) => (b.length > a.length ? b : a),
   139→                    (st!.files ?? [{ index: 0, name: "", length: 0 }])[0],
   140→                  ).index
   141→                : 0)
   142→          : prep.fileIdx;
   143→        const filename = best.behaviorHints?.filename ?? best.parsed?.filename;
   144→        setResolveStatus("Checking video codecs…");
   145→        const plan = await p2pPlan(infoHash, fileIdx, filename).catch(() => null);
   146→        if (!alive) return;
   147→        if (!plan) {
   148→          setPhase("error");
   149→          setError("Torrent engine could not serve this file. Try another stream in the picker.");
   150→          return;
   151→        }
   152→        setCurrent((c) => ({
   153→          ...c,
   154→          url: plan.url,
   155→          streamTitle: best.title ?? c.streamTitle,
   156→          streamDescription: best.description,
   157→          p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   158→        }));
   159→        setPhase("loading");
   160→      } catch (e) {
   161→        if (!alive) return;
   162→        setPhase("error");
   163→        setError(e instanceof Error ? e.message : "Failed to resolve streams");
   164→      }
   165→    })();
   166→    return () => {
   167→      alive = false;
   168→    };
   169→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
   170→
   171→  const setPayload = useCallback((p: PlayerPayload) => {
   172→    setCurrent(p);
   173→    setPhase(p.url ? "loading" : "resolving");
   174→    setError(null);
   175→  }, []);
   176→
   177→  const openSwitcher = useCallback(() => setSwitcherOpen(true), []);
   178→
   179→  // Advance to the next episode in-place (Up Next card / auto-advance / "n" key).
   180→  // Empty url re-enters the resolving path so addons fetch streams for the new episode.
   181→  const handlePlayNext = useCallback(
   182→    (next: { season: number; episode: number; videoId: string; episodeName?: string }) => {
   183→      setCurrent((c) => ({
   184→        ...c,
   185→        url: "",
   186→        season: next.season,
   187→        episode: next.episode,
   188→        videoId: next.videoId,
   189→        episodeName: next.episodeName,
   190→        streamTitle: undefined,
   191→        streamDescription: undefined,
   192→        resumeAt: 0,
   193→      }));
   194→      setPhase("resolving");
   195→      setError(null);
   196→    },
   197→    [],
   198→  );
   199→
   200→  return (
   201→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
   202→      {phase === "resolving" && (
   203→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
   204→          <div className="relative">
   205→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
   206→          </div>
   207→          <p className="text-sm text-ink-muted" aria-live="polite">{resolveStatus}</p>
   208→        </div>
   209→      )}
   210→      {phase === "error" ? (
   211→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   212→          <AlertTriangle className="w-12 h-12 text-danger" />
   213→          <p className="text-sm text-ink-muted max-w-md">{error}</p>
   214→          <div className="flex gap-3">
   215→            <button
   216→              type="button"
   217→              onClick={() => setSwitcherOpen(true)}
   218→              className="rounded-xl bg-accent px-5 py-2.5 text-sm font-bold text-black hover:brightness-110"
   219→            >
   220→              Open stream picker
   221→            </button>
   222→            <button
   223→              type="button"
   224→              onClick={pop}
   225→              className="rounded-xl bg-raised px-5 py-2.5 text-sm font-semibold text-ink hover:bg-raised/70"
   226→            >
   227→              Back
   228→            </button>
   229→          </div>
   230→        </div>
   231→      ) : (
   232→        current.url && (
   233→          <VideoStage
   234→            key={current.url}
   235→            payload={current}
   236→            settings={settings}
   237→            onPhase={handlePhase}
   238→            onError={handleError}
   239→            onPlayNext={handlePlayNext}
   240→            onOpenSwitcher={openSwitcher}
   241→          />
   242→        )
   243→      )}
   244→
   245→      {/* Top bar */}
   246→      {phase !== "error" && (
   247→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   248→          <button
   249→            type="button"
   250→            onClick={() => {
   251→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   252→              pop();
   253→            }}
   254→            className="w-9 h-9 rounded-full bg-black/50 flex items-center justify-center text-ink hover:bg-black/70"
   255→            aria-label="Close player (Esc)"
   256→          >
   257→            <X className="w-5 h-5" />
   258→          </button>
   259→          <div className="min-w-0">
   260→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   261→            {current.episode && (
   262→              <p className="text-xs text-ink-muted">
   263→                S{current.season}:E{current.episode}
   264→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   265→              </p>
   266→            )}
   267→          </div>
   268→          <button
   269→            type="button"
   270→            onClick={() => setSwitcherOpen(true)}
   271→            className="ml-auto flex items-center gap-1.5 rounded-full bg-black/50 hover:bg-black/70 px-3 py-1.5 text-xs font-medium text-ink"
   272→            title="Switch stream (W)"
   273→          >
   274→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   275→          </button>
   276→        </div>
   277→      )}
   278→
   279→      {switcherOpen && (
   280→        <PickerOverlay
   281→          type={current.type}
   282→          id={current.deepLink?.id ?? current.metaId}
   283→          videoId={current.videoId ?? current.deepLink?.videoId}
   284→          season={current.season}
   285→          episode={current.episode}
   286→        />
   287→      )}
   288→    </div>
   289→  );
   290→}

   292→// ---------------- Video stage ----------------
   293→function VideoStage({
   294→  payload,
   295→  settings,
   296→  onPhase,
   297→  onError,
   298→  onPlayNext,
   299→  onOpenSwitcher,
   300→}: {
   301→  payload: PlayerPayload;
   302→  settings: ReturnType<typeof useSettings.getState>["settings"];
   303→  onPhase: (p: "loading" | "playing") => void;
   304→  onError: (msg: string) => void;
   305→  onPlayNext: (next: { season: number; episode: number; videoId: string; episodeName?: string }) => void;
   306→  onOpenSwitcher: () => void;
   307→}) {
   308→  const videoRef = useRef<HTMLVideoElement>(null);
   309→  const containerRef = useRef<HTMLDivElement>(null);
   310→  const hlsRef = useRef<Hls | null>(null);
   311→  const [ready, setReady] = useState(false);
   312→  const [playing, setPlaying] = useState(false);
   313→  const [muted, setMuted] = useState(false);
   314→  const [volume, setVolume] = useState(1);
   315→  const [time, setTime] = useState(0);
   316→  const [duration, setDuration] = useState(0);
   317→  const [buffering, setBuffering] = useState(true);
   318→  const [fullscreen, setFullscreen] = useState(false);
   319→  const [speed, setSpeed] = useState(1);
   320→  const [controlsVisible, setControlsVisible] = useState(true);
   321→  const [volumeHudVisible, setVolumeHudVisible] = useState(false);
   322→  const volumeHudTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   323→  const [subMenuOpen, setSubMenuOpen] = useState(false);
   324→  const [settingsMenuOpen, setSettingsMenuOpen] = useState(false);
   325→  // P2P engine: live swarm stats + one-shot remux fallback when the native stream can't demux
   326→  const [p2pStats, setP2pStats] = useState<{ progress: number; peers: number; downloadSpeed: number; ready: boolean } | null>(null);
   327→  const [srcOverride, setSrcOverride] = useState<string | null>(null);
   328→  const remuxTriedRef = useRef(false);
   329→  const [subtitles, setSubtitles] = useState<LoadedSubtitle[]>([]);
   330→  // Full ranked subtitle pool + how many are loaded — powers the "Load more" affordance
   331→  const subPoolRef = useRef<RawSubtitle[]>([]); // ranked by preferred languages
   332→  const subEffectiveRef = useRef<RawSubtitle[]>([]); // currently displayed ranking (lang filter)
   333→  const subAttemptedRef = useRef(0);
   334→  const [subPoolSize, setSubPoolSize] = useState(0);
   335→  const [subLoadingMore, setSubLoadingMore] = useState(false);
   336→  const [subLangFilter, setSubLangFilter] = useState<string | null>(null);
   337→  // Free-text search over the ranked pool (label / language / provider) — composes
   338→  // with the language chips: chip re-ranks, search then narrows.
   339→  const [subSearch, setSubSearch] = useState(""); // committed (drives the pool view)
   340→  const [subSearchInput, setSubSearchInput] = useState(""); // typing value
   341→  const subViewTokenRef = useRef(0); // invalidates stale parse batches when the view changes
   342→  const subSearchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   343→  // Language chips derived from the pool: chip code + available count
   344→  const subLangChips = useMemo(() => {
   345→    const counts = new Map<string, number>();
   346→    for (const s of subPoolRef.current) {
   347→      const chip = langChip(s.lang);
   348→      counts.set(chip, (counts.get(chip) ?? 0) + 1);
   349→    }
   350→    return Array.from(counts.entries())
   351→      .sort((a, b) => b[1] - a[1])
   352→      .slice(0, 6);
   353→  }, [subPoolSize, subLangFilter]);
   354→  // How many entries the CURRENT view (chip re-rank + search filter) contains
   355→  const [subEffectiveCount, setSubEffectiveCount] = useState(0);
   356→  const subtitlesRef = useRef<LoadedSubtitle[]>([]);
   357→  subtitlesRef.current = subtitles; // latest committed list for stable index math
   358→  const [activeSub, setActiveSub] = useState<number>(-1); // -1 = off
   359→  const [subCue, setSubCue] = useState<SubCue | null>(null);
   360→  const [availableTracks, setAvailableTracks] = useState<{ label: string }[]>([]);
   361→  const [levels, setLevels] = useState<{ index: number; label: string }[]>([]);
   362→  const [levelSel, setLevelSel] = useState<number>(-1); // -1 = adaptive (auto)
   363→  const [autoLevel, setAutoLevel] = useState<number>(-1); // level currently playing in auto mode
   364→  const [activeAudio, setActiveAudio] = useState(0);
   365→  const [bufferedEnd, setBufferedEnd] = useState(0);
   366→  const [nextEp, setNextEp] = useState<{
   367→    season: number;
   368→    episode: number;
   369→    videoId: string;
   370→    episodeName?: string;
   371→  } | null>(null);
   372→  const [upNextDismissed, setUpNextDismissed] = useState(false);
   373→  const controlsTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   374→  const seekHandled = useRef(false);
   375→  // Trakt scrobble state (dedupe: one scrobble per action per 10s)
   376→  const scrobbleStartSent = useRef(false);
   377→  const scrobbleLastSent = useRef<Record<"start" | "pause" | "stop", number>>({
   378→    start: 0,
   379→    pause: 0,
   380→    stop: 0,
   381→  });
   382→
   383→  const resumeAt = useRef(
   384→    payload.resumeAt ??
   385→      (settings.resumePlayback && payload.metaId
   386→        ? resumeMsFor(payload.metaId, payload.videoId) / 1000
   387→        : 0),
   388→  );
   389→
   390→  // ---- P2P swarm stats polling ----
   391→  useEffect(() => {
   392→    if (!payload.p2p) {
   393→      setP2pStats(null);
   394→      return;
   395→    }
   396→    let alive = true;
   397→    const key = payload.p2p.key;
   398→    const tick = async () => {
   399→      try {
   400→        const st = await p2pStatus(key);
   401→        if (!alive) return;
   402→        setP2pStats({
   403→          progress: st.progress ?? 0,
   404→          peers: st.peers ?? 0,
   405→          downloadSpeed: st.downloadSpeed ?? 0,
   406→          ready: st.ready === true,
   407→        });
   408→      } catch {
   409→        if (alive) setP2pStats(null);
   410→      }
   411→    };
   412→    void tick();
   413→    const iv = setInterval(tick, 2500);
   414→    return () => {
   415→      alive = false;
   416→      clearInterval(iv);
   417→    };
   418→  }, [payload.p2p?.key]);
   419→
   420→  // ---- Attach source (HLS or native) ----
   421→  useEffect(() => {
   422→    const video = videoRef.current;
   423→    const url = srcOverride ?? payload.url;
   424→    if (!video || !url) return;
   425→    let cancelled = false;
   426→    const isHls = /\.m3u8(\?|$)/i.test(url) || url.includes("m3u8");
   427→
   428→    const tryPlay = () => {
   429→      video.play().catch(() => {
   430→        /* autoplay blocked; user gesture needed */
   431→      });
   432→    };
   433→
   434→    if (isHls && !video.canPlayType("application/vnd.apple.mpegurl")) {
   435→      const hls = new Hls({ enableWorker: true, maxBufferLength: 30 });
   436→      hlsRef.current = hls;
   437→      hls.loadSource(url);
   438→      hls.attachMedia(video);
   439→      hls.on(Hls.Events.MANIFEST_PARSED, (_evt, data) => {
   440→        if (cancelled) return;
   441→        // Quality levels (video renditions) — dedupe by height, sort hi-fi first
   442→        const seen = new Set<number>();
   443→        const lvls: { index: number; label: string; h: number }[] = [];
   444→        (data?.levels ?? []).forEach((l, i) => {
   445→          const h = typeof l.height === "number" && l.height > 0 ? l.height : 0;
   446→          const label = h > 0 ? `${h}p` : `${Math.round(l.bitrate / 1000)} kbps`;
   447→          const key = h > 0 ? h : Math.round(l.bitrate / 1000);
   448→          if (seen.has(key)) return;
   449→          seen.add(key);
   450→          lvls.push({ index: i, label, h });
   451→        });
   452→        lvls.sort((a, b) => b.h - a.h);
   453→        setLevels(lvls.map(({ index, label }) => ({ index, label })));
   454→        // Alternative audio tracks served by the HLS manifest
   455→        const tracks = hls.audioTracks ?? [];
   456→        if (tracks.length > 1) {
   457→          setAvailableTracks(tracks.map((t, i) => ({ label: t.name || t.lang || `Audio ${i + 1}` })));
   458→        }
   459→        onPhase("loading");
   460→        tryPlay();
   461→      });
   462→      hls.on(Hls.Events.LEVEL_SWITCHED, (_evt, d) => {
   463→        if (!cancelled) setAutoLevel(d.level);
   464→      });
   465→      hls.on(Hls.Events.ERROR, (_evt, data) => {
   466→        if (cancelled || !data.fatal) return;
   467→        if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   468→          hls.startLoad();
   469→        } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   470→          hls.recoverMediaError();
   471→        } else {
   472→          onError("Stream failed to load. It may be offline or blocked by CORS.");
   473→        }
   474→      });
   475→    } else {
   476→      video.src = url;
   477→      onPhase("loading");
   478→      tryPlay();
   479→    }
   480→
   481→    return () => {
   482→      cancelled = true;
   483→      if (hlsRef.current) {
   484→        hlsRef.current.destroy();
   485→        hlsRef.current = null;
   486→      }
   487→    };
   488→  }, [payload.url, srcOverride, onPhase, onError]);
   489→
   490→  // ---- Resume ----
   491→  useEffect(() => {
   492→    const video = videoRef.current;
   493→    if (!video) return;

  1613→function fmtTime(t: number): string {
  1614→  if (!Number.isFinite(t) || t < 0) return "0:00";
  1615→  const h = Math.floor(t / 3600);
  1616→  const m = Math.floor((t % 3600) / 60);
  1617→  const s = Math.floor(t % 60);
  1618→  return h > 0
  1619→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
  1620→    : `${m}:${String(s).padStart(2, "0")}`;
  1621→}
