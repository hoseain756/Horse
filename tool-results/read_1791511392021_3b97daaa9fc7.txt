     3→// Harbor Web — Detail view (Task 30 premium redesign)
     4→// Visual language: Apple TV+ / Disney+ / Netflix-style title page.
     5→//   1. Hero artwork CARD at the top with the official title logo embedded in
     6→//      its bottom scrim — zero floating chips/badges/buttons on the artwork.
     7→//   2. Single-line, non-wrapping metadata row directly below the card:
     8→//      [Rating badge] • [Year] • [Duration] • [Genre tag] (+ cert / source).
     9→//   3. Synopsis right beneath the metadata row (muted, generous line-height).
    10→//   4. Centered action row: primary "Watch Now" CTA + compact expandable
    11→//      menu (watchlist / streams / share) + circular custom-lists button.
    12→//   5. Immersive chrome: the glass dock + floating search are hidden for this
    13→//      page (app-shell immersiveDetail); a semi-transparent back button floats
    14→//      at the top (owned here so deep-linked roots get one too).
    15→// Business logic (meta loading, resume, watchlist, picker push) is unchanged
    16→// from the Harbor port — this is a presentation refactor.

    18→import { AnimatePresence, motion } from "framer-motion";

    40→  return (
    41→    <span aria-hidden className="shrink-0 select-none text-[11px] leading-none text-ink-subtle/60">

    47→/** Semi-transparent top back button (owned by the detail page so it exists
    48→ *  even when a title is deep-linked as the root nav frame). RTL-mirrors. */
    49→function DetailBackButton() {
    50→  const tr = useT();
    51→  const pop = useNav((s) => s.pop);
    52→  const resetTo = useNav((s) => s.resetTo);
    53→  const depth = useNav((s) => s.stack.length);
    54→  const goBack = useCallback(() => {
    55→    if (depth <= 1) resetTo({ kind: "view", view: "home" });
    56→    else pop();
    57→  }, [depth, pop, resetTo]);
    58→  return (
    59→    <button
    60→      type="button"
    61→      onClick={goBack}
    62→      className="harbor-tv-focus fixed top-4 start-4 z-40 flex h-10 items-center gap-2 rounded-full border border-white/15 bg-black/55 ps-3.5 pe-4 text-sm font-medium text-white backdrop-blur-md transition-all hover:scale-[1.03] hover:bg-black/75 active:scale-95 md-state"
    63→      aria-label={tr("goBack")}
    64→    >
    65→      <ArrowLeft className="h-4 w-4 rtl:rotate-180" aria-hidden />
    66→      {tr("back")}
    67→    </button>
    68→  );
    69→}
    70→
    71→export function DetailView({ type, id }: { type: string; id: string }) {
    72→  const tr = useT();
    73→  const { toast } = useToast();
    74→  const push = useNav((s) => s.push);
    75→  const addons = useAddons((s) => s.addons);
    76→  const addonsLoaded = useAddons((s) => s.loaded);
    77→  const settings = useSettings((s) => s.settings);
    78→  // TMDB enrichment (logo, certification, recommendations) — fails soft
    79→  const tmdb = useTmdbEnrichment(type, id, settings.region);
    80→
    81→  const [meta, setMeta] = useState<Meta | null>(null);
    82→  const [loading, setLoading] = useState(true);
    83→  const [error, setError] = useState<string | null>(null);
    84→  const [inList, setInList] = useState(false);
    85→  const [resume, setResume] = useState(0);
    86→  const [cwEntry, setCwEntry] = useState<ReturnType<typeof getCwEntry>>(null);
    87→
    88→  useEffect(() => {
    89→    let alive = true;
    90→    setLoading(true);
    91→    setError(null);
    92→    setMeta(null);
    93→    (async () => {
    94→      try {
    95→        let result = await fetchMeta(type, id);
    96→        // If cinemeta misses, try addon metas
    97→        if (!result && addonsLoaded) {
    98→          for (const addon of addons.filter((a) => a.enabled)) {
    99→            try {
   100→              const m = await fetchAddonMeta(addon, type, id);
   101→              if (m) {
   102→                result = m;
   103→                break;
   104→              }
   105→            } catch {
   106→              /* try next */
   107→            }
   108→          }
   109→        }
   110→        if (!alive) return;
   111→        if (!result) {
   112→          setError(tr("titleNotFound"));
   113→        } else {
   114→          setMeta(result);
   115→          setInList(isInWatchlist(id));
   116→          setResume(resumeMsFor(id));
   117→          setCwEntry(getCwEntry(id));
   118→        }
   119→      } catch (e) {
   120→        if (!alive) return;
   121→        setError(e instanceof Error ? e.message : tr("fetchFailed"));
   122→      } finally {
   123→        if (alive) setLoading(false);
   124→      }
   125→    })();
   126→    return () => {
   127→      alive = false;
   128→    };
   129→    // eslint-disable-next-line react-hooks/exhaustive-deps
   130→  }, [type, id, addons, addonsLoaded]);
   131→
   132→  // Next episode to resume (series)
   133→  const nextVideo: MetaVideo | null = useMemo(() => {
   134→    if (!meta || meta.type !== "series" || !meta.videos?.length) return null;
   135→    if (cwEntry?.season && cwEntry?.episode) {
   136→      const exact = meta.videos.find(
   137→        (v) => v.season === cwEntry.season && v.episode === cwEntry.episode,
   138→      );
   139→      if (exact && resume > 0) return exact;
   140→    }
   141→    return null;
   142→  }, [meta, cwEntry, resume]);
   143→
   144→  const onPlay = useCallback(
   145→    (video?: MetaVideo) => {
   146→      if (!meta) return;
   147→      // Duration hint for the playback timeline (approximate until the real
   148→      // container/ffprobe duration lands): TMDB runtime first (minutes), then
   149→      // the Stremio meta runtime string. Series: per-episode runtime when the
   150→      // addon provides one, else the show average / TMDB episode_run_time.
   151→      const tmdbRuntimeMin =
   152→        meta.type === "series"
   153→          ? (tmdb.details?.episode_run_time?.[0] ?? null)
   154→          : (tmdb.details?.runtime ?? null);
   155→      const runtimeSeconds =
   156→        parseRuntimeToSeconds(video?.duration ?? null) ??
   157→        (tmdbRuntimeMin != null ? Math.round(tmdbRuntimeMin * 60) : null) ??
   158→        parseRuntimeToSeconds(meta.runtime);
   159→      if (settings.instantPlay) {
   160→        push({
   161→          kind: "player",
   162→          payload: {
   163→            url: "", // resolved by picker
   164→            title: meta.name,
   165→            type: meta.type,
   166→            metaId: meta.id,
   167→            poster: meta.poster,
   168→            season: video?.season,
   169→            episode: video?.episode,
   170→            videoId: video?.id,
   171→            episodeName: video?.name ?? video?.title,
   172→            deepLink: { type: meta.type, id: meta.id, videoId: video?.id },
   173→            runtimeSeconds: runtimeSeconds ?? undefined,
   174→          },
   175→        });
   176→      } else {
   177→        push({
   178→          kind: "picker",
   179→          type: meta.type,
   180→          id: meta.id,
   181→          videoId: video?.id,
   182→          season: video?.season,
   183→          episode: video?.episode,
   184→          runtimeSeconds: runtimeSeconds ?? undefined,
   185→        });
   186→      }
   187→    },
   188→    [meta, tmdb.details, push, settings.instantPlay],
   189→  );
   190→
   191→  /* ---------- expandable secondary menu ---------- */
   192→  const [menuOpen, setMenuOpen] = useState(false);
   193→  const menuWrapRef = useRef<HTMLDivElement>(null);
   194→
   195→  // Close on outside pointer + Escape; return focus to the trigger on Esc.
   196→  useEffect(() => {
   197→    if (!menuOpen) return;
   198→    const onDown = (e: PointerEvent) => {
   199→      if (menuWrapRef.current && !menuWrapRef.current.contains(e.target as Node)) {
   200→        setMenuOpen(false);
   201→      }
   202→    };
   203→    const onKey = (e: KeyboardEvent) => {
   204→      if (e.key === "Escape") {
   205→        e.stopPropagation();
   206→        setMenuOpen(false);
   207→        menuWrapRef.current?.querySelector<HTMLButtonElement>("button")?.focus();
   208→      }
   209→    };
   210→    window.addEventListener("pointerdown", onDown);
   211→    window.addEventListener("keydown", onKey, true);
   212→    return () => {
   213→      window.removeEventListener("pointerdown", onDown);
   214→      window.removeEventListener("keydown", onKey, true);
   215→    };
   216→  }, [menuOpen]);
   217→
   218→  const openStreams = useCallback(() => {
   219→    setMenuOpen(false);
   220→    if (!meta) return;
   221→    if (meta.type === "series" && nextVideo) {
   222→      push({
   223→        kind: "picker",
   224→        type: meta.type,
   225→        id: meta.id,
   226→        videoId: nextVideo.id,
   227→        season: nextVideo.season,
   228→        episode: nextVideo.episode,
   229→      });
   230→    } else {
   231→      push({ kind: "picker", type: meta.type, id: meta.id });
   232→    }
   233→  }, [meta, nextVideo, push]);
   234→
   235→  const toggleWatchlistFromMenu = useCallback(() => {
   236→    if (!meta) return;
   237→    setInList(
   238→      toggleWatchlist({
   239→        id: meta.id,
   240→        type: meta.type,
   241→        name: meta.name,
   242→        poster: meta.poster,
   243→        releaseInfo: meta.releaseInfo,
   244→        imdbRating: meta.imdbRating,
   245→      }),
   246→    );
   247→  }, [meta]);
   248→
   249→  const onShare = useCallback(async () => {
   250→    setMenuOpen(false);
   251→    if (!meta) return;
   252→    const url = window.location.href;
   253→    try {
   254→      if (typeof navigator.share === "function") {
   255→        await navigator.share({ title: meta.name, url });
   256→        return; // native sheet finished (dismiss throws AbortError → caught below)
   257→      }
   258→    } catch (e) {
   259→      if (e instanceof DOMException && e.name === "AbortError") return;
   260→      // fall through to the clipboard fallback
   261→    }
   262→    try {
   263→      await navigator.clipboard.writeText(url);
   264→      toast({ title: tr("linkCopied") });
   265→    } catch {
   266→      toast({ title: tr("copyFailed") });
   267→    }
   268→  }, [meta, toast, tr]);
   269→
   270→  // The back button is fixed chrome of this page — render in every state.
   271→  const back = <DetailBackButton />;
   272→
   273→  if (loading) {
   274→    return (
   275→      <div className="px-4 pt-[72px] pb-16 md:px-8 md:pt-20">
   276→        {back}
   277→        {/* hero card skeleton */}
   278→        <div
   279→          className="harbor-skeleton mx-auto aspect-[16/11] max-w-5xl rounded-[28px] sm:aspect-[16/8] md:aspect-[21/8] md:rounded-[36px]"
   280→          aria-hidden
   281→        />
   282→        {/* metadata row skeleton */}
   283→        <div className="mt-5 flex justify-center" aria-hidden>
   284→          <div className="harbor-skeleton h-6 w-72 rounded-full" />
   285→        </div>
   286→        {/* synopsis skeleton */}
   287→        <div className="mx-auto mt-4 max-w-xl space-y-2" aria-hidden>
   288→          <div className="harbor-skeleton mx-auto h-4 w-full rounded-lg" />
   289→          <div className="harbor-skeleton mx-auto h-4 w-3/4 rounded-lg" />
   290→        </div>
   291→        {/* action row skeleton */}
   292→        <div className="mt-7 flex items-center justify-center gap-3" aria-hidden>
   293→          <div className="harbor-skeleton h-[52px] w-44 rounded-full" />
   294→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   295→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   296→        </div>
   297→        <div className="sr-only">{tr("loadingDetails")}</div>
   298→      </div>
   299→    );
   300→  }
   301→
   302→  if (error || !meta) {
   303→    return (
   304→      <div className="mx-auto max-w-xl px-4 pt-28 pb-16 text-center md:pt-32">
   305→        {back}
   306→        <div className="rounded-2xl border border-danger/40 bg-danger/10 px-5 py-4 text-sm text-danger">
   307→          {error ?? tr("titleNotFound")}
   308→        </div>
   309→      </div>
   310→    );
   311→  }
   312→
   313→  const watched = episodeWatchedSet();
   314→
   315→  /* ---- single-line metadata row data (requirement 2) ---- */
   316→  const imdbScore =
   317→    meta.imdbRating && parseFloat(meta.imdbRating) > 0 ? meta.imdbRating : null;
   318→  const year = meta.releaseInfo?.trim() || null;
   319→  const runtime = meta.runtime?.trim() || null;
   320→  const genre = meta.genres?.find((g) => g && g.trim().length > 0) ?? null;
   321→  const hasAnyMeta = imdbScore || year || runtime || genre;
   322→
   323→  return (
   324→    <div className="pb-16">
   325→      {back}
   326→
   327→      {/* ================= 1 · HERO ARTWORK CARD =================
   328→          The official logo lives INSIDE the card's bottom scrim. No chips,
   329→          ratings, text or buttons float over the artwork anymore. */}
   330→      <section className="px-4 pt-[72px] md:px-8 md:pt-20" aria-label={meta.name}>
   331→        <div className="harbor-pop-in relative mx-auto aspect-[16/11] max-w-5xl overflow-hidden rounded-[28px] border border-edge-soft bg-raised shadow-[0_36px_90px_-36px_rgba(0,0,0,0.9)] sm:aspect-[16/8] md:aspect-[21/8] md:rounded-[36px]">
   332→          <PosterImage
   333→            src={tmdb.backdrop ?? meta.background ?? meta.poster}
   334→            alt={meta.name}
   335→            className="absolute inset-0"
   336→            landscape
   337→          />
   338→          {/* Bottom scrim — legibility bed for the embedded logo */}
   339→          <div className="absolute inset-x-0 bottom-0 h-3/5 bg-gradient-to-t from-black/90 via-black/45 to-transparent" />
   340→          {/* Embedded official title logo (sr-only h1 keeps document outline) */}
   341→          <div className="absolute inset-x-0 bottom-0 flex items-end justify-center px-5 pb-5 md:pb-9">
   342→            {tmdb.logo || meta.logo ? (
   343→              <>
   344→                <h1 className="sr-only">{meta.name}</h1>
   345→                <img
   346→                  src={tmdb.logo ?? meta.logo}
   347→                  alt=""
   348→                  className="max-h-14 w-auto max-w-[78%] object-contain drop-shadow-[0_10px_28px_rgba(0,0,0,0.95)] transition-transform duration-500 sm:max-h-16 md:max-h-24"
   349→                  loading="lazy"
   350→                />
   351→              </>
   352→            ) : (
   353→              <h1 className="font-display text-center text-3xl font-bold tracking-tight text-white drop-shadow-[0_6px_20px_rgba(0,0,0,0.9)] md:text-5xl">
   354→                {meta.name}
   355→              </h1>
   356→            )}
   357→          </div>
   358→        </div>
   359→      </section>
   360→
   361→      {/* ============ 2 · SINGLE-LINE METADATA ROW ============
   362→          One horizontal, non-wrapping line directly below the card.
   363→          Scrolls horizontally on the narrowest screens — never wraps. */}
   364→      {hasAnyMeta && (
   365→        <div className="mx-auto mt-4 max-w-3xl">
   366→          {/* Outer scroller + inner w-max row: centers when it fits, scrolls
   367→              from the true start edge when it doesn't (no justify-center
   368→              scroll-trap). Single line — never wraps. */}
   369→          <div className="no-scrollbar overflow-x-auto">
   370→            <div className="flex w-max min-w-full flex-nowrap items-center justify-center gap-2.5 px-4 py-0.5 md:px-6">
   371→              {imdbScore && (
   372→                <span
   373→                  className="flex shrink-0 items-center gap-1.5 rounded-full border border-amber-300/25 bg-amber-400/10 px-2.5 py-1 text-[13px] font-bold text-amber-300 tabular-nums"
   374→                  title="IMDb"
   375→                >
   376→                  <Star className="h-3.5 w-3.5 fill-amber-300" aria-hidden />
   377→                  {imdbScore}
   378→                </span>
   379→              )}
   380→              {imdbScore && year && <MetaBullet />}
   381→              {year && (
   382→                <span className="shrink-0 text-[13px] font-semibold text-ink md:text-sm">
   383→                  {year}
   384→                </span>
   385→              )}
   386→              {year && runtime && <MetaBullet />}
   387→              {runtime && (
   388→                <span
   389→                  dir="ltr" /* addon runtime strings are latin ("58 min", "1h 52m") — isolate so RTL can't reorder to "min 58" */
   390→                  className="shrink-0 text-[13px] font-medium text-ink-muted tabular-nums md:text-sm"
   391→                >
   392→                  {runtime}
   393→                </span>
   394→              )}
   395→              {runtime && genre && <MetaBullet />}
   396→              {genre && (
   397→                <span className="shrink-0 rounded-full border border-edge-soft bg-raised/70 px-2.5 py-1 text-[12px] font-medium text-ink-muted md:text-[13px]">
   398→                  {genre}
   399→                </span>
   400→              )}
   401→              {tmdb.certification && (
   402→                <>
   403→                  <MetaBullet />
   404→                  <span
   405→                    className="shrink-0 rounded-[5px] border border-ink-subtle/40 px-1.5 py-px text-[11px] font-black tracking-wide text-ink-muted"
   406→                    title={tr("contentRating")}
   407→                  >
   408→                    {tmdb.certification}
   409→                  </span>
   410→                </>
   411→              )}
   412→              {meta.addonOrigin && (
   413→                <>
   414→                  <MetaBullet />
   415→                  <span className="shrink-0 text-[12px] text-ink-subtle md:text-[13px]">
   416→                    {/* tr() without vars returns the raw template (keeps the
   417→                        {name} placeholder) so RichVia can bidi-isolate the name */}
   418→                    <RichVia name={meta.addonOrigin.name} template={tr("viaAddon")} />
   419→                  </span>
   420→                </>
   421→              )}
   422→            </div>
   423→          </div>
   424→          {/* Provider rating chips (deep links) — subtle, centered; hidden
   425→              entirely when ratings are off / no provider data. */}
   426→          <div className="mt-2 flex justify-center px-4 md:px-6">
   427→            <RatingsRow
   428→              type={meta.type}
   429→              imdbId={/^tt\d+$/.test(meta.id) ? meta.id : undefined}
   430→              title={meta.name}
   431→              year={meta.releaseInfo}
   432→              genres={meta.genres}
   433→              country={meta.country}
   434→            />
   435→          </div>
   436→        </div>
   437→      )}
   438→
   439→      {/* ================== 3 · SYNOPSIS ================== */}
   440→      {meta.description && (
   441→        <p className="mx-auto mt-4 max-w-2xl px-4 text-center text-sm leading-[1.8] text-ink-muted md:px-6 md:text-[15px]">
   442→          {meta.description}
   443→        </p>
   444→      )}
   445→
   446→      {/* ========== 4 · CENTERED ACTION CONTROLS + EXPANDABLE MENU ========== */}
   447→      <div className="mx-auto mt-6 flex items-center justify-center gap-2.5 px-4 md:gap-3">
   448→        {/* Primary CTA */}
   449→        {meta.type === "series" && nextVideo && resume > 0 ? (
   450→          <button
   451→            type="button"
   452→            onClick={() => onPlay(nextVideo)}
   453→            className="md-btn md-btn-filled md-state harbor-tv-focus !h-[52px] !rounded-full !px-5 !text-[15px] shadow-[0_14px_36px_-14px_var(--md-sys-color-primary)] transition-transform hover:scale-[1.03] active:scale-[0.97] sm:!px-7 md:!px-9"
   454→          >
   455→            <RotateCcw className="md-btn-icon" aria-hidden />
   456→            {tr("resumeSE", { s: nextVideo.season ?? 1, e: nextVideo.episode ?? 1 })}
   457→          </button>
   458→        ) : (
   459→          <button
   460→            type="button"
   461→            onClick={() => onPlay(nextVideo ?? undefined)}
   462→            className="md-btn md-btn-filled md-state harbor-tv-focus !h-[52px] !rounded-full !px-6 !text-[15px] shadow-[0_14px_36px_-14px_var(--md-sys-color-primary)] transition-transform hover:scale-[1.03] active:scale-[0.97] sm:!px-8 md:!px-10"
   463→          >
   464→            <Play className="md-btn-icon fill-current" aria-hidden />
   465→            {tr("watchNow")}
   466→          </button>
   467→        )}
   468→
   469→        {/* Secondary expandable menu (spinner/dropdown button) */}
   470→        <div ref={menuWrapRef} className="relative">
   471→          <button
   472→            type="button"
   473→            onClick={() => setMenuOpen((v) => !v)}
   474→            aria-haspopup="menu"
   475→            aria-expanded={menuOpen}
   476→            aria-label={tr("moreOptions")}
   477→            className="harbor-tv-focus flex h-[52px] w-[52px] items-center justify-center rounded-full border border-edge-soft bg-raised/80 text-ink backdrop-blur-md transition-all hover:scale-[1.04] hover:bg-raised active:scale-95 md-state"
   478→          >
   479→            <motion.span
   480→              animate={{ rotate: menuOpen ? 180 : 0 }}
   481→              transition={{ duration: 0.22, ease: [0.2, 0, 0, 1] }}
   482→              className="flex"
   483→            >
   484→              <ChevronDown className="h-5 w-5" aria-hidden />
   485→            </motion.span>
   486→          </button>
   487→
   488→          <AnimatePresence>
   489→            {menuOpen && (
   490→              <motion.div
   491→                role="menu"
   492→                aria-label={tr("moreOptions")}
   493→                initial={{ opacity: 0, x: "-50%", y: -8, scale: 0.95 }}
   494→                animate={{ opacity: 1, x: "-50%", y: 0, scale: 1 }}

   615→// ---------- Episode list ----------
   616→function EpisodeList({
   617→  meta,
   618→  watched,
   619→  onPlay,
   620→}: {
   621→  meta: Meta;
   622→  watched: Set<string>;
   623→  onPlay: (video: MetaVideo) => void;
   624→}) {
   625→  const tr = useT();
   626→  const videos = useMemo(() => {
   627→    const vids = [...(meta.videos ?? [])];
   628→    // group by season
   629→    return vids;
   630→  }, [meta.videos]);
   631→
   632→  const seasons = useMemo(() => {
   633→    const set = new Set<number>();
   634→    for (const v of videos) {
   635→      if (typeof v.season === "number") set.add(v.season);
   636→    }
   637→    return Array.from(set).sort((a, b) => a - b);
   638→  }, [videos]);
   639→
   640→  const [season, setSeason] = useState<number | null>(null);
   641→  const [sortDesc, setSortDesc] = useState(false);
   642→
   643→  // Default to latest season when none selected (render-time derivation, no effect)
   644→  const effectiveSeason = season ?? (seasons.length > 0 ? seasons[seasons.length - 1] : null);
   645→
   646→  const seasonVideos = useMemo(() => {
   647→    const list = videos.filter((v) => v.season === effectiveSeason);
   648→    return sortDesc ? [...list].reverse() : list;
   649→  }, [videos, effectiveSeason, sortDesc]);
   650→
   651→  if (videos.length === 0) {
   652→    return (
   653→      <div className="mx-auto mt-10 max-w-3xl px-4 text-sm text-ink-subtle md:px-8">
   654→        {tr("noEpisodes")}
   655→      </div>
   656→    );
   657→  }
   658→
   659→  return (
   660→    <div className="mx-auto mt-12 max-w-3xl px-4 md:px-8">
   661→      <div className="mb-3 flex flex-wrap items-center justify-between gap-3">
   662→        <h2 className="md-title-medium text-ink">{tr("episodesTitle")}</h2>
   663→        <button
   664→          type="button"
   665→          onClick={() => setSortDesc((v) => !v)}
   666→          className="md-chip md-state harbor-tv-focus"
   667→        >
   668→          {sortDesc ? tr("newestFirst") : tr("oldestFirst")}
   669→        </button>
   670→      </div>
   671→
   672→      {/* Season selector chips (horizontally scrollable for long-running shows) */}
   673→      {seasons.length > 1 && (
   674→        <div
   675→          className="no-scrollbar mb-3 flex gap-1.5 overflow-x-auto pb-2"
   676→          role="group"
   677→          aria-label={tr("selectSeason")}
   678→        >
   679→          {seasons.map((s) => {
   680→            const active = s === effectiveSeason;
   681→            return (
   682→              <button
   683→                key={s}
   684→                type="button"
   685→                onClick={() => setSeason(s)}
   686→                aria-pressed={active}
   687→                className={cn(
   688→                  "md-chip md-state harbor-tv-focus shrink-0",
   689→                  active && "md-chip-selected",
   690→                )}
   691→              >
   692→                {tr("seasonN", { n: s })}
   693→              </button>
   694→            );
   695→          })}
   696→        </div>
   697→      )}
   698→
   699→      <div className="harbor-scroll max-h-[60vh] space-y-2.5 overflow-y-auto pr-1">
   700→        {seasonVideos.map((v) => {
   701→          const isWatched = watched.has(v.id) || watched.has(`${meta.id}:${v.season}:${v.episode}`);
   702→          return (
   703→            <button
   704→              key={v.id}
   705→              type="button"
   706→              onClick={() => onPlay(v)}
   707→              className={cn(
   708→                "md-state group/ep harbor-tv-focus relative flex w-full items-center gap-4 rounded-[var(--md-sys-shape-corner-medium)] border border-edge-soft p-3 text-left",
   709→                isWatched ? "bg-elevated/40 opacity-60" : "bg-elevated",
   710→              )}
   711→            >
   712→              {/* Accent start bar on hover */}
   713→              <span
   714→                aria-hidden
   715→                className="absolute start-0 top-3 bottom-3 w-[3px] rounded-full bg-accent opacity-0 transition-opacity group-hover/ep:opacity-100"
   716→              />
   717→              <div className="relative h-16 w-28 shrink-0 overflow-hidden rounded-xl bg-raised md:h-20 md:w-36">
   718→                <PosterImage src={v.thumb} alt={v.name ?? v.title ?? ""} className="absolute inset-0" landscape />
   719→                {isWatched && (
   720→                  <span className="absolute top-1 end-1 flex h-5 w-5 items-center justify-center rounded-full border border-accent/50 bg-accent-soft backdrop-blur-sm">
   721→                    <Check className="h-3 w-3 text-accent" />
   722→                  </span>
   723→                )}
   724→              </div>
   725→              <div className="min-w-0 flex-1">
   726→                {/* dir=auto: latin titles render "E1 Title"; Arabic titles render
   727→                    RTL with the latin E-prefix bidi-isolated (bdi) */}
   728→                <p dir="auto" className="harbor-clamp-1 md-title-small text-ink">
   729→                  <bdi className="me-2 text-ink-subtle">E{v.episode}</bdi>
   730→                  {v.name ?? v.title ?? tr("episodeN", { n: v.episode ?? 0 })}
   731→                </p>
   732→                {v.released && <p className="md-body-small mt-0.5 text-ink-subtle">{new Date(v.released).toLocaleDateString()}</p>}
   733→                {(v.overview ?? v.description) && (
   734→                  <p className="md-body-small harbor-clamp-2 mt-1 leading-relaxed text-ink-muted">
   735→                    {v.overview ?? v.description}
   736→                  </p>
   737→                )}
   738→              </div>
   739→              <Play className="h-4 w-4 shrink-0 text-accent transition-transform group-hover/ep:scale-110" aria-hidden />
   740→            </button>
   741→          );
   742→        })}
   743→      </div>
   744→    </div>
   745→  );
   746→}

   773→// ---------- More like this rail (genre-based, Cinemeta top catalog) ----------
   774→
   775→function SimilarRail({
   776→  type,
   777→  genres,
   778→  currentId,
   779→}: {
   780→  type: string;
   781→  genres?: string[];
   782→  currentId: string;
   783→}) {
   784→  const tr = useT();
   785→  const push = useNav((s) => s.push);
   786→  const genre = genres?.find((g) => g && g.trim().length > 0);
   787→  // Data is keyed by the genre it was fetched for, so a genre change renders a
   788→  // skeleton without a synchronous setState inside the effect.
   789→  const [data, setData] = useState<{ genre: string; items: Meta[] } | null>(null);
   790→
   791→  useEffect(() => {
   792→    if (!genre) return;
   793→    let alive = true;
   794→    (async () => {
   795→      try {
   796→        const res = await fetchCinemetaCatalog(
   797→          type === "series" ? "series" : "movie",
   798→          "top",
   799→          { genre },
   800→        );
   801→        if (!alive) return;
   802→        setData({ genre, items: res.filter((m) => m.id !== currentId).slice(0, 14) });
   803→      } catch {
   804→        if (alive) setData({ genre, items: [] });
   805→      }
   806→    })();
   807→    return () => {
   808→      alive = false;
   809→    };
   810→  }, [type, genre, currentId]);
   811→
   812→  if (!genre) return null;
   813→  const items: Meta[] | null = data && data.genre === genre ? data.items : null;
   814→
   815→  return (
   816→    <section className="mt-12" aria-label={tr("moreLikeThis")}>
   817→      <div className="mb-3 flex items-center gap-2 px-4 md:px-8">
   818→        <Layers className="h-5 w-5 text-accent" aria-hidden />
   819→        <h2 className="md-title-medium text-ink">{tr("moreLikeThis")}</h2>
   820→        <span className="md-chip md-label-small !h-6 !px-2 cursor-default!">{genre}</span>
   821→      </div>
   822→      {items === null ? (
   823→        <div className="flex gap-3 overflow-hidden px-4 md:px-8" aria-hidden>
   824→          {Array.from({ length: 8 }).map((_, i) => (
   825→            <div key={i} className="harbor-skeleton aspect-[2/3] w-[112px] shrink-0 rounded-xl md:w-[136px]" />
   826→          ))}
   827→        </div>
   828→      ) : items.length === 0 ? null : (
   829→        /* Hover headroom: same pt-3/-mt-3 contract as TmdbRecsRail above */
   830→        <div className="harbor-scroll-x overflow-x-auto flex gap-3 px-4 pb-2 pt-3 -mt-3 md:px-8">
   831→          {items.map((m) => (
   832→            <div key={`${m.id}`} className="w-[104px] shrink-0 md:w-[126px]">
   833→              <MetaCard
   834→                meta={m}
   835→                onOpen={() => push({ kind: "detail", type: m.type || type, id: m.id })}
   836→              />
   837→            </div>
   838→          ))}
   839→        </div>
   840→      )}
   841→    </section>
   842→  );
   843→}
