    52→    52→function dayKey(t: number): string {
    53→    53→  const d = new Date(t);
    54→    54→  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`;
    55→    55→}

    57→    57→function computeStats(history: HistoryEntry[]): Stats {
    58→    58→  const totalMs = (e: HistoryEntry) => Math.max(0, Math.min(e.positionMs || e.durationMs, e.durationMs || e.positionMs));
    59→    59→  let watchMs = 0;
    60→    60→  let movies = 0;
    61→    61→  let episodes = 0;
    62→    62→  const byDay = new Map<string, number>();
    63→    63→  const byTitle = new Map<string, { name: string; ms: number; poster?: string; type: string; count: number }>();
    64→    64→  const days = new Set<string>();
    65→    65→
    66→    66→  for (const e of history) {
    67→    67→    const ms = totalMs(e);
    68→    68→    watchMs += ms;
    69→    69→    const dk = dayKey(e.t);
    70→    70→    days.add(dk);
    71→    71→    byDay.set(dk, (byDay.get(dk) ?? 0) + ms);
    72→    72→    if (e.type === "series" && e.episode) episodes++;
    73→    73→    else movies++;
    74→    74→    const prev = byTitle.get(e.id);
    75→    75→    byTitle.set(e.id, {
    76→    76→      name: e.name,
    77→    77→      ms: (prev?.ms ?? 0) + ms,
    78→    78→      poster: prev?.poster ?? e.poster,
    79→    79→      type: e.type,
    80→    80→      count: (prev?.count ?? 0) + 1,
    81→    81→    });
    82→    82→  }
    83→    83→
    84→    84→  // last 14 days series
    85→    85→  const daily: { date: string; ms: number }[] = [];
    86→    86→  const now = new Date();
    87→    87→  for (let i = 13; i >= 0; i--) {
    88→    88→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    89→    89→    const key = dayKey(d.getTime());
    90→    90→    daily.push({ date: key, ms: byDay.get(key) ?? 0 });
    91→    91→  }
    92→    92→
    93→    93→  // last 365 days heatmap series (day 364 days ago -> today, oldest first)
    94→    94→  const heat365: { date: string; ms: number }[] = [];
    95→    95→  for (let i = 364; i >= 0; i--) {
    96→    96→    const d = new Date(now.getFullYear(), now.getMonth(), now.getDate() - i);
    97→    97→    const key = dayKey(d.getTime());
    98→    98→    heat365.push({ date: key, ms: byDay.get(key) ?? 0 });
    99→    99→  }
   100→   100→
   101→   101→  // Badge stats — computed honestly from the same history
   102→   102→  let maxEpisodesPerDay = 0;
   103→   103→  let lateNightSessions = 0;
   104→   104→  const episodesPerDay = new Map<string, number>(); // `${dayKey}|${id}` -> same-series episode count
   105→   105→  for (const e of history) {
   106→   106→    if (new Date(e.t).getHours() < 5) lateNightSessions++;
   107→   107→    if (e.type === "series" && e.episode) {
   108→   108→      const k = `${dayKey(e.t)}|${e.id}`;
   109→   109→      const n = (episodesPerDay.get(k) ?? 0) + 1;
   110→   110→      episodesPerDay.set(k, n);
   111→   111→      if (n > maxEpisodesPerDay) maxEpisodesPerDay = n;
   112→   112→    }
   113→   113→  }
   114→   114→  let maxSessions = 0;
   115→   115→  for (const v of byTitle.values()) {
   116→   116→    if (v.count > maxSessions) maxSessions = v.count;
   117→   117→  }
   118→   118→
   119→   119→  let bestDay: { date: string; ms: number } | null = null;
   120→   120→  for (const [date, ms] of byDay) {
   121→   121→    if (!bestDay || ms > bestDay.ms) bestDay = { date, ms };
   122→   122→  }
   123→   123→
   124→   124→  const topTitles = Array.from(byTitle.entries())
   125→   125→    .map(([id, v]) => ({ id, ...v }))
   126→   126→    .sort((a, b) => b.ms - a.ms)
   127→   127→    .slice(0, 6);
   128→   128→
   129→   129→  return {
   130→   130→    totalEntries: history.length,
   131→   131→    uniqueTitles: byTitle.size,
   132→   132→    movies,
   133→   133→    episodes,
   134→   134→    watchMs,
   135→   135→    activeDays: days.size,
   136→   136→    bestDay,
   137→   137→    daily,
   138→   138→    heat365,
   139→   139→    maxEpisodesPerDay,
   140→   140→    lateNightSessions,
   141→   141→    maxSessions,
   142→   142→    topTitles,
   143→   143→    topGenres: [], // filled async
   144→   144→  };
   145→   145→}

   147→   147→function fmtDuration(ms: number): string {
   148→   148→  const hours = Math.floor(ms / 3_600_000);
   149→   149→  const mins = Math.floor((ms % 3_600_000) / 60_000);
   150→   150→  const secs = Math.floor((ms % 60_000) / 1000);
   151→   151→  if (hours >= 1) return `${hours}h ${mins}m`;
   152→   152→  if (mins >= 1) return `${mins}m`;
   153→   153→  return `${secs}s`;
   154→   154→}

   156→   156→/**

   158→   158→ * color string usable in inline color-mix() styles. Falls back to a pleasant
   159→   159→ * warm amber if the variable can't be read.

   161→   161→function resolvedAccent(): string {

   163→   163→    const v = getComputedStyle(document.documentElement).getPropertyValue("--color-accent-var").trim();
   164→   164→    return v || "#f0b429";
   165→   165→  } catch {
   166→   166→    return "#f0b429";
   167→   167→  }

   170→   170→// ---------- 365-day heatmap (GitHub style) ----------
   171→   171→
   172→   172→const WEEKDAY_ROWS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"]; // row 0 = Monday
   173→   173→const MONTHS_SHORT = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   174→   174→
   175→   175→type HeatCell = { key: string; label: string; level: number; future: boolean };
   176→   176→
   177→   177→function heatLevel(ms: number, max: number): number {
   178→   178→  if (ms <= 0) return 0;
   179→   179→  if (max <= 0) return 4;
   180→   180→  if (ms <= max * 0.25) return 1;
   181→   181→  if (ms <= max * 0.5) return 2;
   182→   182→  if (ms <= max * 0.75) return 3;
   183→   183→  return 4;
   184→   184→}
   185→   185→
   186→   186→/** 0 = raised/empty (class-driven); 1-4 = accent tints via color-mix. */
   187→   187→function levelBg(level: number, accent: string): string | undefined {
   188→   188→  switch (level) {
   189→   189→    case 1:
   190→   190→      return `color-mix(in srgb, ${accent} 25%, transparent)`;
   191→   191→    case 2:
   192→   192→      return `color-mix(in srgb, ${accent} 45%, transparent)`;
   193→   193→    case 3:
   194→   194→      return `color-mix(in srgb, ${accent} 70%, transparent)`;
   195→   195→    case 4:
   196→   196→      return accent;
   197→   197→    default:
   198→   198→      return undefined;
   199→   199→  }
   200→   200→}

   202→   202→function HeatmapGrid({ heat, accent }: { heat: { date: string; ms: number }[]; accent: string }) {
   203→   203→  const { columns, monthLabels } = useMemo(() => {
   204→   204→    const now = new Date();
   205→   205→    const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   206→   206→    const start = new Date(today.getFullYear(), today.getMonth(), today.getDate() - 364);
   207→   207→    const dow = (start.getDay() + 6) % 7; // 0 = Monday
   208→   208→    const gridStart = new Date(start.getFullYear(), start.getMonth(), start.getDate() - dow);
   209→   209→    const byDate = new Map(heat.map((h) => [h.date, h.ms]));
   210→   210→    let max = 0;
   211→   211→    for (const h of heat) if (h.ms > max) max = h.ms;
   212→   212→
   213→   213→    const cols: HeatCell[][] = [];
   214→   214→    const monthLabels: (string | null)[] = [];
   215→   215→    let prevMonth = -1;
   216→   216→    const cursor = new Date(gridStart);
   217→   217→    while (cursor <= today) {
   218→   218→      const col: HeatCell[] = [];
   219→   219→      let firstInRangeMonth = -1;
   220→   220→      for (let r = 0; r < 7; r++) {
   221→   221→        if (cursor > today) {
   222→   222→          col.push({ key: `f${cols.length}-${r}`, label: "", level: 0, future: true });
   223→   223→        } else {
   224→   224→          if (cursor.getTime() >= start.getTime() && firstInRangeMonth === -1) {
   225→   225→            firstInRangeMonth = cursor.getMonth();
   226→   226→          }
   227→   227→          const key = dayKey(cursor.getTime());
   228→   228→          const ms = byDate.get(key) ?? 0;
   229→   229→          col.push({
   230→   230→            key,
   231→   231→            label: `${MONTHS_SHORT[cursor.getMonth()]} ${cursor.getDate()} · ${fmtDuration(ms)}`,
   232→   232→            level: heatLevel(ms, max),
   233→   233→            future: false,
   234→   234→          });
   235→   235→        }
   236→   236→        cursor.setDate(cursor.getDate() + 1);
   237→   237→      }
   238→   238→      if (firstInRangeMonth !== -1 && firstInRangeMonth !== prevMonth) {
   239→   239→        monthLabels.push(MONTHS_SHORT[firstInRangeMonth]);
   240→   240→        prevMonth = firstInRangeMonth;
   241→   241→      } else {
   242→   242→        monthLabels.push(null);
   243→   243→      }
   244→   244→      cols.push(col);
   245→   245→    }
   246→   246→    return { columns: cols, monthLabels };
   247→   247→  }, [heat]);
   248→   248→
   249→   249→  const todayKey = dayKey(Date.now());
   250→   250→
   251→   251→  return (
   252→   252→    <div className="flex" role="img" aria-label="365-day watch activity heatmap">
   253→   253→      {/* Weekday gutter (Mon/Wed/Fri only) */}
   254→   254→      <div className="mr-1.5 shrink-0 pt-[14px]" aria-hidden>
   255→   255→        <div className="flex flex-col gap-[2px]">
   256→   256→          {WEEKDAY_ROWS.map((wd, i) => (
   257→   257→            <div key={wd} className="flex h-[10px] w-6 items-center justify-end">
   258→   258→              {(i === 0 || i === 2 || i === 4) && (
   259→   259→                <span className="text-[9px] leading-none text-ink-subtle">{wd}</span>
   260→   260→              )}
   261→   261→            </div>
   262→   262→          ))}
   263→   263→        </div>
   264→   264→      </div>
   265→   265→      {/* Grid — horizontally scrollable on narrow screens */}
   266→   266→      <div className="harbor-scroll flex-1 overflow-x-auto pb-1">
   267→   267→        <div className="w-max">
   268→   268→          <div className="mb-[3px] flex h-[11px] gap-[2px]">
   269→   269→            {monthLabels.map((m, i) => (
   270→   270→              <div key={i} className="relative h-[11px] w-[10px] shrink-0">
   271→   271→                {m && (
   272→   272→                  <span className="absolute left-0 top-0 whitespace-nowrap text-[9px] leading-[11px] text-ink-subtle">
   273→   273→                    {m}
   274→   274→                  </span>
   275→   275→                )}
   276→   276→              </div>
   277→   277→            ))}
   278→   278→          </div>
   279→   279→          <div className="flex gap-[2px]">
   280→   280→            {columns.map((col, ci) => (
   281→   281→              <div key={ci} className="flex flex-col gap-[2px]">
   282→   282→                {col.map((cell) =>
   283→   283→                  cell.future ? (
   284→   284→                    <div key={cell.key} className="h-[10px] w-[10px]" />
   285→   285→                  ) : (
   286→   286→                    <div
   287→   287→                      key={cell.key}
   288→   288→                      className={cn(
   289→   289→                        "h-[10px] w-[10px] rounded-[2px]",
   290→   290→                        cell.level === 0 && "border border-edge-soft bg-raised",
   291→   291→                        cell.key === todayKey && "ring-1 ring-accent/80",
   292→   292→                      )}
   293→   293→                      style={cell.level > 0 ? { backgroundColor: levelBg(cell.level, accent) } : undefined}
   294→   294→                      title={cell.label}
   295→   295→                    />
   296→   296→                  ),
   297→   297→                )}
   298→   298→              </div>
   299→   299→            ))}
   300→   300→          </div>
   301→   301→        </div>
   302→   302→      </div>
   303→   303→    </div>
   304→   304→  );
   305→   305→}

   437→   437→    setSharing(true);

   444→   444→        activeDays: stats.activeDays,
   445→   445→        topTitles: stats.topTitles

   447→   447→          .map((t) => ({ name: t.name, poster: t.poster, type: t.type })),
