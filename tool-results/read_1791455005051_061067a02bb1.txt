    54→/** Classified playback failure — replaces the old one-size-fits-all string. */
    55→export type PlaybackError = {
    56→  message: string; // localized primary message
    57→  cls: FailureClass | "torrent";

    62→
    63→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    64→  const pop = useNav((s) => s.pop);
    65→  const replace = useNav((s) => s.replace);
    66→  const settings = useSettings((s) => s.settings);
    67→  const { toast } = useToast();
    68→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    69→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    70→  const [error, setError] = useState<PlaybackError | null>(null);
    71→  const [switcherOpen, setSwitcherOpen] = useState(false);
    72→  const [resolveStatus, setResolveStatus] = useState<string>("Finding the best stream…");
    73→  // Auto-fallback state: ranked candidates from the resolve path + attempts used
    74→  const candidatesRef = useRef<Stream[]>([]);
    75→  const fallbackAttemptsRef = useRef(0);
    76→  const currentRef = useRef<PlayerPayload>(payload);
    77→  currentRef.current = current;
    78→
    79→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    80→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    81→
    82→  // Auto-fallback: when a chosen stream fails after the proxy/convert ladder,
    83→  // try the next best playable candidate (max 3 automatic attempts).
    84→  const tryFallback = useCallback(async (): Promise<boolean> => {
    85→    if (fallbackAttemptsRef.current >= 3) return false;
    86→    const cur = currentRef.current;
    87→    if (!/^tt\d+$|^kitsu|^demo/.test(cur.metaId) && !cur.stream) return false;
    88→    fallbackAttemptsRef.current += 1;
    89→    const lang = useSettings.getState().settings.uiLanguage;
    90→    toast({ title: homeT("tryingAnother", lang) });
    91→    try {
    92→      let candidates = candidatesRef.current;
    93→      if (candidates.length === 0) {
    94→        const { useAddons } = await import("@/lib/harbor/store");
    95→        await serverCapabilities();
    96→        const targetId = cur.videoId ?? cur.metaId;
    97→        const results = await fetchStreams(useAddons.getState().addons, cur.type, targetId);
    98→        candidates = runPipeline(results);
    99→        candidatesRef.current = candidates;
   100→      }
   101→      const usedKey = `${cur.stream?.infoHash ?? ""}|${cur.stream?.url ?? cur.url}`;
   102→      const pool = candidates
   103→        .map((s, i) => ({ s, i, c: classifyStream(s) }))
   104→        .filter(
   105→          ({ s, c }) =>
   106→            `${s.infoHash ?? ""}|${s.url ?? ""}` !== usedKey &&
   107→            !!s.url && // fallback targets direct-URL streams only (torrents join via the resolver)
   108→            c.verdict !== "unplayable" &&
   109→            c.verdict !== "external",
   110→        )
   111→        .sort((a, b) => verdictRank(a.c.verdict) - verdictRank(b.c.verdict) || a.i - b.i);
   112→      const next = pool[0];
   113→      if (!next) return false;
   114→      candidatesRef.current = candidates.filter((s) => `${s.infoHash ?? ""}|${s.url ?? ""}` !== `${next.s.infoHash ?? ""}|${next.s.url ?? ""}`);
   115→      setCurrent((c) => ({
   116→        ...c,
   117→        url: next.s.url!,
   118→        stream: next.s,
   119→        streamTitle: next.s.title ?? c.streamTitle,
   120→        streamDescription: next.s.description,
   121→        p2p: undefined,
   122→      }));
   123→      setPhase("loading");
   124→      setError(null);
   125→      return true;
   126→    } catch {
   127→      return false;
   128→    }
   129→  }, [toast]);
   130→
   131→  const handleError = useCallback(
   132→    (err: PlaybackError) => {
   133→      void tryFallback().then((fell) => {
   134→        if (fell) return;
   135→        setPhase("error");
   136→        setError(err);
   137→      });
   138→    },
   139→    [tryFallback],
   140→  );
   141→
   142→  /** Wrap a plain message into a classified PlaybackError (resolve-path failures). */
   143→  const setErrorMsg = useCallback((msg: string, cls: PlaybackError["cls"] = "fatal") => {
   144→    setError({ message: msg, cls, code: null, host: null, canConvert: false });
   145→  }, []);
   146→
   147→  // Resolve stream via addons when payload.url is empty (instant play path)
   148→  useEffect(() => {
   149→    if (current.url || phase !== "resolving") return;
   150→    let alive = true;
   151→    setResolveStatus("Finding the best stream…");
   152→    (async () => {
   153→      const { useAddons: addons } = await import("@/lib/harbor/store");
   154→      try {
   155→        // Series opened without an episode id (bare tt-id): stream addons need
   156→        // "id:season:episode" — resolve the default video from meta first.
   157→        let targetId = current.videoId ?? current.metaId;
   158→        if (current.type === "series" && !targetId.includes(":")) {
   159→          const meta = await fetchMeta("series", current.metaId).catch(() => null);
   160→          const vid = defaultVideoId(meta);
   161→          if (vid) {
   162→            targetId = vid;
   163→            if (alive) {
   164→              const vidMeta = (meta?.videos ?? []).find((v) => v.id === vid);
   165→              setCurrent((c) => ({
   166→                ...c,
   167→                videoId: vid,
   168→                season: c.season ?? vidMeta?.season,
   169→                episode: c.episode ?? vidMeta?.episode,
   170→                episodeName: c.episodeName ?? vidMeta?.name ?? vidMeta?.title,
   171→              }));
   172→            }
   173→          }
   174→        }
   175→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
   176→        if (!alive) return;
   177→        await serverCapabilities();
   178→        const ranked = runPipeline(results);
   179→        candidatesRef.current = ranked;
   180→        // Prefer browser-playable candidates: verdict-ranked URL streams first
   181→        // (direct > proxy > convert; VideoStage escalates automatically), then
   182→        // torrents whose parsed video codec isn't HEVC — unless conversion is
   183→        // available server-side, which rescues HEVC releases too.
   184→        const transcodeAvail = cachedTranscodeSupported();
   185→        const p2pCandidate = (s: Stream) =>
   186→          !!s.infoHash && !s.url && (transcodeAvail || s.parsed?.codec !== "HEVC");
   187→        const urlBest = ranked
   188→          .map((s, i) => ({ s, i, c: classifyStream(s) }))
   189→          .filter(({ s, c }) => s.url && c.verdict !== "unplayable" && c.verdict !== "external")
   190→          .sort((a, b) => verdictRank(a.c.verdict) - verdictRank(b.c.verdict) || a.i - b.i)[0];
   191→        const best = urlBest?.s ?? ranked.find(p2pCandidate) ?? ranked[0];
   192→        if (!best) {
   193→          setPhase("error");
   194→          setError({ message: "No streams found from your addons.", cls: "fatal", code: null, host: null, canConvert: false });
   195→          return;
   196→        }
   197→        if (best.url) {
   198→          setCurrent((c) => ({
   199→            ...c,
   200→            url: best.url!,
   201→            stream: best,
   202→            streamTitle: best.title ?? c.streamTitle,
   203→            streamDescription: best.description,
   204→          }));
   205→          setPhase("loading");
   206→          return;
   207→        }
   208→        // Best candidate is a torrent → try the server-side P2P engine before giving up
   209→        if (!best.infoHash || !useSettings.getState().settings.p2pEnabled) {
   210→          setPhase("error");
   211→          setErrorMsg(
   212→            ranked.length > 0
   213→              ? "No browser-playable stream found. Open the stream picker to choose manually — torrent streams play via P2P or unlock with debrid."
   214→              : "No streams found from your addons.",
   215→            "torrent",
   216→          );
   217→          return;
   218→        }
   219→        const { p2pPrepare, p2pStatus, p2pPlan } = await import("@/lib/harbor/p2p");
   220→        const infoHash = best.infoHash.toLowerCase();
   221→        setResolveStatus("Joining the torrent swarm…");
   222→        const prep = await p2pPrepare(infoHash, {
   223→          fileIdx: best.fileIdx,
   224→          filename: best.behaviorHints?.filename ?? best.parsed?.filename,
   225→        });
   226→        for (let i = 0; prep.pending && i < 40 && alive; i++) {
   227→          await new Promise((r) => setTimeout(r, 2500));
   228→          setResolveStatus((s) => (s.startsWith("Finding peers") ? s : `Finding peers… (${i * 2.5 | 0}s)`));
   229→          const st = await p2pStatus(infoHash).catch(() => null);
   230→          if (st?.peers) setResolveStatus(`Finding peers… (${st.peers} found)`);
   231→          if (st?.ready) break;
   232→        }
   233→        if (!alive) return;
   234→        const st = prep.pending ? await p2pStatus(infoHash).catch(() => null) : null;
   235→        if (prep.pending && !st?.ready) {
   236→          setPhase("error");
   237→          setErrorMsg(
   238→            "No peers responded for the best torrent. Open the stream picker to pick another source, or connect debrid for instant cached streams.",
   239→            "torrent",
   240→          );
   241→          return;
   242→        }
   243→        const fileIdx = prep.pending
   244→          ? (Number.isInteger(best.fileIdx)
   245→              ? best.fileIdx!
   246→              : (st?.files ?? []).length > 0
   247→                ? (st!.files ?? []).reduce(
   248→                    (a, b) => (b.length > a.length ? b : a),
   249→                    (st!.files ?? [{ index: 0, name: "", length: 0 }])[0],
   250→                  ).index
   251→                : 0)
   252→          : prep.fileIdx;
   253→        const filename = best.behaviorHints?.filename ?? best.parsed?.filename;
   254→        setResolveStatus("Checking video codecs…");
   255→        await refreshP2pCapabilities(); // engine-level HEVC conversion flag
   256→        const plan = await p2pPlan(infoHash, fileIdx, filename).catch(() => null);
   257→        if (!alive) return;
   258→        if (!plan) {
   259→          setPhase("error");
   260→          setError({
   261→            message: "Torrent engine could not serve this file. Try another stream in the picker.",
   262→            cls: "torrent",
   263→            code: "P2P_NO_PLAN",
   264→            host: "P2P swarm",
   265→            canConvert: false,
   266→          });
   267→          return;
   268→        }
   269→        setCurrent((c) => ({
   270→          ...c,
   271→          url: plan.url,
   272→          stream: best,
   273→          streamTitle: best.title ?? c.streamTitle,
   274→          streamDescription: best.description,
   275→          p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   276→        }));
   277→        setPhase("loading");
   278→      } catch (e) {
   279→        if (!alive) return;
   280→        setPhase("error");
   281→        setError({
   282→          message: e instanceof Error ? e.message : "Failed to resolve streams",
   283→          cls: "fatal",
   284→          code: null,
   285→          host: null,
   286→          canConvert: false,
   287→        });
   288→      }
   289→    })();
   290→    return () => {
   291→      alive = false;
   292→    };
   293→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
   294→
   295→  const setPayload = useCallback((p: PlayerPayload) => {
   296→    setCurrent(p);
   297→    setPhase(p.url ? "loading" : "resolving");
   298→    setError(null);
   299→  }, []);
   300→
   301→  const openSwitcher = useCallback(() => setSwitcherOpen(true), []);
   302→
   303→  // Convert-and-play after a final codec failure (or from the ask dialog):
   304→  // re-signs the original upstream through /api/transcode and reattaches.
   305→  const convertAndPlay = useCallback(async () => {
   306→    const cur = currentRef.current;
   307→    const upstream = cur.stream?.url ?? cur.url;
   308→    if (!upstream || !/^https?:\/\//i.test(upstream)) return;
   309→    const lang = useSettings.getState().settings.uiLanguage;
   310→    try {
   311→      const signed = await signSource({ ...(cur.stream ?? { url: upstream }), url: upstream }, "transcode");
   312→      setPhase("loading");
   313→      setError(null);
   314→      setCurrent((c) => ({ ...c, url: signed.url, stream: cur.stream }));
   315→    } catch {
   316→      setPhase("error");
   317→      setError({
   318→        message: homeT("convertingNeeds", lang),
   319→        cls: "codec",
   320→        code: null,
   321→        host: null,
   322→        canConvert: false,
   323→      });
   324→    }
   325→  }, []);
   326→
   327→  // Advance to the next episode in-place (Up Next card / auto-advance / "n" key).
   328→  // Empty url re-enters the resolving path so addons fetch streams for the new episode.
   329→  const handlePlayNext = useCallback(
   330→    (next: { season: number; episode: number; videoId: string; episodeName?: string }) => {
   331→      setCurrent((c) => ({
   332→        ...c,
   333→        url: "",
   334→        season: next.season,
   335→        episode: next.episode,
   336→        videoId: next.videoId,
   337→        episodeName: next.episodeName,
   338→        streamTitle: undefined,
   339→        streamDescription: undefined,
   340→        resumeAt: 0,
   341→      }));
   342→      setPhase("resolving");
   343→      setError(null);
   344→    },
   345→    [],
   346→  );
   347→
   348→  return (
   349→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
   350→      {phase === "resolving" && (
   351→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
   352→          <div className="relative">
   353→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
   354→          </div>
   355→          <p className="text-sm text-ink-muted" aria-live="polite">{resolveStatus}</p>
   356→        </div>
   357→      )}
   358→      {phase === "error" && error ? (
   359→        <PlaybackErrorPanel
   360→          error={error}
   361→          lang={settings.uiLanguage}
   362→          onConvert={error.canConvert ? () => void convertAndPlay() : undefined}
   363→          onRetry={() => {
   364→            setError(null);
   365→            setPhase(currentRef.current.url ? "loading" : "resolving");
   366→            // Force a fresh attach even for the same URL
   367→            setCurrent((c) => ({ ...c }));
   368→          }}
   369→          onPicker={() => setSwitcherOpen(true)}
   370→          onBack={pop}
   371→        />
   372→      ) : (
   373→        current.url && (
   374→          <VideoStage
   375→            key={current.url}
   376→            payload={current}
   377→            settings={settings}
   378→            onPhase={handlePhase}
   379→            onError={handleError}
   380→            onPlayNext={handlePlayNext}
   381→            onOpenSwitcher={openSwitcher}
   382→          />
   383→        )
   384→      )}
   385→
   386→      {/* Top bar */}
   387→      {phase !== "error" && (
   388→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   389→          <button
   390→            type="button"
   391→            onClick={() => {
   392→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   393→              pop();
   394→            }}
   395→            className="md-icon-btn md-state h-12! w-12! text-white!"
   396→            aria-label="Close player (Esc)"
   397→          >
   398→            <X className="w-5 h-5" />
   399→          </button>
   400→          <div className="min-w-0">
   401→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   402→            {current.episode && (
   403→              <p className="text-xs text-ink-muted">
   404→                S{current.season}:E{current.episode}
   405→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   406→              </p>
   407→            )}
   408→          </div>
   409→          <button
   410→            type="button"
   411→            onClick={() => setSwitcherOpen(true)}
   412→            className="md-state ml-auto flex items-center gap-1.5 rounded-full bg-black/50 px-3 py-1.5 text-xs font-medium text-ink"
   413→            title="Switch stream (W)"
   414→          >
   415→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   416→          </button>
   417→        </div>
   418→      )}
   419→
   420→      {switcherOpen && (
   421→        <PickerOverlay
   422→          type={current.type}
   423→          id={current.deepLink?.id ?? current.metaId}
   424→          videoId={current.videoId ?? current.deepLink?.videoId}
   425→          season={current.season}
   426→          episode={current.episode}
   427→        />
   428→      )}
   429→    </div>
   430→  );
   431→}

   433→// ---------------- Classified error panel ----------------
   434→function PlaybackErrorPanel({
   435→  error,
   436→  lang,
   437→  onConvert,
   438→  onRetry,
   439→  onPicker,
   440→  onBack,
   441→}: {
   442→  error: PlaybackError;
   443→  lang: string;
   444→  onConvert?: () => void;
   445→  onRetry: () => void;
   446→  onPicker: () => void;
   447→  onBack: () => void;
   448→}) {
   449→  const { toast } = useToast();
   450→  const [showTech, setShowTech] = useState(false);
   451→  const copyDiagnostics = () => {
   452→    const lines = [
   453→      `class: ${error.cls}`,
   454→      `code: ${error.code ?? "n/a"}`,
   455→      `host: ${error.host ?? "n/a"}`,
   456→      `user-agent: ${navigator.userAgent}`,
   457→      `page: ${location.host}${location.pathname}`,
   458→    ];
   459→    void navigator.clipboard
   460→      .writeText(lines.join("\n"))
   461→      .then(() => toast({ title: homeT("diagnosticsCopied", lang) }))
   462→      .catch(() => {
   463→        /* clipboard denied */
   464→      });
   465→  };
   466→  return (
   467→    <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   468→      <AlertTriangle className="w-12 h-12 text-danger" />
   469→      <p className="md-title-medium text-ink">{homeT("playbackErrorTitle", lang)}</p>
   470→      <p className="text-sm text-ink-muted max-w-md" role="alert">{error.message}</p>
   471→      <div className="flex gap-3 flex-wrap justify-center">
   472→        {onConvert && (
   473→          <button type="button" onClick={onConvert} className="md-btn md-btn-filled md-state">
   474→            {homeT("convertAndPlay", lang)}
   475→          </button>
   476→        )}
   477→        <button type="button" onClick={onRetry} className="md-btn md-btn-tonal md-state">
   478→          {homeT("retry", lang)}
   479→        </button>
   480→        <button type="button" onClick={onPicker} className="md-btn md-btn-tonal md-state">
   481→          {homeT("pickAnother", lang)}
   482→        </button>
   483→        <button type="button" onClick={onBack} className="md-btn md-btn-text md-state">
   484→          Back
   485→        </button>
   486→      </div>
   487→      <div className="mt-1 max-w-md w-full">
   488→        <button
   489→          type="button"
   490→          onClick={() => setShowTech((v) => !v)}
   491→          className="text-xs text-ink-subtle underline underline-offset-2 hover:text-ink-muted"
   492→          aria-expanded={showTech}
   493→        >
   494→          {homeT("showTechnical", lang)}
   495→        </button>
   496→        {showTech && (
   497→          <div className="mt-2 rounded-xl border border-edge-soft bg-raised/60 p-3 text-start text-[11px] text-ink-subtle space-y-1" dir="ltr">
   498→            <p>{homeT("errClass", lang)}: {error.cls}</p>
   499→            <p>{homeT("errCode", lang)}: {error.code ?? "n/a"}</p>
   500→            <p className="break-all">{homeT("sourceHost", lang)}: {error.host ?? "n/a"}</p>
   501→            <button
   502→              type="button"
   503→              onClick={copyDiagnostics}
   504→              className="mt-1 rounded-full border border-edge-soft px-2.5 py-1 text-[10px] font-semibold text-ink-muted hover:text-ink"
   505→            >
   506→              {homeT("copyDiagnostics", lang)}
   507→            </button>
   508→          </div>
