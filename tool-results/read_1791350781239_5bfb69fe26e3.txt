   119→   119→/// Resume WebView2 after TrySuspend / SetIsVisible(false). Safe no-op if not suspended.

   121→   121→fn resume_webview_if_needed(app: &tauri::AppHandle) {
   122→   122→    use tauri::Manager;
   123→   123→    if !WEBVIEW_SUSPENDED.load(std::sync::atomic::Ordering::SeqCst) {
   124→   124→        return;
   125→   125→    }
   126→   126→    let Some(window) = app.get_webview_window("main") else {
   127→   127→        return;
   128→   128→    };
   129→   129→    let res = window.with_webview(|webview| unsafe {
   130→   130→        use webview2_com::Microsoft::Web::WebView2::Win32::ICoreWebView2_3;
   131→   131→        use windows::core::Interface;
   132→   132→        let controller = webview.controller();
   133→   133→        if let Ok(core) = controller.CoreWebView2() {
   134→   134→            if let Ok(c3) = core.cast::<ICoreWebView2_3>() {
   135→   135→                let _ = c3.Resume();
   136→   136→            }
   137→   137→        }
   138→   138→        let _ = controller.SetIsVisible(true);
   139→   139→    });
   140→   140→    if res.is_ok() {
   141→   141→        WEBVIEW_SUSPENDED.store(false, std::sync::atomic::Ordering::SeqCst);
   142→   142→        eprintln!("[harbor::webview] auto-resumed after suspend");
   143→   143→    }
   144→   144→}

   170→   170→#[cfg(windows)]
   171→   171→unsafe extern "system" fn maxguard_subclass_proc(
   172→   172→    hwnd: windows::Win32::Foundation::HWND,
   173→   173→    msg: u32,
   174→   174→    wparam: windows::Win32::Foundation::WPARAM,
   175→   175→    lparam: windows::Win32::Foundation::LPARAM,
   176→   176→    _id: usize,
   177→   177→    _data: usize,
   178→   178→) -> windows::Win32::Foundation::LRESULT {
   179→   179→    use windows::Win32::Graphics::Gdi::{
   180→   180→        GetMonitorInfoW, MonitorFromWindow, MONITORINFO, MONITOR_DEFAULTTONEAREST,
   181→   181→    };
   182→   182→    use windows::Win32::UI::Shell::DefSubclassProc;
   183→   183→    use windows::Win32::UI::WindowsAndMessaging::{MINMAXINFO, WM_GETMINMAXINFO};
   184→   184→    let res = DefSubclassProc(hwnd, msg, wparam, lparam);
   185→   185→    if msg == WM_GETMINMAXINFO {
   186→   186→        let monitor = MonitorFromWindow(hwnd, MONITOR_DEFAULTTONEAREST);
   187→   187→        let mut mi = MONITORINFO {
   188→   188→            cbSize: std::mem::size_of::<MONITORINFO>() as u32,
   189→   189→            ..Default::default()
   190→   190→        };
   191→   191→        if GetMonitorInfoW(monitor, &mut mi).as_bool() {
   192→   192→            let mmi = &mut *(lparam.0 as *mut MINMAXINFO);
   193→   193→            mmi.ptMaxPosition.x = mi.rcWork.left - mi.rcMonitor.left;
   194→   194→            mmi.ptMaxPosition.y = mi.rcWork.top - mi.rcMonitor.top;
   195→   195→            mmi.ptMaxSize.x = mi.rcWork.right - mi.rcWork.left;
   196→   196→            mmi.ptMaxSize.y = mi.rcWork.bottom - mi.rcWork.top;
   197→   197→        }
   198→   198→    }
   199→   199→    res
   200→   200→}

   336→   336→fn ensure_window_on_screen(app: &tauri::AppHandle) {
   337→   337→    use tauri::Manager;
   338→   338→    let Some(window) = app.get_webview_window("main") else {
   339→   339→        return;
   340→   340→    };
   341→   341→    let (pos, size) = match (window.outer_position(), window.outer_size()) {
   342→   342→        (Ok(p), Ok(s)) => (p, s),
   343→   343→        _ => return,
   344→   344→    };
   345→   345→    let monitors = match window.available_monitors() {
   346→   346→        Ok(m) if !m.is_empty() => m,
   347→   347→        _ => return,
   348→   348→    };
   349→   349→    let ww = size.width as i32;
   350→   350→    let wh = size.height as i32;
   351→   351→    let on_screen = monitors.iter().any(|m| {
   352→   352→        let mp = m.position();
   353→   353→        let ms = m.size();
   354→   354→        pos.x < mp.x + ms.width as i32
   355→   355→            && pos.x + ww > mp.x
   356→   356→            && pos.y < mp.y + ms.height as i32
   357→   357→            && pos.y + wh > mp.y
   358→   358→    });
   359→   359→    if on_screen {
   360→   360→        return;
   361→   361→    }
   362→   362→    let target = window
   363→   363→        .primary_monitor()
   364→   364→        .ok()
   365→   365→        .flatten()
   366→   366→        .or_else(|| monitors.into_iter().next());
   367→   367→    let Some(mon) = target else {
   368→   368→        return;
   369→   369→    };
   370→   370→    let mp = mon.position();
   371→   371→    let ms = mon.size();
   372→   372→    let cx = mp.x + (ms.width as i32 - ww).max(0) / 2;
   373→   373→    let cy = mp.y + (ms.height as i32 - wh).max(0) / 2;
   374→   374→    let _ = window.set_position(tauri::PhysicalPosition::new(cx, cy));
   375→   375→    eprintln!("[harbor::window] launched off-screen; recentered to {},{}", cx, cy);
   376→   376→}
