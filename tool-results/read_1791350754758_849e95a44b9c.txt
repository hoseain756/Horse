   173→   173→function useViewPreloader() {
   174→   174→  useEffect(() => {
   175→   175→    if (typeof window === "undefined") return;
   176→   176→    let cancelled = false;
   177→   177→    const win = window as Window & {
   178→   178→      requestIdleCallback?: (cb: () => void, opts?: { timeout?: number }) => number;
   179→   179→      cancelIdleCallback?: (id: number) => void;
   180→   180→    };
   181→   181→    const schedule = (cb: () => void, timeout: number) =>
   182→   182→      typeof win.requestIdleCallback === "function"
   183→   183→        ? win.requestIdleCallback(cb, { timeout })
   184→   184→        : window.setTimeout(cb, Math.min(timeout, 800));
   185→   185→
   186→   186→    // Priority: Movies/Shows chunks first — they were lazy and felt slower than Anime.
   187→   187→    const priorityId = schedule(() => {
   188→   188→      if (cancelled) return;
   189→   189→      void importMovies();
   190→   190→      void importShows();
   191→   191→      void importAnime();
   192→   192→      void importDiscover();
   193→   193→      void importDetail();
   194→   194→      void importPlayPicker();
   195→   195→      void importPlayer();
   196→   196→    }, 1200);
   197→   197→
   198→   198→    const restId = schedule(() => {
   199→   199→      if (cancelled) return;
   200→   200→      void importSettings();
   201→   201→      void importAddons();
   202→   202→      void importPerson();
   203→   203→      void importFilter();
   204→   204→      void importCalendar();
   205→   205→      void importLive();
   206→   206→      void importQueue();
   207→   207→      void importAward();
   208→   208→      void importAnimeAward();
   209→   209→      void importService();
   210→   210→      void importMatchDetail();
   211→   211→      void importOnboarding();
   212→   212→      void importLibrary();
   213→   213→      void importCatalogs();
   214→   214→      void importKids();
   215→   215→      void importVod();
   216→   216→      void importDownloads();
   217→   217→    }, 2800);
   218→   218→
   219→   219→    return () => {
   220→   220→      cancelled = true;
   221→   221→      if (typeof win.cancelIdleCallback === "function") {
   222→   222→        win.cancelIdleCallback(priorityId as number);
   223→   223→        win.cancelIdleCallback(restId as number);
   224→   224→      } else {
   225→   225→        window.clearTimeout(priorityId);
   226→   226→        window.clearTimeout(restId);
   227→   227→      }
   228→   228→    };
   229→   229→  }, []);
   230→   230→}

   249→   249→function useKeepAlive(active: boolean, requested: boolean, pin = false): boolean {
   250→   250→  const [mounted, setMounted] = useState(active && requested);
   251→   251→  useEffect(() => {
   252→   252→    if (!requested) {
   253→   253→      setMounted(false);
   254→   254→      return;
   255→   255→    }
   256→   256→    if (active || pin) {
   257→   257→      setMounted(true);
   258→   258→      return;
   259→   259→    }
   260→   260→    const t = setTimeout(() => setMounted(false), KEEP_ALIVE_MS);
   261→   261→    return () => clearTimeout(t);
   262→   262→  }, [active, requested, pin]);
   263→   263→  return requested && (mounted || active || pin);
   264→   264→}

   266→   266→function useIdleEvict(active: boolean, pin = false): boolean {
   267→   267→  const [alive, setAlive] = useState(active);
   268→   268→  const [pressure, setPressure] = useState(false);
   269→   269→  useEffect(() => subscribeMemoryPressure(setPressure), []);
   270→   270→  useEffect(() => {
   271→   271→    if (active || pin) {
   272→   272→      setAlive(true);
   273→   273→      return;
   274→   274→    }
   275→   275→    if (!alive) return;
   276→   276→    const t = setTimeout(() => setAlive(false), pressure ? PRESSURE_EVICT_MS : IDLE_EVICT_MS);
   277→   277→    return () => clearTimeout(t);
   278→   278→  }, [active, alive, pressure, pin]);
   279→   279→  return alive || active || pin;
   280→   280→}

   386→   386→function TogetherLocationPublisher() {
   387→   387→  const { topKind, meta, personId, picker, player, service, addonDetailId } = useView();
   388→   388→  const { snapshot, sendPresence } = useTogether();
   389→   389→  const inSession = snapshot.state === "joined";
   390→   390→  const participantsCount = snapshot.participants.length;
   391→   391→  useEffect(() => {
   392→   392→    if (!inSession) return;
   393→   393→    const location = computeLocation();
   394→   394→    sendPresence(location ?? undefined);
   395→   395→    const id = window.setInterval(() => sendPresence(location ?? undefined), 6000);
   396→   396→    return () => window.clearInterval(id);
   397→   397→    function computeLocation(): import("@/lib/together/protocol").ParticipantLocation | null {
   398→   398→      const metaToLoc = (m: import("@/lib/cinemeta").Meta) => ({
   399→   399→        id: m.id,
   400→   400→        type: (m.type === "series" ? "series" : "movie") as "movie" | "series",
   401→   401→        name: m.name,
   402→   402→        poster: m.poster,
   403→   403→        background: m.background,
   404→   404→        releaseInfo: m.releaseInfo,
   405→   405→        logo: m.logo,
   406→   406→      });
   407→   407→      if (player) {
   408→   408→        return {
   409→   409→          kind: "player" as const,

   479→   479→function parseDeepLinkEpisode(videoId?: string): { season: number; episode: number } | undefined {
   480→   480→  if (!videoId) return undefined;
   481→   481→  const parts = videoId.split(":");
   482→   482→  if (parts.length < 3) return undefined;
   483→   483→  const season = Number(parts[parts.length - 2]);
   484→   484→  const episode = Number(parts[parts.length - 1]);
   485→   485→  if (!Number.isFinite(season) || !Number.isFinite(episode)) return undefined;
   486→   486→  return { season, episode };
