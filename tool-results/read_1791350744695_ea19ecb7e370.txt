   173→function useViewPreloader() {
   174→  useEffect(() => {
   175→    if (typeof window === "undefined") return;
   176→    let cancelled = false;
   177→    const win = window as Window & {
   178→      requestIdleCallback?: (cb: () => void, opts?: { timeout?: number }) => number;
   179→      cancelIdleCallback?: (id: number) => void;
   180→    };
   181→    const schedule = (cb: () => void, timeout: number) =>
   182→      typeof win.requestIdleCallback === "function"
   183→        ? win.requestIdleCallback(cb, { timeout })
   184→        : window.setTimeout(cb, Math.min(timeout, 800));
   185→
   186→    // Priority: Movies/Shows chunks first — they were lazy and felt slower than Anime.
   187→    const priorityId = schedule(() => {
   188→      if (cancelled) return;
   189→      void importMovies();
   190→      void importShows();
   191→      void importAnime();
   192→      void importDiscover();
   193→      void importDetail();
   194→      void importPlayPicker();
   195→      void importPlayer();
   196→    }, 1200);
   197→
   198→    const restId = schedule(() => {
   199→      if (cancelled) return;
   200→      void importSettings();
   201→      void importAddons();
   202→      void importPerson();
   203→      void importFilter();
   204→      void importCalendar();
   205→      void importLive();
   206→      void importQueue();
   207→      void importAward();
   208→      void importAnimeAward();
   209→      void importService();
   210→      void importMatchDetail();
   211→      void importOnboarding();
   212→      void importLibrary();
   213→      void importCatalogs();
   214→      void importKids();
   215→      void importVod();
   216→      void importDownloads();
   217→    }, 2800);
   218→
   219→    return () => {
   220→      cancelled = true;
   221→      if (typeof win.cancelIdleCallback === "function") {
   222→        win.cancelIdleCallback(priorityId as number);
   223→        win.cancelIdleCallback(restId as number);
   224→      } else {
   225→        window.clearTimeout(priorityId);
   226→        window.clearTimeout(restId);
   227→      }
   228→    };
   229→  }, []);
   230→}

   249→function useKeepAlive(active: boolean, requested: boolean, pin = false): boolean {
   250→  const [mounted, setMounted] = useState(active && requested);
   251→  useEffect(() => {
   252→    if (!requested) {
   253→      setMounted(false);
   254→      return;
   255→    }
   256→    if (active || pin) {
   257→      setMounted(true);
   258→      return;
   259→    }
   260→    const t = setTimeout(() => setMounted(false), KEEP_ALIVE_MS);
   261→    return () => clearTimeout(t);
   262→  }, [active, requested, pin]);
   263→  return requested && (mounted || active || pin);
   264→}

   266→function useIdleEvict(active: boolean, pin = false): boolean {
   267→  const [alive, setAlive] = useState(active);
   268→  const [pressure, setPressure] = useState(false);
   269→  useEffect(() => subscribeMemoryPressure(setPressure), []);
   270→  useEffect(() => {
   271→    if (active || pin) {
   272→      setAlive(true);
   273→      return;
   274→    }
   275→    if (!alive) return;
   276→    const t = setTimeout(() => setAlive(false), pressure ? PRESSURE_EVICT_MS : IDLE_EVICT_MS);
   277→    return () => clearTimeout(t);
   278→  }, [active, alive, pressure, pin]);
   279→  return alive || active || pin;
   280→}

   386→function TogetherLocationPublisher() {
   387→  const { topKind, meta, personId, picker, player, service, addonDetailId } = useView();
   388→  const { snapshot, sendPresence } = useTogether();
   389→  const inSession = snapshot.state === "joined";
   390→  const participantsCount = snapshot.participants.length;
   391→  useEffect(() => {
   392→    if (!inSession) return;
   393→    const location = computeLocation();
   394→    sendPresence(location ?? undefined);
   395→    const id = window.setInterval(() => sendPresence(location ?? undefined), 6000);
   396→    return () => window.clearInterval(id);
   397→    function computeLocation(): import("@/lib/together/protocol").ParticipantLocation | null {
   398→      const metaToLoc = (m: import("@/lib/cinemeta").Meta) => ({
   399→        id: m.id,
   400→        type: (m.type === "series" ? "series" : "movie") as "movie" | "series",
   401→        name: m.name,
   402→        poster: m.poster,
   403→        background: m.background,
   404→        releaseInfo: m.releaseInfo,
   405→        logo: m.logo,
   406→      });
   407→      if (player) {
   408→        return {
   409→          kind: "player" as const,
   410→          meta: metaToLoc(player.meta),
   411→          episode: player.episode
   412→            ? {
   413→                season: player.episode.season,
   414→                episode: player.episode.episode,
   415→                name: player.episode.name,
   416→              }
   417→            : undefined,
   418→        };
   419→      }
   420→      if (picker) {
   421→        return {
   422→          kind: "picker" as const,
   423→          meta: metaToLoc(picker.meta),
   424→          episode: picker.episode
   425→            ? {
   426→                season: picker.episode.season,
   427→                episode: picker.episode.episode,
   428→                name: picker.episode.name,
   429→              }
   430→            : undefined,
   431→        };
   432→      }
   433→      if (topKind === "meta" && meta) return { kind: "meta" as const, meta: metaToLoc(meta) };
   434→      if (topKind === "person" && personId != null) return { kind: "person" as const, personId };
   435→      if (topKind === "service" && service) return { kind: "service" as const, service };
   436→      if (topKind === "addon-detail" && addonDetailId)
   437→        return { kind: "addon-detail" as const, addonId: addonDetailId };
   438→      if (topKind === "home") return { kind: "home" };
   439→      if (topKind === "discover") return { kind: "discover" };
   440→      if (topKind === "anime") return { kind: "anime" };
   441→      if (topKind === "queue") return { kind: "queue" };
   442→      if (topKind === "addons") return { kind: "addons" };
   443→      if (topKind === "library") return { kind: "home" };
   444→      if (topKind === "settings") return { kind: "settings" };
   445→      return null;
   446→    }
   447→  }, [
   448→    inSession,
   449→    sendPresence,
   450→    topKind,
   451→    meta?.id,
   452→    personId,
   453→    picker?.meta.id,
   454→    picker?.episode?.season,
   455→    picker?.episode?.episode,
   456→    player?.meta.id,
   457→    player?.episode?.season,
   458→    player?.episode?.episode,
   459→    service,
   460→    addonDetailId,
   461→    participantsCount,
   462→  ]);
   463→  return null;
   464→}

   479→function parseDeepLinkEpisode(videoId?: string): { season: number; episode: number } | undefined {
   480→  if (!videoId) return undefined;
   481→  const parts = videoId.split(":");
   482→  if (parts.length < 3) return undefined;
   483→  const season = Number(parts[parts.length - 2]);
   484→  const episode = Number(parts[parts.length - 1]);
   485→  if (!Number.isFinite(season) || !Number.isFinite(episode)) return undefined;
   486→  return { season, episode };
   487→}
