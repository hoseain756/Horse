    31→type Phase = "resolving" | "loading" | "playing" | "error";

    33→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    34→  const pop = useNav((s) => s.pop);
    35→  const replace = useNav((s) => s.replace);
    36→  const settings = useSettings((s) => s.settings);
    37→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    38→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    39→  const [error, setError] = useState<string | null>(null);
    40→  const [switcherOpen, setSwitcherOpen] = useState(false);
    41→  const [resolveStatus, setResolveStatus] = useState<string>("Finding the best stream…");
    42→
    43→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    44→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    45→  const handleError = useCallback((msg: string) => {
    46→    setPhase("error");
    47→    setError(msg);
    48→  }, []);
    49→
    50→  // Resolve stream via addons when payload.url is empty (instant play path)
    51→  useEffect(() => {
    52→    if (current.url || phase !== "resolving") return;
    53→    let alive = true;
    54→    setResolveStatus("Finding the best stream…");
    55→    (async () => {
    56→      const { useAddons: addons } = await import("@/lib/harbor/store");
    57→      try {
    58→        // Series opened without an episode id (bare tt-id): stream addons need
    59→        // "id:season:episode" — resolve the default video from meta first.
    60→        let targetId = current.videoId ?? current.metaId;
    61→        if (current.type === "series" && !targetId.includes(":")) {
    62→          const meta = await fetchMeta("series", current.metaId).catch(() => null);
    63→          const vid = defaultVideoId(meta);
    64→          if (vid) {
    65→            targetId = vid;
    66→            if (alive) {
    67→              const vidMeta = (meta?.videos ?? []).find((v) => v.id === vid);
    68→              setCurrent((c) => ({
    69→                ...c,
    70→                videoId: vid,
    71→                season: c.season ?? vidMeta?.season,
    72→                episode: c.episode ?? vidMeta?.episode,
    73→                episodeName: c.episodeName ?? vidMeta?.name ?? vidMeta?.title,
    74→              }));
    75→            }
    76→          }
    77→        }
    78→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
    79→        if (!alive) return;
    80→        const ranked = runPipeline(results);
    81→        // Prefer browser-playable candidates: direct URLs first, then torrents whose
    82→        // parsed video codec isn't HEVC (browsers can't decode it — pick the best
    83→        // H.264/VP9/AV1 release instead of a 4K HEVC one that would dead-end).
    84→        const p2pCandidate = (s: Stream) =>
    85→          !!s.infoHash && !s.url && s.parsed?.codec !== "HEVC";
    86→        const best = ranked.find((s) => s.url) ?? ranked.find(p2pCandidate) ?? ranked[0];
    87→        if (!best) {
    88→          setPhase("error");
    89→          setError("No streams found from your addons.");
    90→          return;
    91→        }
    92→        if (best.url) {
    93→          setCurrent((c) => ({
    94→            ...c,
    95→            url: best.url!,
    96→            streamTitle: best.title ?? c.streamTitle,
    97→            streamDescription: best.description,
    98→          }));
    99→          setPhase("loading");
   100→          return;
   101→        }
   102→        // Best candidate is a torrent → try the server-side P2P engine before giving up
   103→        if (!best.infoHash || !useSettings.getState().settings.p2pEnabled) {
   104→          setPhase("error");
   105→          setError(
   106→            ranked.length > 0
   107→              ? "No browser-playable stream found. Open the stream picker to choose manually — torrent streams play via P2P or unlock with debrid."
   108→              : "No streams found from your addons.",
   109→          );
   110→          return;
   111→        }
   112→        const { p2pPrepare, p2pStatus, p2pPlan } = await import("@/lib/harbor/p2p");
   113→        const infoHash = best.infoHash.toLowerCase();
   114→        setResolveStatus("Joining the torrent swarm…");
   115→        const prep = await p2pPrepare(infoHash, {
   116→          fileIdx: best.fileIdx,
   117→          filename: best.behaviorHints?.filename ?? best.parsed?.filename,
   118→        });
   119→        for (let i = 0; prep.pending && i < 40 && alive; i++) {
   120→          await new Promise((r) => setTimeout(r, 2500));
   121→          setResolveStatus((s) => (s.startsWith("Finding peers") ? s : `Finding peers… (${i * 2.5 | 0}s)`));
   122→          const st = await p2pStatus(infoHash).catch(() => null);
   123→          if (st?.peers) setResolveStatus(`Finding peers… (${st.peers} found)`);
   124→          if (st?.ready) break;
   125→        }
   126→        if (!alive) return;
   127→        const st = prep.pending ? await p2pStatus(infoHash).catch(() => null) : null;
   128→        if (prep.pending && !st?.ready) {
   129→          setPhase("error");
   130→          setError(
   131→            "No peers responded for the best torrent. Open the stream picker to pick another source, or connect debrid for instant cached streams.",
   132→          );
   133→          return;
   134→        }
   135→        const fileIdx = prep.pending
   136→          ? (Number.isInteger(best.fileIdx)
   137→              ? best.fileIdx!
   138→              : (st?.files ?? []).length > 0
   139→                ? (st!.files ?? []).reduce(
   140→                    (a, b) => (b.length > a.length ? b : a),
   141→                    (st!.files ?? [{ index: 0, name: "", length: 0 }])[0],
   142→                  ).index
   143→                : 0)
   144→          : prep.fileIdx;
   145→        const filename = best.behaviorHints?.filename ?? best.parsed?.filename;
   146→        setResolveStatus("Checking video codecs…");
   147→        const plan = await p2pPlan(infoHash, fileIdx, filename).catch(() => null);
   148→        if (!alive) return;
   149→        if (!plan) {
   150→          setPhase("error");
   151→          setError("Torrent engine could not serve this file. Try another stream in the picker.");
   152→          return;
   153→        }
   154→        setCurrent((c) => ({
   155→          ...c,
   156→          url: plan.url,
   157→          streamTitle: best.title ?? c.streamTitle,
   158→          streamDescription: best.description,
   159→          p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   160→        }));
   161→        setPhase("loading");
   162→      } catch (e) {
   163→        if (!alive) return;
   164→        setPhase("error");
   165→        setError(e instanceof Error ? e.message : "Failed to resolve streams");
   166→      }
   167→    })();
   168→    return () => {
   169→      alive = false;
   170→    };
   171→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
   172→
   173→  const setPayload = useCallback((p: PlayerPayload) => {
   174→    setCurrent(p);
   175→    setPhase(p.url ? "loading" : "resolving");
   176→    setError(null);
   177→  }, []);
   178→
   179→  const openSwitcher = useCallback(() => setSwitcherOpen(true), []);
   180→
   181→  // Advance to the next episode in-place (Up Next card / auto-advance / "n" key).
   182→  // Empty url re-enters the resolving path so addons fetch streams for the new episode.
   183→  const handlePlayNext = useCallback(
   184→    (next: { season: number; episode: number; videoId: string; episodeName?: string }) => {
   185→      setCurrent((c) => ({
   186→        ...c,
   187→        url: "",
   188→        season: next.season,
   189→        episode: next.episode,
   190→        videoId: next.videoId,
   191→        episodeName: next.episodeName,
   192→        streamTitle: undefined,
   193→        streamDescription: undefined,
   194→        resumeAt: 0,
   195→      }));
   196→      setPhase("resolving");
   197→      setError(null);
   198→    },
   199→    [],
   200→  );
   201→
   202→  return (
   203→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
   204→      {phase === "resolving" && (
   205→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
   206→          <div className="relative">
   207→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
   208→          </div>
   209→          <p className="text-sm text-ink-muted" aria-live="polite">{resolveStatus}</p>
   210→        </div>
   211→      )}
   212→      {phase === "error" ? (
   213→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   214→          <AlertTriangle className="w-12 h-12 text-danger" />
   215→          <p className="text-sm text-ink-muted max-w-md">{error}</p>
   216→          <div className="flex gap-3">
   217→            <button
   218→              type="button"
   219→              onClick={() => setSwitcherOpen(true)}
   220→              className="rounded-xl bg-accent px-5 py-2.5 text-sm font-bold text-black hover:brightness-110"
   221→            >
   222→              Open stream picker
   223→            </button>
   224→            <button
   225→              type="button"
   226→              onClick={pop}
   227→              className="rounded-xl bg-raised px-5 py-2.5 text-sm font-semibold text-ink hover:bg-raised/70"
   228→            >
   229→              Back
   230→            </button>
   231→          </div>
   232→        </div>
   233→      ) : (
   234→        current.url && (
   235→          <VideoStage
   236→            key={current.url}
   237→            payload={current}
   238→            settings={settings}
   239→            onPhase={handlePhase}
   240→            onError={handleError}
   241→            onPlayNext={handlePlayNext}
   242→            onOpenSwitcher={openSwitcher}
   243→          />
   244→        )
   245→      )}
   246→
   247→      {/* Top bar */}
   248→      {phase !== "error" && (
   249→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   250→          <button
   251→            type="button"
   252→            onClick={() => {
   253→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   254→              pop();
   255→            }}
   256→            className="w-9 h-9 rounded-full bg-black/50 flex items-center justify-center text-ink hover:bg-black/70"
   257→            aria-label="Close player (Esc)"
   258→          >
   259→            <X className="w-5 h-5" />
   260→          </button>
   261→          <div className="min-w-0">
   262→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   263→            {current.episode && (
   264→              <p className="text-xs text-ink-muted">
   265→                S{current.season}:E{current.episode}
   266→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   267→              </p>
   268→            )}
   269→          </div>
   270→          <button
   271→            type="button"
   272→            onClick={() => setSwitcherOpen(true)}
   273→            className="ml-auto flex items-center gap-1.5 rounded-full bg-black/50 hover:bg-black/70 px-3 py-1.5 text-xs font-medium text-ink"
   274→            title="Switch stream (W)"
   275→          >
   276→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   277→          </button>
   278→        </div>
   279→      )}
   280→
   281→      {switcherOpen && (
   282→        <PickerOverlay
   283→          type={current.type}
   284→          id={current.deepLink?.id ?? current.metaId}
   285→          videoId={current.videoId ?? current.deepLink?.videoId}
   286→          season={current.season}
   287→          episode={current.episode}
   288→        />
   289→      )}
   290→    </div>
   291→  );
   292→}

   294→// ---------------- Video stage ----------------
   295→function VideoStage({
   296→  payload,
   297→  settings,
   298→  onPhase,
   299→  onError,
   300→  onPlayNext,
   301→  onOpenSwitcher,
   302→}: {
   303→  payload: PlayerPayload;
   304→  settings: ReturnType<typeof useSettings.getState>["settings"];
   305→  onPhase: (p: "loading" | "playing") => void;
   306→  onError: (msg: string) => void;
   307→  onPlayNext: (next: { season: number; episode: number; videoId: string; episodeName?: string }) => void;
   308→  onOpenSwitcher: () => void;
   309→}) {
   310→  const { toast } = useToast();
   311→  const videoRef = useRef<HTMLVideoElement>(null);
   312→  const containerRef = useRef<HTMLDivElement>(null);
   313→  const hlsRef = useRef<Hls | null>(null);
   314→  // Single source of truth for subtitle rendering — the custom overlay. This
   315→  // manager guarantees no native <track>/TextTrack layer ever paints alongside
   316→  // it (the root cause of the duplicate-subtitles bug).
   317→  const subManagerRef = useRef<SubtitleManager | null>(null);
   318→  if (!subManagerRef.current) subManagerRef.current = new SubtitleManager();
   319→  // Player-height-based subtitle scaling (measured via ResizeObserver)
   320→  const [stageHeight, setStageHeight] = useState(0);
   321→  const [ready, setReady] = useState(false);
   322→  const [playing, setPlaying] = useState(false);
   323→  const [muted, setMuted] = useState(false);
   324→  const [volume, setVolume] = useState(1);
   325→  const [time, setTime] = useState(0);
   326→  const [duration, setDuration] = useState(0);
   327→  const [buffering, setBuffering] = useState(true);
   328→  const [fullscreen, setFullscreen] = useState(false);
   329→  const [speed, setSpeed] = useState(1);
   330→  const [controlsVisible, setControlsVisible] = useState(true);
   331→  const [volumeHudVisible, setVolumeHudVisible] = useState(false);
   332→  const volumeHudTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   333→  const [subMenuOpen, setSubMenuOpen] = useState(false);
   334→  const [settingsMenuOpen, setSettingsMenuOpen] = useState(false);
   335→  // P2P engine: live swarm stats + one-shot remux fallback when the native stream can't demux
   336→  const [p2pStats, setP2pStats] = useState<{ progress: number; peers: number; downloadSpeed: number; ready: boolean } | null>(null);
   337→  const [srcOverride, setSrcOverride] = useState<string | null>(null);
   338→  const remuxTriedRef = useRef(false);
   339→  const [subtitles, setSubtitles] = useState<LoadedSubtitle[]>([]);
   340→  // Full ranked subtitle pool + how many are loaded — powers the "Load more" affordance
   341→  const subPoolRef = useRef<RawSubtitle[]>([]); // ranked by preferred languages
   342→  const subEffectiveRef = useRef<RawSubtitle[]>([]); // currently displayed ranking (lang filter)
   343→  const subAttemptedRef = useRef(0);
   344→  const [subPoolSize, setSubPoolSize] = useState(0);
   345→  const [subLoadingMore, setSubLoadingMore] = useState(false);
   346→  const [subLangFilter, setSubLangFilter] = useState<string | null>(null);
   347→  // Free-text search over the ranked pool (label / language / provider) — composes
   348→  // with the language chips: chip re-ranks, search then narrows.
   349→  const [subSearch, setSubSearch] = useState(""); // committed (drives the pool view)
   350→  const [subSearchInput, setSubSearchInput] = useState(""); // typing value
   351→  const subViewTokenRef = useRef(0); // invalidates stale parse batches when the view changes
   352→  const subSearchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   353→  // Language chips derived from the pool: chip code + available count
   354→  const subLangChips = useMemo(() => {
   355→    const counts = new Map<string, number>();
   356→    for (const s of subPoolRef.current) {
   357→      const chip = langChip(s.lang);
   358→      counts.set(chip, (counts.get(chip) ?? 0) + 1);
   359→    }
   360→    return Array.from(counts.entries())
   361→      .sort((a, b) => b[1] - a[1])
   362→      .slice(0, 6);
   363→  }, [subPoolSize, subLangFilter]);
   364→  // How many entries the CURRENT view (chip re-rank + search filter) contains
   365→  const [subEffectiveCount, setSubEffectiveCount] = useState(0);
   366→  const subtitlesRef = useRef<LoadedSubtitle[]>([]);
   367→  subtitlesRef.current = subtitles; // latest committed list for stable index math
   368→  const [activeSub, setActiveSub] = useState<number>(-1); // -1 = off
   369→  const [subCue, setSubCue] = useState<SubCue | null>(null);
   370→  const [availableTracks, setAvailableTracks] = useState<{ label: string }[]>([]);
   371→  const [levels, setLevels] = useState<{ index: number; label: string }[]>([]);
   372→  const [levelSel, setLevelSel] = useState<number>(-1); // -1 = adaptive (auto)
   373→  const [autoLevel, setAutoLevel] = useState<number>(-1); // level currently playing in auto mode
   374→  const [activeAudio, setActiveAudio] = useState(0);
   375→  const [bufferedEnd, setBufferedEnd] = useState(0);
   376→  const [nextEp, setNextEp] = useState<{
   377→    season: number;
   378→    episode: number;
   379→    videoId: string;
   380→    episodeName?: string;
   381→  } | null>(null);
   382→  const [upNextDismissed, setUpNextDismissed] = useState(false);
   383→  const controlsTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   384→  const seekHandled = useRef(false);
   385→  // Trakt scrobble state (dedupe: one scrobble per action per 10s)
   386→  const scrobbleStartSent = useRef(false);
   387→  const scrobbleLastSent = useRef<Record<"start" | "pause" | "stop", number>>({
   388→    start: 0,
   389→    pause: 0,
   390→    stop: 0,
   391→  });
   392→
   393→  const resumeAt = useRef(
   394→    payload.resumeAt ??
   395→      (settings.resumePlayback && payload.metaId
   396→        ? resumeMsFor(payload.metaId, payload.videoId) / 1000
   397→        : 0),
   398→  );
   399→
   400→  // ---- P2P swarm stats polling ----
   401→  useEffect(() => {
   402→    if (!payload.p2p) {
   403→      setP2pStats(null);
   404→      return;
   405→    }
   406→    let alive = true;
   407→    const key = payload.p2p.key;
   408→    const tick = async () => {
   409→      try {
   410→        const st = await p2pStatus(key);
   411→        if (!alive) return;
   412→        setP2pStats({
   413→          progress: st.progress ?? 0,
   414→          peers: st.peers ?? 0,
   415→          downloadSpeed: st.downloadSpeed ?? 0,
   416→          ready: st.ready === true,
   417→        });
   418→      } catch {
   419→        if (alive) setP2pStats(null);
   420→      }
   421→    };
   422→    void tick();
   423→    const iv = setInterval(tick, 2500);
   424→    return () => {
   425→      alive = false;
   426→      clearInterval(iv);
   427→    };
   428→  }, [payload.p2p?.key]);
   429→
   430→  // ---- Attach source (HLS or native) ----
   431→  useEffect(() => {
   432→    const video = videoRef.current;
   433→    const url = srcOverride ?? payload.url;
   434→    if (!video || !url) return;
   435→    let cancelled = false;
   436→    const isHls = /\.m3u8(\?|$)/i.test(url) || url.includes("m3u8");
   437→
   438→    const tryPlay = () => {
   439→      video.play().catch(() => {
   440→        /* autoplay blocked; user gesture needed */
   441→      });
   442→    };
   443→
   444→    if (isHls && !video.canPlayType("application/vnd.apple.mpegurl")) {
   445→      const hls = new Hls({
   446→        enableWorker: true,
   447→        maxBufferLength: 30,
   448→        // Subtitle rendering belongs to our custom overlay ONLY — hls.js must
   449→        // never flip embedded subtitle tracks to showing on its own.
   450→        renderTextTracksNatively: false,
   451→      });
   452→      hlsRef.current = hls;
   453→      hls.loadSource(url);
   454→      hls.attachMedia(video);
   455→      hls.on(Hls.Events.MANIFEST_PARSED, (_evt, data) => {
   456→        if (cancelled) return;
   457→        // Quality levels (video renditions) — dedupe by height, sort hi-fi first
   458→        const seen = new Set<number>();
   459→        const lvls: { index: number; label: string; h: number }[] = [];
   460→        (data?.levels ?? []).forEach((l, i) => {
   461→          const h = typeof l.height === "number" && l.height > 0 ? l.height : 0;
   462→          const label = h > 0 ? `${h}p` : `${Math.round(l.bitrate / 1000)} kbps`;
   463→          const key = h > 0 ? h : Math.round(l.bitrate / 1000);
   464→          if (seen.has(key)) return;
   465→          seen.add(key);
   466→          lvls.push({ index: i, label, h });
   467→        });
   468→        lvls.sort((a, b) => b.h - a.h);
   469→        setLevels(lvls.map(({ index, label }) => ({ index, label })));
   470→        // Alternative audio tracks served by the HLS manifest
   471→        const tracks = hls.audioTracks ?? [];
   472→        if (tracks.length > 1) {
   473→          setAvailableTracks(tracks.map((t, i) => ({ label: t.name || t.lang || `Audio ${i + 1}` })));
   474→        }
   475→        onPhase("loading");
   476→        tryPlay();
   477→      });
   478→      hls.on(Hls.Events.LEVEL_SWITCHED, (_evt, d) => {
   479→        if (!cancelled) setAutoLevel(d.level);
   480→      });
   481→      hls.on(Hls.Events.ERROR, (_evt, data) => {
   482→        if (cancelled || !data.fatal) return;
