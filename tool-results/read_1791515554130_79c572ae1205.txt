    47→    47→/** Semi-transparent top back button (owned by the detail page so it exists
    48→    48→ *  even when a title is deep-linked as the root nav frame). RTL-mirrors. */
    49→    49→function DetailBackButton() {
    50→    50→  const tr = useT();
    51→    51→  const pop = useNav((s) => s.pop);
    52→    52→  const resetTo = useNav((s) => s.resetTo);
    53→    53→  const depth = useNav((s) => s.stack.length);
    54→    54→  const goBack = useCallback(() => {
    55→    55→    if (depth <= 1) resetTo({ kind: "view", view: "home" });
    56→    56→    else pop();
    57→    57→  }, [depth, pop, resetTo]);
    58→    58→  return (
    59→    59→    <button
    60→    60→      type="button"
    61→    61→      onClick={goBack}
    62→    62→      className="harbor-tv-focus fixed top-4 start-4 z-40 flex h-10 items-center gap-2 rounded-full border border-white/15 bg-black/55 ps-3.5 pe-4 text-sm font-medium text-white backdrop-blur-md transition-all hover:scale-[1.03] hover:bg-black/75 active:scale-95 md-state"
    63→    63→      aria-label={tr("goBack")}
    64→    64→    >
    65→    65→      <ArrowLeft className="h-4 w-4 rtl:rotate-180" aria-hidden />
    66→    66→      {tr("back")}
    67→    67→    </button>
    68→    68→  );
    69→    69→}

    71→    71→export function DetailView({ type, id }: { type: string; id: string }) {
    72→    72→  const tr = useT();
    73→    73→  const { toast } = useToast();
    74→    74→  const push = useNav((s) => s.push);
    75→    75→  const addons = useAddons((s) => s.addons);
    76→    76→  const addonsLoaded = useAddons((s) => s.loaded);
    77→    77→  const settings = useSettings((s) => s.settings);
    78→    78→  // TMDB enrichment (logo, certification, recommendations) — fails soft
    79→    79→  const tmdb = useTmdbEnrichment(type, id, settings.region);
    80→    80→
    81→    81→  const [meta, setMeta] = useState<Meta | null>(null);
    82→    82→  const [loading, setLoading] = useState(true);
    83→    83→  const [error, setError] = useState<string | null>(null);
    84→    84→  const [inList, setInList] = useState(false);
    85→    85→  const [resume, setResume] = useState(0);
    86→    86→  const [cwEntry, setCwEntry] = useState<ReturnType<typeof getCwEntry>>(null);
    87→    87→
    88→    88→  useEffect(() => {
    89→    89→    let alive = true;
    90→    90→    setLoading(true);
    91→    91→    setError(null);
    92→    92→    setMeta(null);
    93→    93→    (async () => {
    94→    94→      try {
    95→    95→        let result = await fetchMeta(type, id);
    96→    96→        // If cinemeta misses, try addon metas
    97→    97→        if (!result && addonsLoaded) {
    98→    98→          for (const addon of addons.filter((a) => a.enabled)) {
    99→    99→            try {
   100→   100→              const m = await fetchAddonMeta(addon, type, id);
   101→   101→              if (m) {
   102→   102→                result = m;
   103→   103→                break;
   104→   104→              }
   105→   105→            } catch {
   106→   106→              /* try next */
   107→   107→            }
   108→   108→          }
   109→   109→        }
   110→   110→        if (!alive) return;
   111→   111→        if (!result) {
   112→   112→          setError(tr("titleNotFound"));
   113→   113→        } else {
   114→   114→          setMeta(result);
   115→   115→          setInList(isInWatchlist(id));
   116→   116→          setResume(resumeMsFor(id));
   117→   117→          setCwEntry(getCwEntry(id));
   118→   118→        }
   119→   119→      } catch (e) {
   120→   120→        if (!alive) return;
   121→   121→        setError(e instanceof Error ? e.message : tr("fetchFailed"));
   122→   122→      } finally {
   123→   123→        if (alive) setLoading(false);
   124→   124→      }
   125→   125→    })();
   126→   126→    return () => {
   127→   127→      alive = false;
   128→   128→    };
   129→   129→    // eslint-disable-next-line react-hooks/exhaustive-deps
   130→   130→  }, [type, id, addons, addonsLoaded]);
   131→   131→
   132→   132→  // Next episode to resume (series)
   133→   133→  const nextVideo: MetaVideo | null = useMemo(() => {
   134→   134→    if (!meta || meta.type !== "series" || !meta.videos?.length) return null;
   135→   135→    if (cwEntry?.season && cwEntry?.episode) {
   136→   136→      const exact = meta.videos.find(
   137→   137→        (v) => v.season === cwEntry.season && v.episode === cwEntry.episode,
   138→   138→      );
   139→   139→      if (exact && resume > 0) return exact;
   140→   140→    }
   141→   141→    return null;
   142→   142→  }, [meta, cwEntry, resume]);
   143→   143→
   144→   144→  const onPlay = useCallback(
   145→   145→    (video?: MetaVideo) => {
   146→   146→      if (!meta) return;
   147→   147→      // Duration hint for the playback timeline (approximate until the real
   148→   148→      // container/ffprobe duration lands): TMDB runtime first (minutes), then
   149→   149→      // the Stremio meta runtime string. Series: per-episode runtime when the
   150→   150→      // addon provides one, else the show average / TMDB episode_run_time.
   151→   151→      const tmdbRuntimeMin =
   152→   152→        meta.type === "series"
   153→   153→          ? (tmdb.details?.episode_run_time?.[0] ?? null)
   154→   154→          : (tmdb.details?.runtime ?? null);
   155→   155→      const runtimeSeconds =
   156→   156→        parseRuntimeToSeconds(video?.duration ?? null) ??
   157→   157→        (tmdbRuntimeMin != null ? Math.round(tmdbRuntimeMin * 60) : null) ??
   158→   158→        parseRuntimeToSeconds(meta.runtime);
   159→   159→      if (settings.instantPlay) {
   160→   160→        push({
   161→   161→          kind: "player",
   162→   162→          payload: {
   163→   163→            url: "", // resolved by picker
   164→   164→            title: meta.name,
   165→   165→            type: meta.type,
   166→   166→            metaId: meta.id,
   167→   167→            poster: meta.poster,
   168→   168→            season: video?.season,
   169→   169→            episode: video?.episode,
   170→   170→            videoId: video?.id,
   171→   171→            episodeName: video?.name ?? video?.title,
   172→   172→            deepLink: { type: meta.type, id: meta.id, videoId: video?.id },
   173→   173→            runtimeSeconds: runtimeSeconds ?? undefined,
   174→   174→          },
   175→   175→        });
   176→   176→      } else {
   177→   177→        push({
   178→   178→          kind: "picker",
   179→   179→          type: meta.type,
   180→   180→          id: meta.id,
   181→   181→          videoId: video?.id,
   182→   182→          season: video?.season,
   183→   183→          episode: video?.episode,
   184→   184→          runtimeSeconds: runtimeSeconds ?? undefined,
   185→   185→        });
   186→   186→      }
   187→   187→    },
   188→   188→    [meta, tmdb.details, push, settings.instantPlay],
   189→   189→  );
   190→   190→
   191→   191→  /* ---------- expandable secondary menu ---------- */
   192→   192→  const [menuOpen, setMenuOpen] = useState(false);
   193→   193→  const menuWrapRef = useRef<HTMLDivElement>(null);
   194→   194→
   195→   195→  // Close on outside pointer + Escape; return focus to the trigger on Esc.
   196→   196→  useEffect(() => {
   197→   197→    if (!menuOpen) return;
   198→   198→    const onDown = (e: PointerEvent) => {
   199→   199→      if (menuWrapRef.current && !menuWrapRef.current.contains(e.target as Node)) {
   200→   200→        setMenuOpen(false);
   201→   201→      }
   202→   202→    };
   203→   203→    const onKey = (e: KeyboardEvent) => {
   204→   204→      if (e.key === "Escape") {
   205→   205→        e.stopPropagation();
   206→   206→        setMenuOpen(false);
   207→   207→        menuWrapRef.current?.querySelector<HTMLButtonElement>("button")?.focus();
   208→   208→      }
   209→   209→    };
   210→   210→    window.addEventListener("pointerdown", onDown);
   211→   211→    window.addEventListener("keydown", onKey, true);
   212→   212→    return () => {
   213→   213→      window.removeEventListener("pointerdown", onDown);
   214→   214→      window.removeEventListener("keydown", onKey, true);
   215→   215→    };
   216→   216→  }, [menuOpen]);
   217→   217→
   218→   218→  const openStreams = useCallback(() => {
   219→   219→    setMenuOpen(false);
   220→   220→    if (!meta) return;
   221→   221→    if (meta.type === "series" && nextVideo) {
   222→   222→      push({
   223→   223→        kind: "picker",
   224→   224→        type: meta.type,
   225→   225→        id: meta.id,
   226→   226→        videoId: nextVideo.id,
   227→   227→        season: nextVideo.season,
   228→   228→        episode: nextVideo.episode,
   229→   229→      });
   230→   230→    } else {
   231→   231→      push({ kind: "picker", type: meta.type, id: meta.id });
   232→   232→    }
   233→   233→  }, [meta, nextVideo, push]);
   234→   234→
   235→   235→  const toggleWatchlistFromMenu = useCallback(() => {
   236→   236→    if (!meta) return;
   237→   237→    setInList(
   238→   238→      toggleWatchlist({
   239→   239→        id: meta.id,
   240→   240→        type: meta.type,
   241→   241→        name: meta.name,
   242→   242→        poster: meta.poster,
   243→   243→        releaseInfo: meta.releaseInfo,
   244→   244→        imdbRating: meta.imdbRating,
   245→   245→      }),
   246→   246→    );
   247→   247→  }, [meta]);
   248→   248→
   249→   249→  const onShare = useCallback(async () => {
   250→   250→    setMenuOpen(false);
   251→   251→    if (!meta) return;
   252→   252→    const url = window.location.href;
   253→   253→    try {
   254→   254→      if (typeof navigator.share === "function") {
   255→   255→        await navigator.share({ title: meta.name, url });
   256→   256→        return; // native sheet finished (dismiss throws AbortError → caught below)
   257→   257→      }
   258→   258→    } catch (e) {
   259→   259→      if (e instanceof DOMException && e.name === "AbortError") return;
   260→   260→      // fall through to the clipboard fallback
   261→   261→    }
   262→   262→    try {
   263→   263→      await navigator.clipboard.writeText(url);
   264→   264→      toast({ title: tr("linkCopied") });
   265→   265→    } catch {
   266→   266→      toast({ title: tr("copyFailed") });
   267→   267→    }
   268→   268→  }, [meta, toast, tr]);
   269→   269→
   270→   270→  // The back button is fixed chrome of this page — render in every state.

   275→   275→      <div className="px-4 pt-[72px] pb-16 md:px-8 md:pt-20">
   276→   276→        {back}
   277→   277→        {/* hero card skeleton */}

   289→   289→          <div className="harbor-skeleton mx-auto h-4 w-3/4 rounded-lg" />
   290→   290→        </div>
   291→   291→        {/* action row skeleton */}
   292→   292→        <div className="mt-7 flex items-center justify-center gap-3" aria-hidden>
   293→   293→          <div className="harbor-skeleton h-[52px] w-44 rounded-full" />
   294→   294→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   295→   295→          <div className="harbor-skeleton h-[52px] w-[52px] rounded-full" />
   296→   296→        </div>
   297→   297→        <div className="sr-only">{tr("loadingDetails")}</div>

   317→   317→  const runtime = meta.runtime?.trim() || null;
   318→   318→  const genre = meta.genres?.find((g) => g && g.trim().length > 0) ?? null;

   323→   323→      {back}
   324→   324→
   325→   325→      {/* ================= 1 · HERO ARTWORK CARD =================
   326→   326→          The official logo lives INSIDE the card's bottom scrim. No chips,
   327→   327→          ratings, text or buttons float over the artwork anymore. */}
   328→   328→      <section className="px-4 pt-[72px] md:px-8 md:pt-20" aria-label={meta.name}>
   329→   329→        <div className="harbor-pop-in relative mx-auto aspect-[16/11] max-w-5xl overflow-hidden rounded-[28px] border border-edge-soft bg-raised shadow-[0_36px_90px_-36px_rgba(0,0,0,0.9)] sm:aspect-[16/8] md:aspect-[21/8] md:rounded-[36px]">
   330→   330→          <PosterImage
   331→   331→            src={tmdb.backdrop ?? meta.background ?? meta.poster}
   332→   332→            alt={meta.name}
   333→   333→            className="absolute inset-0"
   334→   334→            landscape
   335→   335→          />
   336→   336→          {/* Bottom scrim — legibility bed for the embedded logo */}
   337→   337→          <div className="absolute inset-x-0 bottom-0 h-3/5 bg-gradient-to-t from-black/90 via-black/45 to-transparent" />
   338→   338→          {/* Embedded official title logo (sr-only h1 keeps document outline) */}
   339→   339→          <div className="absolute inset-x-0 bottom-0 flex items-end justify-center px-5 pb-5 md:pb-9">
   340→   340→            {tmdb.logo || meta.logo ? (
   341→   341→              <>
   342→   342→                <h1 className="sr-only">{meta.name}</h1>
   343→   343→                <img
   344→   344→                  src={tmdb.logo ?? meta.logo}
   345→   345→                  alt=""
   346→   346→                  className="max-h-14 w-auto max-w-[78%] object-contain drop-shadow-[0_10px_28px_rgba(0,0,0,0.95)] transition-transform duration-500 sm:max-h-16 md:max-h-24"
   347→   347→                  loading="lazy"
   348→   348→                />
   349→   349→              </>
   350→   350→            ) : (
   351→   351→              <h1 className="font-display text-center text-3xl font-bold tracking-tight text-white drop-shadow-[0_6px_20px_rgba(0,0,0,0.9)] md:text-5xl">

   359→   359→      {/* ============ 2 · SINGLE-LINE METADATA ROW ============

   364→   364→          {/* Outer scroller + inner w-max row: centers when it fits, scrolls
   365→   365→              from the true start edge when it doesn't (no justify-center

   647→   647→function SimilarRail({
   648→   648→  type,
   649→   649→  genres,
   650→   650→  currentId,
   651→   651→}: {
   652→   652→  type: string;
   653→   653→  genres?: string[];
   654→   654→  currentId: string;
   655→   655→}) {
   656→   656→  const tr = useT();
   657→   657→  const push = useNav((s) => s.push);
   658→   658→  const genre = genres?.find((g) => g && g.trim().length > 0);
   659→   659→  // Data is keyed by the genre it was fetched for, so a genre change renders a
   660→   660→  // skeleton without a synchronous setState inside the effect.
   661→   661→  const [data, setData] = useState<{ genre: string; items: Meta[] } | null>(null);
   662→   662→
   663→   663→  useEffect(() => {
   664→   664→    if (!genre) return;
   665→   665→    let alive = true;
   666→   666→    (async () => {
   667→   667→      try {
   668→   668→        const res = await fetchCinemetaCatalog(
   669→   669→          type === "series" ? "series" : "movie",
   670→   670→          "top",
   671→   671→          { genre },
   672→   672→        );
   673→   673→        if (!alive) return;
   674→   674→        setData({ genre, items: res.filter((m) => m.id !== currentId).slice(0, 14) });
   675→   675→      } catch {
   676→   676→        if (alive) setData({ genre, items: [] });
   677→   677→      }
   678→   678→    })();
   679→   679→    return () => {
   680→   680→      alive = false;
   681→   681→    };
   682→   682→  }, [type, genre, currentId]);
   683→   683→
   684→   684→  if (!genre) return null;
   685→   685→  const items: Meta[] | null = data && data.genre === genre ? data.items : null;
   686→   686→
   687→   687→  return (
   688→   688→    <section className="mt-12" aria-label={tr("moreLikeThis")}>
   689→   689→      <div className="mb-3 flex items-center gap-2 px-4 md:px-8">
   690→   690→        <Layers className="h-5 w-5 text-accent" aria-hidden />
   691→   691→        <h2 className="md-title-medium text-ink">{tr("moreLikeThis")}</h2>
   692→   692→        <span className="md-chip md-label-small !h-6 !px-2 cursor-default!">{genre}</span>
   693→   693→      </div>
   694→   694→      {items === null ? (
   695→   695→        <div className="flex gap-3 overflow-hidden px-4 md:px-8" aria-hidden>
   696→   696→          {Array.from({ length: 8 }).map((_, i) => (
   697→   697→            <div key={i} className="harbor-skeleton aspect-[2/3] w-[112px] shrink-0 rounded-xl md:w-[136px]" />
   698→   698→          ))}
   699→   699→        </div>
   700→   700→      ) : items.length === 0 ? null : (
   701→   701→        /* Hover headroom: same pt-3/-mt-3 contract as TmdbRecsRail above */
   702→   702→        <div className="harbor-scroll-x overflow-x-auto flex gap-3 px-4 pb-2 pt-3 -mt-3 md:px-8">
   703→   703→          {items.map((m) => (
   704→   704→            <div key={`${m.id}`} className="w-[104px] shrink-0 md:w-[126px]">
   705→   705→              <MetaCard
   706→   706→                meta={m}
   707→   707→                onOpen={() => push({ kind: "detail", type: m.type || type, id: m.id })}
   708→   708→              />
   709→   709→            </div>
   710→   710→          ))}
   711→   711→        </div>
   712→   712→      )}
   713→   713→    </section>
   714→   714→  );
   715→   715→}
