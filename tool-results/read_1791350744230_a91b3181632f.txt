   299→function pushFrame(cur: Frame[], next: Frame): Frame[] {
   300→  const out = [...cur, next];
   301→  if (out.length <= STACK_MAX) return out;
   302→  return [out[0], ...out.slice(out.length - STACK_MAX + 1)];
   303→}

   312→function frameKey(f: Frame): string {
   313→  switch (f.kind) {
   314→    case "home":
   315→      return "home";
   316→    case "settings":
   317→      return "settings";
   318→    case "anime":
   319→      return "anime";
   320→    case "discover":
   321→      return "discover";
   322→    case "catalogs":
   323→      return "catalogs";
   324→    case "addons":
   325→      return "addons";
   326→    case "addon-detail":
   327→      return `addon-detail:${f.id}`;
   328→    case "calendar":
   329→      return "calendar";
   330→    case "wrapped":
   331→      return "wrapped";
   332→    case "queue":
   333→      return "queue";
   334→    case "movies":
   335→      return "movies";
   336→    case "shows":
   337→      return "shows";
   338→    case "kids":
   339→      return "kids";
   340→    case "library":
   341→      return "library";
   342→    case "live":
   343→      return "live";
   344→    case "vod":
   345→      return "vod";
   346→    case "downloads":
   347→      return "downloads";
   348→    case "service":
   349→      return `service:${f.service}`;
   350→    case "meta":
   351→      return `meta:${f.meta.id}`;
   352→    case "episode-detail":
   353→      return `episode-detail:${f.seriesId}:${f.season}:${f.episode}`;
   354→    case "person":
   355→      return `person:${f.id}`;
   356→    case "collection":
   357→      return `collection:${f.id}`;
   358→    case "collections":
   359→      return "collections";
   360→    case "filter":
   361→      return `filter:${f.filter.kind}:${f.filter.mediaType}:${"name" in f.filter ? f.filter.name : f.filter.value}`;
   362→    case "grid":
   363→      return `grid:${f.grid.title}`;
   364→    case "award":
   365→      return `award:${f.awardType}`;
   366→    case "anime-award":
   367→      return `anime-award:${f.sourceId}`;
   368→    case "picker": {
   369→      const a = typeof f.attempt === "number" ? `:a${f.attempt}` : "";
   370→      return f.episode
   371→        ? `picker:${f.meta.id}:${f.episode.season}:${f.episode.episode}${a}`
   372→        : `picker:${f.meta.id}${a}`;
   373→    }
   374→    case "player":
   375→      return `player:${f.src.meta.id}:${f.src.url.slice(-32)}`;
   376→    case "match-detail":
   377→      return `match-detail:${f.game.id}`;
   378→  }
   379→}

   381→function syncFrameKey(f: Frame): string {
   382→  if (f.kind === "player") {
   383→    const id = f.src.meta.id || `local:${f.src.url.slice(-32)}`;
   384→    const ep = f.src.episode;
   385→    return ep ? `player:${id}:${ep.season}:${ep.episode}` : `player:${id}`;
   386→  }
   387→  if (f.kind === "picker") {
   388→    return f.episode
   389→      ? `picker:${f.meta.id}:${f.episode.season}:${f.episode.episode}`
   390→      : `picker:${f.meta.id}`;
   391→  }
   392→  return frameKey(f);
   393→}

   395→function lastOfKind<K extends Frame["kind"]>(
   396→  frames: Frame[],
   397→  kind: K,
   398→): Extract<Frame, { kind: K }> | undefined {
   399→  for (let i = frames.length - 1; i >= 0; i--) {
   400→    const f = frames[i];
   401→    if (f.kind === kind) return f as Extract<Frame, { kind: K }>;
   402→  }
   403→  return undefined;
   404→}

  1107→function anchorOffsetIn(scrollEl: HTMLElement, anchor: HTMLElement): number {
  1108→  return (
  1109→    anchor.getBoundingClientRect().top - scrollEl.getBoundingClientRect().top + scrollEl.scrollTop
  1110→  );
  1111→}

  1113→function pickAnchor(el: HTMLElement, scrollTop: number): { key: string; delta: number } | null {
  1114→  const anchors = el.querySelectorAll<HTMLElement>("[data-scroll-anchor]");
  1115→  if (anchors.length === 0) return null;
  1116→  let best: { key: string; offset: number } | null = null;
  1117→  for (const a of anchors) {
  1118→    const k = a.dataset.scrollAnchor;
  1119→    if (!k) continue;
  1120→    const off = anchorOffsetIn(el, a);
  1121→    if (off > scrollTop + 1) continue;
  1122→    if (!best || off > best.offset) best = { key: k, offset: off };
  1123→  }
  1124→  if (!best) {
  1125→    const first = anchors[0];
  1126→    const k = first.dataset.scrollAnchor;
  1127→    if (!k) return null;
  1128→    return { key: k, delta: scrollTop - anchorOffsetIn(el, first) };
  1129→  }
  1130→  return { key: best.key, delta: scrollTop - best.offset };
  1131→}

  1133→function targetForSnap(el: HTMLElement, snap: ScrollSnapshot): number | null {
  1134→  if (snap.anchor) {
  1135→    const sel = `[data-scroll-anchor="${CSS.escape(snap.anchor)}"]`;
  1136→    const anchorEl = el.querySelector<HTMLElement>(sel);
  1137→    if (anchorEl) return Math.max(0, anchorOffsetIn(el, anchorEl) + snap.delta);
  1138→  }
  1139→  return snap.fallback >= 0 ? snap.fallback : null;
  1140→}

  1142→export function useScrollMemory(
  1143→  key: string,
  1144→  ref: RefObject<HTMLElement | null>,
  1145→  active: boolean = true,
  1146→) {
  1147→  const { rememberScroll, recallScroll } = useView();
  1148→
  1149→  useEffect(() => {
  1150→    const onReset = (e: Event) => {
  1151→      const detail = (e as CustomEvent<{ view?: string }>).detail;
  1152→      if (!detail?.view) return;
  1153→      if (detail.view !== key) return;
  1154→      const el = ref.current;
  1155→      if (!el) return;
  1156→      el.scrollTo({ top: 0, left: 0, behavior: "auto" });
  1157→    };
  1158→    window.addEventListener("harbor:scroll-top", onReset);
  1159→    return () => window.removeEventListener("harbor:scroll-top", onReset);
  1160→  }, [key, ref]);
  1161→
  1162→  useLayoutEffect(() => {
  1163→    const el = ref.current;
  1164→    if (!el || !active) return;
  1165→
  1166→    let restoring = true;
  1167→    let settleId: number | null = null;
  1168→    let saveTimer: number | null = null;
  1169→
  1170→    const cancelSettle = () => {
  1171→      if (settleId !== null) {
  1172→        clearTimeout(settleId);
  1173→        settleId = null;
  1174→      }
  1175→    };
  1176→
  1177→    const tryRestore = () => {
  1178→      if (!restoring) return;
  1179→      const snap = recallScroll(key);
  1180→      if (!snap) {
  1181→        restoring = false;
  1182→        cancelSettle();
  1183→        return;
  1184→      }
  1185→      if (el.clientHeight === 0) return;
