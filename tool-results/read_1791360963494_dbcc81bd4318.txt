   283→   283→function ThemePanel() {
   284→   284→  const settings = useSettings((s) => s.settings);
   285→   285→  const update = useSettings((s) => s.update);
   286→   286→  const fileRef = useRef<HTMLInputElement>(null);
   287→   287→  const { toast } = useToast();
   288→   288→  const [studioOpen, setStudioOpen] = useState(false);
   289→   289→  const [userThemes, setUserThemes] = useState<UserTheme[]>([]);
   290→   290→
   291→   291→  // Load saved themes after mount (client-only storage; avoids hydration mismatch)
   292→   292→  useEffect(() => {
   293→   293→    const t = setTimeout(() => setUserThemes(loadUserThemes()), 0);
   294→   294→    return () => clearTimeout(t);
   295→   295→  }, []);
   296→   296→
   297→   297→  const setPreset = (preset: string) => {
   298→   298→    update({ theme: { ...settings.theme, preset, customColors: null, customName: null } });
   299→   299→  };
   300→   300→
   301→   301→  const applyUserTheme = (t: UserTheme) => {
   302→   302→    update({
   303→   303→      theme: {
   304→   304→        ...settings.theme,
   305→   305→        preset: "custom",
   306→   306→        customColors: t.colors,
   307→   307→        fontPair: t.fontPair,
   308→   308→        customLayout: t.layout,
   309→   309→        customCardStyle: t.cardStyle,
   310→   310→        customButtonStyle: t.buttonStyle,
   311→   311→        customName: t.name,
   312→   312→      },
   313→   313→    });
   314→   314→  };
   315→   315→
   316→   316→  const removeUserTheme = (id: string) => {
   317→   317→    const next = deleteUserTheme(id);
   318→   318→    setUserThemes(next);
   319→   319→    // If the deleted theme was active, fall back to the default preset
   320→   320→    if (settings.theme.preset === "custom") {
   321→   321→      const stillExists = next.some((t) => t.name === settings.theme.customName);
   322→   322→      if (!stillExists) {
   323→   323→        update({ theme: { ...settings.theme, preset: "cool-grey", customColors: null, customName: null } });
   324→   324→      }
   325→   325→    }
   326→   326→  };
   327→   327→
   328→   328→  const onBgUpload = (file: File) => {
   329→   329→    if (!file.type.startsWith("image/")) return;
   330→   330→    if (file.size > 3 * 1024 * 1024) {
   331→   331→      toast({ title: "Image too large", description: "Use an image under 3 MB.", variant: "destructive" });
   332→   332→      return;
   333→   333→    }
   334→   334→    const reader = new FileReader();
   335→   335→    reader.onload = () => {
   336→   336→      update({ theme: { ...settings.theme, backgroundImage: reader.result as string } });
   337→   337→    };
   338→   338→    reader.readAsDataURL(file);
   339→   339→  };
   340→   340→
   341→   341→  return (
   342→   342→    <div className="space-y-7 max-w-4xl">
   343→   343→      {/* Theme Studio */}
   344→   344→      <div className="rounded-3xl border border-accent/30 bg-accent-soft/50 p-5">
   345→   345→        <div className="flex items-center justify-between gap-4 flex-wrap">
   346→   346→          <div className="min-w-0">
   347→   347→            <h2 className="font-display text-lg font-bold text-ink flex items-center gap-2">
   348→   348→              <Brush className="w-[18px] h-[18px] text-accent" /> Theme Studio
   349→   349→            </h2>
   350→   350→            <p className="text-xs text-ink-muted mt-0.5">
   351→   351→              Build a fully custom palette, fonts and layout — with live preview.
   352→   352→              {settings.theme.preset === "custom" && settings.theme.customName && (
   353→   353→                <span className="text-accent font-medium"> Active: {settings.theme.customName}</span>
   354→   354→              )}
   355→   355→            </p>
   356→   356→          </div>
   357→   357→          <Button onClick={() => setStudioOpen(true)} className="bg-accent text-black hover:brightness-110">
   358→   358→            <Brush className="w-4 h-4 mr-1.5" /> Open Theme Studio
   359→   359→          </Button>
   360→   360→        </div>
   361→   361→        {userThemes.length > 0 && (
   362→   362→          <div className="mt-4">
   363→   363→            <p className="text-[10px] uppercase tracking-wide text-ink-subtle mb-2">Your saved themes</p>
   364→   364→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2">
   365→   365→              {userThemes.map((t) => (
   366→   366→                <div
   367→   367→                  key={t.id}
   368→   368→                  className={cn(
   369→   369→                    "group relative rounded-xl border p-2.5 cursor-pointer transition-all hover:scale-[1.02]",
   370→   370→                    settings.theme.preset === "custom" && settings.theme.customName === t.name
   371→   371→                      ? "border-accent"
   372→   372→                      : "border-edge-soft",
   373→   373→                  )}
   374→   374→                  style={{ background: t.colors.canvas }}
   375→   375→                  onClick={() => applyUserTheme(t)}
   376→   376→                  role="button"
   377→   377→                  tabIndex={0}
   378→   378→                  onKeyDown={(e) => e.key === "Enter" && applyUserTheme(t)}
   379→   379→                  aria-label={`Apply theme ${t.name}`}
   380→   380→                >
   381→   381→                  <div className="flex items-center justify-between">
   382→   382→                    <div className="flex gap-1">
   383→   383→                      {[t.colors.canvas, t.colors.accent, t.colors.elevated].map((c, i) => (
   384→   384→                        <span key={i} className="w-3.5 h-3.5 rounded-full border border-white/20" style={{ background: c }} />
   385→   385→                      ))}
   386→   386→                    </div>
   387→   387→                    <button
   388→   388→                      type="button"
   389→   389→                      onClick={(e) => {
   390→   390→                        e.stopPropagation();
   391→   391→                        removeUserTheme(t.id);
   392→   392→                      }}
   393→   393→                      className="opacity-0 group-hover:opacity-100 text-white/50 hover:text-danger transition-all"
   394→   394→                      aria-label={`Delete theme ${t.name}`}
   395→   395→                    >
   396→   396→                      <Trash2 className="w-3.5 h-3.5" />
   397→   397→                    </button>
   398→   398→                  </div>
   399→   399→                  <p className="text-xs font-semibold mt-1.5 truncate" style={{ color: t.colors.accent }}>
   400→   400→                    {t.name}
   401→   401→                  </p>
   402→   402→                  <p className="text-[10px] text-white/50 capitalize">{t.layout} · {t.fontPair.split("-")[0]}</p>
   403→   403→                </div>
   404→   404→              ))}
   405→   405→            </div>
   406→   406→          </div>
   407→   407→        )}
   408→   408→      </div>
   409→   409→
   410→   410→      <div>
   411→   411→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Theme presets</h2>
   412→   412→        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3">
   413→   413→          {THEME_PRESETS.map((p) => (
   414→   414→            <button
   415→   415→              key={p.id}
   416→   416→              type="button"
   417→   417→              onClick={() => setPreset(p.id)}
   418→   418→              className={cn(
   419→   419→                "harbor-tv-focus rounded-2xl border-2 p-3 text-left transition-all hover:scale-[1.02]",
   420→   420→                settings.theme.preset === p.id ? "border-accent" : "border-edge-soft",
   421→   421→              )}
   422→   422→              style={{ background: p.canvas }}
   423→   423→              aria-pressed={settings.theme.preset === p.id}
   424→   424→            >
   425→   425→              <div className="flex items-center justify-between mb-2.5">
   426→   426→                <div className="flex gap-1.5">
   427→   427→                  {p.swatch.map((c) => (
   428→   428→                    <span key={c} className="w-4 h-4 rounded-full border border-white/20" style={{ background: c }} />
   429→   429→                  ))}

   566→   566→function DataPanel() {
   567→   567→  const { toast } = useToast();
   568→   568→  const settings = useSettings((s) => s.settings);
   569→   569→  const fileRef = useRef<HTMLInputElement>(null);
   570→   570→
   571→   571→  const exportBackup = () => {
   572→   572→    const data: Record<string, string> = {};
   573→   573→    for (let i = 0; i < window.localStorage.length; i++) {
   574→   574→      const key = window.localStorage.key(i);
   575→   575→      if (key?.startsWith("harbor-web.")) {
   576→   576→        const v = window.localStorage.getItem(key);
   577→   577→        if (v) data[key] = v;
   578→   578→      }
   579→   579→    }
   580→   580→    const backup = {
   581→   581→      format: "harbor-web-backup",
   582→   582→      version: 1,
   583→   583→      exportedAt: new Date().toISOString(),
   584→   584→      data,
   585→   585→    };
   586→   586→    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: "application/json" });
   587→   587→    const url = URL.createObjectURL(blob);
   588→   588→    const a = document.createElement("a");
   589→   589→    a.href = url;
   590→   590→    a.download = `harbor-web-backup-${new Date().toISOString().slice(0, 10)}.harbx`;
   591→   591→    a.click();
   592→   592→    URL.revokeObjectURL(url);
   593→   593→    toast({ title: "Backup exported", description: `${Object.keys(data).length} keys saved.` });
   594→   594→  };
   595→   595→
   596→   596→  const importBackup = async (file: File) => {
   597→   597→    try {
   598→   598→      const text = await file.text();
   599→   599→      const parsed = JSON.parse(text) as { format?: string; data?: Record<string, string> };
   600→   600→      if (parsed.format !== "harbor-web-backup" || !parsed.data) {
   601→   601→        throw new Error("Not a Harbor Web backup file");
   602→   602→      }
   603→   603→      let restored = 0;
   604→   604→      for (const [key, value] of Object.entries(parsed.data)) {
   605→   605→        if (key.startsWith("harbor-web.")) {
   606→   606→          window.localStorage.setItem(key, value);
   607→   607→          restored++;
   608→   608→        }
   609→   609→      }
   610→   610→      toast({ title: "Backup restored", description: `${restored} keys restored. Reloading…` });
   611→   611→      setTimeout(() => window.location.reload(), 900);
   612→   612→    } catch (e) {
   613→   613→      toast({
   614→   614→        title: "Restore failed",
   615→   615→        description: e instanceof Error ? e.message : "Invalid file",
   616→   616→        variant: "destructive",
   617→   617→      });
   618→   618→    }
   619→   619→  };
   620→   620→
   621→   621→  const clearData = () => {
   622→   622→    const keys: string[] = [];
   623→   623→    for (let i = 0; i < window.localStorage.length; i++) {
   624→   624→      const key = window.localStorage.key(i);
   625→   625→      if (key?.startsWith("harbor-web.")) keys.push(key);
   626→   626→    }
   627→   627→    keys.forEach((k) => window.localStorage.removeItem(k));
   628→   628→    toast({ title: "Local data cleared", description: "Reloading…" });
   629→   629→    setTimeout(() => window.location.reload(), 900);
   630→   630→  };
   631→   631→
   632→   632→  return (
   633→   633→    <div className="space-y-3 max-w-3xl">
   634→   634→      <CloudSyncCard />
   635→   635→      <SettingRow title="Export backup" description={`Save settings, addons, watchlist (${getWatchlist().length} items) to a .harbx file`}>
   636→   636→        <Button onClick={exportBackup} className="bg-accent text-black hover:brightness-110">
   637→   637→          Export
   638→   638→        </Button>
   639→   639→      </SettingRow>
   640→   640→      <SettingRow title="Restore backup" description="Import a .harbx backup file">
   641→   641→        <Button variant="outline" onClick={() => fileRef.current?.click()} className="border-edge text-ink-muted">
   642→   642→          Restore
   643→   643→        </Button>
   644→   644→        <input
   645→   645→          ref={fileRef}
   646→   646→          type="file"
   647→   647→          accept=".harbx,.json"
   648→   648→          className="hidden"
   649→   649→          onChange={(e) => e.target.files?.[0] && importBackup(e.target.files[0])}
   650→   650→          aria-label="Import backup file"
   651→   651→        />
   652→   652→      </SettingRow>
   653→   653→      <SettingRow title="Clear local data" description="Remove all Harbor Web data from this browser">
   654→   654→        <Button variant="outline" onClick={clearData} className="border-danger/40 text-danger hover:bg-danger/10">
   655→   655→          Clear
   656→   656→        </Button>
   657→   657→      </SettingRow>
   658→   658→      <SettingRow
   659→   659→        title="Current settings size"
   660→   660→        description={`${(JSON.stringify(settings).length / 1024).toFixed(1)} KB in localStorage`}
   661→   661→      >
   662→   662→        <span />
   663→   663→      </SettingRow>

   900→   900→      alive = false;

   902→   902→    };

   927→   927→    }

   930→   930→  if (!loaded) return null;

   952→   952→              toast({ title: "Trakt disconnected" });

   955→   955→          >
   956→   956→            <Unplug className="w-3.5 h-3.5" /> Disconnect
   957→   957→          </button>

   960→   960→
   961→   961→      {!auth && phase !== "connecting" && (

   966→   966→                Client ID

  1016→  1016→
  1017→  1017→      {phase === "connecting" && device && (

  1051→  1051→
