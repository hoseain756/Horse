    80→/** "{Series|Movie|Anime}" for the hero meta line. Animation-genre series are
    81→ *  labelled Anime (matches the app's Anime section semantics). */
    82→export function metaTypeLabel(type: string, genres: string[] | undefined, lang: string): string {
    83→  const ar = isArabic(lang);
    84→  const list = (genres ?? []).map((g) => g.toLowerCase());
    85→  if (type !== "movie" && (list.includes("anime") || list.includes("animation"))) {
    86→    return ar ? "أنمي" : "Anime";
    87→  }
    88→  if (type === "movie") return ar ? "فيلم" : "Movie";
    89→  return ar ? "مسلسل" : "Series";
    90→}

   104→/** "1h 20m left" / "تبقى ١س ٢٠د"-style remaining time (movies). */
   105→export function remainingLabel(ms: number, lang: string): string | null {
   106→  if (!Number.isFinite(ms) || ms <= 60_000) return null;
   107→  const totalMin = Math.round(ms / 60_000);
   108→  const h = Math.floor(totalMin / 60);
   109→  const m = totalMin % 60;
   110→  const ar = isArabic(lang);
   111→  const parts: string[] = [];
   112→  if (h > 0) parts.push(`${h}${ar ? "س" : "h"}`);
   113→  if (m > 0 || h === 0) parts.push(`${m}${ar ? "د" : "m"}`);
   114→  return ar ? `تبقى ${parts.join(" ")}` : `${parts.join(" ")} left`;
   115→}

   117→function daysUntil(releasedMs: number, now: number): number {
   118→  return Math.max(0, Math.ceil((releasedMs - now) / 86_400_000));
   119→}
   120→
   121→/**
   122→ * "Upcoming" badge text with correct plurals via Intl.PluralRules.
   123→ * Arabic: اليوم (0) / غداً (1) / يومان (2) / أيام (3-10) / يوم (11+).
   124→ * English: today / tomorrow / "In {n} day(s)".
   125→ */
   126→export function upcomingLabel(releasedMs: number, lang: string): string | null {
   127→  if (!Number.isFinite(releasedMs)) return null;
   128→  const now = Date.now();
   129→  if (releasedMs <= now) return null;
   130→  const days = daysUntil(releasedMs, now);
   131→  const ar = isArabic(lang);
   132→  if (days === 0) return homeT("airToday", lang);
   133→  if (days === 1) return homeT("airTomorrow", lang);
   134→  if (ar) {
   135→    const cat = new Intl.PluralRules("ar").select(days);
   136→    const unit = cat === "two" ? "يومين" : cat === "few" ? "أيام" : "يوم";
   137→    return homeT("airIn", lang).replace("{n}", String(days)).replace("{unit}", unit);
   138→  }
   139→  const unit = new Intl.PluralRules("en").select(days) === "one" ? "day" : "days";
   140→  return homeT("airIn", lang).replace("{n}", String(days)).replace("{unit}", unit);
   141→}

   400→/** Dev-time missing-key guard: renders the key itself and fails loudly. */
   401→function warnMissing(key: string): string {
   402→  if (process.env.NODE_ENV !== "production") {
   403→    console.error(`[i18n] Missing translation key: "${key}"`);
   404→  }
   405→  return key;
   406→}

   408→/** Interpolate `{var}` placeholders. */
   409→function interpolate(template: string, vars?: Vars): string {
   410→  if (!vars) return template;
   411→  return template.replace(/\{(\w+)\}/g, (m, k: string) =>
   412→    k in vars ? String(vars[k]) : m,
   413→  );
   414→}

   416→/** Translate an app string. Unknown keys fail loudly in dev. */
   417→export function t(key: AppStringKey, lang: string, vars?: Vars): string {
   418→  const entry = APP_STRINGS[key] as { en: string; ar: string; arOther?: string } | undefined;
   419→  if (!entry) return warnMissing(String(key));
   420→  if (!isArabic(lang)) return interpolate(entry.en, vars);
   421→  // Simple English plural shape for Arabic via arOther when {n} ≠ 1
   422→  if (entry.arOther && vars?.n !== undefined && Number(vars.n) !== 1) {
   423→    return interpolate(entry.arOther, vars);
   424→  }
   425→  return interpolate(entry.ar, vars);
   426→}

   432→const NUM_FMT: Record<string, Intl.NumberFormat> = {};
   433→/** Latin-digit number formatting in every locale (digit style stays consistent). */
   434→export function formatNumber(n: number, lang: string): string {
   435→  const tag = isArabic(lang) ? "ar-u-nu-latn" : "en";
   436→  NUM_FMT[tag] ??= new Intl.NumberFormat(tag);
   437→  return NUM_FMT[tag].format(n);
   438→}

   440→const REL_FMT: Record<string, Intl.RelativeTimeFormat> = {};
   441→const REL_UNITS: { limit: number; div: number; unit: Intl.RelativeTimeFormatUnit }[] = [
   442→  { limit: 45_000, div: 1_000, unit: "second" },
   443→  { limit: 45 * 60_000, div: 60_000, unit: "minute" },
   444→  { limit: 22 * 3_600_000, div: 3_600_000, unit: "hour" },
   445→  { limit: 21 * 86_400_000, div: 86_400_000, unit: "day" },
   446→  { limit: 300 * 86_400_000, div: 7 * 86_400_000, unit: "week" },
   447→  { limit: 18 * 31_536_000_000, div: 30 * 86_400_000, unit: "month" },
   448→  { limit: Infinity, div: 365 * 86_400_000, unit: "year" },
   449→];
   450→
   451→/** Localized "منذ 8 دقائق" / "8 minutes ago" via Intl.RelativeTimeFormat. */
   452→export function formatTimeAgo(ms: number, lang: string): string {
   453→  const diff = ms - Date.now(); // negative → past
   454→  const abs = Math.abs(diff);
   455→  const { div, unit } = REL_UNITS.find((u) => abs < u.limit) ?? REL_UNITS[REL_UNITS.length - 1];
   456→  const value = Math.round(diff / div);
   457→  const tag = isArabic(lang) ? "ar" : "en";
   458→  REL_FMT[tag] ??= new Intl.RelativeTimeFormat(tag, { numeric: "auto" });
   459→  return REL_FMT[tag].format(value, unit);
   460→}
