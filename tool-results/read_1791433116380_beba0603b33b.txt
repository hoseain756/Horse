    99→    99→function loadRecents(): RecentRecord[] {
   100→   100→  if (typeof window === "undefined") return [];
   101→   101→  try {
   102→   102→    const raw = window.localStorage.getItem(RECENTS_KEY);
   103→   103→    if (!raw) return [];
   104→   104→    const arr = JSON.parse(raw) as RecentRecord[];
   105→   105→    return Array.isArray(arr) ? arr.filter((r) => r && r.uid && r.action).slice(0, 5) : [];
   106→   106→  } catch {
   107→   107→    return [];
   108→   108→  }
   109→   109→}

   111→   111→function saveRecent(rec: RecentRecord): RecentRecord[] {
   112→   112→  const list = loadRecents().filter((r) => r.uid !== rec.uid);
   113→   113→  const next = [rec, ...list].slice(0, 5);
   114→   114→  try {
   115→   115→    window.localStorage.setItem(RECENTS_KEY, JSON.stringify(next));
   116→   116→  } catch {
   117→   117→    /* ignore */
   118→   118→  }

   151→   151→// ---------- Fuzzy scoring (subsequence + consecutive/word-start bonuses) ----------
   152→   152→function fuzzyScore(query: string, text: string): number | null {
   153→   153→  const q = query.toLowerCase();
   154→   154→  const t = text.toLowerCase();
   155→   155→  if (!q) return 0;
   156→   156→  let score = 0;
   157→   157→  let searchFrom = 0;
   158→   158→  let prevIdx = -2;
   159→   159→  for (let qi = 0; qi < q.length; qi++) {
   160→   160→    const ch = q[qi];
   161→   161→    const idx = t.indexOf(ch, searchFrom);
   162→   162→    if (idx === -1) return null;
   163→   163→    let bonus = 1;
   164→   164→    if (idx === prevIdx + 1) bonus += 6; // consecutive match
   165→   165→    if (idx === 0 || /[\s\-_:.·/(\[]/.test(t[idx - 1])) bonus += 8; // word start
   166→   166→    score += bonus;
   167→   167→    prevIdx = idx;
   168→   168→    searchFrom = idx + 1;
   169→   169→  }
   170→   170→  // Slight preference for shorter targets
   171→   171→  score += Math.max(0, 8 - Math.floor(t.length / 12));
   172→   172→  return score;
   173→   173→}

   315→   315→      run: () => useNav.getState().push({ kind: "detail", type, id }),

   325→   325→function remoteMetaLine(m: Meta): string | undefined {
   326→   326→  const parts: string[] = [];
   327→   327→  if (m.releaseInfo) parts.push(m.releaseInfo); // "2016" / "2016–2019"
   328→   328→  if (m.imdbRating) parts.push(`★ ${m.imdbRating}`);
   329→   329→  return parts.length > 0 ? parts.join(" · ") : undefined;
   330→   330→}

   332→   332→function buildRemoteItems(
   333→   333→  remote: { movies: Meta[]; series: Meta[]; addon: Meta[] },
   334→   334→  libraryKeys: Set<string>,
   335→   335→): PaletteItem[] {
   336→   336→  const out: PaletteItem[] = [];
   337→   337→  // Dedupe against the local library (type:id) — and, since `seen` accumulates,
   338→   338→  // addon results that duplicate a Cinemeta hit shown above them.
   339→   339→  const seen = new Set(libraryKeys);
   340→   340→  const push = (m: Meta, group: string, uid: string, keywords: string) => {
   341→   341→    if (!m?.id || !m.name) return;
   342→   342→    const key = `${m.type}:${m.id}`;
   343→   343→    if (seen.has(key)) return;
   344→   344→    seen.add(key);
   345→   345→    out.push({
   346→   346→      uid,
   347→   347→      icon: m.type === "series" ? Tv : Film,
   348→   348→      label: m.name,
   349→   349→      group,
   350→   350→      keywords,
   351→   351→      thumb: m.poster,
   352→   352→      typeChip: m.type,
   353→   353→      meta: remoteMetaLine(m),
   354→   354→      action: { t: "detail", type: m.type, id: m.id, name: m.name, poster: m.poster },
   355→   355→      run: () => useNav.getState().push({ kind: "detail", type: m.type, id: m.id }),
   356→   356→    });
   357→   357→  };
   358→   358→  for (const m of remote.movies.slice(0, 6)) {
   359→   359→    push(m, "Movies", `cine:movie:${m.id}`, `${m.type} search remote content`);
   360→   360→  }
   361→   361→  for (const m of remote.series.slice(0, 6)) {
   362→   362→    push(m, "Series", `cine:series:${m.id}`, `${m.type} search remote content`);
   363→   363→  }
   364→   364→  for (const m of remote.addon.slice(0, 8)) {
   365→   365→    push(
   366→   366→      m,
   367→   367→      "Addon results",
   368→   368→      `cine:addon:${m.type}:${m.id}`,
   369→   369→      `${m.type} addon search remote content`,
   370→   370→    );
   371→   371→  }
   372→   372→  return out;
   373→   373→}

   391→   391→function itemFromRecent(rec: RecentRecord, settings: Settings, canInstall: boolean): PaletteItem | null {
   392→   392→  const a = rec.action;
   393→   393→  switch (a.t) {
   394→   394→    case "view": {
   395→   395→      const nav = NAV_ITEMS.find((n) => n.id === a.view);
   396→   396→      const view = a.view as View;
   397→   397→      return {
   398→   398→        uid: rec.uid,
   399→   399→        icon: nav?.icon ?? House,
   400→   400→        label: rec.label,
   401→   401→        group: "Recent",
   402→   402→        keywords: nav?.label,
   403→   403→        action: a,
   404→   404→        run: () => useNav.getState().push({ kind: "view", view }),
   405→   405→      };
   406→   406→    }
   407→   407→    case "section": {
   408→   408→      const def = SECTION_DEFS.find((s) => s.id === a.section);
   409→   409→      return {
   410→   410→        uid: rec.uid,
   411→   411→        icon: def?.icon ?? SettingsIcon,
   412→   412→        label: rec.label,
   413→   413→        group: "Recent",
   414→   414→        keywords: def?.keywords,
   415→   415→        action: a,
   416→   416→        run: () => {
   417→   417→          const nav = useNav.getState();
   418→   418→          const top = nav.top();
   419→   419→          if (top?.kind !== "view" || top.view !== "settings") {
   420→   420→            nav.push({ kind: "view", view: "settings" });
   421→   421→          }
   422→   422→          window.setTimeout(() => {
   423→   423→            window.dispatchEvent(new CustomEvent("harbor:settings-section", { detail: a.section }));
   424→   424→          }, 60);
   425→   425→        },
   426→   426→      };
   427→   427→    }
   428→   428→    case "action": {
   429→   429→      if (a.id === "install" && !canInstall) return null;
   430→   430→      const built = buildActionItems(settings, canInstall).find((i) => i.uid === `act:${a.id}`);
   431→   431→      if (!built) return null;
   432→   432→      return { ...built, uid: rec.uid, group: "Recent" };
   433→   433→    }
   434→   434→    case "detail":
   435→   435→      return {
   436→   436→        uid: rec.uid,
   437→   437→        icon: a.type === "series" ? Tv : Film,
   438→   438→        label: rec.label,
   439→   439→        group: "Recent",
   440→   440→        thumb: rec.thumb ?? a.poster,
   441→   441→        typeChip: rec.typeChip ?? a.type,
   442→   442→        action: a,
   443→   443→        run: () => useNav.getState().push({ kind: "detail", type: a.type, id: a.id }),
   444→   444→      };
   445→   445→    case "addon":
   446→   446→      return {
   447→   447→        uid: rec.uid,
   448→   448→        icon: Puzzle,
   449→   449→        label: rec.label,
   450→   450→        group: "Recent",
   451→   451→        thumb: rec.thumb ?? a.logo,
   452→   452→        action: a,
   453→   453→        run: () => useNav.getState().push({ kind: "addon-detail", addonId: a.addonId }),
   454→   454→      };
   455→   455→  }
   456→   456→}

   467→   467→function PaletteSurface({ onClose }: { onClose: () => void }) {
   468→   468→  const settings = useSettings((s) => s.settings);
   469→   469→  const addons = useAddons((s) => s.addons);
   470→   470→  const canInstall = usePwa((s) => s.canInstall);
   471→   471→
   472→   472→  const [query, setQuery] = useState("");
   473→   473→  const [selected, setSelected] = useState(0);
   474→   474→  const [recents, setRecents] = useState<RecentRecord[]>(() => loadRecents());
   475→   475→  // Round 9: remote content search state (Cinemeta + installed addon catalogs)
   476→   476→  // and the query the last completed search was for (drives the "Search
   477→   477→  // everywhere" row so it only shows for a finished, empty current search).
   478→   478→  const [remoteResults, setRemoteResults] = useState<{
   479→   479→    movies: Meta[];
   480→   480→    series: Meta[];
   481→   481→    addon: Meta[];
   482→   482→  }>({ movies: [], series: [], addon: [] });
   483→   483→  const [searching, setSearching] = useState(false);
   484→   484→  const [doneQuery, setDoneQuery] = useState<string | null>(null);
   485→   485→  const inputRef = useRef<HTMLInputElement>(null);
   486→   486→  const runToken = useRef(0);
   487→   487→
   488→   488→  // Focus + lock scroll while mounted (no state resets needed — fresh mount)
   489→   489→  useEffect(() => {
   490→   490→    const t = setTimeout(() => inputRef.current?.focus(), 30);
   491→   491→    document.body.style.overflow = "hidden";
   492→   492→    return () => {
   493→   493→      clearTimeout(t);
   494→   494→      document.body.style.overflow = "";
   495→   495→    };
   496→   496→  }, []);
   497→   497→
   498→   498→  // Round 9: debounced remote content search. The token is incremented on
   499→   499→  // every query change (and on unmount, below) so stale responses never reach
   500→   500→  // state. State resets use 0ms timeouts — the same setState-in-effect-safe
   501→   501→  // pattern the search overlay uses. Player overlays suppress the palette
   502→   502→  // entirely, so a remote search can never run during playback.
   503→   503→  useEffect(() => {
   504→   504→    const q = query.trim();
   505→   505→    const token = ++runToken.current; // invalidates any in-flight response
   506→   506→    if (q.length < 2) {
   507→   507→      const reset = setTimeout(() => {
   508→   508→        setRemoteResults({ movies: [], series: [], addon: [] });
   509→   509→        setSearching(false);
   510→   510→        setDoneQuery(null);
   511→   511→      }, 0);
   512→   512→      return () => clearTimeout(reset);
   513→   513→    }
   514→   514→    const spin = setTimeout(() => setSearching(true), 0);
   515→   515→    const t = setTimeout(() => {
   516→   516→      void Promise.allSettled([
   517→   517→        searchCinemeta(q),
   518→   518→        searchAddonCatalogs(installedAddons(), q),
   519→   519→      ]).then(([cine, addon]) => {
   520→   520→        if (token !== runToken.current) return; // stale response — ignore
   521→   521→        setRemoteResults({
   522→   522→          movies: cine.status === "fulfilled" ? cine.value.movies : [],
   523→   523→          series: cine.status === "fulfilled" ? cine.value.series : [],
   524→   524→          addon: addon.status === "fulfilled" ? addon.value : [],
   525→   525→        });
   526→   526→        setSearching(false);
   527→   527→        setDoneQuery(q);
   528→   528→      });
   529→   529→    }, 320);
   530→   530→    return () => {
   531→   531→      clearTimeout(spin);
   532→   532→      clearTimeout(t);
   533→   533→    };
   534→   534→  }, [query]);
   535→   535→
   536→   536→  // Invalidate any in-flight remote search when the palette unmounts (closes).
   537→   537→  useEffect(() => {
   538→   538→    return () => {
   539→   539→      runToken.current += 1;
   540→   540→    };
   541→   541→  }, []);
   542→   542→
   543→   543→  // Full command pool (only needed when filtering)
   544→   544→  const pool = useMemo<PaletteItem[]>(
   545→   545→    () => [
   546→   546→      ...buildNavItems(settings),
   547→   547→      ...buildSectionItems(),
   548→   548→      ...buildActionItems(settings, canInstall),
   549→   549→      ...buildLibraryItems(),
   550→   550→      ...buildAddonItems(addons),
   551→   551→    ],
   552→   552→    [settings, canInstall, addons],
   553→   553→  );
   554→   554→
   555→   555→  // Round 9: type:id keys of local library rows (library items store uid
   556→   556→  // `detail:{type}:{id}`), used to dedupe remote results against the library.
   557→   557→  const libraryKeys = useMemo(() => {
   558→   558→    const keys = new Set<string>();
   559→   559→    for (const it of pool) {
   560→   560→      if (it.uid.startsWith("detail:")) keys.add(it.uid.slice("detail:".length));
   561→   561→    }
   562→   562→    return keys;
   563→   563→  }, [pool]);
   564→   564→
   565→   565→  const itemsBase = useMemo<PaletteItem[]>(() => {
   566→   566→    const trimmed = query.trim();
   567→   567→    const q = trimmed.toLowerCase();
   568→   568→    if (!q) {
   569→   569→      // Empty query: recents first, then all navigation items. Recents reuse
   570→   570→      // the original item uids, so drop any nav item already shown as a recent
   571→   571→      // (duplicate React keys otherwise).
   572→   572→      const rec = recents
   573→   573→        .map((r) => itemFromRecent(r, settings, canInstall))
   574→   574→        .filter((i): i is PaletteItem => i !== null);
   575→   575→      const seen = new Set(rec.map((i) => i.uid));
   576→   576→      return [...rec, ...buildNavItems(settings).filter((i) => !seen.has(i.uid))];
   577→   577→    }
   578→   578→    const scored: { item: PaletteItem; score: number }[] = [];
   579→   579→    for (const item of pool) {
   580→   580→      let best = fuzzyScore(q, item.label);
   581→   581→      if (item.keywords) {
   582→   582→        const ks = fuzzyScore(q, item.keywords);
   583→   583→        if (ks !== null) best = best === null ? ks * 0.8 : Math.max(best, ks * 0.8);
   584→   584→      }
   585→   585→      if (best !== null) scored.push({ item, score: best });
   586→   586→    }
   587→   587→    scored.sort((a, b) => {
   588→   588→      if (b.score !== a.score) return b.score - a.score;
   589→   589→      const ga = GROUP_ORDER[a.item.group] ?? 9;
   590→   590→      const gb = GROUP_ORDER[b.item.group] ?? 9;
   591→   591→      if (ga !== gb) return ga - gb;
   592→   592→      return a.item.label.localeCompare(b.item.label);
   593→   593→    });
   594→   594→    const local = scored.map((s) => s.item);
   595→   595→    // Round 9: append remote content results after the local scored items.
   596→   596→    // Results are only stored when their response token was current, so they
   597→   597→    // always belong to the most recent completed search (typeahead-fresh).
   598→   598→    if (
   599→   599→      trimmed.length >= 2 &&
   600→   600→      remoteResults.movies.length +
   601→   601→        remoteResults.series.length +
   602→   602→        remoteResults.addon.length >
   603→   603→        0
   604→   604→    ) {
   605→   605→      return [...local, ...buildRemoteItems(remoteResults, libraryKeys)];
   606→   606→    }
   607→   607→    return local;
   608→   608→  }, [query, pool, recents, settings, canInstall, remoteResults, libraryKeys]);
   609→   609→
   610→   610→  // Round 9: when a >= 2-char query matched nothing locally or remotely and
   611→   611→  // that search has finished, offer a hand-off to the full search overlay
   612→   612→  // instead of a dead end (replaces the "No matches" block).
   613→   613→  const searchEverywhereItem = useMemo<PaletteItem | null>(() => {
   614→   614→    const q = query.trim();
   615→   615→    if (q.length < 2 || searching || doneQuery !== q || itemsBase.length > 0) return null;
   616→   616→    return {
   617→   617→      uid: "cine:search-everywhere",
   618→   618→      icon: Search,
   619→   619→      label: `Search everywhere for “${q}”`,
   620→   620→      group: "Search",
   621→   621→      keywords: `${q} everywhere all sources remote search`,
   622→   622→      accent: true,
   623→   623→      kbd: "↵",
   624→   624→      // Order matters (runItem closes the palette before run()): dispatch
   625→   625→      // stores the pending prefill, THEN the overlay opens and consumes it.
   626→   626→      run: () => {
   627→   627→        window.dispatchEvent(new CustomEvent("harbor:prefill-search", { detail: q }));
   628→   628→        useNav.getState().setSearchOpen(true);
   629→   629→      },
   630→   630→    };
   631→   631→  }, [query, searching, doneQuery, itemsBase]);
   632→   632→
   633→   633→  const items = useMemo(
   634→   634→    () => (searchEverywhereItem ? [searchEverywhereItem, ...itemsBase] : itemsBase),
   635→   635→    [searchEverywhereItem, itemsBase],
   636→   636→  );
   637→   637→
   638→   638→  const sel = Math.min(selected, Math.max(0, items.length - 1));
   639→   639→
   640→   640→  // Keep the selected row in view
   641→   641→  useEffect(() => {
   642→   642→    document
   643→   643→      .getElementById(`harbor-palette-opt-${sel}`)
   644→   644→      ?.scrollIntoView({ block: "nearest" });
   645→   645→  }, [sel, items]);
   646→   646→
   647→   647→  const runItem = (item: PaletteItem) => {
   648→   648→    if (item.action) {
   649→   649→      setRecents(
   650→   650→        saveRecent({
   651→   651→          uid: item.uid,
   652→   652→          label: item.label,
   653→   653→          group: item.group,
   654→   654→          thumb: item.thumb,
   655→   655→          typeChip: item.typeChip,
   656→   656→          action: item.action,
   657→   657→        }),
   658→   658→      );
   659→   659→    }
   660→   660→    onClose();
   661→   661→    item.run();
   662→   662→  };
   663→   663→
   664→   664→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   665→   665→    if ((e.ctrlKey || e.metaKey) && e.shiftKey && e.key.toLowerCase() === "p") {
   666→   666→      // Palette shortcut is now Ctrl/Cmd+Shift+P (Ctrl/Cmd+K belongs to the floating search)
   667→   667→      e.preventDefault();
   668→   668→      onClose();
   669→   669→      return;
   670→   670→    }
   671→   671→    if (e.key === "ArrowDown") {
   672→   672→      e.preventDefault();
   673→   673→      setSelected((s) => (items.length ? Math.min(s + 1, items.length - 1) : 0));
   674→   674→    } else if (e.key === "ArrowUp") {
   675→   675→      e.preventDefault();
   676→   676→      setSelected((s) => Math.max(s - 1, 0));
   677→   677→    } else if (e.key === "Enter") {
   678→   678→      e.preventDefault();
   679→   679→      const item = items[sel];
   680→   680→      if (item) runItem(item);
   681→   681→    } else if (e.key === "Escape" || e.key === "Tab") {
   682→   682→      e.preventDefault();
   683→   683→      onClose();
   684→   684→    }
   685→   685→  };
   686→   686→
   687→   687→  return (
   688→   688→    <div
   689→   689→      className="fixed inset-0 z-[300] flex items-start justify-center bg-black/70 px-4 backdrop-blur-sm harbor-scroll"
   690→   690→      role="dialog"
   691→   691→      aria-modal="true"
   692→   692→      aria-label="Command palette"
   693→   693→      onClick={onClose}
   694→   694→    >
   695→   695→      <div
   696→   696→        className="mt-[12vh] w-full max-w-xl overflow-hidden rounded-2xl border border-edge-soft bg-canvas shadow-2xl harbor-pop-in"
   697→   697→        onClick={(e) => e.stopPropagation()}
   698→   698→      >
   699→   699→        {/* Input row */}
   700→   700→        <div className="flex items-center gap-3 border-b border-edge-soft px-4 py-3.5">
   701→   701→          <Search className="h-5 w-5 shrink-0 text-ink-subtle" />
   702→   702→          <input
   703→   703→            ref={inputRef}
   704→   704→            value={query}
   705→   705→            onChange={(e) => {
   706→   706→              setQuery(e.target.value);
   707→   707→              setSelected(0);
   708→   708→            }}
   709→   709→            onKeyDown={onInputKeyDown}
   710→   710→            placeholder="Type a command or search…"
   711→   711→            className="flex-1 bg-transparent text-base text-ink outline-none placeholder:text-ink-subtle"
   712→   712→            aria-label="Command palette search"
   713→   713→            aria-controls="harbor-palette-list"
   714→   714→            aria-activedescendant={items.length > 0 ? `harbor-palette-opt-${sel}` : undefined}
   715→   715→          />
   716→   716→          {searching ? (
   717→   717→            <Loader2
   718→   718→              className="h-4 w-4 shrink-0 animate-spin text-accent"
   719→   719→              role="status"
   720→   720→              aria-label="Searching content"
   721→   721→            />
   722→   722→          ) : null}
   723→   723→          <kbd className="harbor-kbd">esc</kbd>
   724→   724→          <button
   725→   725→            type="button"
   726→   726→            onClick={onClose}
   727→   727→            className="harbor-tv-focus flex h-8 w-8 shrink-0 items-center justify-center rounded-lg text-ink-muted hover:bg-raised hover:text-ink"
   728→   728→            aria-label="Close command palette"
   729→   729→          >
   730→   730→            <X className="h-5 w-5" />
   731→   731→          </button>
   732→   732→        </div>
   733→   733→
   734→   734→        {/* Results */}
   735→   735→        <div id="harbor-palette-list" role="listbox" aria-label="Commands" className="harbor-scroll harbor-palette-scroll p-2">
   736→   736→          {items.length === 0 ? (
   737→   737→            <div className="px-6 py-10 text-center">
   738→   738→              <Search className="mx-auto mb-3 h-8 w-8 text-ink-subtle opacity-50" />
   739→   739→              <p className="text-sm text-ink-muted">No matches for “{query.trim()}”</p>
   740→   740→              <p className="mt-1 text-xs text-ink-subtle">
   741→   741→                Try “settings” or a title from your library
   742→   742→              </p>
   743→   743→            </div>
   744→   744→          ) : (
   745→   745→            items.map((item, i) => {
   746→   746→              const Icon = item.icon;
   747→   747→              return (
   748→   748→                <div
   749→   749→                  key={item.uid}
   750→   750→                  id={`harbor-palette-opt-${i}`}
   751→   751→                  role="option"
   752→   752→                  aria-selected={i === sel}
   753→   753→                  aria-label={item.label}
   754→   754→                  onMouseEnter={() => setSelected(i)}
   755→   755→                  onClick={() => runItem(item)}
   756→   756→                  className={cn(
   757→   757→                    "flex min-h-[44px] cursor-pointer items-center gap-3 rounded-xl px-3 py-2.5 text-sm transition-colors",
   758→   758→                    i === sel ? "bg-raised text-ink" : "text-ink-muted",
   759→   759→                  )}
   760→   760→                >
   761→   761→                  <Icon className="h-4 w-4 shrink-0" />
   762→   762→                  {item.thumb ? (
   763→   763→                    <span className="relative block h-10 w-7 shrink-0 overflow-hidden rounded-md border border-edge-soft ring-1 ring-edge-soft">
   764→   764→                      <PosterImage
   765→   765→                        src={item.thumb}
   766→   766→                        alt=""
   767→   767→                        className="absolute inset-0 h-full w-full object-cover"
   768→   768→                      />
   769→   769→                    </span>
   770→   770→                  ) : null}
   771→   771→                  <span
   772→   772→                    className={cn(
   773→   773→                      "truncate",
   774→   774→                      item.accent && "font-medium text-accent",
   775→   775→                    )}
   776→   776→                  >
   777→   777→                    {item.label}
   778→   778→                  </span>
   779→   779→                  {item.typeChip ? (
   780→   780→                    <span className="shrink-0 rounded-full border border-edge-soft bg-black/30 px-2 py-0.5 text-[10px] uppercase tracking-wide text-ink-subtle">
   781→   781→                      {item.typeChip}
   782→   782→                    </span>
   783→   783→                  ) : null}
   784→   784→                  {item.meta ? (
   785→   785→                    <span className="shrink-0 text-[10px] font-semibold text-ink-subtle">
   786→   786→                      {item.meta}
   787→   787→                    </span>
   788→   788→                  ) : null}
   789→   789→                  <span className="ml-auto hidden shrink-0 text-[10px] uppercase tracking-wide text-ink-subtle sm:inline">
   790→   790→                    {item.group}
   791→   791→                  </span>
   792→   792→                  {item.kbd ? (
   793→   793→                    <kbd className="harbor-kbd ml-auto shrink-0 sm:ml-0">{item.kbd}</kbd>
   794→   794→                  ) : null}
   795→   795→                </div>

   797→   797→            })

   799→   799→        </div>
   800→   800→
   801→   801→        {/* Footer hints */}

   815→   815→            <kbd className="harbor-kbd">ctrl</kbd>

   819→   819→      </div>
   820→   820→    </div>
   821→   821→  );
   822→   822→}
