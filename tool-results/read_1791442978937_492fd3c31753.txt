    25→export function PickerOverlay({

    32→  type: string;

    35→  season?: number;

    41→  const addonsLoaded = useAddons((s) => s.loaded);
    42→  const settings = useSettings((s) => s.settings);
    43→  const { toast } = useToast();
    44→  const debridConfigured = useDebrid((s) => s.apiKey !== null);
    45→  const [debridDialogOpen, setDebridDialogOpen] = useState(false);
    46→

    68→  // to the dialog without touching the mouse. Mouse use (dataset.tv flips off in

    73→  const focusTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    74→
    75→  const scheduleTvFocus = useCallback(() => {
    76→    if (focusTimer.current) clearTimeout(focusTimer.current);

    85→    }, 350);
    86→  }, []);
    87→
    88→  // On open: focus the first row if streams are already there, else the close button.

   180→        }),
   181→      }))
   182→      .filter((t) => t.items.length > 0);
   183→  }, [tiers, filter, query]);

   187→    return (
   188→      <PlayerOverlay
   189→        payload={{
   190→          url: playing.url ?? "",
   191→          title: metaName,
   192→          type,
   193→          metaId: id,
   194→          season,
   195→          episode,
   196→          videoId: targetId,
   197→          streamTitle: playing.title,
   198→          streamDescription: playing.description,
   199→          deepLink: { type, id, videoId: targetId },
   200→          ...(playing.p2p ? { p2p: playing.p2p } : {}),
   201→        }}
   202→      />
   203→    );
   204→  }
   205→

   207→    <div className="fixed inset-0 z-[150] bg-black/80 backdrop-blur-sm flex items-end md:items-center justify-center" role="dialog" aria-modal="true" aria-label="Stream picker">
   208→      <div className="md-sheet rounded-b-none! md:rounded-b-[28px]! w-full md:max-w-3xl md:max-h-[82vh] max-h-[88vh] flex flex-col overflow-hidden shadow-[var(--md-sys-elevation-3)]">
   209→        {/* Header */}
   210→        <div className="flex items-center justify-between gap-3 px-5 py-4 border-b border-edge-soft shrink-0">
   211→          <div className="min-w-0">
   212→            <h2 className="md-title-large text-ink">Choose a stream</h2>
   213→            <p className="text-xs text-ink-subtle truncate">
   214→              {type === "series" ? `Series · ${targetId}` : targetId}
   215→              {season ? ` · S${season}:E${episode}` : ""}
   216→            </p>
   217→          </div>
   218→          <div className="flex items-center gap-1.5 shrink-0">
   219→            <button
   220→              type="button"

   229→              type="button"

   242→          {(
   243→            [
   244→              ["all", "All"],
   245→              ["free", "Free"],
   246→              ["cached", "Cached"],
   247→              ["4k", "4K"],
   248→              ["1080p", "1080p"],

   253→              type="button"
   254→              onClick={() => setFilter(key)}

   256→              className={cn("md-chip md-state", filter === key && "md-chip-selected")}

   258→              {label}

   260→          ))}

   262→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-ink-subtle" />
   263→            <input
   264→              value={query}
   265→              onChange={(e) => setQuery(e.target.value)}

   282→              type="button"

   306→                  </div>
   307→                  <div className="w-16 h-7 rounded-lg bg-white/10 shrink-0" />

   309→              ))}
   310→            </div>

   312→

   350→      </div>
   351→      {debridDialogOpen && <DebridSetupDialog onClose={() => setDebridDialogOpen(false)} />}

   356→// Quality-ladder dot per tier header: emerald top, amber mid, neutral low —
   357→// colors follow Harbor's existing palette conventions (understated).
   358→function tierDotColor(tier: string): string {
   359→  if (tier === "4K HDR" || tier === "4K") return "bg-emerald-400";
   360→  if (tier === "1080p HDR" || tier === "1080p") return "bg-amber-500";
   361→  if (tier === "720p" || tier === "SD") return "bg-zinc-500";
   362→  return "bg-zinc-600";
   363→}
   364→
   365→function StreamRow({
   366→  stream,
   367→  onPick,
   368→  onNeedsSetup,
   369→  configured,
   370→  showQuality,
   371→}: {
   372→  stream: Stream;
   373→  onPick: (s: Stream) => void;
   374→  onNeedsSetup: () => void;
   375→  configured: boolean;
   376→  showQuality: boolean;
   377→}) {
   378→  const { toast } = useToast();
   379→  const debridKey = useDebrid((s) => s.apiKey);
   380→  const debridResolve = useDebrid((s) => s.resolve);
   381→  const [resolving, setResolving] = useState(false);
   382→  // P2P connecting state: null = idle, else seconds elapsed while joining the swarm
   383→  const [p2pPhase, setP2pPhase] = useState<"idle" | "joining" | "planning">("idle");
   384→  const [p2pInfo, setP2pInfo] = useState<{ peers: number; elapsed: number } | null>(null);
   385→  const [err, setErr] = useState<string | null>(null);
   386→  const errTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   387→  const p2pAbort = useRef<AbortController | null>(null);
   388→
   389→  useEffect(() => {
   390→    return () => {
   391→      if (errTimer.current) clearTimeout(errTimer.current);
   392→      p2pAbort.current?.abort();
   393→    };
   394→  }, []);
   395→
   396→  const cached = Object.values(stream.cached ?? {}).some(Boolean);
   397→  const isTorrent = !!stream.infoHash && !stream.url; // torrent without a direct URL → debrid unlock OR P2P
   398→  const p2pEnabled = useSettings((s) => s.settings.p2pEnabled);
   399→  const p = stream.parsed;
   400→  const titleLine = (stream.title ?? stream.description ?? "Stream").split("\n")[0];
   401→  const detailLine = (stream.description ?? "").split("\n").filter(Boolean).slice(1).join(" ").slice(0, 140);
   402→
   403→  const fail = (msg: string) => {
   404→    setErr(msg);
   405→    toast({ title: "Stream unavailable", description: msg, variant: "destructive" });
   406→    if (errTimer.current) clearTimeout(errTimer.current);
   407→    errTimer.current = setTimeout(() => setErr(null), 6000);
   408→  };
   409→
   410→  const unlock = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   411→    e?.stopPropagation();
   412→    e?.preventDefault();
   413→    if (resolving) return;
   414→    if (debridKey === null) {
   415→      onNeedsSetup();
   416→      return;
   417→    }
   418→    setResolving(true);
   419→    setErr(null);
   420→    const res = await debridResolve(
   421→      stream.infoHash!,
   422→      stream.behaviorHints?.filename,
   423→      stream.parsed?.size,
   424→    );
   425→    setResolving(false);
   426→    if ("url" in res && res.url) {
   427→      onPick({
   428→        ...stream,
   429→        url: res.url,
   430→        behaviorHints: {
   431→          ...stream.behaviorHints,
   432→          ...(res.filename ? { filename: res.filename } : {}),
   433→        },
   434→      });
   435→    } else {
   436→      const msg = "error" in res ? res.error : "Unlock failed";
   437→      fail(msg);
   438→    }
   439→  };
   440→
   441→  // Play a torrent through the server-side P2P engine: join swarm → wait for
   442→  // metadata → pick the right file (native container vs ffmpeg remux) → play.
   443→  const playP2p = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   444→    e?.stopPropagation();
   445→    e?.preventDefault();
   446→    if (p2pPhase !== "idle") return;
   447→    if (!useSettings.getState().settings.p2pEnabled) {
   448→      onNeedsSetup();
   449→      return;
   450→    }
   451→    setErr(null);
   452→    setP2pPhase("joining");
   453→    setP2pInfo({ peers: 0, elapsed: 0 });
   454→    const startedAt = Date.now();
   455→    const abort = new AbortController();
   456→    p2pAbort.current = abort;
   457→    const infoHash = stream.infoHash!.toLowerCase();
   458→
   459→    // progress ticker while the swarm join runs
   460→    const ticker = setInterval(() => {
   461→      setP2pInfo((prev) => (prev ? { ...prev, elapsed: Math.round((Date.now() - startedAt) / 1000) } : prev));
   462→    }, 1000);
   463→
   464→    try {
   465→      // 1) prepare (join swarm). 202 = metadata still downloading → poll status.
   466→      const prep = await p2pPrepare(infoHash, {
   467→        fileIdx: stream.fileIdx,
   468→        filename: stream.behaviorHints?.filename ?? stream.parsed?.filename,
   469→      });
   470→
   471→      // 2) poll while discovering (up to ~100s; the engine keeps the swarm alive)
   472→      while (prep.pending) {
   473→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   474→        if (Date.now() - startedAt > 100_000) {
   475→          throw new Error(
   476→            "Could not find peers for this torrent. The swarm may be dead or this network blocks BitTorrent — a debrid service would unlock it instantly.",
   477→          );
   478→        }
   479→        await new Promise((r) => setTimeout(r, 2500));
   480→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   481→        const st = await p2pStatus(infoHash).catch(() => null);
   482→        setP2pInfo((prev) => (prev ? { ...prev, peers: st?.peers ?? prev.peers } : prev));
   483→        if (st?.ready) break;
   484→      }
   485→
   486→      // 3) choose serving mode (native range stream vs ffmpeg remux for mkv)
   487→      setP2pPhase("planning");
   488→      const fileIdx = prep.pending
   489→        ? await (async () => {
   490→            const st = await p2pStatus(infoHash);
   491→            const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   492→            const idx = Number.isInteger(stream.fileIdx)
   493→              ? stream.fileIdx!
   494→              : st.files
   495→                ? pickFileIndex(st.files, filename)
   496→                : 0;
   497→            return idx;
   498→          })()
   499→        : prep.fileIdx;
   500→      const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   501→      let plan: P2pPlaybackPlan;
   502→      try {
   503→        plan = await p2pPlan(infoHash, fileIdx, filename);
   504→      } catch {
   505→        plan = { key: infoHash, fileIdx, url: `/stream/${infoHash}/${fileIdx}`, mode: "unknown" };
   506→      }
   507→      if (plan.mode === "unknown" && plan.codec?.video === "hevc") {
   508→        throw new Error(
   509→          `This file uses ${plan.codec.video.toUpperCase()} video which browsers cannot play — unlock it with debrid instead.`,
   510→        );
   511→      }
   512→      clearInterval(ticker);
   513→      setP2pPhase("idle");
   514→      setP2pInfo(null);
   515→      onPick({
   516→        ...stream,
   517→        url: plan.url,
   518→        behaviorHints: {
   519→          ...stream.behaviorHints,
   520→          filename: filename ?? plan.codec?.filename,
   521→        },
   522→        p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   523→      });
   524→    } catch (e) {
   525→      clearInterval(ticker);
   526→      setP2pPhase("idle");
   527→      setP2pInfo(null);
   528→      if ((e as Error)?.name === "AbortError") return;
   529→      fail(e instanceof Error ? e.message : "P2P stream failed");
   530→    }
   531→  };
   532→
   533→  const rowActivate = () => {
   534→    if (isTorrent) {
   535→      if (debridKey !== null) void unlock();
   536→      else if (p2pEnabled) void playP2p();
   537→      else onNeedsSetup();
   538→    } else {
   539→      onPick(stream);
   540→    }
   541→  };
   542→
   543→  const busy = resolving || p2pPhase !== "idle";
   544→  const direct = !!stream.url && !isTorrent;
   545→
   546→  return (
   547→    <div
   548→      role="button"
   549→      tabIndex={0}
   550→      onClick={rowActivate}
   551→      onKeyDown={(e) => {
   552→        if (e.key === "Enter" || e.key === " ") {
   553→          e.preventDefault();
   554→          rowActivate();
   555→        }
   556→      }}
   557→      aria-label={`${titleLine} — play stream`}
   558→      className={cn(
   559→        "harbor-tv-focus harbor-picker-row md-state relative w-full flex flex-wrap items-center gap-3.5 rounded-[var(--md-sys-shape-corner-medium)] border p-3 text-left transition-colors cursor-pointer",
   560→        cached ? "border-emerald-500/30 bg-emerald-500/5" : "border-edge-soft bg-[var(--md-sys-color-surface-container)]",
   561→        direct && "border-l-2 border-l-emerald-500/50",
   562→      )}
   563→    >
   564→      {busy && (
   565→        <span className="absolute inset-0 z-10 rounded-[var(--md-sys-shape-corner-medium)] bg-black/60 backdrop-blur-[2px] flex flex-col items-center justify-center gap-1" aria-live="polite">
   566→          {resolving ? (
   567→            <Loader2 className="w-5 h-5 text-accent animate-spin" />
   568→          ) : (
   569→            <>
   570→              <span className="flex items-center gap-2 text-xs font-semibold text-ink">
   571→                <Wifi className="w-4 h-4 text-accent animate-pulse" aria-hidden />
   572→                {p2pPhase === "planning" ? "Preparing file…" : "Joining swarm…"}
   573→              </span>
   574→              <span className="text-[10px] text-ink-subtle">
   575→                {p2pInfo?.peers ? `${p2pInfo.peers} peers` : "looking for peers"} · {p2pInfo?.elapsed ?? 0}s
   576→              </span>
   577→            </>
   578→          )}
   579→        </span>
   580→      )}
   581→      <div className="flex flex-col items-center gap-1 w-14 shrink-0">
   582→        <span className={cn("rounded-lg px-2 py-1 text-[11px] font-bold", cached ? "bg-emerald-500/20 text-emerald-300" : "bg-raised text-ink-muted")}>
   583→          {p?.resolution ?? "—"}
   584→        </span>
   585→        {p?.hdrFormat && (
   586→          <span className="rounded px-1.5 py-0.5 text-[9px] font-bold bg-accent-soft text-accent">{p.hdrFormat}</span>

   589→      <div className="min-w-0 flex-1">

   600→            {p?.seeders !== undefined && <span className="flex items-center gap-1"><Signal className="w-3 h-3" />{p.seeders} seeders</span>}

   606→          {isTorrent && (
   607→            <span className="rounded border border-edge-soft bg-raised px-1 py-0.5 text-[9px] font-bold text-ink-muted">
   608→              DEBRID
   609→            </span>
   610→          )}
   611→          {isTorrent && p2pEnabled && (
   612→            <span className="rounded border border-accent/40 bg-accent/10 px-1 py-0.5 text-[9px] font-bold text-accent flex items-center gap-1">
   613→              <Network className="w-2.5 h-2.5" aria-hidden /> P2P
   614→            </span>
   615→          )}
   616→          {stream.addonName ?? "addon"}
   617→        </p>
   618→        {err && (
   619→          <p className="basis-full w-full text-xs text-danger mt-1" role="alert">
   620→            {err}
   621→          </p>
   622→        )}
   623→      </div>
   624→      {isTorrent && (
   625→        <div className="flex items-center gap-1.5 shrink-0">
   626→          {p2pEnabled && (
   627→            <button
   628→              type="button"
   629→              onClick={(e) => void playP2p(e)}
   630→              disabled={busy}
   631→              className={cn(
   632→                "harbor-tv-focus md-btn md-state h-9! px-3.5!",
   633→                debridKey === null ? "md-btn-filled" : "md-btn-tonal",
   634→              )}
   635→              aria-label={debridKey === null ? "Play via P2P" : "Play via P2P torrent engine"}
   636→              title={debridKey === null ? "Play free via the built-in P2P engine" : "Play via the built-in P2P engine (slower)"}
   637→            >
   638→              {p2pPhase !== "idle" ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Network className="w-3.5 h-3.5" aria-hidden />}
   639→              {p2pPhase !== "idle" ? "Connecting…" : "Play"}
   640→            </button>
   641→          )}
   642→          <button
   643→            type="button"
   644→            onClick={(e) => void unlock(e)}
   645→            disabled={resolving}
   646→            className={cn(
   647→              "harbor-tv-focus md-btn md-state h-9! px-3.5!",
   648→              debridKey === null ? "md-btn-tonal" : "md-btn-filled",
   649→            )}
   650→            aria-label="Unlock with debrid"
   651→            title={debridKey === null ? "Connect a debrid service for instant links" : "Unlock with your debrid service"}
   652→          >
   653→            {resolving ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <KeyRound className="w-3.5 h-3.5" aria-hidden />}
   654→            {resolving ? "Unlocking…" : debridKey === null ? "Debrid" : "Unlock"}
   655→          </button>
   656→        </div>
   657→      )}
   658→    </div>
   659→  );

   662→// Choose the torrent file index client-side when metadata arrives via polling
   663→function pickFileIndex(files: { index: number; name: string; length: number }[], filename?: string): number {
   664→  const VIDEO_RE = /\.(mp4|m4v|webm|mkv|avi|mov|ts|flv|wmv|mpg|mpeg)$/i;
   665→  const videos = files.filter((f) => VIDEO_RE.test(f.name) || f.length > 100 * 1024 * 1024);
   666→  const pool = videos.length > 0 ? videos : files;
   667→  if (filename) {
   668→    const want = filename.toLowerCase().replace(/[\s._-]+/g, "").slice(0, 24);
   669→    const byName = pool.find((f) => f.name.toLowerCase().replace(/[\s._-]+/g, "").includes(want));
   670→    if (byName) return byName.index;
   671→  }
   672→  return pool.reduce((a, b) => (b.length > a.length ? b : a)).index;
   673→}
