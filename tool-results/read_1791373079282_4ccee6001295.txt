    25→type Phase = "resolving" | "loading" | "playing" | "error";
    26→
    27→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    28→  const pop = useNav((s) => s.pop);
    29→  const replace = useNav((s) => s.replace);
    30→  const settings = useSettings((s) => s.settings);
    31→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    32→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    33→  const [error, setError] = useState<string | null>(null);
    34→  const [switcherOpen, setSwitcherOpen] = useState(false);
    35→
    36→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    37→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    38→  const handleError = useCallback((msg: string) => {
    39→    setPhase("error");
    40→    setError(msg);
    41→  }, []);
    42→
    43→  // Resolve stream via addons when payload.url is empty (instant play path)
    44→  useEffect(() => {
    45→    if (current.url || phase !== "resolving") return;
    46→    let alive = true;
    47→    (async () => {
    48→      const { useAddons: addons } = await import("@/lib/harbor/store");
    49→      try {
    50→        const targetId = current.videoId ?? current.metaId;
    51→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
    52→        if (!alive) return;
    53→        const ranked = runPipeline(results);
    54→        const best = ranked.find((s) => s.url) ?? ranked[0];
    55→        if (!best || !best.url) {
    56→          setPhase("error");
    57→          setError(
    58→            ranked.length > 0
    59→              ? "No browser-playable stream found. Open the stream picker to choose manually (torrent streams require a debrid-cached source)."
    60→              : "No streams found from your addons.",
    61→          );
    62→          return;
    63→        }
    64→        setCurrent((c) => ({
    65→          ...c,
    66→          url: best.url!,
    67→          streamTitle: best.title ?? c.streamTitle,
    68→          streamDescription: best.description,
    69→        }));
    70→        setPhase("loading");
    71→      } catch (e) {
    72→        if (!alive) return;
    73→        setPhase("error");
    74→        setError(e instanceof Error ? e.message : "Failed to resolve streams");
    75→      }
    76→    })();
    77→    return () => {
    78→      alive = false;
    79→    };
    80→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
    81→
    82→  const setPayload = useCallback((p: PlayerPayload) => {
    83→    setCurrent(p);
    84→    setPhase(p.url ? "loading" : "resolving");
    85→    setError(null);
    86→  }, []);
    87→
    88→  return (
    89→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
    90→      {phase === "resolving" && (
    91→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
    92→          <div className="relative">
    93→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
    94→          </div>
    95→          <p className="text-sm text-ink-muted">Finding the best stream…</p>
    96→        </div>
    97→      )}
    98→      {phase === "error" ? (
    99→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   100→          <AlertTriangle className="w-12 h-12 text-danger" />
   101→          <p className="text-sm text-ink-muted max-w-md">{error}</p>
   102→          <div className="flex gap-3">
   103→            <button
   104→              type="button"
   105→              onClick={() => setSwitcherOpen(true)}
   106→              className="rounded-xl bg-accent px-5 py-2.5 text-sm font-bold text-black hover:brightness-110"
   107→            >
   108→              Open stream picker
   109→            </button>
   110→            <button
   111→              type="button"
   112→              onClick={pop}
   113→              className="rounded-xl bg-raised px-5 py-2.5 text-sm font-semibold text-ink hover:bg-raised/70"
   114→            >
   115→              Back
   116→            </button>
   117→          </div>
   118→        </div>
   119→      ) : (
   120→        current.url && (
   121→          <VideoStage
   122→            key={current.url}
   123→            payload={current}
   124→            settings={settings}
   125→            onPhase={handlePhase}
   126→            onError={handleError}
   127→          />
   128→        )
   129→      )}
   130→
   131→      {/* Top bar */}
   132→      {phase !== "error" && (
   133→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   134→          <button
   135→            type="button"
   136→            onClick={() => {
   137→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   138→              pop();
   139→            }}
   140→            className="w-9 h-9 rounded-full bg-black/50 flex items-center justify-center text-ink hover:bg-black/70"
   141→            aria-label="Close player (Esc)"
   142→          >
   143→            <X className="w-5 h-5" />
   144→          </button>
   145→          <div className="min-w-0">
   146→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   147→            {current.episode && (
   148→              <p className="text-xs text-ink-muted">
   149→                S{current.season}:E{current.episode}
   150→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   151→              </p>
   152→            )}
   153→          </div>
   154→          <button
   155→            type="button"
   156→            onClick={() => setSwitcherOpen(true)}
   157→            className="ml-auto flex items-center gap-1.5 rounded-full bg-black/50 hover:bg-black/70 px-3 py-1.5 text-xs font-medium text-ink"
   158→            title="Switch stream (W)"
   159→          >
   160→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   161→          </button>
   162→        </div>
   163→      )}
   164→
   165→      {switcherOpen && (
   166→        <PickerOverlay
   167→          type={current.type}
   168→          id={current.deepLink?.id ?? current.metaId}
   169→          videoId={current.videoId ?? current.deepLink?.videoId}
   170→          season={current.season}
   171→          episode={current.episode}
   172→        />
   173→      )}
   174→    </div>
   175→  );
   176→}

   178→// ---------------- Video stage ----------------
   179→function VideoStage({
   180→  payload,
   181→  settings,
   182→  onPhase,
   183→  onError,
   184→}: {
   185→  payload: PlayerPayload;
   186→  settings: ReturnType<typeof useSettings.getState>["settings"];
   187→  onPhase: (p: "loading" | "playing") => void;
   188→  onError: (msg: string) => void;
   189→}) {
   190→  const videoRef = useRef<HTMLVideoElement>(null);
   191→  const containerRef = useRef<HTMLDivElement>(null);
   192→  const hlsRef = useRef<Hls | null>(null);
   193→  const [ready, setReady] = useState(false);
   194→  const [playing, setPlaying] = useState(false);
   195→  const [muted, setMuted] = useState(false);
   196→  const [volume, setVolume] = useState(1);
   197→  const [time, setTime] = useState(0);
   198→  const [duration, setDuration] = useState(0);
   199→  const [buffering, setBuffering] = useState(true);
   200→  const [fullscreen, setFullscreen] = useState(false);
   201→  const [speed, setSpeed] = useState(1);
   202→  const [controlsVisible, setControlsVisible] = useState(true);
   203→  const [volumeHudVisible, setVolumeHudVisible] = useState(false);
   204→  const volumeHudTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   205→  const [subMenuOpen, setSubMenuOpen] = useState(false);
   206→  const [settingsMenuOpen, setSettingsMenuOpen] = useState(false);
   207→  const [subtitles, setSubtitles] = useState<LoadedSubtitle[]>([]);
   208→  const [activeSub, setActiveSub] = useState<number>(-1); // -1 = off
   209→  const [subCue, setSubCue] = useState<SubCue | null>(null);
   210→  const [availableTracks, setAvailableTracks] = useState<{ label: string }[]>([]);
   211→  const controlsTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   212→  const seekHandled = useRef(false);
   213→  // Trakt scrobble state (dedupe: one scrobble per action per 10s)
   214→  const scrobbleStartSent = useRef(false);
   215→  const scrobbleLastSent = useRef<Record<"start" | "pause" | "stop", number>>({
   216→    start: 0,
   217→    pause: 0,
   218→    stop: 0,
   219→  });
   220→
   221→  const resumeAt = useRef(
   222→    payload.resumeAt ??
   223→      (settings.resumePlayback && payload.metaId
   224→        ? resumeMsFor(payload.metaId, payload.videoId) / 1000
   225→        : 0),
   226→  );
   227→
   228→  // ---- Attach source (HLS or native) ----
   229→  useEffect(() => {
   230→    const video = videoRef.current;
   231→    if (!video || !payload.url) return;
   232→    let cancelled = false;
   233→    const isHls = /\.m3u8(\?|$)/i.test(payload.url) || payload.url.includes("m3u8");
   234→
   235→    const tryPlay = () => {
   236→      video.play().catch(() => {
   237→        /* autoplay blocked; user gesture needed */
   238→      });
   239→    };
   240→
   241→    if (isHls && !video.canPlayType("application/vnd.apple.mpegurl")) {
   242→      const hls = new Hls({ enableWorker: true, maxBufferLength: 30 });
   243→      hlsRef.current = hls;
   244→      hls.loadSource(payload.url);
   245→      hls.attachMedia(video);
   246→      hls.on(Hls.Events.MANIFEST_PARSED, () => {
   247→        if (cancelled) return;
   248→        onPhase("loading");
   249→        tryPlay();
   250→      });
   251→      hls.on(Hls.Events.ERROR, (_evt, data) => {
   252→        if (cancelled || !data.fatal) return;
   253→        if (data.type === Hls.ErrorTypes.NETWORK_ERROR) {
   254→          hls.startLoad();
   255→        } else if (data.type === Hls.ErrorTypes.MEDIA_ERROR) {
   256→          hls.recoverMediaError();
   257→        } else {
   258→          onError("Stream failed to load. It may be offline or blocked by CORS.");
   259→        }
   260→      });
   261→    } else {
   262→      video.src = payload.url;
   263→      onPhase("loading");
   264→      tryPlay();
   265→    }
   266→
   267→    return () => {
   268→      cancelled = true;
   269→      if (hlsRef.current) {
   270→        hlsRef.current.destroy();
   271→        hlsRef.current = null;
   272→      }
   273→    };
   274→  }, [payload.url, onPhase, onError]);
   275→
   276→  // ---- Resume ----
   277→  useEffect(() => {
   278→    const video = videoRef.current;
   279→    if (!video) return;
   280→    if (resumeAt.current > 1 && !seekHandled.current) {
   281→      const seek = () => {
   282→        if (!seekHandled.current && video.duration > 0 && resumeAt.current < video.duration - 5) {
   283→          video.currentTime = resumeAt.current;
   284→        }
   285→        seekHandled.current = true;
   286→      };
   287→      if (video.readyState >= 1) seek();
   288→      else video.addEventListener("loadedmetadata", seek, { once: true });
   289→    }
   290→  }, [ready]);
   291→
   292→  // ---- Media event handlers ----
   293→  useEffect(() => {
   294→    const video = videoRef.current;
   295→    if (!video) return;
   296→    const onTime = () => {
   297→      setTime(video.currentTime);
   298→      if (video.duration > 0) {
   299→        setDuration(video.duration);
   300→        // persist CW every 5s
   301→        saveProgressThrottled(payload, video.currentTime, video.duration, settings);
   302→      }
   303→    };
   304→    const onLoaded = () => {
   305→      setDuration(video.duration);
   306→      setReady(true);
   307→      setBuffering(false);
   308→      // populate audio tracks
   309→      updateTrackList(video, setAvailableTracks);
   310→    };
   311→    const onPlay = () => setPlaying(true);
   312→    const onPause = () => setPlaying(false);
   313→    const onWaiting = () => setBuffering(true);
   314→    const onPlaying = () => {
   315→      setBuffering(false);
   316→      onPhase("playing");
   317→    };
   318→    // Named onVideoError so it never shadows the onError prop from the closure
   319→    const onVideoError = () => {
   320→      onError("This stream can't be played in the browser (codec or CORS). Try another stream via the stream picker.");
   321→    };
   322→    const onVolume = () => {
   323→      setMuted(video.muted);
   324→      setVolume(video.volume);
   325→      setVolumeHudVisible(true);
   326→      if (volumeHudTimer.current) clearTimeout(volumeHudTimer.current);
   327→      volumeHudTimer.current = setTimeout(() => setVolumeHudVisible(false), 1400);
   328→    };
   329→    video.addEventListener("timeupdate", onTime);
   330→    video.addEventListener("loadedmetadata", onLoaded);
   331→    video.addEventListener("play", onPlay);
   332→    video.addEventListener("pause", onPause);
   333→    video.addEventListener("waiting", onWaiting);
   334→    video.addEventListener("playing", onPlaying);
   335→    video.addEventListener("error", onVideoError);
   336→    video.addEventListener("volumechange", onVolume);
   337→    return () => {
   338→      video.removeEventListener("timeupdate", onTime);
   339→      video.removeEventListener("loadedmetadata", onLoaded);
   340→      video.removeEventListener("play", onPlay);
   341→      video.removeEventListener("pause", onPause);
   342→      video.removeEventListener("waiting", onWaiting);
   343→      video.removeEventListener("playing", onPlaying);
   344→      video.removeEventListener("error", onVideoError);
   345→      video.removeEventListener("volumechange", onVolume);
   346→    };
   347→  }, [payload, settings, onPhase, onError]);
   348→
   349→  // ---- Finalize watch progress on unmount + when tab is hidden/closed (Harbor flush parity) ----
   350→  useEffect(() => {
   351→    const flush = () => {
   352→      const video = videoRef.current;
   353→      if (video && video.duration > 0) {
   354→        saveProgress(payload, video.currentTime, video.duration, settings, true);
   355→      }
   356→    };
   357→    const onVisibility = () => {
   358→      if (document.visibilityState === "hidden") flush();
   359→    };
   360→    document.addEventListener("visibilitychange", onVisibility);
   361→    window.addEventListener("pagehide", flush);
   362→    return () => {
   363→      document.removeEventListener("visibilitychange", onVisibility);
   364→      window.removeEventListener("pagehide", flush);
   365→      flush();
   366→    };
   367→  }, []);
   368→
   369→  // ---- Trakt scrobble (fire-and-forget; demo/unmatched streams never scrobble) ----
   370→  useEffect(() => {
   371→    const video = videoRef.current;
   372→    if (!video) return;
   373→    if (!/^tt\d+$/.test(payload.metaId)) return;
   374→    const type: "movie" | "episode" = payload.type === "series" ? "episode" : "movie";
   375→    if (type === "episode" && (payload.season === undefined || payload.episode === undefined)) return;
   376→
   377→    scrobbleStartSent.current = false;
   378→    scrobbleLastSent.current = { start: 0, pause: 0, stop: 0 };
   379→
   380→    const send = (action: "start" | "pause" | "stop") => {
   381→      const now = Date.now();
   382→      if (now - scrobbleLastSent.current[action] < 10_000) return; // dedupe per action
   383→      scrobbleLastSent.current[action] = now;
   384→      if (action === "start") scrobbleStartSent.current = true;
   385→      void useTrakt.getState().scrobble(action, {
   386→        progress: video.duration > 0 ? Math.min(100, Math.max(0, (video.currentTime / video.duration) * 100)) : 0,
   387→        type,
   388→        imdbId: payload.metaId,
   389→        season: payload.season,
   390→        episode: payload.episode,
   391→      });
   392→    };
   393→
   394→    let playStart = 0;
   395→    let cumulativeMs = 0;
   396→    const foldElapsed = () => {
   397→      if (playStart > 0) {
   398→        cumulativeMs += Date.now() - playStart;
   399→        playStart = 0;
   400→      }
   401→    };
   402→
   403→    const onPlaying = () => {
   404→      if (playStart === 0) playStart = Date.now();
   405→      const total = cumulativeMs + (Date.now() - playStart);
   406→      if (total > 3000 && !scrobbleStartSent.current) send("start");
   407→    };
   408→    const onPause = () => {
   409→      foldElapsed();
   410→      if (video.ended) return; // ended handler sends the final stop
   411→      if (scrobbleStartSent.current) send("pause");
   412→    };
   413→    const onEnded = () => {
   414→      foldElapsed();
   415→      if (scrobbleStartSent.current) send("stop");
   416→    };
   417→    const onPageHide = () => {
   418→      if (scrobbleStartSent.current) send("stop");
   419→    };
   420→    const onVisibility = () => {
   421→      if (document.visibilityState === "hidden" && scrobbleStartSent.current) send("stop");
   422→    };
   423→
   424→    video.addEventListener("playing", onPlaying);
   425→    video.addEventListener("pause", onPause);
   426→    video.addEventListener("ended", onEnded);
   427→    window.addEventListener("pagehide", onPageHide);
   428→    document.addEventListener("visibilitychange", onVisibility);
   429→    return () => {
   430→      // Unmount or stream switch: close the scrobble session
   431→      if (scrobbleStartSent.current) send("stop");
   432→      video.removeEventListener("playing", onPlaying);
   433→      video.removeEventListener("pause", onPause);
   434→      video.removeEventListener("ended", onEnded);
   435→      window.removeEventListener("pagehide", onPageHide);
   436→      document.removeEventListener("visibilitychange", onVisibility);
   437→    };
   438→  }, [payload]);
   439→
   440→  // ---- Subtitle cue rendering ----
   441→  useEffect(() => {
   442→    const video = videoRef.current;
   443→    if (!video) return;
   444→    const onTime = () => {
   445→      if (activeSub < 0 || subtitles[activeSub] === undefined) {
   446→        setSubCue(null);
   447→        return;
   448→      }
   449→      setSubCue(findActiveCue(subtitles[activeSub].cues, video.currentTime));
   450→    };
   451→    video.addEventListener("timeupdate", onTime);
   452→    return () => video.removeEventListener("timeupdate", onTime);
   453→  }, [activeSub, subtitles]);
   454→
   455→  // ---- Auto-load preferred subtitles ----
   456→  useEffect(() => {
   457→    if (payload.type === "movie" || payload.type === "series") {
   458→      let alive = true;
   459→      (async () => {
   460→        try {
   461→          const { useAddons: addons } = await import("@/lib/harbor/store");
   462→          const subs = await fetchSubtitles(
   463→            addons.getState().addons,
   464→            payload.type,
   465→            payload.metaId,
   466→            payload.videoId ?? payload.deepLink?.videoId,
   467→          );
   468→          if (!alive || subs.length === 0) return;
   469→          const preferred = settings.preferredSubLangs;
   470→          // score: preferred language first
   471→          const sorted = [...subs].sort((a, b) => subScore(b, preferred) - subScore(a, preferred));
   472→          const top = sorted.slice(0, 8);
   473→          const loaded: LoadedSubtitle[] = [];
   474→          for (const s of top) {
   475→            try {
   476→              const res = await fetchAndParseSubtitle(s.url, s.m ?? s.lang ?? "Subtitle", s.lang);
   477→              loaded.push(res);
   478→            } catch {
   479→              /* skip bad sub */
   480→            }
   481→          }
   482→          if (!alive || loaded.length === 0) return;
   483→          setSubtitles(loaded);
   484→          if (!settings.subtitlesOffByDefault) setActiveSub(0);
   485→        } catch {
   486→          /* subtitles optional */
   487→        }
   488→      })();
   489→      return () => {
   490→        alive = false;
   491→      };
   492→    }
   493→     
   494→  }, [payload.metaId, payload.videoId]);
   495→
   496→  // ---- Fullscreen tracking ----
   497→  useEffect(() => {
   498→    const onFs = () => setFullscreen(!!document.fullscreenElement);
   499→    document.addEventListener("fullscreenchange", onFs);
   500→    return () => document.removeEventListener("fullscreenchange", onFs);
   501→  }, []);
   502→
   503→  // ---- Controls auto-hide ----
   504→  const bumpControls = useCallback(() => {
   505→    setControlsVisible(true);
   506→    if (controlsTimer.current) clearTimeout(controlsTimer.current);
   507→    controlsTimer.current = setTimeout(() => {
   508→      if (videoRef.current && !videoRef.current.paused) {
   509→        setControlsVisible(false);
   510→        setSubMenuOpen(false);
   511→        setSettingsMenuOpen(false);
   512→      }
   513→    }, 3200);
   514→  }, []);
   515→
   516→  useEffect(() => {
   517→    bumpControls();
   518→    return () => {
   519→      if (controlsTimer.current) clearTimeout(controlsTimer.current);
   520→    };
   521→  }, [bumpControls]);
   522→
   523→  // ---- Keyboard shortcuts (ported from Harbor hotkeys.ts) ----
   524→  useEffect(() => {
   525→    const onKey = (e: KeyboardEvent) => {
   526→      const video = videoRef.current;
   527→      if (!video) return;
   528→      const target = e.target as HTMLElement;
   529→      if (target.tagName === "INPUT" || target.tagName === "TEXTAREA") return;
   530→      bumpControls();
   531→      switch (e.key) {
   532→        case " ":
   533→          e.preventDefault();
   534→          if (video.paused) video.play();
   535→          else video.pause();
   536→          break;
   537→        case "f":
   538→          toggleFullscreen();
   539→          break;
   540→        case "u":
   541→          togglePip();
   542→          break;
   543→        case "m":
   544→          video.muted = !video.muted;
   545→          break;
   546→        case "ArrowLeft":

   971→function fmtTime(t: number): string {
   972→  if (!Number.isFinite(t) || t < 0) return "0:00";
   973→  const h = Math.floor(t / 3600);
   974→  const m = Math.floor((t % 3600) / 60);
   975→  const s = Math.floor(t % 60);
   976→  return h > 0
   977→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
   978→    : `${m}:${String(s).padStart(2, "0")}`;
   979→}

   981→function subScore(s: RawSubtitle, preferred: string[]): number {
   982→  let score = 0;
   983→  const lang = (s.lang ?? "").toLowerCase();
   984→  preferred.forEach((p, i) => {
   985→    if (lang.includes(p.toLowerCase().split(" ")[0])) score += 100 - i * 10;
   986→  });
   987→  return score;

   990→let lastSave = 0;
   991→let lastHistorySave = 0;
   992→function saveProgressThrottled(
   993→  payload: PlayerPayload,
   994→  position: number,
   995→  duration: number,
   996→  settings: ReturnType<typeof useSettings.getState>["settings"],
   997→) {
   998→  const now = Date.now();
   999→  if (now - lastSave < 5000) return;
  1000→  lastSave = now;
  1001→  saveProgress(payload, position, duration, settings, false);
  1002→  // Record into history every 30s of progress so stats survive reloads/crashes
  1003→  if (now - lastHistorySave >= 30_000) {
  1004→    lastHistorySave = now;
  1005→    pushHistory({
  1006→      id: payload.metaId,
  1007→      type: payload.type === "series" ? "series" : "movie",
  1008→      name: payload.title,
  1009→      poster: payload.poster,
  1010→      videoId: payload.videoId,
  1011→      season: payload.season,
  1012→      episode: payload.episode,
  1013→      episodeName: payload.episodeName,
  1014→      positionMs: Math.round(position * 1000),
  1015→      durationMs: Math.round(duration * 1000),
  1016→    });
  1017→  }
  1018→}

  1020→function saveProgress(
  1021→  payload: PlayerPayload,
  1022→  position: number,
  1023→  duration: number,
  1024→  settings: ReturnType<typeof useSettings.getState>["settings"],
  1025→  final: boolean,
  1026→) {
  1027→  if (duration <= 0) return;
  1028→  const entry = {
  1029→    id: payload.metaId,
  1030→    type: payload.type === "series" ? ("series" as const) : ("movie" as const),
  1031→    name: payload.title,
  1032→    poster: payload.poster,
  1033→    season: payload.season,
  1034→    episode: payload.episode,
  1035→    videoId: payload.videoId,
  1036→    episodeName: payload.episodeName,
  1037→    positionMs: Math.round(position * 1000),
  1038→    durationMs: Math.round(duration * 1000),
  1039→    t: Date.now(),
  1040→  };
  1041→  const progress = position / duration;
  1042→  if (!final || progress < 0.92) {
  1043→    upsertCw(entry);
  1044→  }
  1045→  if (final) {
  1046→    pushHistory(entry);
  1047→  }
  1048→}
