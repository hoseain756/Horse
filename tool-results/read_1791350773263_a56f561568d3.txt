  1702→  const root = document.documentElement;
  1703→  for (const [k, v] of Object.entries(resolveTokens(theme))) {
  1704→    root.style.setProperty(k, v);

  1707→  const fontPairId = preset?.fontPair ?? theme.fontPair;

  1709→  if (theme.customFontId) {

  1712→    root.style.setProperty("--font-sans", `${custom}, ${pair.sans}`);
