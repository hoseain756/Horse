   498→function PosterDockPreview({ transitionMs }: { transitionMs: number }) {
   499→  const trackRef = useRef<HTMLDivElement>(null);
   500→  const frameRef = useRef<number | null>(null);
   501→  const pointerXRef = useRef<number | null>(null);
   502→
   503→  const update = () => {
   504→    frameRef.current = null;
   505→    const track = trackRef.current;
   506→    const pointerX = pointerXRef.current;
   507→    if (!track || pointerX === null) return;
   508→
   509→    updatePosterDock({
   510→      track,
   511→      pointerX,
   512→      cellWidth: 80,
   513→      gap: 12,
   514→      scrollPosition: 0,
   515→      rtl: false,
   516→      transitionMs,
   517→    });
   518→  };
   519→
   520→  const schedule = (pointerX: number) => {
   521→    pointerXRef.current = pointerX;
   522→    if (frameRef.current === null) frameRef.current = requestAnimationFrame(update);
   523→  };
   524→
   525→  useEffect(
   526→    () => () => {
   527→      if (frameRef.current !== null) cancelAnimationFrame(frameRef.current);
   528→      if (trackRef.current) resetPosterDock(trackRef.current);
   529→    },
   530→    [],
   531→  );
   532→
   533→  return (

   624→function PxField({
   625→  value,
   626→  min,
   627→  max,
   628→  onCommit,
   629→}: {
   630→  value: number;
   631→  min: number;
   632→  max: number;
   633→  onCommit: (v: number) => void;
   634→}) {
   635→  const [editing, setEditing] = useState(false);
   636→  const [draft, setDraft] = useState(String(value));
   637→  useEffect(() => {
   638→    if (!editing) setDraft(String(value));
   639→  }, [value, editing]);
   640→  const commit = () => {
   641→    const n = Math.max(min, Math.min(max, Math.round(Number(draft) || value)));
   642→    onCommit(n);
   643→    setEditing(false);
   644→  };
   645→  if (editing) {
   646→    return (
   647→      <input
   648→        type="number"
   649→        autoFocus
   650→        value={draft}
   651→        min={min}
   652→        max={max}
   653→        onChange={(e) => setDraft(e.target.value)}
   654→        onBlur={commit}
   655→        onKeyDown={(e) => {
   656→          if (e.key === "Enter") commit();
   657→          else if (e.key === "Escape") setEditing(false);
   658→        }}
   659→        className="w-14 rounded-md border border-ink bg-canvas px-1.5 py-0.5 text-[12px] tabular-nums text-ink outline-none"
   660→      />
   661→    );
   662→  }
   663→  return (
   664→    <button
   665→      type="button"
   666→      onClick={() => setEditing(true)}
   667→      title="Click to edit"
   668→      className="rounded px-1 py-0.5 tabular-nums text-ink-muted transition-colors hover:bg-raised hover:text-ink"
   669→    >
   670→      {value}px
   671→    </button>
   672→  );
   673→}

   675→const POSTER_SIZES = [
   676→  { value: "compact", label: "Compact", scale: 0.8 },
   677→  { value: "dense", label: "Dense", scale: 0.9 },
   678→  { value: "standard", label: "Standard", scale: 1 },
   679→  { value: "balanced", label: "Balanced", scale: 1.15 },
   680→  { value: "comfort", label: "Comfort", scale: 1.3 },
   681→  { value: "large", label: "Large", scale: 1.5 },
   682→] as const;
   683→
   684→function posterSizeKey(scale: number): string {
   685→  let best: (typeof POSTER_SIZES)[number] = POSTER_SIZES[0];
   686→  for (const p of POSTER_SIZES) {
   687→    if (Math.abs(p.scale - scale) < Math.abs(best.scale - scale)) best = p;
   688→  }
   689→  return best.value;
