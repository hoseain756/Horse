   246→      <div className="md-sheet rounded-b-none! md:rounded-b-[28px]! w-full md:max-w-3xl md:max-h-[82vh] max-h-[88vh] flex flex-col overflow-hidden shadow-[var(--md-sys-elevation-3)]">

   338→            >
   339→              Connect

   343→
   344→        {/* Body */}
   345→        <div ref={bodyRef} className="harbor-picker-body flex-1 overflow-y-auto harbor-scroll px-5 py-4 space-y-5">

   347→            <div className="space-y-3">

   360→                  <div className="w-16 h-7 rounded-lg bg-white/10 shrink-0" />

   365→

   378→
   379→          {!loading &&
   380→            filteredTiers.map(({ tier, items }) => (
   381→              <section key={tier}>
   382→                <div className="flex items-center gap-2 mb-2.5">
   383→                  <span aria-hidden className={cn("h-1.5 w-1.5 rounded-full shrink-0", tierDotColor(tier))} />
   384→                  <h3 className="text-xs font-bold uppercase tracking-widest text-ink-muted">{tier}</h3>

   391→                      key={`${s.addonId}-${i}`}
   392→                      stream={s}

   396→                      showQuality={settings.showQualityInfo}
   397→                    />
   398→                  ))}
   399→                </div>
   400→              </section>
   401→            ))}
   402→        </div>
   403→      </div>
   404→      {debridDialogOpen && <DebridSetupDialog onClose={() => setDebridDialogOpen(false)} />}

   409→// Quality-ladder dot per tier header: emerald top, amber mid, neutral low —

   414→  if (tier === "720p" || tier === "SD") return "bg-zinc-500";
   415→  return "bg-zinc-600";
   416→}

   455→function StreamRow({
   456→  stream,
   457→  onPick,
   458→  onNeedsSetup,
   459→  configured,
   460→  showQuality,
   461→}: {
   462→  stream: Stream;
   463→  onPick: (s: Stream) => void;
   464→  onNeedsSetup: () => void;
   465→  configured: boolean;
   466→  showQuality: boolean;
   467→}) {
   468→  const { toast } = useToast();
   469→  const debridKey = useDebrid((s) => s.apiKey);
   470→  const debridResolve = useDebrid((s) => s.resolve);
   471→  const [resolving, setResolving] = useState(false);
   472→  // P2P connecting state: null = idle, else seconds elapsed while joining the swarm
   473→  const [p2pPhase, setP2pPhase] = useState<"idle" | "joining" | "planning">("idle");
   474→  const [p2pInfo, setP2pInfo] = useState<{ peers: number; elapsed: number } | null>(null);
   475→  const [err, setErr] = useState<string | null>(null);
   476→  const errTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   477→  const p2pAbort = useRef<AbortController | null>(null);
   478→
   479→  useEffect(() => {
   480→    return () => {
   481→      if (errTimer.current) clearTimeout(errTimer.current);
   482→      p2pAbort.current?.abort();
   483→    };
   484→  }, []);
   485→
   486→  const cached = Object.values(stream.cached ?? {}).some(Boolean);
   487→  const isTorrent = !!stream.infoHash && !stream.url; // torrent without a direct URL → debrid unlock OR P2P
   488→  const p2pEnabled = useSettings((s) => s.settings.p2pEnabled);
   489→  const p = stream.parsed;
   490→  // Playback compatibility badge (computed BEFORE the user picks)
   491→  const cls: StreamClass = classifyStream(stream);
   492→  const badgeInfo = BADGE_STYLES[cls.badge];
   493→  const titleLine = (stream.title ?? stream.description ?? "Stream").split("\n")[0];
   494→  const detailLine = (stream.description ?? "").split("\n").filter(Boolean).slice(1).join(" ").slice(0, 140);
   495→
   496→  const fail = (msg: string) => {
   497→    setErr(msg);
   498→    toast({ title: "Stream unavailable", description: msg, variant: "destructive" });
   499→    if (errTimer.current) clearTimeout(errTimer.current);
   500→    errTimer.current = setTimeout(() => setErr(null), 6000);
   501→  };
   502→
   503→  const unlock = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   504→    e?.stopPropagation();
   505→    e?.preventDefault();
   506→    if (resolving) return;
   507→    if (debridKey === null) {
   508→      onNeedsSetup();
   509→      return;
   510→    }
   511→    setResolving(true);
   512→    setErr(null);
   513→    const res = await debridResolve(
   514→      stream.infoHash!,
   515→      stream.behaviorHints?.filename,
   516→      stream.parsed?.size,
   517→    );
   518→    setResolving(false);
   519→    if ("url" in res && res.url) {
   520→      onPick({
   521→        ...stream,
   522→        url: res.url,
   523→        behaviorHints: {
   524→          ...stream.behaviorHints,
   525→          ...(res.filename ? { filename: res.filename } : {}),
   526→        },
   527→      });
   528→    } else {
   529→      const msg = "error" in res ? res.error : "Unlock failed";
   530→      fail(msg);
   531→    }
   532→  };
   533→
   534→  // Play a torrent through the server-side P2P engine: join swarm → wait for
   535→  // metadata → pick the right file (native container vs ffmpeg remux) → play.
   536→  const playP2p = async (e?: React.MouseEvent | React.KeyboardEvent) => {
   537→    e?.stopPropagation();
   538→    e?.preventDefault();
   539→    if (p2pPhase !== "idle") return;
   540→    if (!useSettings.getState().settings.p2pEnabled) {
   541→      onNeedsSetup();
   542→      return;
   543→    }
   544→    setErr(null);
   545→    setP2pPhase("joining");
   546→    setP2pInfo({ peers: 0, elapsed: 0 });
   547→    const startedAt = Date.now();
   548→    const abort = new AbortController();
   549→    p2pAbort.current = abort;
   550→    const infoHash = stream.infoHash!.toLowerCase();
   551→
   552→    // progress ticker while the swarm join runs
   553→    const ticker = setInterval(() => {
   554→      setP2pInfo((prev) => (prev ? { ...prev, elapsed: Math.round((Date.now() - startedAt) / 1000) } : prev));
   555→    }, 1000);
   556→
   557→    try {
   558→      // 1) prepare (join swarm). 202 = metadata still downloading → poll status.
   559→      const prep = await p2pPrepare(infoHash, {
   560→        fileIdx: stream.fileIdx,
   561→        filename: stream.behaviorHints?.filename ?? stream.parsed?.filename,
   562→      });
   563→
   564→      // 2) poll while discovering (up to ~100s; the engine keeps the swarm alive)
   565→      while (prep.pending) {
   566→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   567→        if (Date.now() - startedAt > 100_000) {
   568→          throw new Error(
   569→            "Could not find peers for this torrent. The swarm may be dead or this network blocks BitTorrent — a debrid service would unlock it instantly.",
   570→          );
   571→        }
   572→        await new Promise((r) => setTimeout(r, 2500));
   573→        if (abort.signal.aborted) throw new DOMException("cancelled", "AbortError");
   574→        const st = await p2pStatus(infoHash).catch(() => null);
   575→        setP2pInfo((prev) => (prev ? { ...prev, peers: st?.peers ?? prev.peers } : prev));
   576→        if (st?.ready) break;
   577→      }
   578→
   579→      // 3) choose serving mode (native range stream vs ffmpeg remux for mkv)
   580→      setP2pPhase("planning");
   581→      await refreshP2pCapabilities(); // engine-level HEVC conversion flag
   582→      const fileIdx = prep.pending
   583→        ? await (async () => {
   584→            const st = await p2pStatus(infoHash);
   585→            const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   586→            const idx = Number.isInteger(stream.fileIdx)
   587→              ? stream.fileIdx!
   588→              : st.files
   589→                ? pickFileIndex(st.files, filename)
   590→                : 0;
   591→            return idx;
   592→          })()
   593→        : prep.fileIdx;
   594→      const filename = stream.behaviorHints?.filename ?? stream.parsed?.filename;
   595→      let plan: P2pPlaybackPlan;
   596→      try {
   597→        plan = await p2pPlan(infoHash, fileIdx, filename);
   598→      } catch {
   599→        plan = { key: infoHash, fileIdx, url: `/stream/${infoHash}/${fileIdx}`, mode: "unknown" };
   600→      }
   601→      if (plan.mode === "unknown" && plan.codec?.video === "hevc") {
   602→        throw new Error(
   603→          `This file uses ${plan.codec.video.toUpperCase()} video which browsers cannot play — unlock it with debrid instead.`,
   604→        );
   605→      }
   606→      clearInterval(ticker);
   607→      setP2pPhase("idle");
   608→      setP2pInfo(null);
   609→      onPick({
   610→        ...stream,
   611→        url: plan.url,
   612→        behaviorHints: {
   613→          ...stream.behaviorHints,
   614→          filename: filename ?? plan.codec?.filename,
   615→        },
   616→        p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   617→      });
   618→    } catch (e) {
   619→      clearInterval(ticker);
   620→      setP2pPhase("idle");
   621→      setP2pInfo(null);
   622→      if ((e as Error)?.name === "AbortError") return;
   623→      fail(e instanceof Error ? e.message : "P2P stream failed");
   624→    }
   625→  };
   626→
   627→  const rowActivate = () => {
   628→    // External destinations (YouTube ids / openInBrowser links) never enter the player
   629→    if (cls.verdict === "external") {
   630→      const target = cls.ytId
   631→        ? `https://www.youtube.com/watch?v=${cls.ytId}`
   632→        : cls.externalUrl;
   633→      if (target) window.open(target, "_blank", "noopener,noreferrer");
   634→      return;
   635→    }
   636→    if (isTorrent) {
   637→      if (debridKey !== null) void unlock();
   638→      else if (p2pEnabled) void playP2p();
   639→      else onNeedsSetup();
   640→    } else {
   641→      onPick(stream);
   642→    }
   643→  };
   644→
   645→  const busy = resolving || p2pPhase !== "idle";
   646→  const direct = !!stream.url && !isTorrent;
   647→
   648→  return (
   649→    <div
   650→      role="button"
   651→      tabIndex={0}
   652→      onClick={rowActivate}
   653→      onKeyDown={(e) => {
   654→        if (e.key === "Enter" || e.key === " ") {
   655→          e.preventDefault();
   656→          rowActivate();
   657→        }
   658→      }}
   659→      aria-label={`${titleLine} — play stream`}
   660→      className={cn(
   661→        "harbor-tv-focus harbor-picker-row md-state relative w-full flex flex-wrap items-center gap-3.5 rounded-[var(--md-sys-shape-corner-medium)] border p-3 text-left transition-colors cursor-pointer",
   662→        cached ? "border-emerald-500/30 bg-emerald-500/5" : "border-edge-soft bg-[var(--md-sys-color-surface-container)]",
   663→        direct && "border-l-2 border-l-emerald-500/50",
   664→      )}
   665→    >
   666→      {busy && (
   667→        <span className="absolute inset-0 z-10 rounded-[var(--md-sys-shape-corner-medium)] bg-black/60 backdrop-blur-[2px] flex flex-col items-center justify-center gap-1" aria-live="polite">
   668→          {resolving ? (
   669→            <Loader2 className="w-5 h-5 text-accent animate-spin" />
   670→          ) : (
   671→            <>
   672→              <span className="flex items-center gap-2 text-xs font-semibold text-ink">
   673→                <Wifi className="w-4 h-4 text-accent animate-pulse" aria-hidden />
   674→                {p2pPhase === "planning" ? "Preparing file…" : "Joining swarm…"}
   675→              </span>
   676→              <span className="text-[10px] text-ink-subtle">
   677→                {p2pInfo?.peers ? `${p2pInfo.peers} peers` : "looking for peers"} · {p2pInfo?.elapsed ?? 0}s
   678→              </span>
   679→            </>
   680→          )}
   681→        </span>
   682→      )}
   683→      <div className="flex flex-col items-center gap-1 w-14 shrink-0">
   684→        <span className={cn("rounded-lg px-2 py-1 text-[11px] font-bold", cached ? "bg-emerald-500/20 text-emerald-300" : "bg-raised text-ink-muted")}>
   685→          {p?.resolution ?? "—"}
   686→        </span>
   687→        {p?.hdrFormat && (
   688→          <span className="rounded px-1.5 py-0.5 text-[9px] font-bold bg-accent-soft text-accent">{p.hdrFormat}</span>
   689→        )}
   690→        {/* Playability badge: where this stream will actually play */}
   691→        <span
   692→          className={cn(
   693→            "rounded-full px-1.5 py-0.5 text-[8.5px] font-bold uppercase tracking-wide inline-flex items-center gap-1",
   694→            badgeInfo.className,
   695→          )}
   696→          title={cls.reasons.length > 0 ? cls.reasons.join(" · ") : badgeInfo.title}
   697→        >
   698→          <span aria-hidden className={cn("h-1.5 w-1.5 rounded-full", badgeInfo.dot)} />
   699→          {badgeInfo.label}
   700→        </span>
   701→      </div>
   702→      <div className="min-w-0 flex-1">
   703→        <p className="harbor-clamp-1 text-sm font-medium text-ink">{titleLine}</p>
   704→        {detailLine && !showQuality && (
   705→          <p className="harbor-clamp-1 text-xs text-ink-subtle mt-0.5">{detailLine}</p>
   706→        )}
   707→        {showQuality && (
   708→          <div className="flex items-center gap-2.5 mt-1 text-[10px] text-ink-subtle flex-wrap">
   709→            {p?.codec && (
   710→              <span className="flex items-center gap-1"><FileVideo className="w-3 h-3" />{p.codec}</span>
   711→            )}
   712→            {p?.size && <span className="flex items-center gap-1"><HardDrive className="w-3 h-3" />{formatSize(p.size)}</span>}
   713→            {p?.seeders !== undefined && <span className="flex items-center gap-1"><Signal className="w-3 h-3" />{p.seeders} seeders</span>}
   714→            {p?.source && <span className="flex items-center gap-1"><Gauge className="w-3 h-3" />{p.source}</span>}
   715→          </div>
   716→        )}
   717→        <p className="text-[10px] text-ink-subtle mt-0.5 flex items-center gap-1.5 flex-wrap">
   718→          {cached && <Zap className="w-3 h-3 text-emerald-400" />}
   719→          {isTorrent && (
   720→            <span className="rounded border border-edge-soft bg-raised px-1 py-0.5 text-[9px] font-bold text-ink-muted">
   721→              DEBRID
   722→            </span>
   723→          )}
   724→          {isTorrent && p2pEnabled && (
   725→            <span className="rounded border border-accent/40 bg-accent/10 px-1 py-0.5 text-[9px] font-bold text-accent flex items-center gap-1">
   726→              <Network className="w-2.5 h-2.5" aria-hidden /> P2P
   727→            </span>
   728→          )}
   729→          {stream.addonName ?? "addon"}
   730→        </p>
   731→        {err && (
   732→          <p className="basis-full w-full text-xs text-danger mt-1" role="alert">
   733→            {err}
   734→          </p>
   735→        )}
   736→      </div>
   737→      {isTorrent && (
   738→        <div className="flex items-center gap-1.5 shrink-0">
   739→          {p2pEnabled && (
   740→            <button
   741→              type="button"
   742→              onClick={(e) => void playP2p(e)}
   743→              disabled={busy}
   744→              className={cn(
   745→                "harbor-tv-focus md-btn md-state h-9! px-3.5!",
   746→                debridKey === null ? "md-btn-filled" : "md-btn-tonal",
   747→              )}
   748→              aria-label={debridKey === null ? "Play via P2P" : "Play via P2P torrent engine"}
   749→              title={debridKey === null ? "Play free via the built-in P2P engine" : "Play via the built-in P2P engine (slower)"}
   750→            >
   751→              {p2pPhase !== "idle" ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Network className="w-3.5 h-3.5" aria-hidden />}
   752→              {p2pPhase !== "idle" ? "Connecting…" : "Play"}
   753→            </button>
   754→          )}
   755→          <button
   756→            type="button"
   757→            onClick={(e) => void unlock(e)}
   758→            disabled={resolving}
   759→            className={cn(
   760→              "harbor-tv-focus md-btn md-state h-9! px-3.5!",
   761→              debridKey === null ? "md-btn-tonal" : "md-btn-filled",
   762→            )}
   763→            aria-label="Unlock with debrid"
   764→            title={debridKey === null ? "Connect a debrid service for instant links" : "Unlock with your debrid service"}
   765→          >
   766→            {resolving ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <KeyRound className="w-3.5 h-3.5" aria-hidden />}
   767→            {resolving ? "Unlocking…" : debridKey === null ? "Debrid" : "Unlock"}
   768→          </button>
   769→        </div>
   770→      )}
   771→    </div>
   772→  );
   773→}

   775→// Choose the torrent file index client-side when metadata arrives via polling
   776→function pickFileIndex(files: { index: number; name: string; length: number }[], filename?: string): number {
   777→  const VIDEO_RE = /\.(mp4|m4v|webm|mkv|avi|mov|ts|flv|wmv|mpg|mpeg)$/i;
   778→  const videos = files.filter((f) => VIDEO_RE.test(f.name) || f.length > 100 * 1024 * 1024);
   779→  const pool = videos.length > 0 ? videos : files;
   780→  if (filename) {
   781→    const want = filename.toLowerCase().replace(/[\s._-]+/g, "").slice(0, 24);
   782→    const byName = pool.find((f) => f.name.toLowerCase().replace(/[\s._-]+/g, "").includes(want));
   783→    if (byName) return byName.index;
   784→  }
   785→  return pool.reduce((a, b) => (b.length > a.length ? b : a)).index;
   786→}
