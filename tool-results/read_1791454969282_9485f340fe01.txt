    43→    43→export function readRecents(): string[] {
    44→    44→  if (typeof window === "undefined") return [];
    45→    45→  try {
    46→    46→    const arr = JSON.parse(window.localStorage.getItem(RECENTS_KEY) ?? "[]") as unknown;
    47→    47→    return Array.isArray(arr) ? arr.filter((x): x is string => typeof x === "string").slice(0, MAX_RECENTS) : [];
    48→    48→  } catch {
    49→    49→    return [];
    50→    50→  }
    51→    51→}

    53→    53→export function writeRecent(q: string) {
    54→    54→  if (typeof window === "undefined") return;
    55→    55→  const trimmed = q.trim();
    56→    56→  if (!trimmed) return;
    57→    57→  const next = [trimmed, ...readRecents().filter((r) => r.toLowerCase() !== trimmed.toLowerCase())].slice(0, MAX_RECENTS);
    58→    58→  try {
    59→    59→    window.localStorage.setItem(RECENTS_KEY, JSON.stringify(next));
    60→    60→  } catch {
    61→    61→    /* quota — ignore */
    62→    62→  }
    63→    63→}

    65→    65→// A flattened result row (used for keyboard navigation)
    66→    66→type FlatRow =
    67→    67→  | { kind: "meta"; id: string; meta: Meta; group: string }
    68→    68→  | { kind: "person"; id: string; name: string; sub?: string | null; img?: string | null; url?: string }
    69→    69→  | { kind: "action"; id: string; action: "ai" | "see-all" }
    70→    70→  | { kind: "recent"; id: string; query: string }
    71→    71→  | { kind: "destination"; id: string; entry: HubEntry };

    73→    73→/** Quick Access destinations matching the query (EN + Arabic labels/keywords).
    74→    74→ *  Second way to reach the pages moved from the sidebar into Settings.
    75→    75→ *  Exported for the full-screen SearchOverlay (round-22 parity). */
    76→    76→export function matchingDestinations(q: string, lang: string): HubEntry[] {
    77→    77→  const needle = q.trim().toLowerCase();
    78→    78→  if (needle.length < 2) return [];
    79→    79→  const ar = /^ar(-|_|$)/i.test(lang);
    80→    80→  return HUB_ENTRIES.filter((e) => {
    81→    81→    const label = hubLabel(e, lang).toLowerCase();
    82→    82→    const words = [...e.keywordsEn, ...e.keywordsAr, label];
    83→    83→    return words.some((w) => w.toLowerCase().includes(needle));
    84→    84→  }).slice(0, ar ? 4 : 3);
    85→    85→}

    89→    89→  const openFullSearch = useNav((s) => s.setSearchOpen);
    90→    90→  const uiLanguage = useSettings((s) => s.settings.uiLanguage);
    91→    91→
    92→    92→  const [expanded, setExpanded] = useState(false);

    94→    94→  const [query, setQuery] = useState("");
    95→    95→  const [phase, setPhase] = useState<Phase>("idle");
    96→    96→  const [movies, setMovies] = useState<Meta[]>([]);
    97→    97→  const [series, setSeries] = useState<Meta[]>([]);

   100→   100→  const [recents, setRecents] = useState<string[]>([]);
   101→   101→  const [trending, setTrending] = useState<Meta[]>([]);
   102→   102→  const [activeIdx, setActiveIdx] = useState(-1);
   103→   103→  const runToken = useRef(0);
   104→   104→  const debounce = useRef<ReturnType<typeof setTimeout> | null>(null);

   106→   106→  const inputRef = useRef<HTMLInputElement>(null);
   107→   107→  const listRef = useRef<HTMLDivElement>(null);
   108→   108→
   109→   109→  // Focus bridge + outside click + Esc handled here

   112→   112→      setExpanded(true);
   113→   113→      setHidden(false); // hotkey reveal: never focus an off-screen bar

   116→   116→    };

   118→   118→      if (!wrapRef.current?.contains(e.target as Node)) {
   119→   119→        setExpanded(false);
   120→   120→        setActiveIdx(-1);
   121→   121→      }
   122→   122→    };

   124→   124→    document.addEventListener("mousedown", onPointer);
   125→   125→    return () => {
   126→   126→      window.removeEventListener("harbor:focus-floating-search", onFocus);
   127→   127→      document.removeEventListener("mousedown", onPointer);
   128→   128→    };
   129→   129→  }, []);
   130→   130→
   131→   131→  // Scroll-direction visibility: scrolling DOWN hides the bar (content first),
   132→   132→  // scrolling UP (or returning near the top) reveals it again. While the bar
   133→   133→  // is expanded/focused it always stays visible.
   134→   134→  useEffect(() => {
   135→   135→    let lastY = window.scrollY;
   136→   136→    const onScroll = () => {
   137→   137→      const y = window.scrollY;
   138→   138→      const dy = y - lastY;
   139→   139→      lastY = y;
   140→   140→      if (expanded) return; // never hide while the user is searching
   141→   141→      if (y > 90 && dy > 4) setHidden(true);
   142→   142→      else if (dy < -4 || y <= 90) setHidden(false);
   143→   143→    };
   144→   144→    window.addEventListener("scroll", onScroll, { passive: true });
   145→   145→    return () => window.removeEventListener("scroll", onScroll);
   146→   146→  }, [expanded]);
   147→   147→
   148→   148→  // Trending suggestions (idle state) — fetched once per session, errors are silent
   149→   149→  useEffect(() => {
   150→   150→    let alive = true;
   151→   151→    (async () => {
   152→   152→      const [mv, sr] = await Promise.allSettled([
   153→   153→        fetchCinemetaCatalog("movie", "trending"),
   154→   154→        fetchCinemetaCatalog("series", "trending"),
   155→   155→      ]);
   156→   156→      if (!alive) return;
   157→   157→      const items = [
   158→   158→        ...(mv.status === "fulfilled" ? mv.value : []),
   159→   159→        ...(sr.status === "fulfilled" ? sr.value : []),
   160→   160→      ].slice(0, 8);
   161→   161→      setTrending(items);
   162→   162→    })();
   163→   163→    return () => {
   164→   164→      alive = false;
   165→   165→    };
   166→   166→  }, []);
   167→   167→
   168→   168→  const runInstant = useCallback(async (q: string) => {
   169→   169→    const token = ++runToken.current;
   170→   170→    setPhase("searching");
   171→   171→    try {
   172→   172→      const jobs: Promise<void>[] = [];
   173→   173→      jobs.push(
   174→   174→        searchCinemeta(q).then((r) => {
   175→   175→          if (token !== runToken.current) return;
   176→   176→          setMovies(r.movies);
   177→   177→          setSeries(r.series);
   178→   178→        }),
   179→   179→      );
   180→   180→      jobs.push(
   181→   181→        searchAddonCatalogs(installedAddons(), q).then((r) => {
   182→   182→          if (token !== runToken.current) return;
   183→   183→          setAddonHits(r);
   184→   184→        }),
   185→   185→      );
   186→   186→      if (tmdbEnabled()) {
   187→   187→        jobs.push(
   188→   188→          tmdbMultiSearch(q).then((r) => {
   189→   189→            if (token !== runToken.current) return;
   190→   190→            setPeople(r.person.slice(0, MAX_PER_GROUP));
   191→   191→          }),
   192→   192→        );
   193→   193→      } else {
   194→   194→        setPeople([]);
   195→   195→      }
   196→   196→      const res = await Promise.allSettled(jobs);
   197→   197→      if (token !== runToken.current) return;
   198→   198→      const allFailed = res.every((r) => r.status === "rejected");
   199→   199→      setPhase(allFailed && q.trim().length >= 2 ? "error" : "idle");
   200→   200→    } catch {
   201→   201→      if (token === runToken.current) setPhase("error");
   202→   202→    }
   203→   203→  }, []);
   204→   204→

   206→   206→  // (never synchronously in the effect body) to satisfy the set-state lint rule.

   208→   208→    if (debounce.current) clearTimeout(debounce.current);
   209→   209→    const q = query.trim();

   211→   211→    if (q.length < 2) {
   212→   212→      debounce.current = setTimeout(() => {
   213→   213→        if (token !== runToken.current) return;
   214→   214→        setMovies([]);
   215→   215→        setSeries([]);
   216→   216→        setAddonHits([]);
   217→   217→        setPeople([]);
   218→   218→        setPhase("idle");
   219→   219→      }, 0);
   220→   220→    } else {
   221→   221→      debounce.current = setTimeout(() => void runInstant(q), 250);
   222→   222→    }
   223→   223→    return () => {
   224→   224→      if (debounce.current) clearTimeout(debounce.current);
   225→   225→    };
   226→   226→  }, [query, runInstant]);
   227→   227→
   228→   228→  const openMeta = useCallback(
   229→   229→    (m: Meta) => {
   230→   230→      writeRecent(query || m.name);
   231→   231→      setRecents(readRecents());
   232→   232→      setExpanded(false);
   233→   233→      setActiveIdx(-1);
   234→   234→      push({ kind: "detail", type: m.type, id: m.id });
   235→   235→    },
   236→   236→    [push, query],
   237→   237→  );
   238→   238→
   239→   239→  const openFull = useCallback(

   246→   246→    },
   247→   247→    [openFullSearch],
   248→   248→  );
   249→   249→
   250→   250→  // Flatten the dropdown for keyboard navigation
   251→   251→  const rows = useMemo<FlatRow[]>(() => {
   252→   252→    const out: FlatRow[] = [];
   253→   253→    const q = query.trim();
   254→   254→    if (q.length < 2) {
   255→   255→      for (const r of recents) out.push({ kind: "recent", id: `recent:${r}`, query: r });
   256→   256→      for (const m of trending) out.push({ kind: "meta", id: `trend:${m.type}:${m.id}`, meta: m, group: "Trending" });
   257→   257→      return out;
   258→   258→    }
   259→   259→    for (const m of movies.slice(0, MAX_PER_GROUP)) out.push({ kind: "meta", id: `m:${m.id}`, meta: m, group: "Movies" });
   260→   260→    for (const m of series.slice(0, MAX_PER_GROUP)) out.push({ kind: "meta", id: `s:${m.id}`, meta: m, group: "Series" });
   261→   261→    for (const e of matchingDestinations(q, uiLanguage)) out.push({ kind: "destination", id: `dest:${e.id}`, entry: e });
   262→   262→    for (const m of people) out.push({ kind: "person", id: `p:${m.id}`, name: m.name, sub: m.known_for_department, img: m.profile_path, url: m.url });

   266→   266→  }, [query, movies, series, addonHits, people, recents, trending, uiLanguage]);
   267→   267→
   268→   268→  const onInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {

   273→   273→    } else if (e.key === "ArrowUp") {
   274→   274→      e.preventDefault();
   275→   275→      e.stopPropagation();
   276→   276→      setActiveIdx((i) => Math.max(-1, i - 1));

   280→   280→      if (row?.kind === "meta") openMeta(row.meta);

   283→   283→        inputRef.current?.focus();
   284→   284→      } else if (row?.kind === "destination") {
   285→   285→        setExpanded(false);
   286→   286→        setActiveIdx(-1);

   297→   297→      e.stopPropagation();

   299→   299→        setQuery("");
   300→   300→      } else {
   301→   301→        setExpanded(false);
   302→   302→        setActiveIdx(-1);

   304→   304→      }
   305→   305→    }
   306→   306→  };
   307→   307→
   308→   308→  // Keep the active option in view

   312→   312→    el?.scrollIntoView({ block: "nearest" });
   313→   313→  }, [activeIdx]);

   316→   316→  const searching = phase === "searching";
   317→   317→  const showDropdown = expanded;

   331→   331→        expanded && "md:w-[42rem]",
   332→   332→        // Scrolled-down: slide the bar fully off-screen (keeps md centering)
   333→   333→        hidden && "-translate-y-[160%] opacity-0 pointer-events-none",
   334→   334→      )}
   335→   335→      aria-hidden={hidden || undefined}

   342→   342→      <button
   343→   343→        type="button"
   344→   344→        onClick={() => openFullSearch(true)}
   345→   345→        className="md:hidden md-state harbor-tv-focus flex h-12 w-12 items-center justify-center rounded-full harbor-glass border border-edge-soft shadow-[0_10px_36px_-12px_rgba(0,0,0,0.65),0_2px_10px_rgba(0,0,0,0.3)]"

   350→   350→
   351→   351→      {/* md+ docked bar (behavior unchanged) */}

   358→   358→          expanded && "shadow-[0_18px_50px_-12px_rgba(0,0,0,0.75)]",
