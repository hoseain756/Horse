   283→function ThemePanel() {
   284→  const settings = useSettings((s) => s.settings);
   285→  const update = useSettings((s) => s.update);
   286→  const fileRef = useRef<HTMLInputElement>(null);
   287→  const { toast } = useToast();
   288→  const [studioOpen, setStudioOpen] = useState(false);
   289→  const [userThemes, setUserThemes] = useState<UserTheme[]>([]);
   290→
   291→  // Load saved themes after mount (client-only storage; avoids hydration mismatch)
   292→  useEffect(() => {
   293→    const t = setTimeout(() => setUserThemes(loadUserThemes()), 0);
   294→    return () => clearTimeout(t);
   295→  }, []);
   296→
   297→  const setPreset = (preset: string) => {
   298→    update({ theme: { ...settings.theme, preset, customColors: null, customName: null } });
   299→  };
   300→
   301→  const applyUserTheme = (t: UserTheme) => {
   302→    update({
   303→      theme: {
   304→        ...settings.theme,
   305→        preset: "custom",
   306→        customColors: t.colors,
   307→        fontPair: t.fontPair,
   308→        customLayout: t.layout,
   309→        customCardStyle: t.cardStyle,
   310→        customButtonStyle: t.buttonStyle,
   311→        customName: t.name,
   312→      },
   313→    });
   314→  };
   315→
   316→  const removeUserTheme = (id: string) => {
   317→    const next = deleteUserTheme(id);
   318→    setUserThemes(next);
   319→    // If the deleted theme was active, fall back to the default preset
   320→    if (settings.theme.preset === "custom") {
   321→      const stillExists = next.some((t) => t.name === settings.theme.customName);
   322→      if (!stillExists) {
   323→        update({ theme: { ...settings.theme, preset: "cool-grey", customColors: null, customName: null } });
   324→      }
   325→    }
   326→  };
   327→
   328→  const onBgUpload = (file: File) => {
   329→    if (!file.type.startsWith("image/")) return;
   330→    if (file.size > 3 * 1024 * 1024) {
   331→      toast({ title: "Image too large", description: "Use an image under 3 MB.", variant: "destructive" });
   332→      return;
   333→    }
   334→    const reader = new FileReader();
   335→    reader.onload = () => {
   336→      update({ theme: { ...settings.theme, backgroundImage: reader.result as string } });
   337→    };
   338→    reader.readAsDataURL(file);
   339→  };
   340→
   341→  return (
   342→    <div className="space-y-7 max-w-4xl">
   343→      {/* Theme Studio */}
   344→      <div className="rounded-3xl border border-accent/30 bg-accent-soft/50 p-5">
   345→        <div className="flex items-center justify-between gap-4 flex-wrap">
   346→          <div className="min-w-0">
   347→            <h2 className="font-display text-lg font-bold text-ink flex items-center gap-2">
   348→              <Brush className="w-[18px] h-[18px] text-accent" /> Theme Studio
   349→            </h2>
   350→            <p className="text-xs text-ink-muted mt-0.5">
   351→              Build a fully custom palette, fonts and layout — with live preview.
   352→              {settings.theme.preset === "custom" && settings.theme.customName && (
   353→                <span className="text-accent font-medium"> Active: {settings.theme.customName}</span>
   354→              )}
   355→            </p>
   356→          </div>
   357→          <Button onClick={() => setStudioOpen(true)} className="bg-accent text-black hover:brightness-110">
   358→            <Brush className="w-4 h-4 mr-1.5" /> Open Theme Studio
   359→          </Button>
   360→        </div>
   361→        {userThemes.length > 0 && (
   362→          <div className="mt-4">
   363→            <p className="text-[10px] uppercase tracking-wide text-ink-subtle mb-2">Your saved themes</p>
   364→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2">
   365→              {userThemes.map((t) => (
   366→                <div
   367→                  key={t.id}
   368→                  className={cn(
   369→                    "group relative rounded-xl border p-2.5 cursor-pointer transition-all hover:scale-[1.02]",
   370→                    settings.theme.preset === "custom" && settings.theme.customName === t.name
   371→                      ? "border-accent"
   372→                      : "border-edge-soft",
   373→                  )}
   374→                  style={{ background: t.colors.canvas }}
   375→                  onClick={() => applyUserTheme(t)}
   376→                  role="button"
   377→                  tabIndex={0}
   378→                  onKeyDown={(e) => e.key === "Enter" && applyUserTheme(t)}
   379→                  aria-label={`Apply theme ${t.name}`}
   380→                >
   381→                  <div className="flex items-center justify-between">
   382→                    <div className="flex gap-1">
   383→                      {[t.colors.canvas, t.colors.accent, t.colors.elevated].map((c, i) => (
   384→                        <span key={i} className="w-3.5 h-3.5 rounded-full border border-white/20" style={{ background: c }} />
   385→                      ))}
   386→                    </div>
   387→                    <button
   388→                      type="button"
   389→                      onClick={(e) => {
   390→                        e.stopPropagation();
   391→                        removeUserTheme(t.id);
   392→                      }}
   393→                      className="opacity-0 group-hover:opacity-100 text-white/50 hover:text-danger transition-all"
   394→                      aria-label={`Delete theme ${t.name}`}
   395→                    >
   396→                      <Trash2 className="w-3.5 h-3.5" />
   397→                    </button>
   398→                  </div>
   399→                  <p className="text-xs font-semibold mt-1.5 truncate" style={{ color: t.colors.accent }}>
   400→                    {t.name}
   401→                  </p>
   402→                  <p className="text-[10px] text-white/50 capitalize">{t.layout} · {t.fontPair.split("-")[0]}</p>
   403→                </div>
   404→              ))}
   405→            </div>
   406→          </div>
   407→        )}
   408→      </div>
   409→
   410→      <div>
   411→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Theme presets</h2>
   412→        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3">
   413→          {THEME_PRESETS.map((p) => (
   414→            <button
   415→              key={p.id}
   416→              type="button"
   417→              onClick={() => setPreset(p.id)}
   418→              className={cn(
   419→                "harbor-tv-focus rounded-2xl border-2 p-3 text-left transition-all hover:scale-[1.02]",
   420→                settings.theme.preset === p.id ? "border-accent" : "border-edge-soft",
   421→              )}
   422→              style={{ background: p.canvas }}
   423→              aria-pressed={settings.theme.preset === p.id}
   424→            >
   425→              <div className="flex items-center justify-between mb-2.5">
   426→                <div className="flex gap-1.5">
   427→                  {p.swatch.map((c) => (
   428→                    <span key={c} className="w-4 h-4 rounded-full border border-white/20" style={{ background: c }} />
   429→                  ))}
   430→                </div>
   431→                {settings.theme.preset === p.id && (
   432→                  <span className="w-5 h-5 rounded-full bg-accent flex items-center justify-center">
   433→                    <Check className="w-3 h-3 text-black" />
   434→                  </span>
   435→                )}
   436→              </div>
   437→              <p className="text-sm font-semibold" style={{ color: p.accent }}>
   438→                {p.name}
   439→              </p>
   440→              <p className="text-[11px] text-white/50 capitalize">{p.layout} layout</p>
   441→            </button>
   442→          ))}
   443→        </div>
   444→      </div>
   445→
   446→      <div>
   447→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Font pairing</h2>
   448→        <div className="flex flex-wrap gap-2">
   449→          {(Object.keys(FONT_PAIRS) as FontPairId[]).map((id) => (
   450→            <button
   451→              key={id}
   452→              type="button"
   453→              onClick={() => {
   454→                update({ theme: { ...settings.theme, fontPair: id } });
   455→                // live preview: apply font immediately
   456→                const pair = FONT_PAIRS[id];
   457→                document.documentElement.style.setProperty("--font-display-var", pair.display);
   458→                document.documentElement.style.setProperty("--font-sans-var", pair.sans);
   459→              }}
   460→              className={cn(
   461→                "harbor-tv-focus rounded-xl border px-4 py-2.5 text-sm transition-colors",
   462→                settings.theme.fontPair === id ? "border-accent bg-accent-soft text-ink" : "border-edge-soft bg-elevated text-ink-muted hover:text-ink",
   463→              )}
   464→              style={{ fontFamily: FONT_PAIRS[id].display }}
   465→            >
   466→              {FONT_PAIRS[id].name}
   467→            </button>
   468→          ))}
   469→        </div>
   470→      </div>
   471→
   472→      <div>
   473→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Custom background</h2>
   474→        <div className="flex items-center gap-3 flex-wrap">
   475→          <Button variant="outline" onClick={() => fileRef.current?.click()} className="border-edge text-ink-muted">
   476→            Upload image
   477→          </Button>
   478→          {settings.theme.backgroundImage && (
   479→            <Button
   480→              variant="outline"
   481→              onClick={() => update({ theme: { ...settings.theme, backgroundImage: null } })}
   482→              className="border-edge text-ink-muted"
   483→            >
   484→              <RotateCcw className="w-4 h-4 mr-1" /> Remove
   485→            </Button>
   486→          )}
   487→          <div className="flex items-center gap-2">
   488→            <span className="text-xs text-ink-subtle">Dim</span>
   489→            <div className="w-32">
   490→              <Slider
   491→                value={[settings.theme.backgroundDim]}
   492→                min={0}
   493→                max={1}
   494→                step={0.05}
   495→                onValueChange={([v]) => update({ theme: { ...settings.theme, backgroundDim: v } })}
   496→              />
   497→            </div>
   498→          </div>
   499→          <input
   500→            ref={fileRef}
   501→            type="file"
   502→            accept="image/*"
   503→            className="hidden"
   504→            onChange={(e) => e.target.files?.[0] && onBgUpload(e.target.files[0])}
   505→            aria-label="Upload background image"
   506→          />
   507→        </div>
   508→      </div>
   509→
   510→      <ThemeStudio
   511→        open={studioOpen}
   512→        onClose={() => {
   513→          setStudioOpen(false);
   514→          // Restore the persisted theme in case the draft preview diverged, and refresh the saved list
   515→          applyTheme(useSettings.getState().settings.theme);
   516→          setUserThemes(loadUserThemes());
   517→        }}
   518→      />

   566→function DataPanel() {
   567→  const { toast } = useToast();
   568→  const settings = useSettings((s) => s.settings);
   569→  const fileRef = useRef<HTMLInputElement>(null);
   570→
   571→  const exportBackup = () => {
   572→    const data: Record<string, string> = {};
   573→    for (let i = 0; i < window.localStorage.length; i++) {
   574→      const key = window.localStorage.key(i);
   575→      if (key?.startsWith("harbor-web.")) {
   576→        const v = window.localStorage.getItem(key);
   577→        if (v) data[key] = v;
   578→      }
   579→    }
   580→    const backup = {
   581→      format: "harbor-web-backup",
   582→      version: 1,
   583→      exportedAt: new Date().toISOString(),
   584→      data,
   585→    };
   586→    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: "application/json" });
   587→    const url = URL.createObjectURL(blob);
   588→    const a = document.createElement("a");
   589→    a.href = url;
   590→    a.download = `harbor-web-backup-${new Date().toISOString().slice(0, 10)}.harbx`;
   591→    a.click();
   592→    URL.revokeObjectURL(url);
   593→    toast({ title: "Backup exported", description: `${Object.keys(data).length} keys saved.` });
   594→  };
   595→
   596→  const importBackup = async (file: File) => {
   597→    try {
   598→      const text = await file.text();
   599→      const parsed = JSON.parse(text) as { format?: string; data?: Record<string, string> };
   600→      if (parsed.format !== "harbor-web-backup" || !parsed.data) {
   601→        throw new Error("Not a Harbor Web backup file");
   602→      }
   603→      let restored = 0;
   604→      for (const [key, value] of Object.entries(parsed.data)) {
   605→        if (key.startsWith("harbor-web.")) {
   606→          window.localStorage.setItem(key, value);
   607→          restored++;
   608→        }
   609→      }
   610→      toast({ title: "Backup restored", description: `${restored} keys restored. Reloading…` });
   611→      setTimeout(() => window.location.reload(), 900);
   612→    } catch (e) {
   613→      toast({
   614→        title: "Restore failed",
   615→        description: e instanceof Error ? e.message : "Invalid file",
   616→        variant: "destructive",
   617→      });
   618→    }
   619→  };
   620→
   621→  const clearData = () => {
   622→    const keys: string[] = [];
   623→    for (let i = 0; i < window.localStorage.length; i++) {
   624→      const key = window.localStorage.key(i);
   625→      if (key?.startsWith("harbor-web.")) keys.push(key);
   626→    }
   627→    keys.forEach((k) => window.localStorage.removeItem(k));
   628→    toast({ title: "Local data cleared", description: "Reloading…" });
   629→    setTimeout(() => window.location.reload(), 900);
   630→  };
   631→
   632→  return (
   633→    <div className="space-y-3 max-w-3xl">
   634→      <CloudSyncCard />
   635→      <SettingRow title="Export backup" description={`Save settings, addons, watchlist (${getWatchlist().length} items) to a .harbx file`}>
   636→        <Button onClick={exportBackup} className="bg-accent text-black hover:brightness-110">
   637→          Export
   638→        </Button>
   639→      </SettingRow>
   640→      <SettingRow title="Restore backup" description="Import a .harbx backup file">
   641→        <Button variant="outline" onClick={() => fileRef.current?.click()} className="border-edge text-ink-muted">
   642→          Restore
   643→        </Button>
   644→        <input
   645→          ref={fileRef}
   646→          type="file"
   647→          accept=".harbx,.json"
   648→          className="hidden"
   649→          onChange={(e) => e.target.files?.[0] && importBackup(e.target.files[0])}
   650→          aria-label="Import backup file"
   651→        />
   652→      </SettingRow>
   653→      <SettingRow title="Clear local data" description="Remove all Harbor Web data from this browser">
   654→        <Button variant="outline" onClick={clearData} className="border-danger/40 text-danger hover:bg-danger/10">
   655→          Clear
   656→        </Button>
   657→      </SettingRow>
   658→      <SettingRow
   659→        title="Current settings size"
   660→        description={`${(JSON.stringify(settings).length / 1024).toFixed(1)} KB in localStorage`}
   661→      >
   662→        <span />
   663→      </SettingRow>
   664→    </div>
   665→  );
   666→}

   856→function TraktCard() {
   857→  const { toast } = useToast();
   858→  const auth = useTrakt((s) => s.auth);
   859→  const phase = useTrakt((s) => s.phase);
   860→  const device = useTrakt((s) => s.device);
   861→  const error = useTrakt((s) => s.error);
   862→  const importing = useTrakt((s) => s.importing);
   863→  const loaded = useTrakt((s) => s.loaded);
   864→  const load = useTrakt((s) => s.load);
   865→  const connect = useTrakt((s) => s.connect);
   866→  const pollOnce = useTrakt((s) => s.pollOnce);
   867→  const cancelConnect = useTrakt((s) => s.cancelConnect);
   868→  const disconnect = useTrakt((s) => s.disconnect);
   869→  const importWatchlist = useTrakt((s) => s.importWatchlist);
   870→
   871→  const [clientId, setClientId] = useState("");
   872→  const [clientSecret, setClientSecret] = useState("");
   873→  const [connecting, setConnecting] = useState(false);
   874→  const [wlCount, setWlCount] = useState<number | null>(null);
   875→
   876→  useEffect(() => {
   877→    load();
   878→  }, [load]);
   879→
   880→  // Poll Trakt while a device code is active
   881→  useEffect(() => {
   882→    if (phase !== "connecting" || !device) return;
   883→    let alive = true;
   884→    let timer: ReturnType<typeof setTimeout> | null = null;
   885→    const tick = async () => {
   886→      if (!alive) return;
   887→      const result = await pollOnce();
   888→      if (!alive) return;
   889→      if (result === "authorized") {
   890→        toast({ title: "Trakt connected", description: "You can now import your watchlist." });
   891→        return; // stop polling
   892→      }
   893→      if (result === "pending") {
   894→        timer = setTimeout(tick, 5000);
   895→      }
   896→      // "failed" → phase reset by store; stop
   897→    };
   898→    timer = setTimeout(tick, 5000);
   899→    return () => {
   900→      alive = false;
   901→      if (timer) clearTimeout(timer);
   902→    };
   903→  }, [phase, device, pollOnce, toast]);
   904→
   905→  const startConnect = async () => {
   906→    if (!clientId.trim() || !clientSecret.trim()) {
   907→      toast({ title: "Both fields are required", variant: "destructive" });
   908→      return;

   910→    setConnecting(true);
   911→    const ok = await connect(clientId, clientSecret);
   912→    setConnecting(false);
   913→    if (ok) toast({ title: "Enter the code on Trakt", description: "Approve access to continue." });
   914→    else toast({ title: "Could not start Trakt flow", description: useTrakt.getState().error ?? undefined, variant: "destructive" });
   915→  };

   918→    const added = await importWatchlist();

   920→      setWlCount(added);
   921→      toast({
   922→        title: added > 0 ? `Imported ${added} ${added === 1 ? "title" : "titles"}` : "Watchlist already up to date",
   923→        description: added > 0 ? "Merged into your Library watchlist." : undefined,
   924→      });
   925→    } else {
   926→      toast({ title: "Import failed", description: useTrakt.getState().error ?? undefined, variant: "destructive" });
   927→    }
   928→  };
   929→
   930→  if (!loaded) return null;
   931→
   932→  return (
   933→    <div className="rounded-2xl border border-edge-soft bg-elevated p-5">
   934→      <div className="flex items-start justify-between gap-4 flex-wrap mb-1">
   935→        <div className="flex items-center gap-3">
   936→          <span className="w-10 h-10 rounded-xl bg-accent-soft flex items-center justify-center">
   937→            <Plug className="w-5 h-5 text-accent" />
   938→          </span>
   939→          <div>
   940→            <h3 className="text-sm font-bold text-ink">Trakt.tv</h3>
   941→            <p className="text-xs text-ink-subtle">
   942→              {auth ? `Connected${auth.username ? ` as @${auth.username}` : ""}` : "Import your Trakt watchlist"}
   943→            </p>
   944→          </div>
   945→        </div>
   946→        {auth && (
   947→          <button
   948→            type="button"
   949→            onClick={() => {
   950→              disconnect();
   951→              setWlCount(null);
   952→              toast({ title: "Trakt disconnected" });
   953→            }}
   954→            className="harbor-tv-focus flex items-center gap-1.5 rounded-xl bg-raised px-3.5 py-2 text-xs font-semibold text-ink-muted hover:text-danger transition-colors"
   955→          >

   972→                placeholder="Your Trakt app client id"
