    50→export function SettingsView() {
    51→  const [section, setSection] = useState<Section>("basics");
    52→
    53→  // Round 8: command palette (and other callers) can jump straight to a section
    54→  useEffect(() => {
    55→    const h = (e: Event) => {
    56→      const id = (e as CustomEvent<string>).detail;
    57→      if (SECTIONS.some((s) => s.id === id)) setSection(id as Section);
    58→    };
    59→    window.addEventListener("harbor:settings-section", h);
    60→    return () => window.removeEventListener("harbor:settings-section", h);
    61→  }, []);
    62→
    63→  return (
    64→    <div className="pt-20 md:pt-14 pb-16 px-4 md:px-8 max-w-6xl">
    65→      <div className="flex items-center gap-3 mb-7">
    66→        <SettingsIcon className="w-6 h-6 text-accent" />
    67→        <h1 className="font-display text-2xl md:text-3xl font-bold text-ink">Settings</h1>
    68→      </div>
    69→
    70→      <div className="harbor-scroll-x overflow-x-auto flex gap-2 mb-7 pb-1">
    71→        {SECTIONS.map((s) => {
    72→          const Icon = s.icon;
    73→          return (
    74→            <button
    75→              key={s.id}
    76→              type="button"
    77→              onClick={() => setSection(s.id)}
    78→              className={cn(
    79→                "harbor-tv-focus flex items-center gap-2 shrink-0 rounded-full px-4 py-2 text-sm font-medium transition-colors",
    80→                section === s.id ? "bg-accent text-black" : "bg-raised text-ink-muted hover:text-ink",
    81→              )}
    82→            >
    83→              <Icon className="w-4 h-4" /> {s.label}
    84→            </button>
    85→          );
    86→        })}
    87→      </div>
    88→
    89→      {section === "basics" && <BasicsPanel />}
    90→      {section === "player" && <PlayerPanel />}
    91→      {section === "theme" && <ThemePanel />}
    92→      {section === "language" && <LanguagePanel />}
    93→      {section === "integrations" && <IntegrationsPanel />}
    94→      {section === "data" && <DataPanel />}
    95→      {section === "about" && <AboutPanel />}
    96→    </div>
    97→  );
    98→}

   302→function ThemePanel() {
   303→  const settings = useSettings((s) => s.settings);
   304→  const update = useSettings((s) => s.update);
   305→  const fileRef = useRef<HTMLInputElement>(null);
   306→  const { toast } = useToast();
   307→  const [studioOpen, setStudioOpen] = useState(false);
   308→  const [userThemes, setUserThemes] = useState<UserTheme[]>([]);
   309→
   310→  // Load saved themes after mount (client-only storage; avoids hydration mismatch)
   311→  useEffect(() => {
   312→    const t = setTimeout(() => setUserThemes(loadUserThemes()), 0);
   313→    return () => clearTimeout(t);
   314→  }, []);
   315→
   316→  const setPreset = (preset: string) => {
   317→    update({ theme: { ...settings.theme, preset, customColors: null, customName: null } });
   318→  };
   319→
   320→  const applyUserTheme = (t: UserTheme) => {
   321→    update({
   322→      theme: {
   323→        ...settings.theme,
   324→        preset: "custom",
   325→        customColors: t.colors,
   326→        fontPair: t.fontPair,
   327→        customLayout: t.layout,
   328→        customCardStyle: t.cardStyle,
   329→        customButtonStyle: t.buttonStyle,
   330→        customName: t.name,
   331→      },
   332→    });
   333→  };
   334→
   335→  const removeUserTheme = (id: string) => {
   336→    const next = deleteUserTheme(id);
   337→    setUserThemes(next);
   338→    // If the deleted theme was active, fall back to the default preset
   339→    if (settings.theme.preset === "custom") {
   340→      const stillExists = next.some((t) => t.name === settings.theme.customName);
   341→      if (!stillExists) {
   342→        update({ theme: { ...settings.theme, preset: "cool-grey", customColors: null, customName: null } });
   343→      }
   344→    }
   345→  };
   346→
   347→  const onBgUpload = (file: File) => {
   348→    if (!file.type.startsWith("image/")) return;
   349→    if (file.size > 3 * 1024 * 1024) {
   350→      toast({ title: "Image too large", description: "Use an image under 3 MB.", variant: "destructive" });
   351→      return;
   352→    }
   353→    const reader = new FileReader();
   354→    reader.onload = () => {
   355→      update({ theme: { ...settings.theme, backgroundImage: reader.result as string } });
   356→    };
   357→    reader.readAsDataURL(file);
   358→  };
   359→
   360→  return (
   361→    <div className="space-y-7 max-w-4xl">
   362→      {/* Theme Studio */}
   363→      <div className="rounded-3xl border border-accent/30 bg-accent-soft/50 p-5">
   364→        <div className="flex items-center justify-between gap-4 flex-wrap">
   365→          <div className="min-w-0">
   366→            <h2 className="font-display text-lg font-bold text-ink flex items-center gap-2">
   367→              <Brush className="w-[18px] h-[18px] text-accent" /> Theme Studio
   368→            </h2>
   369→            <p className="text-xs text-ink-muted mt-0.5">
   370→              Build a fully custom palette, fonts and layout — with live preview.
   371→              {settings.theme.preset === "custom" && settings.theme.customName && (
   372→                <span className="text-accent font-medium"> Active: {settings.theme.customName}</span>
   373→              )}
   374→            </p>
   375→          </div>
   376→          <Button onClick={() => setStudioOpen(true)} className="bg-accent text-black hover:brightness-110">
   377→            <Brush className="w-4 h-4 mr-1.5" /> Open Theme Studio
   378→          </Button>
   379→        </div>
   380→        {userThemes.length > 0 && (
   381→          <div className="mt-4">
   382→            <p className="text-[10px] uppercase tracking-wide text-ink-subtle mb-2">Your saved themes</p>
   383→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2">
   384→              {userThemes.map((t) => (
   385→                <div
   386→                  key={t.id}
   387→                  className={cn(
   388→                    "group relative rounded-xl border p-2.5 cursor-pointer transition-all hover:scale-[1.02]",
   389→                    settings.theme.preset === "custom" && settings.theme.customName === t.name
   390→                      ? "border-accent"
   391→                      : "border-edge-soft",
   392→                  )}
   393→                  style={{ background: t.colors.canvas }}
   394→                  onClick={() => applyUserTheme(t)}
   395→                  role="button"
   396→                  tabIndex={0}
   397→                  onKeyDown={(e) => e.key === "Enter" && applyUserTheme(t)}
   398→                  aria-label={`Apply theme ${t.name}`}
   399→                >
   400→                  <div className="flex items-center justify-between">
   401→                    <div className="flex gap-1">
   402→                      {[t.colors.canvas, t.colors.accent, t.colors.elevated].map((c, i) => (
   403→                        <span key={i} className="w-3.5 h-3.5 rounded-full border border-white/20" style={{ background: c }} />
   404→                      ))}
   405→                    </div>
   406→                    <button
   407→                      type="button"
   408→                      onClick={(e) => {
   409→                        e.stopPropagation();
   410→                        removeUserTheme(t.id);
   411→                      }}
   412→                      className="opacity-0 group-hover:opacity-100 text-white/50 hover:text-danger transition-all"
   413→                      aria-label={`Delete theme ${t.name}`}
   414→                    >
   415→                      <Trash2 className="w-3.5 h-3.5" />
   416→                    </button>
   417→                  </div>
   418→                  <p className="text-xs font-semibold mt-1.5 truncate" style={{ color: t.colors.accent }}>
   419→                    {t.name}
   420→                  </p>
   421→                  <p className="text-[10px] text-white/50 capitalize">{t.layout} · {t.fontPair.split("-")[0]}</p>
   422→                </div>
   423→              ))}
   424→            </div>
   425→          </div>
   426→        )}
   427→      </div>
   428→
   429→      <div>
   430→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Theme presets</h2>
   431→        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3">
   432→          {THEME_PRESETS.map((p) => (
   433→            <button
   434→              key={p.id}
   435→              type="button"
   436→              onClick={() => setPreset(p.id)}
   437→              className={cn(
   438→                "harbor-tv-focus rounded-2xl border-2 p-3 text-left transition-all hover:scale-[1.02]",
   439→                settings.theme.preset === p.id ? "border-accent" : "border-edge-soft",
   440→              )}
   441→              style={{ background: p.canvas }}
   442→              aria-pressed={settings.theme.preset === p.id}
   443→            >
   444→              <div className="flex items-center justify-between mb-2.5">
   445→                <div className="flex gap-1.5">
   446→                  {p.swatch.map((c) => (
   447→                    <span key={c} className="w-4 h-4 rounded-full border border-white/20" style={{ background: c }} />
   448→                  ))}
   449→                </div>
   450→                {settings.theme.preset === p.id && (
   451→                  <span className="w-5 h-5 rounded-full bg-accent flex items-center justify-center">
   452→                    <Check className="w-3 h-3 text-black" />
   453→                  </span>
   454→                )}
   455→              </div>
   456→              <p className="text-sm font-semibold" style={{ color: p.accent }}>
   457→                {p.name}
   458→              </p>
   459→              <p className="text-[11px] text-white/50 capitalize">{p.layout} layout</p>
   460→            </button>
   461→          ))}
   462→        </div>
   463→      </div>
   464→
   465→      <div>
   466→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Font pairing</h2>
   467→        <div className="flex flex-wrap gap-2">
   468→          {(Object.keys(FONT_PAIRS) as FontPairId[]).map((id) => (
   469→            <button
   470→              key={id}
   471→              type="button"
   472→              onClick={() => {
   473→                update({ theme: { ...settings.theme, fontPair: id } });
   474→                // live preview: apply font immediately
   475→                const pair = FONT_PAIRS[id];
   476→                document.documentElement.style.setProperty("--font-display-var", pair.display);

   478→              }}
   479→              className={cn(
   480→                "harbor-tv-focus rounded-xl border px-4 py-2.5 text-sm transition-colors",
   481→                settings.theme.fontPair === id ? "border-accent bg-accent-soft text-ink" : "border-edge-soft bg-elevated text-ink-muted hover:text-ink",
   482→              )}
   483→              style={{ fontFamily: FONT_PAIRS[id].display }}
   484→            >
   485→              {FONT_PAIRS[id].name}
   486→            </button>
   487→          ))}
   488→        </div>
   489→      </div>
   490→
   491→      <div>
   492→        <h2 className="text-sm font-semibold text-ink-muted uppercase tracking-wide mb-3">Custom background</h2>
   493→        <div className="flex items-center gap-3 flex-wrap">
   494→          <Button variant="outline" onClick={() => fileRef.current?.click()} className="border-edge text-ink-muted">
   495→            Upload image
   496→          </Button>
   497→          {settings.theme.backgroundImage && (
   498→            <Button
   499→              variant="outline"
   500→              onClick={() => update({ theme: { ...settings.theme, backgroundImage: null } })}
   501→              className="border-edge text-ink-muted"

   503→              <RotateCcw className="w-4 h-4 mr-1" /> Remove

   542→function LanguagePanel() {
   543→  const settings = useSettings((s) => s.settings);

   545→  const LANGS = ["English", "Spanish", "French", "German", "Japanese", "Korean", "Chinese", "Arabic", "Hindi", "Portuguese", "Russian", "Italian"];
   546→  const preferred = settings.preferredSubLangs;
   547→
   548→  const toggleLang = (lang: string) => {
   549→    // Read live store state (not the hook snapshot) so rapid successive toggles
   550→    // within one render batch can't silently drop each other's changes.
   551→    const cur = useSettings.getState().settings.preferredSubLangs;
   552→    const set = new Set(cur);
   553→    if (set.has(lang)) set.delete(lang);
   554→    else set.add(lang);
   555→    update({ preferredSubLangs: Array.from(set) });
   556→  };
   557→
   558→  // Move a language in the priority list (index-based; bounds-safe)
   559→  const move = (from: number, to: number) => {
   560→    const cur = useSettings.getState().settings.preferredSubLangs;
   561→    if (from === to || to < 0 || to >= cur.length) return;
   562→    const next = [...cur];
   563→    const [item] = next.splice(from, 1);
   564→    next.splice(to, 0, item);
   565→    update({ preferredSubLangs: next });
   566→  };
   567→
   568→  // Drag-to-reorder state (HTML5 DnD; buttons below cover keyboard/touch)
   569→  const [dragIdx, setDragIdx] = useState<number | null>(null);
   570→  const [overIdx, setOverIdx] = useState<number | null>(null);
   571→  const dropOn = (to: number) => {
   572→    if (dragIdx !== null) move(dragIdx, to);
   573→    setDragIdx(null);
   574→    setOverIdx(null);
   575→  };
   576→
   577→  return (
   578→    <div className="space-y-3 max-w-3xl">
   579→      <div className="rounded-2xl border border-edge-soft bg-elevated px-4 py-4">

   696→function DataPanel() {
   697→  const { toast } = useToast();
   698→  const settings = useSettings((s) => s.settings);
   699→  const fileRef = useRef<HTMLInputElement>(null);
   700→
   701→  const exportBackup = () => {
   702→    const data: Record<string, string> = {};
   703→    for (let i = 0; i < window.localStorage.length; i++) {
   704→      const key = window.localStorage.key(i);
   705→      if (key?.startsWith("harbor-web.")) {
   706→        const v = window.localStorage.getItem(key);
   707→        if (v) data[key] = v;
   708→      }
   709→    }
   710→    const backup = {
   711→      format: "harbor-web-backup",
   712→      version: 1,
   713→      exportedAt: new Date().toISOString(),
   714→      data,
   715→    };
   716→    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: "application/json" });
   717→    const url = URL.createObjectURL(blob);
   718→    const a = document.createElement("a");
   719→    a.href = url;
   720→    a.download = `harbor-web-backup-${new Date().toISOString().slice(0, 10)}.harbx`;
   721→    a.click();
   722→    URL.revokeObjectURL(url);
   723→    toast({ title: "Backup exported", description: `${Object.keys(data).length} keys saved.` });
   724→  };
   725→
   726→  const importBackup = async (file: File) => {
   727→    try {
   728→      const text = await file.text();
   729→      const parsed = JSON.parse(text) as { format?: string; data?: Record<string, string> };
   730→      if (parsed.format !== "harbor-web-backup" || !parsed.data) {
   731→        throw new Error("Not a Harbor Web backup file");
   732→      }
   733→      let restored = 0;
   734→      for (const [key, value] of Object.entries(parsed.data)) {
   735→        if (key.startsWith("harbor-web.")) {
   736→          window.localStorage.setItem(key, value);
   737→          restored++;
   738→        }
   739→      }
   740→      toast({ title: "Backup restored", description: `${restored} keys restored. Reloading…` });
   741→      setTimeout(() => window.location.reload(), 900);
   742→    } catch (e) {
   743→      toast({
   744→        title: "Restore failed",
   745→        description: e instanceof Error ? e.message : "Invalid file",
   746→        variant: "destructive",
   747→      });
   748→    }
   749→  };
   750→
   751→  const clearData = () => {
   752→    const keys: string[] = [];
   753→    for (let i = 0; i < window.localStorage.length; i++) {
   754→      const key = window.localStorage.key(i);
   755→      if (key?.startsWith("harbor-web.")) keys.push(key);
   756→    }
   757→    keys.forEach((k) => window.localStorage.removeItem(k));
   758→    toast({ title: "Local data cleared", description: "Reloading…" });
   759→    setTimeout(() => window.location.reload(), 900);
   760→  };
   761→
   762→  return (
   763→    <div className="space-y-3 max-w-3xl">
   764→      <CloudSyncCard />
   765→      <SettingRow title="Export backup" description={`Save settings, addons, watchlist (${getWatchlist().length} items) to a .harbx file`}>
   766→        <Button onClick={exportBackup} className="bg-accent text-black hover:brightness-110">
   767→          Export
   768→        </Button>
   769→      </SettingRow>
   770→      <SettingRow title="Restore backup" description="Import a .harbx backup file">
   771→        <Button variant="outline" onClick={() => fileRef.current?.click()} className="border-edge text-ink-muted">
   772→          Restore
   773→        </Button>
   774→        <input
   775→          ref={fileRef}
   776→          type="file"
   777→          accept=".harbx,.json"
   778→          className="hidden"
   779→          onChange={(e) => e.target.files?.[0] && importBackup(e.target.files[0])}
   780→          aria-label="Import backup file"
   781→        />
   782→      </SettingRow>
   783→      <SettingRow title="Clear local data" description="Remove all Harbor Web data from this browser">
   784→        <Button variant="outline" onClick={clearData} className="border-danger/40 text-danger hover:bg-danger/10">
   785→          Clear
   786→        </Button>
   787→      </SettingRow>
   788→      <SettingRow
   789→        title="Current settings size"
   790→        description={`${(JSON.stringify(settings).length / 1024).toFixed(1)} KB in localStorage`}
   791→      >
   792→        <span />
   793→      </SettingRow>
   794→    </div>
   795→  );
   796→}
