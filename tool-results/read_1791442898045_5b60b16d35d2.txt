    33→    33→export function PlayerOverlay({ payload }: { payload: PlayerPayload }) {
    34→    34→  const pop = useNav((s) => s.pop);
    35→    35→  const replace = useNav((s) => s.replace);
    36→    36→  const settings = useSettings((s) => s.settings);
    37→    37→  const [current, setCurrent] = useState<PlayerPayload>(payload);
    38→    38→  const [phase, setPhase] = useState<Phase>(payload.url ? "loading" : "resolving");
    39→    39→  const [error, setError] = useState<string | null>(null);
    40→    40→  const [switcherOpen, setSwitcherOpen] = useState(false);
    41→    41→  const [resolveStatus, setResolveStatus] = useState<string>("Finding the best stream…");
    42→    42→
    43→    43→  // Stable callbacks — identity must never change, otherwise VideoStage re-attaches HLS every render
    44→    44→  const handlePhase = useCallback((p: "loading" | "playing") => setPhase(p), []);
    45→    45→  const handleError = useCallback((msg: string) => {
    46→    46→    setPhase("error");
    47→    47→    setError(msg);
    48→    48→  }, []);
    49→    49→
    50→    50→  // Resolve stream via addons when payload.url is empty (instant play path)
    51→    51→  useEffect(() => {
    52→    52→    if (current.url || phase !== "resolving") return;
    53→    53→    let alive = true;
    54→    54→    setResolveStatus("Finding the best stream…");
    55→    55→    (async () => {
    56→    56→      const { useAddons: addons } = await import("@/lib/harbor/store");
    57→    57→      try {
    58→    58→        // Series opened without an episode id (bare tt-id): stream addons need
    59→    59→        // "id:season:episode" — resolve the default video from meta first.
    60→    60→        let targetId = current.videoId ?? current.metaId;
    61→    61→        if (current.type === "series" && !targetId.includes(":")) {
    62→    62→          const meta = await fetchMeta("series", current.metaId).catch(() => null);
    63→    63→          const vid = defaultVideoId(meta);
    64→    64→          if (vid) {
    65→    65→            targetId = vid;
    66→    66→            if (alive) {
    67→    67→              const vidMeta = (meta?.videos ?? []).find((v) => v.id === vid);
    68→    68→              setCurrent((c) => ({
    69→    69→                ...c,
    70→    70→                videoId: vid,
    71→    71→                season: c.season ?? vidMeta?.season,
    72→    72→                episode: c.episode ?? vidMeta?.episode,
    73→    73→                episodeName: c.episodeName ?? vidMeta?.name ?? vidMeta?.title,
    74→    74→              }));
    75→    75→            }
    76→    76→          }
    77→    77→        }
    78→    78→        const results = await fetchStreams(addons.getState().addons, current.type, targetId);
    79→    79→        if (!alive) return;
    80→    80→        const ranked = runPipeline(results);
    81→    81→        // Prefer browser-playable candidates: direct URLs first, then torrents whose
    82→    82→        // parsed video codec isn't HEVC (browsers can't decode it — pick the best
    83→    83→        // H.264/VP9/AV1 release instead of a 4K HEVC one that would dead-end).
    84→    84→        const p2pCandidate = (s: Stream) =>
    85→    85→          !!s.infoHash && !s.url && s.parsed?.codec !== "HEVC";
    86→    86→        const best = ranked.find((s) => s.url) ?? ranked.find(p2pCandidate) ?? ranked[0];
    87→    87→        if (!best) {
    88→    88→          setPhase("error");
    89→    89→          setError("No streams found from your addons.");
    90→    90→          return;
    91→    91→        }
    92→    92→        if (best.url) {
    93→    93→          setCurrent((c) => ({
    94→    94→            ...c,
    95→    95→            url: best.url!,
    96→    96→            streamTitle: best.title ?? c.streamTitle,
    97→    97→            streamDescription: best.description,
    98→    98→          }));
    99→    99→          setPhase("loading");
   100→   100→          return;
   101→   101→        }
   102→   102→        // Best candidate is a torrent → try the server-side P2P engine before giving up
   103→   103→        if (!best.infoHash || !useSettings.getState().settings.p2pEnabled) {
   104→   104→          setPhase("error");
   105→   105→          setError(
   106→   106→            ranked.length > 0
   107→   107→              ? "No browser-playable stream found. Open the stream picker to choose manually — torrent streams play via P2P or unlock with debrid."
   108→   108→              : "No streams found from your addons.",
   109→   109→          );
   110→   110→          return;
   111→   111→        }
   112→   112→        const { p2pPrepare, p2pStatus, p2pPlan } = await import("@/lib/harbor/p2p");
   113→   113→        const infoHash = best.infoHash.toLowerCase();
   114→   114→        setResolveStatus("Joining the torrent swarm…");
   115→   115→        const prep = await p2pPrepare(infoHash, {
   116→   116→          fileIdx: best.fileIdx,
   117→   117→          filename: best.behaviorHints?.filename ?? best.parsed?.filename,
   118→   118→        });
   119→   119→        for (let i = 0; prep.pending && i < 40 && alive; i++) {
   120→   120→          await new Promise((r) => setTimeout(r, 2500));
   121→   121→          setResolveStatus((s) => (s.startsWith("Finding peers") ? s : `Finding peers… (${i * 2.5 | 0}s)`));
   122→   122→          const st = await p2pStatus(infoHash).catch(() => null);
   123→   123→          if (st?.peers) setResolveStatus(`Finding peers… (${st.peers} found)`);
   124→   124→          if (st?.ready) break;
   125→   125→        }
   126→   126→        if (!alive) return;
   127→   127→        const st = prep.pending ? await p2pStatus(infoHash).catch(() => null) : null;
   128→   128→        if (prep.pending && !st?.ready) {
   129→   129→          setPhase("error");
   130→   130→          setError(
   131→   131→            "No peers responded for the best torrent. Open the stream picker to pick another source, or connect debrid for instant cached streams.",
   132→   132→          );
   133→   133→          return;
   134→   134→        }
   135→   135→        const fileIdx = prep.pending
   136→   136→          ? (Number.isInteger(best.fileIdx)
   137→   137→              ? best.fileIdx!
   138→   138→              : (st?.files ?? []).length > 0
   139→   139→                ? (st!.files ?? []).reduce(
   140→   140→                    (a, b) => (b.length > a.length ? b : a),
   141→   141→                    (st!.files ?? [{ index: 0, name: "", length: 0 }])[0],
   142→   142→                  ).index
   143→   143→                : 0)
   144→   144→          : prep.fileIdx;
   145→   145→        const filename = best.behaviorHints?.filename ?? best.parsed?.filename;
   146→   146→        setResolveStatus("Checking video codecs…");
   147→   147→        const plan = await p2pPlan(infoHash, fileIdx, filename).catch(() => null);
   148→   148→        if (!alive) return;
   149→   149→        if (!plan) {
   150→   150→          setPhase("error");
   151→   151→          setError("Torrent engine could not serve this file. Try another stream in the picker.");
   152→   152→          return;
   153→   153→        }
   154→   154→        setCurrent((c) => ({
   155→   155→          ...c,
   156→   156→          url: plan.url,
   157→   157→          streamTitle: best.title ?? c.streamTitle,
   158→   158→          streamDescription: best.description,
   159→   159→          p2p: { key: plan.key, fileIdx: plan.fileIdx, infoHash, mode: plan.mode },
   160→   160→        }));
   161→   161→        setPhase("loading");
   162→   162→      } catch (e) {
   163→   163→        if (!alive) return;
   164→   164→        setPhase("error");
   165→   165→        setError(e instanceof Error ? e.message : "Failed to resolve streams");
   166→   166→      }
   167→   167→    })();
   168→   168→    return () => {
   169→   169→      alive = false;
   170→   170→    };
   171→   171→  }, [current.url, current.type, current.videoId, current.metaId, phase]);
   172→   172→
   173→   173→  const setPayload = useCallback((p: PlayerPayload) => {
   174→   174→    setCurrent(p);
   175→   175→    setPhase(p.url ? "loading" : "resolving");
   176→   176→    setError(null);
   177→   177→  }, []);
   178→   178→
   179→   179→  const openSwitcher = useCallback(() => setSwitcherOpen(true), []);
   180→   180→
   181→   181→  // Advance to the next episode in-place (Up Next card / auto-advance / "n" key).
   182→   182→  // Empty url re-enters the resolving path so addons fetch streams for the new episode.
   183→   183→  const handlePlayNext = useCallback(
   184→   184→    (next: { season: number; episode: number; videoId: string; episodeName?: string }) => {
   185→   185→      setCurrent((c) => ({
   186→   186→        ...c,
   187→   187→        url: "",
   188→   188→        season: next.season,
   189→   189→        episode: next.episode,
   190→   190→        videoId: next.videoId,
   191→   191→        episodeName: next.episodeName,
   192→   192→        streamTitle: undefined,
   193→   193→        streamDescription: undefined,
   194→   194→        resumeAt: 0,
   195→   195→      }));
   196→   196→      setPhase("resolving");
   197→   197→      setError(null);
   198→   198→    },
   199→   199→    [],
   200→   200→  );
   201→   201→
   202→   202→  return (
   203→   203→    <div className="fixed inset-0 z-[180] bg-black" role="dialog" aria-modal="true" aria-label={`Playing ${current.title}`}>
   204→   204→      {phase === "resolving" && (
   205→   205→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4">
   206→   206→          <div className="relative">
   207→   207→            <Loader2 className="w-12 h-12 text-accent animate-spin" />
   208→   208→          </div>
   209→   209→          <p className="text-sm text-ink-muted" aria-live="polite">{resolveStatus}</p>
   210→   210→        </div>
   211→   211→      )}
   212→   212→      {phase === "error" ? (
   213→   213→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 px-6 text-center">
   214→   214→          <AlertTriangle className="w-12 h-12 text-danger" />
   215→   215→          <p className="text-sm text-ink-muted max-w-md">{error}</p>
   216→   216→          <div className="flex gap-3">
   217→   217→            <button type="button" onClick={() => setSwitcherOpen(true)} className="md-btn md-btn-filled md-state">
   218→   218→              Open stream picker
   219→   219→            </button>
   220→   220→            <button type="button" onClick={pop} className="md-btn md-btn-tonal md-state">
   221→   221→              Back
   222→   222→            </button>
   223→   223→          </div>
   224→   224→        </div>
   225→   225→      ) : (
   226→   226→        current.url && (
   227→   227→          <VideoStage
   228→   228→            key={current.url}
   229→   229→            payload={current}
   230→   230→            settings={settings}
   231→   231→            onPhase={handlePhase}
   232→   232→            onError={handleError}
   233→   233→            onPlayNext={handlePlayNext}
   234→   234→            onOpenSwitcher={openSwitcher}
   235→   235→          />
   236→   236→        )
   237→   237→      )}
   238→   238→
   239→   239→      {/* Top bar */}
   240→   240→      {phase !== "error" && (
   241→   241→        <div className="absolute top-0 inset-x-0 z-20 flex items-center gap-3 bg-gradient-to-b from-black/80 to-transparent px-4 py-3 pb-10 transition-opacity">
   242→   242→          <button
   243→   243→            type="button"
   244→   244→            onClick={() => {
   245→   245→              if (settings.playerConfirmLeave && !window.confirm("Leave playback?")) return;
   246→   246→              pop();
   247→   247→            }}
   248→   248→            className="md-icon-btn md-state h-12! w-12! text-white!"
   249→   249→            aria-label="Close player (Esc)"
   250→   250→          >
   251→   251→            <X className="w-5 h-5" />
   252→   252→          </button>
   253→   253→          <div className="min-w-0">
   254→   254→            <p className="harbor-clamp-1 text-sm font-semibold text-ink">{current.title}</p>
   255→   255→            {current.episode && (
   256→   256→              <p className="text-xs text-ink-muted">
   257→   257→                S{current.season}:E{current.episode}
   258→   258→                {current.episodeName ? ` · ${current.episodeName}` : ""}
   259→   259→              </p>
   260→   260→            )}
   261→   261→          </div>
   262→   262→          <button
   263→   263→            type="button"
   264→   264→            onClick={() => setSwitcherOpen(true)}
   265→   265→            className="md-state ml-auto flex items-center gap-1.5 rounded-full bg-black/50 px-3 py-1.5 text-xs font-medium text-ink"
   266→   266→            title="Switch stream (W)"
   267→   267→          >
   268→   268→            <ArrowLeftRight className="w-3.5 h-3.5" /> Streams
   269→   269→          </button>
   270→   270→        </div>
   271→   271→      )}
   272→   272→
   273→   273→      {switcherOpen && (
   274→   274→        <PickerOverlay
   275→   275→          type={current.type}
   276→   276→          id={current.deepLink?.id ?? current.metaId}
   277→   277→          videoId={current.videoId ?? current.deepLink?.videoId}
   278→   278→          season={current.season}
   279→   279→          episode={current.episode}
   280→   280→        />
   281→   281→      )}
   282→   282→    </div>
   283→   283→  );
   284→   284→}

   286→   286→// ---------------- Video stage ----------------
   287→   287→function VideoStage({
   288→   288→  payload,
   289→   289→  settings,
   290→   290→  onPhase,
   291→   291→  onError,
   292→   292→  onPlayNext,
   293→   293→  onOpenSwitcher,
   294→   294→}: {
   295→   295→  payload: PlayerPayload;
   296→   296→  settings: ReturnType<typeof useSettings.getState>["settings"];
   297→   297→  onPhase: (p: "loading" | "playing") => void;
   298→   298→  onError: (msg: string) => void;
   299→   299→  onPlayNext: (next: { season: number; episode: number; videoId: string; episodeName?: string }) => void;
   300→   300→  onOpenSwitcher: () => void;
   301→   301→}) {

   303→   303→  const videoRef = useRef<HTMLVideoElement>(null);
   304→   304→  const containerRef = useRef<HTMLDivElement>(null);
   305→   305→  const hlsRef = useRef<Hls | null>(null);
   306→   306→  // Single source of truth for subtitle rendering — the custom overlay. This
   307→   307→  // manager guarantees no native <track>/TextTrack layer ever paints alongside
   308→   308→  // it (the root cause of the duplicate-subtitles bug).
   309→   309→  const subManagerRef = useRef<SubtitleManager | null>(null);
   310→   310→  if (!subManagerRef.current) subManagerRef.current = new SubtitleManager();
   311→   311→  // Player-height-based subtitle scaling (measured via ResizeObserver)
   312→   312→  const [stageHeight, setStageHeight] = useState(0);
   313→   313→  const [ready, setReady] = useState(false);
   314→   314→  const [playing, setPlaying] = useState(false);
   315→   315→  const [muted, setMuted] = useState(false);
   316→   316→  const [volume, setVolume] = useState(1);
   317→   317→  const [time, setTime] = useState(0);
   318→   318→  const [duration, setDuration] = useState(0);
   319→   319→  const [buffering, setBuffering] = useState(true);
   320→   320→  const [fullscreen, setFullscreen] = useState(false);
   321→   321→  const [speed, setSpeed] = useState(1);
   322→   322→  const [controlsVisible, setControlsVisible] = useState(true);
   323→   323→  const [volumeHudVisible, setVolumeHudVisible] = useState(false);
   324→   324→  const volumeHudTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   325→   325→  const [subMenuOpen, setSubMenuOpen] = useState(false);
   326→   326→  const [settingsMenuOpen, setSettingsMenuOpen] = useState(false);
   327→   327→  // P2P engine: live swarm stats + one-shot remux fallback when the native stream can't demux
   328→   328→  const [p2pStats, setP2pStats] = useState<{ progress: number; peers: number; downloadSpeed: number; ready: boolean } | null>(null);
   329→   329→  const [srcOverride, setSrcOverride] = useState<string | null>(null);
   330→   330→  const remuxTriedRef = useRef(false);
   331→   331→  const [subtitles, setSubtitles] = useState<LoadedSubtitle[]>([]);
   332→   332→  // Full ranked subtitle pool + how many are loaded — powers the "Load more" affordance
   333→   333→  const subPoolRef = useRef<RawSubtitle[]>([]); // ranked by preferred languages
   334→   334→  const subEffectiveRef = useRef<RawSubtitle[]>([]); // currently displayed ranking (lang filter)
   335→   335→  const subAttemptedRef = useRef(0);
   336→   336→  const [subPoolSize, setSubPoolSize] = useState(0);
   337→   337→  const [subLoadingMore, setSubLoadingMore] = useState(false);
   338→   338→  const [subLangFilter, setSubLangFilter] = useState<string | null>(null);
   339→   339→  // Free-text search over the ranked pool (label / language / provider) — composes
   340→   340→  // with the language chips: chip re-ranks, search then narrows.
   341→   341→  const [subSearch, setSubSearch] = useState(""); // committed (drives the pool view)
   342→   342→  const [subSearchInput, setSubSearchInput] = useState(""); // typing value
   343→   343→  const subViewTokenRef = useRef(0); // invalidates stale parse batches when the view changes
   344→   344→  const subSearchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   345→   345→  // Language chips derived from the pool: chip code + available count
   346→   346→  const subLangChips = useMemo(() => {
   347→   347→    const counts = new Map<string, number>();
   348→   348→    for (const s of subPoolRef.current) {
   349→   349→      const chip = langChip(s.lang);
   350→   350→      counts.set(chip, (counts.get(chip) ?? 0) + 1);
   351→   351→    }
   352→   352→    return Array.from(counts.entries())
   353→   353→      .sort((a, b) => b[1] - a[1])
   354→   354→      .slice(0, 6);
   355→   355→  }, [subPoolSize, subLangFilter]);
   356→   356→  // How many entries the CURRENT view (chip re-rank + search filter) contains
   357→   357→  const [subEffectiveCount, setSubEffectiveCount] = useState(0);
   358→   358→  const subtitlesRef = useRef<LoadedSubtitle[]>([]);
   359→   359→  subtitlesRef.current = subtitles; // latest committed list for stable index math
   360→   360→  const [activeSub, setActiveSub] = useState<number>(-1); // -1 = off
   361→   361→  const [subCue, setSubCue] = useState<SubCue | null>(null);
   362→   362→  const [availableTracks, setAvailableTracks] = useState<{ label: string }[]>([]);
   363→   363→  const [levels, setLevels] = useState<{ index: number; label: string }[]>([]);
   364→   364→  const [levelSel, setLevelSel] = useState<number>(-1); // -1 = adaptive (auto)
   365→   365→  const [autoLevel, setAutoLevel] = useState<number>(-1); // level currently playing in auto mode
   366→   366→  const [activeAudio, setActiveAudio] = useState(0);
   367→   367→  const [bufferedEnd, setBufferedEnd] = useState(0);
   368→   368→  const [nextEp, setNextEp] = useState<{
   369→   369→    season: number;
   370→   370→    episode: number;
   371→   371→    videoId: string;
   372→   372→    episodeName?: string;
   373→   373→  } | null>(null);
   374→   374→  const [upNextDismissed, setUpNextDismissed] = useState(false);
   375→   375→  const controlsTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   376→   376→  const seekHandled = useRef(false);
   377→   377→  // Trakt scrobble state (dedupe: one scrobble per action per 10s)
   378→   378→  const scrobbleStartSent = useRef(false);
   379→   379→  const scrobbleLastSent = useRef<Record<"start" | "pause" | "stop", number>>({
   380→   380→    start: 0,
   381→   381→    pause: 0,
   382→   382→    stop: 0,
   383→   383→  });
   384→   384→
   385→   385→  const resumeAt = useRef(
   386→   386→    payload.resumeAt ??
   387→   387→      (settings.resumePlayback && payload.metaId
   388→   388→        ? resumeMsFor(payload.metaId, payload.videoId) / 1000
   389→   389→        : 0),
   390→   390→  );
   391→   391→
   392→   392→  // ---- P2P swarm stats polling ----
   393→   393→  useEffect(() => {
   394→   394→    if (!payload.p2p) {
   395→   395→      setP2pStats(null);
   396→   396→      return;
   397→   397→    }
   398→   398→    let alive = true;
   399→   399→    const key = payload.p2p.key;
   400→   400→    const tick = async () => {
   401→   401→      try {
   402→   402→        const st = await p2pStatus(key);
   403→   403→        if (!alive) return;
   404→   404→        setP2pStats({
   405→   405→          progress: st.progress ?? 0,

   808→   808→  const applySubView = useCallback(async (chip: string | null, search: string) => {

   822→   822→          `${subLabel(s)} ${s.lang ?? ""} ${s.addonName ?? s.source ?? ""}`
   823→   823→            .toLowerCase()
   824→   824→            .includes(query),
   825→   825→        )
   826→   826→      : afterChip;
   827→   827→    subEffectiveRef.current = effective;
   828→   828→    setSubEffectiveCount(effective.length);
   829→   829→    subAttemptedRef.current = 0;
   830→   830→    setSubtitles([]);
   831→   831→    setActiveSub(-1);
   832→   832→    const batch = effective.slice(0, 8);
   833→   833→    subAttemptedRef.current = batch.length;
   834→   834→    if (batch.length === 0) return;

   836→   836→    const loaded = await parseSubBatch(batch);
   837→   837→    if (token !== subViewTokenRef.current) return; // superseded by a newer view
   838→   838→    if (loaded.length > 0) {
   839→   839→      setSubtitles((prev) => mergeSubtitlesUnique(prev, loaded));
   840→   840→      setActiveSub(0);

  1705→  1705→  const h = Math.floor(t / 3600);

  1708→  1708→  return h > 0
  1709→  1709→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`

  1721→  1721→// Fetch + parse a batch of subtitle entries, skipping failures.

  1724→  1724→async function parseSubBatch(
  1725→  1725→  batch: RawSubtitle[],
  1726→  1726→  existingKeys?: Set<string>,
  1727→  1727→): Promise<LoadedSubtitle[]> {
  1728→  1728→  const seen = existingKeys ?? new Set<string>();
  1729→  1729→  const loaded: LoadedSubtitle[] = [];
  1730→  1730→  for (const s of batch) {
  1731→  1731→    const label = subLabel(s);

  1733→  1733→    if (seen.has(key)) continue;
  1734→  1734→    seen.add(key);
  1735→  1735→    try {
  1736→  1736→      loaded.push(
  1737→  1737→        await fetchAndParseSubtitle(s.url, label, s.lang, {
  1738→  1738→          source: s.addonName ?? s.source,
