  1703→function fmtTime(t: number): string {
  1704→  if (!Number.isFinite(t) || t < 0) return "0:00";
  1705→  const h = Math.floor(t / 3600);
  1706→  const m = Math.floor((t % 3600) / 60);
  1707→  const s = Math.floor(t % 60);
  1708→  return h > 0
  1709→    ? `${h}:${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`
  1710→    : `${m}:${String(s).padStart(2, "0")}`;
  1711→}

  1721→// Fetch + parse a batch of subtitle entries, skipping failures.
  1722→// Dedupe within the batch itself (two providers often serve the same file)
  1723→// and against `existingKeys` so re-parses can never double-commit a track.
  1724→async function parseSubBatch(
  1725→  batch: RawSubtitle[],
  1726→  existingKeys?: Set<string>,
  1727→): Promise<LoadedSubtitle[]> {
  1728→  const seen = existingKeys ?? new Set<string>();
  1729→  const loaded: LoadedSubtitle[] = [];
  1730→  for (const s of batch) {
  1731→    const label = subLabel(s);
  1732→    const key = subtitleDedupeKey({ url: s.url, lang: s.lang, label });
  1733→    if (seen.has(key)) continue;
  1734→    seen.add(key);
  1735→    try {
  1736→      loaded.push(
  1737→        await fetchAndParseSubtitle(s.url, label, s.lang, {
  1738→          source: s.addonName ?? s.source,
  1739→          aiTranslated: s.ai_translated === true || s.ai_translated === "true",
  1740→        }),
  1741→      );
  1742→    } catch {
  1743→      /* skip bad sub */
  1744→    }
  1745→  }
  1746→  return loaded;

  1752→  payload: PlayerPayload,
  1753→  position: number,
  1754→  duration: number,
  1755→  settings: ReturnType<typeof useSettings.getState>["settings"],
  1756→) {

  1772→      episodeName: payload.episodeName,

  1774→      durationMs: Math.round(duration * 1000),

  1780→  payload: PlayerPayload,
  1781→  position: number,
  1782→  duration: number,
  1783→  settings: ReturnType<typeof useSettings.getState>["settings"],
  1784→  final: boolean,
  1785→) {

  1788→    id: payload.metaId,
  1789→    type: payload.type === "series" ? ("series" as const) : ("movie" as const),
  1790→    name: payload.title,

  1792→    season: payload.season,
  1793→    episode: payload.episode,

  1795→    episodeName: payload.episodeName,
  1796→    positionMs: Math.round(position * 1000),
  1797→    durationMs: Math.round(duration * 1000),
  1798→    t: Date.now(),
  1799→  };
  1800→  const progress = position / duration;
  1801→  if (!final || progress < 0.92) {
  1802→    upsertCw(entry);
  1803→  }
  1804→  if (final) {
  1805→    pushHistory(entry);
